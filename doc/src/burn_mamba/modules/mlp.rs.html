1//! The SwiGLU feed-forward block that some checkpoints interleave between SSM
2//! mixers (`mamba_ssm`'s `GatedMLP`, selected there by `d_intermediate > 0`).
3//!
4//! ```text
5//!   [v | g] = fc1(x)            split of the `2·hidden` output down the last dim
6//!   y       = fc2(v ⊙ silu(g))
7//! ```
8//!
9//! The value half comes first and the gate half second — the order the reference
10//! `y.chunk(2, dim=-1)` produces, and therefore the order the checkpoint's single
11//! fused `fc1` matrix is stored in.
12//!
13//! `hidden` is rounded **up** to a multiple of [`GatedMlpConfig::multiple_of`], so
14//! the config figure and the checkpoint shape can legitimately disagree: the
15//! Mamba-3 MIMO 187m `config.json` says `d_intermediate: 1264`, while its
16//! `fc1.weight` is `[2·1280, 768]`.
17//!
18//! Point-wise in the sequence, so `forward` and `step` are the same map applied at
19//! different ranks — hence the rank-generic body.
20
21use crate::modules::Silu;
22use crate::modules::split_into;
23use burn::nn::{Linear, LinearConfig};
24use burn::prelude::*;
25
26/// Configuration to create a [`GatedMlp`].
27#[derive(Config, Debug)]
28pub struct GatedMlpConfig {
29    /// Input and output width (the residual stream), `d_model`.
30    pub d_model: usize,
31    /// Inner width **before** rounding — `d_intermediate` in the reference
32    /// configs. Rounded up to a multiple of [`Self::multiple_of`] by
33    /// [`Self::hidden`].
34    pub d_intermediate: usize,
35    /// Round [`Self::d_intermediate`] up to a multiple of this.
36    #[config(default = 128)]
37    pub multiple_of: usize,
38    /// Whether `fc1`/`fc2` carry a bias term.
39    #[config(default = false)]
40    pub has_proj_bias: bool,
41}
42
43impl GatedMlpConfig {
44    /// The realised inner width: [`Self::d_intermediate`] rounded up to a
45    /// multiple of [`Self::multiple_of`].
46    pub fn hidden(&self) -> usize {
47        self.d_intermediate.div_ceil(self.multiple_of) * self.multiple_of
48    }
49
50    /// Initialize a new [`GatedMlp`] module.
51    pub fn init(&self, device: &Device) -> GatedMlp {
52        let hidden = self.hidden();
53        GatedMlp {
54            fc1: LinearConfig::new(self.d_model, 2 * hidden)
55                .with_bias(self.has_proj_bias)
56                .init(device),
57            fc2: LinearConfig::new(hidden, self.d_model)
58                .with_bias(self.has_proj_bias)
59                .init(device),
60            activation: Silu::new(),
61        }
62    }
63}
64
65/// A SwiGLU feed-forward block: `fc2(v ⊙ silu(g))` where `[v | g] = fc1(x)`.
66///
67/// Should be created using the [`GatedMlpConfig`] configuration.
68#[derive(Module, Debug)]
69pub struct GatedMlp {
70    /// Fused value+gate up-projection, `d_model → 2·hidden`.
71    pub fc1: Linear,
72    /// Down-projection, `hidden → d_model`.
73    pub fc2: Linear,
74    /// The gate activation (SiLU).
75    pub activation: Silu,
76}
77
78impl GatedMlp {
79    /// The inner width `hidden` (inferred from `fc2`).
80    pub fn hidden(&self) -> usize {
81        let [hidden, _d_model] = self.fc2.weight.dims();
82        hidden
83    }
84
85    /// Applies the forward pass on the input tensor.
86    ///
87    /// # Shapes
88    /// - input `x`: `[..., d_model]`
89    /// - output: `[..., d_model]`
90    pub fn forward<const D: usize>(&self, x: Tensor<D>) -> Tensor<D> {
91        let hidden = self.hidden();
92        let [value, gate] = split_into(self.fc1.forward(x), [hidden, hidden], D - 1);
93        self.fc2.forward(value * self.activation.forward(gate))
94    }
95}
