11/// A single Pre-LN block wrapper computing `M(RMSNorm(x))` — the residual is
12/// **not** applied here. The enclosing [`Layers`](crate::modules::Layers) owns
13/// that decision (add the input back, suppress it on the first/last layer, or
14/// thread it through Multi-Gate streams), so no input clone / zero-add is wasted
15/// when no residual is wanted.
16///
17/// With [`Self::mlp`] set the layer additionally runs a second Pre-LN sub-block,
18/// the SwiGLU feed-forward of `mamba_ssm`'s `d_intermediate > 0` checkpoints. It
19/// has a residual of its own, *inside* the layer, which is the reason the
20/// methods below return the layer's **total delta** rather than the mixer output:
21///
22/// ```text
23///   h₁ = M(norm(x))                     the mixer sub-block
24///   h₂ = mlp(norm2(x + h₁))             the feed-forward sub-block
25///   return h₁ + h₂                      so that Layers' `x + delta` is
26///                                       (x + h₁) + h₂ — both residuals
27/// ```
28///
29/// Folding it this way keeps [`Layers`] the single owner of the *outer* residual
30/// (and of the `ignore_first/last_residual` ablations, which therefore govern
31/// only that outer add — the feed-forward's inner residual is intrinsic to the
32/// sub-block and always applies). Without an `mlp` the delta is just `h₁` and
33/// nothing changes for Mamba-1/2 or for Mamba-3 checkpoints that carry no MLP.
34///
35/// May carry its own [`ClassLatent`]s, placed from a [`ClassCursor`]: `step`
36/// splices them around the token it is given, while in `forward` the caller
37/// splices them first (via [`Self::insert_latents`]) so the residual it adds
38/// sees the same lengthened sequence; [`Self::prime`] steps the ones waiting for
39/// the next token *without* that token. They are independent of any class
40/// latents on the enclosing [`Layers`].
41#[derive(Module, Debug)]
42pub struct Layer<M: Module> {
43    /// Pre-norm applied before the inner block.
44    pub norm: RmsNorm,
45    /// The inner Mamba-x SSM block.
46    pub mamba_block: M,
47    /// Pre-norm of the feed-forward sub-block. `Some` exactly when [`Self::mlp`]
48    /// is (`norm2` in the reference checkpoints).
49    pub norm2: Option<RmsNorm>,
50    /// Optional SwiGLU feed-forward sub-block run after the mixer, with its own
51    /// residual. `None` ⇒ the layer is mixer-only.
52    pub mlp: Option<GatedMlp>,
53    /// Positions of this layer's class latents (empty ⇒ none).
54    #[module(skip)]
55    pub class_latents: Vec<ClassLatent>,
56    /// The class-latent embeddings, `[num_class_latents, d_model]` (`None` ⇒ none).
57    pub class_latents_emb: Option<Param<Tensor<2>>>,
58}
59
60impl<M: MambaBlock> Layer<M> {
61    /// Splice this layer's class latents into the chunk `x` (no-op when there
62    /// are none), advancing `class` past it.
63    ///
64    /// Public so a caller driving a bare [`Layer`] can lengthen the sequence
65    /// itself (and add the matching residual) before calling [`Self::forward`].
66    /// `None` cursors ⇒ this chunk is the whole sequence. [`Layers`] splices its
67    /// layers' latents itself, since under
68    /// [`MultiGate`](crate::modules::MultiGate) residuals the same rows must
69    /// also enter the carried streams.
70    pub fn insert_latents(&self, x: Tensor<3>, class: Option<&mut ClassCursor>) -> Tensor<3> {
71        let mut whole = ClassCursor::whole(x.dims()[1]);
72        let cursor = class.unwrap_or(&mut whole);
73        insert_class_markers(
74            x,
75            &self.class_latents,
76            self.class_latents_emb.as_ref(),
77            cursor,
78            "Layer",
79        )
80    }
81
82    /// The layer input, kept only when the feed-forward sub-block needs it for
83    /// its inner residual — otherwise `None`, so the mixer-only path still moves
84    /// `x` straight into the pre-norm with no clone.
85    fn mlp_residual<const D: usize>(&self, x: &Tensor<D>) -> Option<Tensor<D>> {
86        self.mlp.as_ref().map(|_| x.clone())
87    }
88
89    /// Completes the layer's total delta: `h₁ ↦ h₁ + mlp(norm2(x + h₁))`.
90    ///
91    /// `residual` is whatever [`Self::mlp_residual`] captured, so a `None` here
92    /// means there is no feed-forward and the delta is the mixer output alone.
93    fn add_mlp_delta<const D: usize>(
94        &self,
95        residual: Option<Tensor<D>>,
96        h1: Tensor<D>,
97    ) -> Tensor<D> {
98        let Some(mlp) = self.mlp.as_ref() else {
99            return h1;
100        };
101        let x = residual.expect("`mlp_residual` captures the input whenever `mlp` is present");
102        let norm2 = self
103            .norm2
104            .as_ref()
105            .expect("`norm2` is allocated alongside `mlp`");
106        let h2 = mlp.forward(norm2.forward(x + h1.clone()));
107        h1 + h2
108    }
109
110    /// Full-sequence Pre-LN block **without** the outer residual: the layer's
111    /// total delta `M(RMSNorm(x))`, plus the feed-forward sub-block's own
112    /// contribution when [`Self::mlp`] is set (see the type docs).
113    ///
114    /// The caller owns any class-latent insertion ([`Self::insert_latents`]) and
115    /// the outer residual.
116    pub fn forward(
117        &self,
118        x: Tensor<3>,
119        cache: Option<M::Cache>,
120        ssd_path: M::SsdPath,
121    ) -> (Tensor<3>, M::Cache) {
122        let residual = self.mlp_residual(&x);
123        let normed = self.norm.forward(x);
124        let (h1, cache) = self.mamba_block.block_forward(normed, cache, ssd_path);
125        (self.add_mlp_delta(residual, h1), cache)
126    }
127
128    /// Single-token Pre-LN block step **without** the residual.
129    ///
130    /// `class` is this layer's own class-latent cursor. With `Some`, every
131    /// latent whose position falls on this token is stepped around it — before
132    /// it (`Start`/`Middle`/`Custom`, which precede a token) or after it (`End`,
133    /// which closes the sequence) — each a step of its own. What comes back is
134    /// the **last** token the step emitted (see
135    /// [`ClassCursors`](crate::utils::ClassCursors)): the user token, unless an
136    /// `End` latent follows it, that latent being then the sequence's true last
137    /// token. With `None` no class latents are injected — and `Middle`/`End`
138    /// latents panic (their positions need the full sequence length). The
139    /// residual is the caller's responsibility.
140    pub fn step(
141        &self,
142        x: Tensor<2>,
143        cache: Option<M::Cache>,
144        class: Option<&mut ClassCursor>,
145    ) -> (Tensor<2>, M::Cache) {
146        let Some(cursor) = class else {
147            assert_full_len_known(&self.class_latents, None, "Layer");
148            return self.step_one(x, cache);
149        };
150        let plan = class_chunk_plan(&self.class_latents, 1, cursor, "Layer");
151        if plan.is_empty() {
152            return self.step_one(x, cache);
153        }
154        // `at == 0` ⇒ the latent precedes the user token, `at == 1` ⇒ it is an
155        // `End` closing the sequence, and follows it.
156        let [batch, d_model] = x.dims();
157        let row = |i: usize| class_row(self.class_latents_emb.as_ref(), i, batch, d_model);
158        let (before, after): (Vec<_>, Vec<_>) = plan.into_iter().partition(|&(at, _)| at == 0);
159        let mut cache = cache;
160        for (_, i) in before {
161            let (_discard, c) = self.step_one(row(i), cache);
162            cache = Some(c);
163        }
164        let (mut out, mut cache) = self.step_one(x, cache);
165        for (_, i) in after {
166            // A closing `End` *is* the sequence's last token — its output, not
167            // the user token's, is what this step produced.
168            let (o, c) = self.step_one(row(i), Some(cache));
169            out = o;
170            cache = c;
171        }
172        (out, cache)
173    }
174
175    /// Step the class latents this layer has waiting for its next token — with
176    /// **no** token of its own, so nothing but class data is consumed.
177    ///
178    /// This is [`Self::step`]'s opening half on its own (see
179    /// [`ClassCursors`](crate::utils::ClassCursors)): the latents that would
180    /// have preceded the next token are stepped now, in the same order, so a
181    /// `prime` followed by a `step` runs exactly the sequence that `step` alone
182    /// would have. `End` latents are never primed — closing the sequence, they
183    /// belong to the step carrying its last token.
184    ///
185    /// Returns the **last** latent stepped, as the pair `(delta, latent)` — this
186    /// layer's own embedding row alongside the delta it produced, since the
187    /// caller has no other way to complete the residual (`delta + latent`, as it
188    /// does with the token it hands to [`Self::step`]). `None` ⇒ nothing was
189    /// waiting, and the cache comes back exactly as it went in (`None` included:
190    /// a layer that stepped nothing has the state it already had).
191    pub fn prime(
192        &self,
193        batch: usize,
194        cache: Option<M::Cache>,
195        class: Option<&mut ClassCursor>,
196    ) -> (Option<(Tensor<2>, Tensor<2>)>, Option<M::Cache>) {
197        let Some(cursor) = class else {
198            // No cursor ⇒ nothing is injected, exactly as in a `None` step.
199            assert_full_len_known(&self.class_latents, None, "Layer");
200            return (None, cache);
201        };
202        let plan = class_prime_plan(&self.class_latents, 0, cursor, "Layer");
203        if plan.is_empty() {
204            return (None, cache);
205        }
206        let width = class_emb_width(self.class_latents_emb.as_ref());
207        let mut cache = cache;
208        let mut last = None;
209        for (_at, i) in plan {
210            let row = class_row(self.class_latents_emb.as_ref(), i, batch, width);
211            let (out, c) = self.step_one(row.clone(), cache);
212            last = Some((out, row));
213            cache = Some(c);
214        }
215        (last, cache)
216    }
217
218    /// The actual one-token work: no class injection, no outer residual.
219    ///
220    /// `pub(crate)` for [`Layers`]'s cascade, which places this layer's class
221    /// latents from the stack-wide [`ClassCursors`](crate::utils::ClassCursors)
222    /// itself and so must bypass [`Self::step`]'s cursorless guard (that guard
223    /// rejects `Middle`/`End`, which the cascade has already resolved).
224    pub(crate) fn step_one(&self, x: Tensor<2>, cache: Option<M::Cache>) -> (Tensor<2>, M::Cache) {
225        let residual = self.mlp_residual(&x);
226        let normed = self.norm.forward(x);
227        let (h1, cache) = self.mamba_block.block_step(normed, cache);
228        (self.add_mlp_delta(residual, h1), cache)
229    }
230
231    /// Stationary fixed point of the Pre-LN block under a constant token,
232    /// **without** the residual: the `step` counterpart of infinitely many
233    /// identical tokens (closed form, no cache — see
234    /// [`MambaBlock::block_step_infinite`]). Cursorless: class latents are not
235    /// injected (`Middle`/`End` latents panic, as in a `None`-cursor `step`).
236    /// The feed-forward sub-block is point-wise, so it composes with the limit:
237    /// once the mixer output settles, `x + h₁` is constant and so is `h₂`.
238    pub fn step_infinite(&self, x: Tensor<2>) -> Tensor<2> {
239        assert_full_len_known(&self.class_latents, None, "Layer");
240        let residual = self.mlp_residual(&x);
241        let h1 = self.mamba_block.block_step_infinite(self.norm.forward(x));
242        self.add_mlp_delta(residual, h1)
243    }
244}
