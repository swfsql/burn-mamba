1//! # Pooled SSM-state moments (for state participation ratios)
2//!
3//! [`StateMoments`] carries the **exact** first and second moments of every
4//! per-token SSM state `hₜ ∈ ℝ^{per_head_dim × state_rank}` of a `forward`
5//! pass, treating each state **row** (one `(token, per_head_dim)` pair) as a
6//! sample in `ℝ^{state_rank}` — the same sample convention a token-by-token
7//! `step` loop reading the cache would produce.
8//!
9//! The moments are all a participation ratio (PR, a differentiable effective
10//! rank) needs: with `Σ` the sample covariance, `PR = (tr Σ)² / tr(Σ²)` and
11//! both traces derive from `Σ hhᵀ`, `Σ h`, and the sample count. Storing raw
12//! **sums** (not averages) makes moments *composable*: [`StateMoments::merge`]
13//! pools across forward calls (streaming chunks, eval batches) and the PR of
14//! the merged moments is the exact pooled PR.
15
16use burn::prelude::*;
17
18/// Raw (un-normalised) first/second moments of the per-token SSM states of
19/// one block's `forward` pass, pooled over tokens and `per_head_dim` rows.
20///
21/// Produced by `forward_with_state_moments` (currently Mamba-2 only) without
22/// ever materialising the per-token states — see
23/// `Mamba2SsdInput::state_moments`.
24#[derive(Debug, Clone)]
25pub struct StateMoments {
26    /// Second-moment (Gram) sum `Σₜ hₜᵀ hₜ` — the `ᵀ` contraction pools the
27    /// `per_head_dim` rows, the `Σₜ` the (unpadded) tokens.
28    ///
29    /// # Shape
30    /// - `[batch, nheads, state_rank, state_rank]`
31    pub m2_bhrr: Tensor<4>,
32    /// First-moment sum `Σₜ Σₚ hₜ[p, :]`.
33    ///
34    /// # Shape
35    /// - `[batch, nheads, state_rank]`
36    pub m1_bhr: Tensor<3>,
37    /// Samples pooled into each `(batch, head)` slice:
38    /// `valid_tokens · per_head_dim` (grows additively under [`Self::merge`]).
39    pub count: usize,
40}
41
42impl StateMoments {
43    /// Pool two moment sets (e.g. consecutive streamed `forward` calls, or
44    /// separate eval batches). PR of the merged moments is the exact PR of
45    /// the union of samples.
46    pub fn merge(self, other: Self) -> Self {
47        assert_eq!(
48            self.m2_bhrr.dims(),
49            other.m2_bhrr.dims(),
50            "merged state moments must share [batch, nheads, state_rank]"
51        );
52        Self {
53            m2_bhrr: self.m2_bhrr + other.m2_bhrr,
54            m1_bhr: self.m1_bhr + other.m1_bhr,
55            count: self.count + other.count,
56        }
57    }
58
59    /// Fold the batch dimension into the samples (batch-pooled moments with
60    /// `batch = 1`), matching diagnostics that treat every
61    /// `(token, batch, per_head_dim)` triple as one sample.
62    pub fn pool_batch(self) -> Self {
63        let [batch, _h, _r, _r2] = self.m2_bhrr.dims();
64        Self {
65            m2_bhrr: self.m2_bhrr.sum_dim(0),
66            m1_bhr: self.m1_bhr.sum_dim(0),
67            count: self.count * batch,
68        }
69    }
70
71    /// Participation ratio `(tr Σ)² / tr(Σ²)` of the sample covariance, per
72    /// `(batch, head)` slice; `center` subtracts the sample mean (`Σ` becomes
73    /// the centered covariance instead of the raw second moment).
74    ///
75    /// Differentiable (two traces, no eigendecomposition).
76    ///
77    /// # Shape
78    /// - output: `[batch, nheads]`
79    pub fn pr(&self, center: bool) -> Tensor<2> {
80        let [batch, nheads, state_rank, _] = self.m2_bhrr.dims();
81        assert!(self.count > 0, "state moments hold no samples");
82        let device = self.m2_bhrr.device();
83        let samples = self.count as f32;
84
85        let sigma_bhrr = {
86            let m2_bhrr = self.m2_bhrr.clone() / samples;
87            if center {
88                let mu_bhr = self.m1_bhr.clone() / samples;
89                let outer_bhrr =
90                    mu_bhr.clone().unsqueeze_dim::<4>(3) * mu_bhr.unsqueeze_dim::<4>(2);
91                m2_bhrr - outer_bhrr
92            } else {
93                m2_bhrr
94            }
95        };
96
97        // tr Σ via an identity mask; tr(Σ²) = ‖Σ‖²_F (Σ is symmetric).
98        let eye_11rr = Tensor::<2>::eye(state_rank, &device).unsqueeze::<4>();
99        let tr_bh = (sigma_bhrr.clone() * eye_11rr.clone())
100            .sum_dim(3)
101            .sum_dim(2)
102            .reshape([batch, nheads]);
103        // `PR = (tr Σ)² / tr(Σ²)`. Computed via the trace-normalised
104        // `Σ̂ = Σ / tr(Σ).detach()`, keeping **both** traces of `Σ̂`:
105        //
106        //     PR = tr(Σ̂)² / tr(Σ̂²).
107        //
108        // With `c = tr(Σ).detach()` a frozen scalar, the `c²` cancels between
109        // numerator and denominator, so this equals `tr(Σ)²/tr(Σ²)` **as a
110        // function of Σ** — identical value *and* exact gradient — while every
111        // differentiated intermediate stays O(1). Two subtleties that a naive
112        // rewrite gets wrong:
113        //   - Keep the numerator `tr(Σ̂)²`: it is numerically 1, but with the
114        //     normaliser detached its *gradient* w.r.t. Σ is not zero — it
115        //     carries PR's rank-reducing (trace-tangential) component.
116        //     Collapsing to `1/tr(Σ̂²)` drops it, leaving only the radial
117        //     (magnitude) direction, orthogonal to ∇PR — a penalty that no
118        //     longer reduces rank (see `pr_gradient_matches_direct_formula`).
119        //   - Differentiating *through* `tr(Σ)` instead (no detach) is
120        //     value/grad-correct but puts `tr(Σ²)²` in the backward, which
121        //     underflows fp32 to 0 (→ NaN gradient) once the state magnitude
122        //     `tr(Σ) ≲ 1e-11` — which weight decay drives it toward. The
123        //     detached O(1) form is finite at every representable magnitude
124        //     (see `pr_gradient_finite_as_magnitude_shrinks`).
125        //
126        // Two floors, for two quantities at very different scales. The
127        // normaliser `tr Σ` is a *magnitude* that weight decay drives down to —
128        // and below — `div_eps` (the crate's O(1)-calibrated negligibility
129        // threshold): flooring it there would corrupt PR across the live
130        // operating range, so it is floored only at the dtype's smallest
131        // positive normal (`finfo().min_positive`), firing solely for an
132        // all-zero state (`Σ ≡ 0`, e.g. a dead head — then `Σ̂ = 0/ε = 0`, no
133        // `0/0`). `tr(Σ̂²)` is scale-normalised (`∈ [1/r, 1]` for any nonzero Σ)
134        // and nears zero only for `Σ ≡ 0`, so `div_eps(dtype)` is the correct
135        // dtype-aware guard there (cf. `MseLoss`'s fp16 path).
136        let dtype = self.m2_bhrr.dtype();
137        let min_positive = dtype.finfo().expect("state moments are a float dtype").min_positive;
138        let scale_bh = tr_bh.clamp_min(min_positive).detach();
139        let sigma_hat = sigma_bhrr / scale_bh.reshape([batch, nheads, 1, 1]);
140        let tr1_hat_bh = (sigma_hat.clone() * eye_11rr)
141            .sum_dim(3)
142            .sum_dim(2)
143            .reshape([batch, nheads]);
144        let tr2_hat_bh = sigma_hat
145            .powf_scalar(2.0)
146            .sum_dim(3)
147            .sum_dim(2)
148            .reshape([batch, nheads])
149            .clamp_min(crate::utils::div_eps(dtype));
150        tr1_hat_bh.clone() * tr1_hat_bh / tr2_hat_bh
151    }
152
153    /// Raw uncentered state magnitude `tr Σ = trace(m2)/count` per
154    /// `(batch, head)` — the mean squared state magnitude `⟨‖h‖²⟩`, which is
155    /// [`Self::pr`]'s numerator scale. Reported alongside PR to tell a genuine
156    /// rank-1 state (`PR → 1`, magnitude healthy) apart from a state
157    /// collapsing toward zero (where `pr`'s `1e-12` denominator clamp drags
158    /// the ratio below its true floor of 1).
159    ///
160    /// # Shape
161    /// - output: `[batch, nheads]`
162    pub fn trace(&self) -> Tensor<2> {
163        let [batch, nheads, state_rank, _] = self.m2_bhrr.dims();
164        assert!(self.count > 0, "state moments hold no samples");
165        let eye_11rr = Tensor::<2>::eye(state_rank, &self.m2_bhrr.device()).unsqueeze::<4>();
166        (self.m2_bhrr.clone() * eye_11rr)
167            .sum_dim(3)
168            .sum_dim(2)
169            .reshape([batch, nheads])
170            / self.count as f32
171    }
172}
