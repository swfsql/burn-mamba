1//! The row end: where a row reads each cache field that keeps its last
2//! samples.
3//!
4//! Some cache fields keep the last `width` samples of a row, along one axis:
5//!
6//! - the conv window of Mamba-1 and Mamba-2 (`width = conv_kernel`),
7//! - the tap FIFO of Mamba-3 (`width = lag`),
8//! - the `ln Λ` and `c` carries of the Mamba-3 positive systems
9//!   (`width = 1`).
10//!
11//! A call gives these fields for the next call. Each field is the last `width`
12//! entries before the row end of each row, in the samples of the call after a
13//! `start` field:
14//!
15//! ```text
16//!   [ start | samples of the call ]  →  the `width` entries before the row end
17//! ```
18//!
19//! `start` has `width` entries. It holds the samples from before the call, so
20//! a call that is shorter than `width` also gives `width` entries. A
21//! [`RowEnd`] does each such read of one call. The code builds it once for
22//! each call, from the padding and the resets of the call.
23//!
24//! # Padding
25//!
26//! Under right padding ([`crate::padding`]), the row end of a row is one past
27//! its last real position. A row with no real position reads `start`. Each row
28//! has its own row end. So the read is a gather at an offset for each row,
29//! with a fixed shape for all lengths.
30//!
31//! # Resets
32//!
33//! A packed call ([`crate::packing`]) returns the cache of the last segment of
34//! each row, and that segment starts from the origin cache. So for a row with
35//! a reset in the call, `start` is the origin field. Each entry from before the
36//! start of the last segment comes from `start`. For a row with no reset,
37//! `start` is the incoming field.
38//!
39//! Let `s` be the first position of the last segment of a row, or `0` for a
40//! row with no reset. Let `e` be the row end. Then:
41//!
42//! - In `[start | last segment]`, the read starts at `e′ = max(e − s, 0)`, so
43//!   its entry `j < width` is at `i = e′ + j`.
44//! - In `[start | samples]`, that entry is at `i` for `i < width` (an entry of
45//!   `start`), else at `i + s`.
46//! - With no reset, `s = 0`, and the read is `[e, e + width)` of
47//!   `[start | samples]`.
48//!
49//! A last segment that holds only padding has `e ≤ s`, and its read is
50//! `start`: the field of the origin.
51//!
52//! # One `narrow`
53//!
54//! The gather and the `cat` cost memory. A read is one `narrow`, with no
55//! gather, when no row of the call has padding and one of these conditions is
56//! true:
57//!
58//! - The call has no resets.
59//! - `width` is not more than the positions of one token.
60//!
61//! The row end of each row is then the end of the call. A segment holds at
62//! least one token of the call, so a read of one token or less stays in the
63//! last segment. A Mamba-3 read has `width ≤ u`, because a tap reads at most
64//! `u` positions back (`lag ≤ u`) and a positive carry reads 1. A packed
65//! Mamba-2 call still gathers. Its last segment can be shorter than
66//! `conv_kernel`, because a packed call requires only
67//! `chunk_len ≥ conv_kernel − 1`.
68//!
69//! When the call holds at least `width` positions, that `narrow` reads the
70//! samples alone, with no `cat`.
71//!
72//! In the shape suffixes of this module, `w` is the `width` of a read.
73

76/// The row-end reader of one call: it reads each "last samples" cache field
77/// at the row end of each row. See the module header.
78pub(crate) struct RowEnd {
79    /// `[batch]`: the row end of each row. `None` when no row of the call has
80    /// padding. Each row end is then the end of the call.
81    end_b: Option<Tensor<1, Int>>,
82    /// `[batch, nchunks]` (`true` at a chunk that starts a segment), and the
83    /// positions of one chunk. `None` for a call with no resets.
84    resets: Option<(Tensor<2, Bool>, usize)>,
85    /// The positions of one token: `u` for Mamba-3, `1` for Mamba-1 and
86    /// Mamba-2.
87    micro_steps: usize,
88}
89
90impl RowEnd {
91    /// The reader of one call, from its padding (`pad_bs`, `true` at a padded
92    /// position) and its resets. `resets` is `(reset_bn, chunk_len)`: `true`
93    /// at a chunk that starts a segment, and the positions of one chunk.
94    /// `micro_steps` is the positions of one token.
95    ///
96    /// # Shapes
97    /// - `pad_bs`   : `[batch, len]`
98    /// - `reset_bn` : `[batch, nchunks]`
99    pub fn new(
100        pad_bs: Option<&Tensor<2, Bool>>,
101        resets: Option<(&Tensor<2, Bool>, usize)>,
102        micro_steps: usize,
103    ) -> Self {
104        Self {
105            end_b: pad_bs.map(row_end_b),
106            resets: resets.map(|(reset_bn, chunk_len)| (reset_bn.clone(), chunk_len)),
107            micro_steps,
108        }
109    }
110
111    /// The last `width` entries of each row before its row end, along `axis`,
112    /// from `[start | samples]`. Axis 0 is the batch.
113    ///
114    /// `start` is `origin` for a row with a reset in the call, else
115    /// `incoming`. Each one has `width` entries along `axis`. A call with no
116    /// resets does not read `origin`.
117    ///
118    /// # Panics
119    ///
120    /// If the call has resets and `origin` is `None`.
121    pub fn read<const D: usize>(
122        &self,
123        samples: Tensor<D>,
124        axis: usize,
125        width: usize,
126        incoming: Tensor<D>,
127        origin: Option<Tensor<D>>,
128    ) -> Tensor<D> {
129        let reset_bn = self.resets.as_ref().map(|(reset_bn, _)| reset_bn);
130        assert!(
131            reset_bn.is_none() || origin.is_some(),
132            "a row-end read in a call with resets needs the origin field: the last segment of a row \
133             with a reset starts from it"
134        );
135        self.read_after(samples, axis, width, || match reset_bn.zip(origin) {
136            None => incoming,
137            Some((reset_bn, origin)) => origin_where_reset(incoming, origin, reset_bn),
138        })
139    }
140
141    /// [`Self::read`] with zeros for `start` in each row. The decay of the
142    /// slots of the tap FIFO reads this way, because the incoming slots
143    /// already carry their decay.
144    #[cfg_attr(not(feature = "mamba3"), allow(dead_code))]
145    pub fn read_after_zeros<const D: usize>(&self, samples: Tensor<D>, axis: usize, width: usize) -> Tensor<D> {
146        let mut dims = samples.dims();
147        dims[axis] = width;
148        let options = (&samples.device(), samples.dtype());
149        self.read_after(samples, axis, width, || Tensor::zeros(dims, options))
150    }
151
152    /// The read from `[start() | samples]`. Only a read that holds an entry
153    /// of `start` calls `start`.
154    fn read_after<const D: usize>(
155        &self,
156        samples: Tensor<D>,
157        axis: usize,
158        width: usize,
159        start: impl FnOnce() -> Tensor<D>,
160    ) -> Tensor<D> {
161        let dims = samples.dims();
162        let [batch, len] = [dims[0], dims[axis]];
163        if self.end_b.is_none() && (self.resets.is_none() || width <= self.micro_steps) {
164            if len >= width {
165                return samples.narrow(axis, len - width, width);
166            }
167            // A call is at least one token. So a call that is shorter than
168            // `width` has no resets here, and `start` is the incoming field.
169            return Tensor::cat(vec![start(), samples], axis).narrow(axis, len, width);
170        }
171        let device = samples.device();
172        let end_bw = match &self.end_b {
173            Some(end_b) => end_b.clone(),
174            None => Tensor::full([batch], len as i64, &device),
175        }
176        .reshape([batch, 1])
177        .expand([batch, width]);
178        let offset_bw = Tensor::<1, Int>::arange(0..width as i64, &device)
179            .reshape([1, width])
180            .expand([batch, width]);
181        let idx_bw = match &self.resets {
182            None => offset_bw + end_bw,
183            Some((reset_bn, chunk_len)) => {
184                let first_bw = last_segment_start_b(reset_bn, *chunk_len)
185                    .reshape([batch, 1])
186                    .expand([batch, width]);
187                // The index in `[start | last segment]`, then in
188                // `[start | samples]`.
189                let local_bw = offset_bw + (end_bw - first_bw.clone()).clamp_min(0);
190                let in_start_bw = local_bw.clone().lower_elem(width as i64);
191                local_bw + first_bw.mask_fill(in_start_bw, 0)
192            }
193        };
194        assert_eq!([batch, width], idx_bw.dims());
195        gather_rows(Tensor::cat(vec![start(), samples], axis), axis, idx_bw)
196    }
197}

199/// `incoming`, where each row with a reset takes its row of `origin`: the
200/// field that the last segment of each row starts from. Axis 0 is the batch.
201fn origin_where_reset<const D: usize>(
202    incoming: Tensor<D>,
203    origin: Tensor<D>,
204    reset_bn: &Tensor<2, Bool>,
205) -> Tensor<D> {
206    let dims = incoming.dims();
207    let mut shape = [1usize; D];
208    shape[0] = dims[0];
209    let reset_rows = reset_bn.clone().any_dim(1).reshape(shape).expand(dims);
210    incoming.mask_where(reset_rows, origin)
211}

213/// The first position of the last segment of each row, `[batch]`. A row with
214/// no reset gives `0`.
215fn last_segment_start_b(reset_bn: &Tensor<2, Bool>, chunk_len: usize) -> Tensor<1, Int> {
216    let [batch, nchunks] = reset_bn.dims();
217    (Tensor::<1, Int>::arange(0..nchunks as i64, &reset_bn.device()) * chunk_len as i64)
218        .reshape([1, nchunks])
219        .expand([batch, nchunks])
220        .mask_fill(reset_bn.clone().bool_not(), 0)
221        .max_dim(1)
222        .reshape([batch])
223}

225/// The row end of each row, `[batch]`: one past its last real position. A row
226/// with no real position gives `0`. In a packed row, padded positions can end
227/// each segment, and only the last real position counts.
228fn row_end_b(pad_bs: &Tensor<2, Bool>) -> Tensor<1, Int> {
229    let [batch, len] = pad_bs.dims();
230    Tensor::<1, Int>::arange(1..len as i64 + 1, &pad_bs.device())
231        .reshape([1, len])
232        .expand([batch, len])
233        .mask_fill(pad_bs.clone(), 0)
234        .max_dim(1)
235        .reshape([batch])
236}

238/// The entries `idx_bw` of `x` along `axis`, one set for each row. Axis 0 is
239/// the batch.
240fn gather_rows<const D: usize>(x: Tensor<D>, axis: usize, idx_bw: Tensor<2, Int>) -> Tensor<D> {
241    let [batch, width] = idx_bw.dims();
242    let mut shape = [1usize; D];
243    shape[0] = batch;
244    shape[axis] = width;
245    let mut dims = x.dims();
246    dims[axis] = width;
247    x.gather(axis, idx_bw.reshape(shape).expand(dims))
248}
