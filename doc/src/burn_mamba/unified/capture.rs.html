1//! Graph-capture support for the caches: the
2//! [`CacheTensors`](burn_stack::modules::CacheTensors) traversal of each
3//! family. With it, a captured `step`
4//! ([`CapturedStep`](burn_stack::utils::graph::CapturedStep)) writes its new
5//! cache **into** the buffers of the old one.
6//!
7//! There is one traversal per cache type, and it visits every tensor field. An
8//! optional field must be present in both caches or in neither. This module
9//! also shows why the stable cache must first be copied into buffers of its
10//! own ([`into_owned_buffers`](burn_stack::modules::CacheTensors::into_owned_buffers)):
11//! the Mamba-3 tap slots come back from `step` as `narrow`s of its own
12//! tensors.
13
14use burn::prelude::*;
15use burn_stack::modules::{CacheTensors, TensorZip};
16
17impl CacheTensors for super::MambaCaches {
18    fn zip_tensors(self, other: Self, z: &mut impl TensorZip) -> Self {
19        match (self, other) {
20            #[cfg(feature = "mamba1")]
21            (Self::Mamba1(a), Self::Mamba1(b)) => Self::Mamba1(a.zip_tensors(b, z)),
22            #[cfg(feature = "mamba2")]
23            (Self::Mamba2(a), Self::Mamba2(b)) => Self::Mamba2(a.zip_tensors(b, z)),
24            #[cfg(feature = "mamba3")]
25            (Self::Mamba3(a), Self::Mamba3(b)) => Self::Mamba3(a.zip_tensors(b, z)),
26            #[allow(unreachable_patterns)]
27            _ => panic!("the two caches differ in family"),
28        }
29    }
30}

49                caches: self.caches.zip_tensors(other.caches, z),

52    }

78#[cfg(feature = "mamba3")]
79mod impl_mamba3 {
80    use super::*;
81    use crate::mamba3::double_ssd::prelude::{Mamba3DoubleSsdCache, Mamba3DoubleSsdCaches};
82    use crate::mamba3::prelude::{Mamba3Cache, Mamba3Caches, RotationState};
83    use crate::mamba3::single_ssd::prelude::{Mamba3SingleSsdCache, Mamba3SingleSsdCaches};
84
85    /// An optional field: both present or both absent.
86    fn zip_option<const D: usize>(
87        a: Option<Tensor<D>>,
88        b: Option<Tensor<D>>,
89        z: &mut impl TensorZip,
90    ) -> Option<Tensor<D>> {
91        match (a, b) {
92            (Some(a), Some(b)) => Some(z.zip(a, b)),
93            (None, None) => None,
94            _ => panic!("the two caches differ in which optional fields they hold"),
95        }
96    }
97
98    fn zip_rotation(a: RotationState, b: RotationState, z: &mut impl TensorZip) -> RotationState {
99        match (a, b) {
100            (RotationState::Real(u), RotationState::Real(_)) => RotationState::Real(u),
101            (RotationState::Angle(a), RotationState::Angle(b)) => RotationState::Angle(z.zip(a, b)),
102            (RotationState::Quaternion(a), RotationState::Quaternion(b)) => {
103                RotationState::Quaternion(z.zip(a, b))
104            }
105            (RotationState::Rotor(a), RotationState::Rotor(b)) => RotationState::Rotor(z.zip(a, b)),
106            _ => panic!("the two caches differ in rotation kind"),
107        }
108    }
109
110    impl CacheTensors for Mamba3DoubleSsdCache {
111        fn zip_tensors(self, other: Self, z: &mut impl TensorZip) -> Self {
112            Self {
113                ssm_bhpr: z.zip(self.ssm_bhpr, other.ssm_bhpr),
114                k_state_bumhr: zip_option(self.k_state_bumhr, other.k_state_bumhr, z),
115                v_state_buhp: zip_option(self.v_state_buhp, other.v_state_buhp, z),
116                rotation: zip_rotation(self.rotation, other.rotation, z),
117                log_precision_bh: zip_option(self.log_precision_bh, other.log_precision_bh, z),
118                tropical_bh: zip_option(self.tropical_bh, other.tropical_bh, z),
119            }
120        }
121    }
122
123    /// The same fields as the double-ssd cache (a lossless move, see
124    /// `mamba3/cache.rs`), so it rides that traversal.
125    impl CacheTensors for Mamba3SingleSsdCache {
126        fn zip_tensors(self, other: Self, z: &mut impl TensorZip) -> Self {
127            Mamba3DoubleSsdCache::from(self)
128                .zip_tensors(other.into(), z)
129                .into()
130        }
131    }
