1//! The log-semiring prefix product that every positive member runs on.
2//!
3//! An element is a 2×2 nonnegative matrix in log coordinates, one per
4//! `[batch, position, head]`. It has two shapes:
5//!
6//! - [`Mobius`](scan::Mobius) — all four entries, acting **projectively** on
7//!   `(ℓ, 0)`. The read is `σ₀ − σ₁`, so any constant shift of the product is
8//!   allowed. The scan applies one after every combine, to keep the entries
9//!   bounded.
10//! - [`Affine`](scan::Affine) — the lower row is fixed at `(−∞, 0)`, so only
11//!   `(a, b)` is carried. The read `lse(a + c₀, b)` is absolute: no shift is
12//!   allowed, and none is necessary.
13//!
14//! [`prefix`](scan::prefix) composes them by Hillis–Steele doubling, the same
15//! schedule as the quaternion scan (`rotation::quat_cumprod`): `⌈log₂ len⌉`
16//! full-width combines. [`fold`](scan::fold) is the sequential left fold, the
17//! reference for [`prefix`](scan::prefix) in the tests.
18

22/// `ln(eᵃ + eᵇ)`, elementwise.
23///
24/// Written as `hi + ln(1 + e^(lo − hi))`, with `hi`/`lo` *selected*, not
25/// computed by `max`/`abs`. So the result is exactly the larger operand when
26/// the other is [`LOG_ZERO`] (an `a + softplus(b − a)` form would round `b`
27/// through `b − LOG_ZERO`). Its gradient on `a` is `σ(a − b)`, also at a tie,
28/// where a `max`-based form gives all of it to one side.
29pub fn lse(a: Tensor<3>, b: Tensor<3>) -> Tensor<3> {
30    let a_wins = a.clone().greater_equal(b.clone());
31    let hi = b.clone().mask_where(a_wins.clone(), a.clone());
32    let lo = a.mask_where(a_wins, b);
33    hi.clone() + (lo - hi).exp().log1p()
34}

36/// The one and zero of the semiring, `(0, LOG_ZERO)`, over `len` positions,
37/// with the shape, device and dtype of `like`. So a scan follows the dtype of
38/// its elements, not the default of the device.
39fn one_and_zero(like: &Tensor<3>, len: usize) -> (Tensor<3>, Tensor<3>) {
40    let [batch, _len, nheads] = like.dims();
41    let device = like.device();
42    let options = (&device, like.dtype());
43    (
44        Tensor::zeros([batch, len, nheads], options),
45        Tensor::full([batch, len, nheads], LOG_ZERO, options),
46    )
47}

49/// An associative element over `[batch, len, nheads]` tensors.
50pub trait Element: Sized + Clone {
51    /// `later ∘ earlier` — apply `earlier` first.
52    fn compose(later: Self, earlier: Self) -> Self;
53    /// The identity over `len` positions, with the batch, heads, device and
54    /// dtype of this element.
55    fn identity_like(&self, len: usize) -> Self;
56    /// `[batch, len, nheads]`.
57    fn dims(&self) -> [usize; 3];
58    /// A run of positions along axis 1.
59    fn narrow(self, start: usize, len: usize) -> Self;
60    /// Concatenate along axis 1.
61    fn cat(parts: Vec<Self>) -> Self;
62}

64/// A projective 2×2 element: `σ ↦ M ⊗ σ` read as `σ₀ − σ₁`.
65#[derive(Clone, Debug)]
66pub struct Mobius {
67    /// `ln M₀₀`, `[batch, len, nheads]`.
68    pub m00: Tensor<3>,
69    /// `ln M₀₁`.
70    pub m01: Tensor<3>,
71    /// `ln M₁₀`.
72    pub m10: Tensor<3>,
73    /// `ln M₁₁`.
74    pub m11: Tensor<3>,
75}
76
77impl Mobius {
78    /// The log-ratio after this (prefix) product, from `ℓ₀` per head:
79    /// `lse(M₀₀ + ℓ₀, M₀₁) − lse(M₁₀ + ℓ₀, M₁₁)`.
80    ///
81    /// # Shapes
82    /// - `carry_bh` : `[batch, nheads]`
83    /// - out        : `[batch, len, nheads]`
84    pub fn apply(self, carry_bh: Tensor<2>) -> Tensor<3> {
85        let carry_b1h = carry_bh.unsqueeze_dim::<3>(1);
86        lse(self.m00 + carry_b1h.clone(), self.m01) - lse(self.m10 + carry_b1h, self.m11)
87    }
88}
89
90impl Element for Mobius {
91    fn compose(later: Self, earlier: Self) -> Self {
92        let (x, y) = (later, earlier);
93        let m00 = lse(x.m00.clone() + y.m00.clone(), x.m01.clone() + y.m10.clone());
94        let m01 = lse(x.m00.clone() + y.m01.clone(), x.m01 + y.m11.clone());
95        let m10 = lse(x.m10.clone() + y.m00, x.m11.clone() + y.m10);
96        let m11 = lse(x.m10 + y.m01, x.m11 + y.m11);
97        // Projective: the read cancels any common shift, in value and in
98        // gradient. So subtract the largest entry (detached, because its
99        // gradient would cancel anyway), and floor the structural zeros again.
100        let shift = m00
101            .clone()
102            .max_pair(m01.clone())
103            .max_pair(m10.clone())
104            .max_pair(m11.clone())
105            .detach();
106        let floor = |t: Tensor<3>| (t - shift.clone()).clamp_min(LOG_ZERO);
107        Mobius {
108            m00: floor(m00),
109            m01: floor(m01),
110            m10: floor(m10),
111            m11: floor(m11),
112        }
113    }
114
115    fn identity_like(&self, len: usize) -> Self {
116        let (one, zero) = one_and_zero(&self.m00, len);
117        Mobius {
118            m00: one.clone(),
119            m01: zero.clone(),
120            m10: zero,
121            m11: one,
122        }
123    }
124
125    fn dims(&self) -> [usize; 3] {
126        self.m00.dims()
127    }
128
129    fn narrow(self, start: usize, len: usize) -> Self {
130        Mobius {
131            m00: self.m00.narrow(1, start, len),
132            m01: self.m01.narrow(1, start, len),
133            m10: self.m10.narrow(1, start, len),
134            m11: self.m11.narrow(1, start, len),
135        }
136    }
137
138    fn cat(parts: Vec<Self>) -> Self {
139        let mut m = [vec![], vec![], vec![], vec![]];
140        for p in parts {
141            m[0].push(p.m00);
142            m[1].push(p.m01);
143            m[2].push(p.m10);
144            m[3].push(p.m11);
145        }
146        let [m00, m01, m10, m11] = m.map(|v| Tensor::cat(v, 1));
147        Mobius { m00, m01, m10, m11 }
148    }
149}

151/// An affine element `c ↦ lse(c + a, b)` — the matrix `[[a, b], [−∞, 0]]`.
152#[derive(Clone, Debug)]
153pub struct Affine {
154    /// The log-gain, `[batch, len, nheads]` (any sign).
155    pub a: Tensor<3>,
156    /// The log-offset.
157    pub b: Tensor<3>,
158}
159
160impl Affine {
161    /// The register after this (prefix) map, from `c₀` per head:
162    /// `lse(a + c₀, b)`.
163    ///
164    /// # Shapes
165    /// - `carry_bh` : `[batch, nheads]`
166    /// - out        : `[batch, len, nheads]`
167    pub fn apply(self, carry_bh: Tensor<2>) -> Tensor<3> {
168        lse(self.a + carry_bh.unsqueeze_dim::<3>(1), self.b)
169    }
170}
171
172impl Element for Affine {
173    fn compose(later: Self, earlier: Self) -> Self {
174        // lse(lse(c + a₁, b₁) + a₂, b₂) = lse(c + (a₁ + a₂), lse(b₁ + a₂, b₂)).
175        Affine {
176            b: lse(earlier.b + later.a.clone(), later.b).clamp_min(LOG_ZERO),
177            a: (earlier.a + later.a).clamp_min(LOG_ZERO),
178        }
179    }
180
181    fn identity_like(&self, len: usize) -> Self {
182        let (one, zero) = one_and_zero(&self.a, len);
183        Affine { a: one, b: zero }
184    }
185
186    fn dims(&self) -> [usize; 3] {
187        self.a.dims()
188    }
189
190    fn narrow(self, start: usize, len: usize) -> Self {
191        Affine {
192            a: self.a.narrow(1, start, len),
193            b: self.b.narrow(1, start, len),
194        }
195    }
196
197    fn cat(parts: Vec<Self>) -> Self {
198        let (a, b) = parts.into_iter().map(|p| (p.a, p.b)).unzip();
199        Affine {
200            a: Tensor::cat(a, 1),
201            b: Tensor::cat(b, 1),
202        }
203    }
204}

206/// Inclusive prefix products `Pₜ = eₜ ∘ ⋯ ∘ e₀` along axis 1, by Hillis–Steele
207/// doubling.
208///
209/// Invariant after the round at `offset`: position `t` holds the product of
210/// the window `[max(t − 2·offset + 1, 0), t]`, where the positions before the
211/// axis are the identity. `⌈log₂ len⌉` rounds cover every window.
212///
213/// # Why doubling, and what it costs
214///
215/// `helpers::prefix_sum` blocks its scan, and its measurements show blocking
216/// faster than doubling at every length. But its blocks run on `cumsum`, a
217/// one-launch in-block scan that this element does not have: the log-semiring
218/// product is not `+`, and the [`Mobius`] one is not even commutative. So an
219/// in-block pass would itself be a doubling or a loop. The schedule is thus
220/// that of `quat_scan`: `⌈log₂ len⌉` full-width rounds over the whole folded
221/// sequence. Each round is about fifty kernels for [`Mobius`] (four [`lse`]s,
222/// the renormalising shift, the new floors) and a dozen for [`Affine`]. The
223/// tensors are `[batch, len, nheads]`, with no `per_head_dim·state_rank`
224/// factor, so the cost that grows with `len` is mostly launches, not bytes.
225///
226/// The backward is plain autodiff, which keeps the intermediates of every
227/// round alive until it runs. A recompute backward like that of `quat_scan` is
228/// still an open item, and its divide-out trick does not port: a log-semiring
229/// element has no inverse. The `a` of [`Affine`] is a plain running sum on the
230/// same rounds, which `prefix_sum` could compute alone.
231pub fn prefix<E: Element>(elements: E) -> E {
232    let [_batch, len, _nheads] = elements.dims();
233    let mut acc = elements;
234    let mut offset = 1usize;
235    while offset < len {
236        let shifted = E::cat(vec![
237            acc.identity_like(offset),
238            acc.clone().narrow(0, len - offset),
239        ]);
240        acc = E::compose(acc, shifted);
241        offset *= 2;
242    }
243    acc
244}

246/// The same prefix products by a sequential left fold: one combine per
247/// position. The tests compare [`prefix`] against it.
248pub fn fold<E: Element>(elements: E) -> E {
249    let [_batch, len, _nheads] = elements.dims();
250    let mut out = vec![elements.clone().narrow(0, 1)];
251    for t in 1..len {
252        let prev = out[t - 1].clone();
253        out.push(E::compose(elements.clone().narrow(t, 1), prev));
254    }
255    E::cat(out)
256}
