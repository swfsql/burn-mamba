1//! The log-semiring prefix product that every positive member runs on.
2//!
3//! An element is a 2×2 nonnegative matrix in log coordinates, one per
4//! `[batch, position, head]`. It has two shapes:
5//!
6//! - [`Mobius`](scan::Mobius) — all four entries, acting **projectively** on
7//!   `(ℓ, 0)`. The read is `σ₀ − σ₁`, so any constant shift of the product is
8//!   allowed. The scan applies one after every combine, to keep the entries
9//!   bounded.
10//! - [`Affine`](scan::Affine) — the lower row is fixed at `(−∞, 0)`, so only
11//!   `(a, b)` is carried. The read `lse(a + c₀, b)` is absolute: no shift is
12//!   allowed, and none is necessary.
13//!
14//! [`prefix`](scan::prefix) composes them by Hillis–Steele doubling, the same
15//! schedule as the quaternion scan (`rotation::quat_cumprod`): `⌈log₂ len⌉`
16//! full-width combines. [`fold`](scan::fold) is the sequential left fold, the
17//! reference for [`prefix`](scan::prefix) in the tests.
18

22/// `ln(eᵃ + eᵇ)`, elementwise.
23///
24/// Written as `hi + ln(1 + e^(lo − hi))`, with `hi`/`lo` *selected*, not
25/// computed by `max`/`abs`. So the result is exactly the larger operand when
26/// the other is [`LOG_ZERO`] (an `a + softplus(b − a)` form would round `b`
27/// through `b − LOG_ZERO`). Its gradient on `a` is `σ(a − b)`, also at a tie,
28/// where a `max`-based form gives all of it to one side.
29pub fn lse(a: Tensor<3>, b: Tensor<3>) -> Tensor<3> {
30    let a_wins = a.clone().greater_equal(b.clone());
31    let hi = b.clone().mask_where(a_wins.clone(), a.clone());
32    let lo = a.mask_where(a_wins, b);
33    hi.clone() + (lo - hi).exp().log1p()
34}

36/// The one and zero of the semiring, `(0, LOG_ZERO)`, over `len` positions,
37/// with the shape, device and dtype of `like`. So a scan follows the dtype of
38/// its elements, not the default of the device.
39fn one_and_zero(like: &Tensor<3>, len: usize) -> (Tensor<3>, Tensor<3>) {
40    let [batch, _len, nheads] = like.dims();
41    let device = like.device();
42    let options = (&device, like.dtype());
43    (
44        Tensor::zeros([batch, len, nheads], options),
45        Tensor::full([batch, len, nheads], LOG_ZERO, options),
46    )
47}

49/// An associative element over `[batch, len, nheads]` tensors.
50pub trait Element: Sized + Clone {
51    /// `later ∘ earlier` — apply `earlier` first.
52    fn compose(later: Self, earlier: Self) -> Self;
53    /// The identity over `len` positions, with the batch, heads, device and
54    /// dtype of this element.
55    fn identity_like(&self, len: usize) -> Self;
56    /// `[batch, len, nheads]`.
57    fn dims(&self) -> [usize; 3];
58    /// A run of positions along axis 1.
59    fn narrow(self, start: usize, len: usize) -> Self;
60    /// Concatenate along axis 1.
61    fn cat(parts: Vec<Self>) -> Self;
62    /// The entries of `other` where `mask_bsh` is `true`, and those of `self`
63    /// elsewhere.
64    fn select(self, mask_bsh: Tensor<3, Bool>, other: Self) -> Self;
65}

67/// A projective 2×2 element: `σ ↦ M ⊗ σ` read as `σ₀ − σ₁`.
68#[derive(Clone, Debug)]
69pub struct Mobius {
70    /// `ln M₀₀`, `[batch, len, nheads]`.
71    pub m00: Tensor<3>,
72    /// `ln M₀₁`.
73    pub m01: Tensor<3>,
74    /// `ln M₁₀`.
75    pub m10: Tensor<3>,
76    /// `ln M₁₁`.
77    pub m11: Tensor<3>,
78}
79
80impl Mobius {
81    /// The log-ratio after this (prefix) product, from `ℓ₀` per head:
82    /// `lse(M₀₀ + ℓ₀, M₀₁) − lse(M₁₀ + ℓ₀, M₁₁)`.
83    ///
84    /// # Shapes
85    /// - `carry_bh` : `[batch, nheads]`
86    /// - out        : `[batch, len, nheads]`
87    pub fn apply(self, carry_bh: Tensor<2>) -> Tensor<3> {
88        self.apply_each(carry_bh.unsqueeze_dim::<3>(1))
89    }
90
91    /// [`Self::apply`] with a carry per position, `[batch, len, nheads]` (or
92    /// `[batch, 1, nheads]`, broadcast).
93    pub fn apply_each(self, carry_bsh: Tensor<3>) -> Tensor<3> {
94        lse(self.m00 + carry_bsh.clone(), self.m01) - lse(self.m10 + carry_bsh, self.m11)
95    }
96}
97
98impl Element for Mobius {
99    fn compose(later: Self, earlier: Self) -> Self {
100        let (x, y) = (later, earlier);
101        let m00 = lse(x.m00.clone() + y.m00.clone(), x.m01.clone() + y.m10.clone());
102        let m01 = lse(x.m00.clone() + y.m01.clone(), x.m01 + y.m11.clone());
103        let m10 = lse(x.m10.clone() + y.m00, x.m11.clone() + y.m10);
104        let m11 = lse(x.m10 + y.m01, x.m11 + y.m11);
105        // Projective: the read cancels any common shift, in value and in
106        // gradient. So subtract the largest entry (detached, because its
107        // gradient would cancel anyway), and floor the structural zeros again.
108        let shift = m00
109            .clone()
110            .max_pair(m01.clone())
111            .max_pair(m10.clone())
112            .max_pair(m11.clone())
113            .detach();
114        let floor = |t: Tensor<3>| (t - shift.clone()).clamp_min(LOG_ZERO);
115        Mobius {
116            m00: floor(m00),
117            m01: floor(m01),
118            m10: floor(m10),
119            m11: floor(m11),
120        }
121    }
122
123    fn identity_like(&self, len: usize) -> Self {
124        let (one, zero) = one_and_zero(&self.m00, len);
125        Mobius {
126            m00: one.clone(),
127            m01: zero.clone(),
128            m10: zero,
129            m11: one,
130        }
131    }
132
133    fn dims(&self) -> [usize; 3] {
134        self.m00.dims()
135    }
136
137    fn narrow(self, start: usize, len: usize) -> Self {
138        Mobius {
139            m00: self.m00.narrow(1, start, len),
140            m01: self.m01.narrow(1, start, len),
141            m10: self.m10.narrow(1, start, len),
142            m11: self.m11.narrow(1, start, len),
143        }
144    }
145
146    fn cat(parts: Vec<Self>) -> Self {
147        let mut m = [vec![], vec![], vec![], vec![]];
148        for p in parts {
149            m[0].push(p.m00);
150            m[1].push(p.m01);
151            m[2].push(p.m10);
152            m[3].push(p.m11);
153        }
154        let [m00, m01, m10, m11] = m.map(|v| Tensor::cat(v, 1));
155        Mobius { m00, m01, m10, m11 }
156    }
157
158    fn select(self, mask_bsh: Tensor<3, Bool>, other: Self) -> Self {
159        Mobius {
160            m00: self.m00.mask_where(mask_bsh.clone(), other.m00),
161            m01: self.m01.mask_where(mask_bsh.clone(), other.m01),
162            m10: self.m10.mask_where(mask_bsh.clone(), other.m10),
163            m11: self.m11.mask_where(mask_bsh, other.m11),
164        }
165    }
166}

168/// An affine element `c ↦ lse(c + a, b)` — the matrix `[[a, b], [−∞, 0]]`.
169#[derive(Clone, Debug)]
170pub struct Affine {
171    /// The log-gain, `[batch, len, nheads]` (any sign).
172    pub a: Tensor<3>,
173    /// The log-offset.
174    pub b: Tensor<3>,
175}
176
177impl Affine {
178    /// The register after this (prefix) map, from `c₀` per head:
179    /// `lse(a + c₀, b)`.
180    ///
181    /// # Shapes
182    /// - `carry_bh` : `[batch, nheads]`
183    /// - out        : `[batch, len, nheads]`
184    pub fn apply(self, carry_bh: Tensor<2>) -> Tensor<3> {
185        self.apply_each(carry_bh.unsqueeze_dim::<3>(1))
186    }
187
188    /// [`Self::apply`] with a carry per position, `[batch, len, nheads]` (or
189    /// `[batch, 1, nheads]`, broadcast).
190    pub fn apply_each(self, carry_bsh: Tensor<3>) -> Tensor<3> {
191        lse(self.a + carry_bsh, self.b)
192    }
193}
194
195impl Element for Affine {
196    fn compose(later: Self, earlier: Self) -> Self {
197        // lse(lse(c + a₁, b₁) + a₂, b₂) = lse(c + (a₁ + a₂), lse(b₁ + a₂, b₂)).
198        Affine {
199            b: lse(earlier.b + later.a.clone(), later.b).clamp_min(LOG_ZERO),
200            a: (earlier.a + later.a).clamp_min(LOG_ZERO),
201        }
202    }
203
204    fn identity_like(&self, len: usize) -> Self {
205        let (one, zero) = one_and_zero(&self.a, len);
206        Affine { a: one, b: zero }
207    }
208
209    fn dims(&self) -> [usize; 3] {
210        self.a.dims()
211    }
212
213    fn narrow(self, start: usize, len: usize) -> Self {
214        Affine {
215            a: self.a.narrow(1, start, len),
216            b: self.b.narrow(1, start, len),
217        }
218    }
219
220    fn cat(parts: Vec<Self>) -> Self {
221        let (a, b) = parts.into_iter().map(|p| (p.a, p.b)).unzip();
222        Affine {
223            a: Tensor::cat(a, 1),
224            b: Tensor::cat(b, 1),
225        }
226    }
227
228    fn select(self, mask_bsh: Tensor<3, Bool>, other: Self) -> Self {
229        Affine {
230            a: self.a.mask_where(mask_bsh.clone(), other.a),
231            b: self.b.mask_where(mask_bsh, other.b),
232        }
233    }
234}

236/// Inclusive prefix products `Pₜ = eₜ ∘ ⋯ ∘ e₀` along axis 1, by Hillis–Steele
237/// doubling.
238///
239/// Invariant after the round at `offset`: position `t` holds the product of
240/// the window `[max(t − 2·offset + 1, 0), t]`, where the positions before the
241/// axis are the identity. `⌈log₂ len⌉` rounds cover every window.
242///
243/// # Why doubling, and what it costs
244///
245/// `helpers::prefix_sum` blocks its scan, and its measurements show blocking
246/// faster than doubling at every length. But its blocks run on `cumsum`, a
247/// one-launch in-block scan that this element does not have: the log-semiring
248/// product is not `+`, and the [`Mobius`] one is not even commutative. So an
249/// in-block pass would itself be a doubling or a loop. The schedule is thus
250/// that of `quat_scan`: `⌈log₂ len⌉` full-width rounds over the whole folded
251/// sequence. Each round is about fifty kernels for [`Mobius`] (four [`lse`]s,
252/// the renormalising shift, the new floors) and a dozen for [`Affine`]. The
253/// tensors are `[batch, len, nheads]`, with no `per_head_dim·state_rank`
254/// factor, so the cost that grows with `len` is mostly launches, not bytes.
255///
256/// The backward is plain autodiff, which keeps the intermediates of every
257/// round alive until it runs. A recompute backward like that of `quat_scan` is
258/// still an open item, and its divide-out trick does not port: a log-semiring
259/// element has no inverse. The `a` of [`Affine`] is a plain running sum on the
260/// same rounds, which `prefix_sum` could compute alone.
261pub fn prefix<E: Element>(elements: E) -> E {
262    let [_batch, len, _nheads] = elements.dims();
263    let mut acc = elements;
264    let mut offset = 1usize;
265    while offset < len {
266        let shifted = E::cat(vec![
267            acc.identity_like(offset),
268            acc.clone().narrow(0, len - offset),
269        ]);
270        acc = E::compose(acc, shifted);
271        offset *= 2;
272    }
273    acc
274}

276/// The segments of a packed row, as a scan over the row reads them.
277#[derive(Clone, Debug)]
278pub struct Restart {
279    /// `[batch, len]`: the first position of the segment of each position
280    /// (`-1` before the first reset of the row).
281    pub start_bs: Tensor<2, Int>,
282    /// `[batch, nheads]`: the carry that each segment starts from (that of
283    /// the origin cache).
284    pub origin_bh: Tensor<2>,
285}
286
287impl Restart {
288    /// The carry of each position, `[batch, len, nheads]`: `carry_bh` before
289    /// the first reset of the row, and the origin carry from there on.
290    pub fn carries(&self, carry_bh: Tensor<2>) -> Tensor<3> {
291        let [batch, len] = self.start_bs.dims();
292        let [_, nheads] = carry_bh.dims();
293        let restarted_bsh = self
294            .start_bs
295            .clone()
296            .greater_equal_elem(0)
297            .unsqueeze_dim::<3>(2)
298            .expand([batch, len, nheads]);
299        carry_bh
300            .unsqueeze_dim::<3>(1)
301            .expand([batch, len, nheads])
302            .mask_where(restarted_bsh, self.origin_bh.clone().unsqueeze_dim::<3>(1).expand([batch, len, nheads]))
303    }
304}

306/// [`prefix`] over the segments of a packed row: each product starts at the
307/// first position of its own segment (`start_bs`, `[batch, len]`; `-1`
308/// before the first reset of the row).
309///
310/// A combine whose older window would reach back past the start takes the
311/// identity instead, as a position does before the first one. So the
312/// invariant of [`prefix`] holds with `max(·, 0)` replaced by
313/// `max(·, start)`.
314pub fn prefix_in_segments<E: Element>(elements: E, start_bs: &Tensor<2, Int>) -> E {
315    let [batch, len, nheads] = elements.dims();
316    let device = start_bs.device();
317    let position_bs = Tensor::<1, Int>::arange(0..len as i64, &device)
318        .reshape([1, len])
319        .expand([batch, len]);
320    let mut acc = elements;
321    let mut offset = 1usize;
322    while offset < len {
323        let shifted = E::cat(vec![
324            acc.identity_like(offset),
325            acc.clone().narrow(0, len - offset),
326        ]);
327        let crosses_bsh = (position_bs.clone() - offset as i64)
328            .lower(start_bs.clone())
329            .unsqueeze_dim::<3>(2)
330            .expand([batch, len, nheads]);
331        let shifted = shifted.select(crosses_bsh, acc.identity_like(len));
332        acc = E::compose(acc, shifted);
333        offset *= 2;
334    }
335    acc
336}

338/// The same prefix products by a sequential left fold: one combine per
339/// position. The tests compare [`prefix`] against it.
340pub fn fold<E: Element>(elements: E) -> E {
341    let [_batch, len, _nheads] = elements.dims();
342    let mut out = vec![elements.clone().narrow(0, 1)];
343    for t in 1..len {
344        let prev = out[t - 1].clone();
345        out.push(E::compose(elements.clone().narrow(t, 1), prev));
346    }
347    E::cat(out)
348}
