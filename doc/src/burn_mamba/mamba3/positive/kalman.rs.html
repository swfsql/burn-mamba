1//! The Kalman gate: a per-head precision `Λ`, scanned from the inputs, and the
2//! decay it computes (see the module header one level up):
3//!
4//! ```text
5//!   Lₜ = Λₜ₋₁ + νₜ                      the left endpoint's late installment
6//!   dₜ = αₜ / (1 + qₜ·αₜ·Lₜ)            the predict
7//!   Λₜ = dₜ·Lₜ + γₜ                      the right endpoint's measurement
8//! ```
9//!
10//! `γ`/`ν` are the trapezoid's two endpoint masses (`γ = Δ`, no `ν`, under
11//! [`Trapezoid::None`](crate::mamba3::trapezoid::Trapezoid::None)). The plant
12//! pays a sample's left-endpoint installment one step late, transported by that
13//! step's decay (`β = ν·α`, with `d` for `α`), so `Λ` **is** the plant's own
14//! recurrence on ones — the total weight of every sample it has written, a
15//! fresh cache's zero tap slot included — and `η/Λ` a weighted mean of what it
16//! wrote. Each step is
17//! still one Möbius map — shift by `ν`, predict, shift by `γ` compose to
18//! `ln [[α(1+qγ), αν + γ(1+qαν)], [αq, 1+qαν]]` — so the scan is unchanged.
19//!
20//! That identity is exact for a **lag-1** tap, which is every pattern at
21//! `u = 1` and the default at any `u`. A lag-`u` tap (the `Vertical*`
22//! patterns at `u > 1`) is transported across its whole gap, `∏ d` over `u`
23//! steps, which reads `u` earlier precisions and so is no 2×2 map; its
24//! installment enters `L` like a lag-1 one, transported by `dₜ` alone, and `Λ`
25//! over-counts the plant by `νₜ·(dₜ − ∏ d)` per step. `Λ` is then an upper
26//! bound on the plant's weight rather than the weight itself; every other
27//! property (the ceiling, the contraction, the stock join) holds unchanged.
28//!
29//! Everything is in log coordinates: `ℓ = ln Λ`, and the decay is formed as a
30//! log-decay correction, `ln dₜ = ln αₜ − softplus(ln qₜ + ln αₜ + ln Lₜ)`, so
31//! that `κ = 0` (`ln q = −∞`, floored to [`LOG_ZERO`]) leaves `ln αₜ`
32//! untouched bit for bit.
33

40/// What the gate reads besides the discretisation: the per-head `ln κ`, the
41/// projected noise (if any) and the carried `ℓ`.
42pub struct GainInput {
43    /// `ln κₕ`, `[nheads]`. `−∞` is the stock block.
44    pub log_kappa_h: Tensor<1>,
45    /// `rₜ`, `[batch, len, nheads]`, under
46    /// [`Gain::KalmanProjectedNoise`](super::Gain::KalmanProjectedNoise).
47    pub noise_bsh: Option<Tensor<3>>,
48    /// `ℓ` before the first position, `[batch, nheads]` — [`LOG_ZERO`] for a
49    /// fresh sequence (`Λ₀ = 0`: no evidence yet).
50    pub carry_bh: Tensor<2>,
51}
52
53/// The gate's per-position quantities.
54pub struct GainOutput {
55    /// `ln dₜ`, which replaces `Δₜ·Aₜ` wherever the block reads a log-decay.
56    pub da_bsh: Tensor<3>,
57    /// `ℓₜ = ln Λₜ` after position `t`'s update, `[batch, len, nheads]`; the
58    /// last one is the next call's carry.
59    pub log_precision_bsh: Tensor<3>,
60}

62/// The step's masses in log coordinates, each `[batch, len, nheads]` and
63/// floored at [`LOG_ZERO`].
64///
65/// Formed from the **pre-activations**, never as `ln` of the masses the plant
66/// multiplies by. A mass underflows to `0` — `softplus`/`σ` of a large
67/// negative pre-activation: below ≈ −104 in f32, ≈ −17 in f16 — and there
68/// `ln`'s forward can be floored but its backward cannot: `∂ ln x/∂x = 1/x`
69/// meets the floor's zero gradient as `0·∞ = NaN` (and a denormal `x`
70/// overflows `1/x` first). `ln softplus` and `ln σ` of the pre-activation have
71/// derivatives in `(0, 1]` everywhere instead.
72pub struct LogMasses {
73    /// `ln Δₜ` — the elapsed time, which paces the doubt `q`.
74    pub dt_bsh: Tensor<3>,
75    /// `ln γₜ` — the right endpoint: this position's own measurement.
76    pub gamma_bsh: Tensor<3>,
77    /// `ln νₜ` — the left endpoint: an earlier sample's installment, paid at
78    /// this position. Every tap together (a two-tap `μ` split sums back to
79    /// `(1 − λ)·Δ`); `None` under
80    /// [`Trapezoid::None`](crate::mamba3::trapezoid::Trapezoid::None).
81    pub nu_bsh: Option<Tensor<3>>,
82}
83
84impl LogMasses {
85    /// The masses of `helpers::trapezoidal_coefficients`, in logs: `Δ =
86    /// clamp(softplus(dt_raw), dt_limit)`, `γ = λ'·Δ`, `ν = (1 − λ')·Δ`, where
87    /// `λ' = σ(lambda_raw)`, set to `1` where `far_open_1s1` closes the far tap.
88    ///
89    /// # Shapes
90    /// - `dt_raw_bsh`, `lambda_raw_bsh` : `[batch, len, nheads]` (`dd_dt + dt_bias`; `λ`'s logit)
91    /// - `far_open_1s1`                 : `[1, len, 1]`, `0` where the far tap is closed
92    pub fn new(
93        dt_raw_bsh: Tensor<3>,
94        dt_limit: (f64, f64),
95        lambda_raw_bsh: Option<Tensor<3>>,
96        far_open_1s1: Option<Tensor<3>>,
97    ) -> Self {
98        // `ln` is monotone, so clamping `ln Δ` is clamping `Δ`.
99        let ln_dt = log_softplus(dt_raw_bsh).clamp_max(dt_limit.1.ln());
100        let ln_dt = if dt_limit.0 > 0.0 {
101            ln_dt.clamp_min(dt_limit.0.ln())
102        } else {
103            ln_dt
104        }
105        .clamp_min(LOG_ZERO);
106        let Some(lambda_raw_bsh) = lambda_raw_bsh else {
107            return LogMasses {
108                gamma_bsh: ln_dt.clone(),
109                dt_bsh: ln_dt,
110                nu_bsh: None,
111            };
112        };
113        // `ln σ(x) = −softplus(−x)` and `ln(1 − σ(x)) = −softplus(x)`.
114        let ln_lambda = -softplus(-lambda_raw_bsh.clone());
115        let ln_one_minus = -softplus(lambda_raw_bsh);
116        // A closed far tap is `λ' = 1`: `ln λ' = 0` and `ln(1 − λ') = ln 0`.
117        let (ln_lambda, ln_one_minus) = match far_open_1s1 {
118            Some(open_1s1) => (
119                ln_lambda * open_1s1.clone(),
120                ln_one_minus * open_1s1.clone() + (-open_1s1 + 1.0) * LOG_ZERO,
121            ),
122            None => (ln_lambda, ln_one_minus),
123        };
124        LogMasses {
125            gamma_bsh: (ln_lambda + ln_dt.clone()).clamp_min(LOG_ZERO),
126            nu_bsh: Some((ln_one_minus + ln_dt.clone()).clamp_min(LOG_ZERO)),
127            dt_bsh: ln_dt,
128        }
129    }
130}

132/// `ln softplus(x)`, finite everywhere with a derivative `σ(x)/softplus(x)` in
133/// `(0, 1]` — including where `softplus(x)` itself underflows to `0`.
134///
135/// Below a knee `softplus(x) = eˣ·(1 − eˣ/2 + …)`, so `ln softplus(x) = x − eˣ/2
136/// + O(e²ˣ)`. The knee keeps `softplus` a normal number above it (f16's
137/// smallest is `e^−9.7`) and the dropped `e²ˣ` term under the dtype's
138/// resolution below it: `−8` in f16, `−20` otherwise.
139pub fn log_softplus<const D: usize>(x: Tensor<D>) -> Tensor<D> {
140    let knee = match x.dtype() {
141        DType::F16 => -8.0,
142        _ => -20.0,
143    };
144    let below = x.clone().lower_elem(knee);
145    // Each branch is fed only its own side of the knee: a branch `mask_where`
146    // discards still backpropagates, and `ln 0` or an overflowing `eˣ` there
147    // would reach the gradient as `0·∞ = NaN`.
148    let above = softplus(x.clone().clamp_min(knee)).log();
149    let x_below = x.clamp_max(knee);
150    let series = x_below.clone() - x_below.exp() * 0.5;
151    above.mask_where(below, series)
152}

154/// The per-position Möbius elements — shift by `ν`, predict, shift by `γ`:
155///
156/// ```text
157///   ln [[α(1 + qγ),  αν + γ(1 + qαν)],
158///       [αq,         1 + qαν       ]]
159/// ```
160///
161/// which is `ln [[α(1+qγ), γ], [αq, 1]]` without a `ν`.
162///
163/// # Shapes
164/// - every input : `[batch, len, nheads]` (`da_bsh` is stock's `ln α = Δ·A`)
165fn elements(da_bsh: &Tensor<3>, ln_q_bsh: &Tensor<3>, masses: &LogMasses) -> Mobius {
166    let floor = |t: Tensor<3>| t.clamp_min(LOG_ZERO);
167    let ln_gamma = masses.gamma_bsh.clone();
168    let m00 = floor(da_bsh.clone() + softplus(ln_q_bsh.clone() + ln_gamma.clone()));
169    let m10 = floor(da_bsh.clone() + ln_q_bsh.clone());
170    let (m01, m11) = match &masses.nu_bsh {
171        None => (ln_gamma, m10.zeros_like()),
172        Some(ln_nu) => {
173            let m11 = softplus(floor(ln_q_bsh.clone() + da_bsh.clone() + ln_nu.clone()));
174            let m01 = scan::lse(floor(da_bsh.clone() + ln_nu.clone()), ln_gamma + m11.clone());
175            (m01, m11)
176        }
177    };
178    Mobius { m00, m01, m10, m11 }
179}

181/// Run the gate over a folded run of positions (the whole sequence in
182/// `forward`, one token's `u` micro-steps in `step`).
183///
184/// # Shapes
185/// - `da_bsh` : `[batch, len, nheads]`, stock's `Δ·A`
186pub fn gate(da_bsh: Tensor<3>, masses: LogMasses, input: GainInput) -> GainOutput {
187    let [batch, len, nheads] = da_bsh.dims();
188    let ln_q = input.log_kappa_h.clone().unsqueeze::<3>() + masses.dt_bsh.clone();
189    let ln_q = match &input.noise_bsh {
190        Some(noise_bsh) => ln_q + noise_bsh.clone(),
191        None => ln_q,
192    };
193    let ln_q = ln_q.clamp_min(LOG_ZERO);
194    let elements = elements(&da_bsh, &ln_q, &masses);
195    let log_precision_bsh = scan::prefix(elements).apply(input.carry_bh.clone());
196    // ℓₜ₋₁: the carry, then every position but the last.
197    let carry_b1h = input.carry_bh.reshape([batch, 1, nheads]);
198    let prev_bsh = if len == 1 {
199        carry_b1h
200    } else {
201        Tensor::cat(vec![carry_b1h, log_precision_bsh.clone().narrow(1, 0, len - 1)], 1)
202    };
203    // `ln Lₜ`: the predict acts on everything known about the samples before
204    // this position, including the installment of one that is paid only now.
205    let ln_l_bsh = match masses.nu_bsh {
206        Some(ln_nu) => scan::lse(prev_bsh, ln_nu),
207        None => prev_bsh,
208    };
209    let da_bsh = da_bsh.clone() - softplus(ln_q + da_bsh + ln_l_bsh);
210    GainOutput {
211        da_bsh,
212        log_precision_bsh,
213    }
