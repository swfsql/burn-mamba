1//! The Kalman gate: a per-head precision `Λ`, scanned from the inputs, and the
2//! decay it computes (see the module header one level up):
3//!
4//! ```text
5//!   Lₜ = Λₜ₋₁ + νₜ                      the left endpoint's late installment
6//!   dₜ = αₜ / (1 + qₜ·αₜ·Lₜ)            the predict
7//!   Λₜ = dₜ·Lₜ + γₜ                      the right endpoint's measurement
8//! ```
9//!
10//! `γ`/`ν` are the two endpoint masses of the trapezoid (`γ = Δ` and no `ν`
11//! under [`Trapezoid::None`](crate::mamba3::trapezoid::Trapezoid::None)). The
12//! plant pays the left-endpoint installment of a sample one step late,
13//! transported by the decay of that step (`β = ν·α`, with `d` for `α`). So `Λ`
14//! **is** the recurrence of the plant on ones: the total weight of every
15//! sample it wrote (the zero tap slot of a fresh cache included). And `η/Λ` is
16//! a weighted mean of what it wrote. Each step is still one Möbius map: shift
17//! by `ν`, predict, and shift by `γ` compose to
18//! `ln [[α(1+qγ), αν + γ(1+qαν)], [αq, 1+qαν]]`. So the scan is unchanged.
19//!
20//! That identity is exact for a **lag-1** tap: every pattern at `u = 1`, and
21//! the default at any `u`. A lag-`u` tap (the `Vertical*` patterns at `u > 1`)
22//! is transported across its whole gap, `∏ d` over `u` steps. That reads `u`
23//! earlier precisions, so it is not a 2×2 map. Its installment enters `L` like
24//! a lag-1 installment, transported by `dₜ` alone, and `Λ` over-counts the
25//! plant by `νₜ·(dₜ − ∏ d)` per step. `Λ` is then an upper bound on the weight
26//! of the plant, not the weight itself. Every other property (the ceiling, the
27//! contraction, the stock join) holds unchanged.
28//!
29//! Everything is in log coordinates: `ℓ = ln Λ`. The decay is a log-decay
30//! correction, `ln dₜ = ln αₜ − softplus(ln qₜ + ln αₜ + ln Lₜ)`. So `κ = 0`
31//! (`ln q = −∞`, floored to [`LOG_ZERO`]) leaves `ln αₜ` unchanged, bit for
32//! bit.
33

40/// What the gate reads in addition to the discretisation: the per-head `ln κ`,
41/// the projected noise (if any) and the carried `ℓ`.
42pub struct GainInput {
43    /// `ln κₕ`, `[nheads]`. `−∞` is the stock block.
44    pub log_kappa_h: Tensor<1>,
45    /// `rₜ`, `[batch, len, nheads]`, under
46    /// [`Gain::KalmanProjectedNoise`](super::Gain::KalmanProjectedNoise).
47    pub noise_bsh: Option<Tensor<3>>,
48    /// `ℓ` before the first position, `[batch, nheads]`: [`LOG_ZERO`] for a
49    /// fresh sequence (`Λ₀ = 0`: no evidence yet).
50    pub carry_bh: Tensor<2>,
51}

53/// The per-position quantities of the gate.
54pub struct GainOutput {
55    /// `ln dₜ`, which replaces `Δₜ·Aₜ` wherever the block reads a log-decay.
56    pub da_bsh: Tensor<3>,
57    /// `ℓₜ = ln Λₜ` after the update of position `t`, `[batch, len, nheads]`.
58    /// The last one is the carry of the next call.
59    pub log_precision_bsh: Tensor<3>,
60}

62/// The masses of the step in log coordinates, each `[batch, len, nheads]` and
63/// floored at [`LOG_ZERO`].
64///
65/// They come from the **pre-activations**, never as `ln` of the masses that
66/// the plant multiplies by. A mass underflows to `0` (`softplus`/`σ` of a
67/// large negative pre-activation: below ≈ −104 in f32, ≈ −17 in f16). There,
68/// the forward of `ln` can be floored, but its backward cannot:
69/// `∂ ln x/∂x = 1/x` meets the zero gradient of the floor as `0·∞ = NaN` (and
70/// a denormal `x` overflows `1/x` first). `ln softplus` and `ln σ` of the
71/// pre-activation have derivatives in `(0, 1]` everywhere.
72pub struct LogMasses {
73    /// `ln Δₜ`: the elapsed time, which paces the doubt `q`.
74    pub dt_bsh: Tensor<3>,
75    /// `ln γₜ`: the right endpoint, the own measurement of this position.
76    pub gamma_bsh: Tensor<3>,
77    /// `ln νₜ`: the left endpoint, the installment of an earlier sample, paid
78    /// at this position. All taps together (a two-tap `μ` split sums back to
79    /// `(1 − λ)·Δ`). `None` under
80    /// [`Trapezoid::None`](crate::mamba3::trapezoid::Trapezoid::None).
81    pub nu_bsh: Option<Tensor<3>>,
82}
83
84impl LogMasses {
85    /// The masses of `helpers::trapezoidal_coefficients`, in logs:
86    /// `Δ = clamp(softplus(dt_raw), dt_limit)`, `γ = λ'·Δ`, `ν = (1 − λ')·Δ`,
87    /// where `λ' = σ(lambda_raw)`, set to `1` where `far_open_1s1` closes the
88    /// far tap.
89    ///
90    /// # Shapes
91    /// - `dt_raw_bsh`, `lambda_raw_bsh` : `[batch, len, nheads]` (`dd_dt + dt_bias`; `λ`'s logit)
92    /// - `far_open_1s1`                 : `[1, len, 1]`, `0` where the far tap is closed
93    pub fn new(
94        dt_raw_bsh: Tensor<3>,
95        dt_limit: (f64, f64),
96        lambda_raw_bsh: Option<Tensor<3>>,
97        far_open_1s1: Option<Tensor<3>>,
98    ) -> Self {
99        // `ln` is monotone, so clamping `ln Δ` is clamping `Δ`.
100        let ln_dt = log_softplus(dt_raw_bsh).clamp_max(dt_limit.1.ln());
101        let ln_dt = if dt_limit.0 > 0.0 {
102            ln_dt.clamp_min(dt_limit.0.ln())
103        } else {
104            ln_dt
105        }
106        .clamp_min(LOG_ZERO);
107        let Some(lambda_raw_bsh) = lambda_raw_bsh else {
108            return LogMasses {
109                gamma_bsh: ln_dt.clone(),
110                dt_bsh: ln_dt,
111                nu_bsh: None,
112            };
113        };
114        // `ln σ(x) = −softplus(−x)` and `ln(1 − σ(x)) = −softplus(x)`.
115        let ln_lambda = -softplus(-lambda_raw_bsh.clone());
116        let ln_one_minus = -softplus(lambda_raw_bsh);
117        // A closed far tap is `λ' = 1`: `ln λ' = 0` and `ln(1 − λ') = ln 0`.
118        let (ln_lambda, ln_one_minus) = match far_open_1s1 {
119            Some(open_1s1) => (
120                ln_lambda * open_1s1.clone(),
121                ln_one_minus * open_1s1.clone() + (-open_1s1 + 1.0) * LOG_ZERO,
122            ),
123            None => (ln_lambda, ln_one_minus),
124        };
125        LogMasses {
126            gamma_bsh: (ln_lambda + ln_dt.clone()).clamp_min(LOG_ZERO),
127            nu_bsh: Some((ln_one_minus + ln_dt.clone()).clamp_min(LOG_ZERO)),
128            dt_bsh: ln_dt,
129        }
130    }
131}

133/// `ln softplus(x)`, finite everywhere, with a derivative
134/// `σ(x)/softplus(x)` in `(0, 1]`. This also holds where `softplus(x)` itself
135/// underflows to `0`.
136///
137/// Below a knee, `softplus(x) = eˣ·(1 − eˣ/2 + …)`, so
138/// `ln softplus(x) = x − eˣ/2 + O(e²ˣ)`. Above the knee, `softplus` stays a
139/// normal number (the smallest f16 normal is `e^−9.7`). Below it, the dropped
140/// `e²ˣ` term is under the resolution of the dtype. The knee is `−8` in f16
141/// and `−20` otherwise.
142pub fn log_softplus<const D: usize>(x: Tensor<D>) -> Tensor<D> {
143    let knee = match x.dtype() {
144        DType::F16 => -8.0,
145        _ => -20.0,
146    };
147    let below = x.clone().lower_elem(knee);
148    // Each branch gets only its own side of the knee. A branch that
149    // `mask_where` discards still backpropagates, and a `ln 0` or an
150    // overflowing `eˣ` there would reach the gradient as `0·∞ = NaN`.
151    let above = softplus(x.clone().clamp_min(knee)).log();
152    let x_below = x.clamp_max(knee);
153    let series = x_below.clone() - x_below.exp() * 0.5;
154    above.mask_where(below, series)
155}

157/// The per-position Möbius elements (shift by `ν`, predict, shift by `γ`):
158///
159/// ```text
160///   ln [[α(1 + qγ),  αν + γ(1 + qαν)],
161///       [αq,         1 + qαν       ]]
162/// ```
163///
164/// which is `ln [[α(1+qγ), γ], [αq, 1]]` without a `ν`.
165///
166/// # Shapes
167/// - every input : `[batch, len, nheads]` (`da_bsh` is the stock `ln α = Δ·A`)
168fn elements(da_bsh: &Tensor<3>, ln_q_bsh: &Tensor<3>, masses: &LogMasses) -> Mobius {
169    let floor = |t: Tensor<3>| t.clamp_min(LOG_ZERO);
170    let ln_gamma = masses.gamma_bsh.clone();
171    let m00 = floor(da_bsh.clone() + softplus(ln_q_bsh.clone() + ln_gamma.clone()));
172    let m10 = floor(da_bsh.clone() + ln_q_bsh.clone());
173    let (m01, m11) = match &masses.nu_bsh {
174        None => (ln_gamma, m10.zeros_like()),
175        Some(ln_nu) => {
176            let m11 = softplus(floor(ln_q_bsh.clone() + da_bsh.clone() + ln_nu.clone()));
177            let m01 = scan::lse(floor(da_bsh.clone() + ln_nu.clone()), ln_gamma + m11.clone());
178            (m01, m11)
179        }
180    };
181    Mobius { m00, m01, m10, m11 }
182}

184/// Run the gate over a folded run of positions (the whole sequence in
185/// `forward`, the `u` micro-steps of one token in `step`).
186///
187/// # Shapes
188/// - `da_bsh` : `[batch, len, nheads]`, the stock `Δ·A`
189pub fn gate(da_bsh: Tensor<3>, masses: LogMasses, input: GainInput) -> GainOutput {
190    let [batch, len, nheads] = da_bsh.dims();
191    let ln_q = input.log_kappa_h.clone().unsqueeze::<3>() + masses.dt_bsh.clone();
192    let ln_q = match &input.noise_bsh {
193        Some(noise_bsh) => ln_q + noise_bsh.clone(),
194        None => ln_q,
195    };
196    let ln_q = ln_q.clamp_min(LOG_ZERO);
197    let elements = elements(&da_bsh, &ln_q, &masses);
198    let log_precision_bsh = scan::prefix(elements).apply(input.carry_bh.clone());
199    // ℓₜ₋₁: the carry, then every position but the last.
200    let carry_b1h = input.carry_bh.reshape([batch, 1, nheads]);
201    let prev_bsh = if len == 1 {
202        carry_b1h
203    } else {
204        Tensor::cat(vec![carry_b1h, log_precision_bsh.clone().narrow(1, 0, len - 1)], 1)
205    };
206    // `ln Lₜ`: the predict acts on all that is known about the samples before
207    // this position, including the installment of a sample that is paid only
208    // now.
209    let ln_l_bsh = match masses.nu_bsh {
210        Some(ln_nu) => scan::lse(prev_bsh, ln_nu),
211        None => prev_bsh,
212    };
213    let da_bsh = da_bsh.clone() - softplus(ln_q + da_bsh + ln_l_bsh);
214    GainOutput {
215        da_bsh,
216        log_precision_bsh,
217    }
