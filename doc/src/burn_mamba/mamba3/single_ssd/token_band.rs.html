1//! # The lag-`u` correction band (single-SSD, [`Vertical`])
2//!
3//! [`Vertical`]: crate::mamba3::trapezoid::Trapezoid::Vertical
4//!
5//! The single-SSD pathway scales the key of sample `s` by the whole collapsed
6//! weight `Δ̃ₛ = γₛ + νₛ₊ₗₐ₉` (`info/mamba-3/trapezoid-as-integration.md` §5).
7//! That is right for every read `t` *after* the tap is paid (`t − s ≥ lag`). It
8//! is wrong for the `lag` reads before it, where the weight must still be
9//! `γₛ`. At lag 1 that exception is only the diagonal, and
10//! [`ssd::diag`](crate::mamba3::single_ssd::ssd::diag) corrects it inside the
11//! kernel. At lag `u` it is a `u`-wide **band** (§9).
12//!
13//! The caller passes the excess in. It is not recomputed from `scale − γ`,
14//! because the scale of a two-tap pattern also holds a **lag-1** installment.
15//! That installment has already landed at every read that the band covers,
16//! except the diagonal, which the kernel replaces anyway. So only the lag-`u`
17//! mass belongs here.
18//!
19//! ## Why the band never has to enter the kernel
20//!
21//! The band would cross chunk boundaries, and the part of it that arrived
22//! through the initial state of the chunk could not be un-weighted. But it
23//! does not have to enter the kernel. The only outputs of `forward` are at the
24//! **last** micro-step of each token (the [read axis](crate::mamba3::product)
25//! of the chunk). For a read at folded position `p = τ·u + (u−1)`, the band
26//! `{p−u+1 … p}` is *exactly token `τ`*. So the correction is one small
27//! contraction per token, applied after the kernel at token resolution. It
28//! needs no mask change, no chunk-length constraint and no cross-chunk term.
29//!
30//! This is the one place where the two pathways do not agree pointwise. They
31//! stay equivalent on **everything that a caller can observe**: the output of
32//! the block and every field of the returned cache. That is all that
33//! `forward_single_ssd` promises. The correction does not apply to the
34//! intermediate `y` at the `u−1` folded positions per token that the read axis
35//! does not read (and does not compute): partial sums on the way to the read,
36//! never a value that the block returns. The **state** is exact at every
37//! position in both pathways. The caches carry it, and a split prefill
38//! continues from it.
39//!
40//! A correction of those positions too is possible, but not free. Their band
41//! reaches back over a token boundary. So part of it arrives through the
42//! initial state of the chunk (or, in the first token of the call, through the
43//! boundary β seed), already weighted by `scale`, where it cannot be
44//! un-weighted. This module does not pay that cost.
45//!
46//! ## The term
47//!
48//! The diagonal of the kernel is already `γ`, so what is left is `j < u−1`:
49//!
50//! ```text
51//!   corr[τ, m_out, h, p] = Σ_{j<u−1} νᵗᵃᵖ[τ,j] · dcy[τ,j]
52//!                          · Σ_{m_in} (C[τ,u−1,m_out]·B[τ,j,m_in]) · V[τ,j,m_in,p]
53//!
54//!   νᵗᵃᵖ[τ,j] = νᶠᵃʳₛ₊ᵤ at s = (τ,j)     dcy[τ,j] = exp(Σ_{r=j+1}^{u−1} da[τ,r])
55//! ```
56//!
57//! `dcy` is the scalar decay from the tapped position to the read. The
58//! relative *rotation* needs no factor, because `C̄`/`B̄` already carry it.
59

62/// The intra-token part of the `lag`-wide correction band, to be **subtracted**
63/// from the single-SSD output at token resolution.
64///
65/// `None` at `micro_steps == 1` (the band is the diagonal, which the kernel
66/// already corrects). Callers use it only at `lag == micro_steps`, under one
67/// of the lag-`u` patterns
68/// ([`Trapezoid::Vertical`](crate::mamba3::trapezoid::Trapezoid::Vertical) and
69/// the two that add a tap to it). `excess_bsh` is the **lag-`u`** mass,
70/// shifted back to the sample that owes it (`νₛ₊ᵤ`), never the whole
71/// `scaleₛ − γₛ`.
72///
73/// # Shapes
74/// - `b_bsmhr`             : `[batch, sequence, mimo_rank, nheads, state_rank]`
75/// - `c_btmhr`             : `[batch, tokens, mimo_rank, nheads, state_rank]` —
76///   already on the read axis, like everywhere else `C` appears
77/// - `v_bsmhp`             : `[batch, sequence, mimo_rank, nheads, per_head_dim]`
78/// - `excess_bsh`, `da_bsh`: `[batch, sequence, nheads]`
79/// - out                   : `[batch, tokens, mimo_rank, nheads, per_head_dim]`
80///
81/// where `sequence = tokens · micro_steps` is the folded axis.
82#[allow(non_snake_case)]
83pub fn token_band_correction(
84    v_bsmhp: Tensor<5>,
85    b_bsmhr: Tensor<5>,
86    c_btmhr: Tensor<5>,
87    excess_bsh: Tensor<3>,
88    da_bsh: Tensor<3>,
89    micro_steps: usize,
90) -> Option<Tensor<5>> {
91    let u = micro_steps;
92    if u == 1 {
93        return None;
94    }
95    let [batch, sequence, mimo_rank, nheads, per_head_dim] = v_bsmhp.dims();
96    let [.., state_rank] = b_bsmhr.dims();
97    let tokens = sequence / u;
98    debug_assert_eq!(tokens, c_btmhr.dims()[1], "C is on the read axis");
99    // The `u−1` tapped positions per token; the `u`-th is the read itself, whose
100    // weight the kernel's γ-diagonal already fixed.
101    let taps = u - 1;
102
103    // ── The per-tap scalar: the unpaid excess, decayed to the read ────────────
104    // `Σ_{r=j+1}^{u−1} da` within the token, i.e. the reverse-exclusive
105    // cumulative log-decay of the token's own window.
106    let da_btuh = da_bsh.reshape([batch, tokens, u, nheads]);
107    let cumulative_btuh = da_btuh.cumsum(2);
108    let total_bt1h = cumulative_btuh.clone().narrow(2, u - 1, 1);
109    let decay_btJh = (total_bt1h - cumulative_btuh).exp().narrow(2, 0, taps);
110    let excess_btJh = excess_bsh
111        .reshape([batch, tokens, u, nheads])
112        .narrow(2, 0, taps);
113    let weight_btJh = excess_btJh * decay_btJh;
114
115    // ── Fuse (tap, mimo_rank): both are outer products into one state ─────────
116    let fused = taps * mimo_rank;
117    let v_btJmhp = v_bsmhp
118        .reshape([batch, tokens, u, mimo_rank, nheads, per_head_dim])
119        .narrow(2, 0, taps);
120    let v_weighted_btKhp = (v_btJmhp * weight_btJh.unsqueeze_dims::<6>(&[3, 5]))
121        .reshape([batch, tokens, fused, nheads, per_head_dim]);
122    let b_btKhr = b_bsmhr
123        .reshape([batch, tokens, u, mimo_rank, nheads, state_rank])
124        .narrow(2, 0, taps)
125        .reshape([batch, tokens, fused, nheads, state_rank]);
126    // The read `C` is the token's last micro-step — the copy carrying the
127    // cumulative rotation the readout happens at, which is the only one it is
128    // ever built at.
129
130    // ── (C · Bᵀ) · V, contracting state_rank then the fused axis ──────────────
131    let c_bthmr = c_btmhr.swap_dims(2, 3);
132    let b_bthrK = b_btKhr.permute([0, 1, 3, 4, 2]);
133    let qk_bthmK = c_bthmr.matmul(b_bthrK);
134    let v_bthKp = v_weighted_btKhp.swap_dims(2, 3);
135    let corr_bthmp = qk_bthmK.matmul(v_bthKp);
136    Some(corr_bthmp.swap_dims(2, 3))
137}
