1//! # Mamba-3 — Single-Pass SSD Forward
2//!
3//! `forward_single_ssd` and `step_single_ssd` on
4//! [`Mamba3`](crate::mamba3::mamba3::Mamba3): the pathway of the **official
5//! Mamba-3 kernels**, Triton (SISO) and Tilelang (MIMO):
6//!
7//! ```text
8//!   scaleₜ = γₜ + νₜ₊ₗₐ₉ (+ νⁱⁿᵗₜ₊₁)
9//!
10//!   forward_single_ssd:    h' = SSD(V_raw, K_scaled = scaleₜ B) with:
11//!                               * strict lower-triangular intra-chunk mask
12//!                               * additive γ-weighted diagonal correction
13//!                               * boundary β seed Σⱼ νⱼ Kⱼ ⊗ xⱼ over the
14//!                                 cache's `lag` tap slots
15//!                               * at lag > 1, the rest of the correction band
16//!                                 ([`crate::mamba3::single_ssd::token_band`])
17//! ```
18//!
19//! `lag` is [`TapPattern::tap_lag`](crate::mamba3::tap_pattern::TapPattern::tap_lag):
20//! `1` for the default tap pattern, `u` for
21//! [`TapPattern::Vertical`](crate::mamba3::tap_pattern::TapPattern::Vertical). A
22//! two-tap pattern adds the term in parentheses: a second shift of the same
23//! shape, because the collapse of §9 leaves one scalar per sample for any
24//! number of taps. That is all it costs here: one pass either way.
25//!
26//! References:
27//! - [`mamba3_siso_fwd.py`](https://github.com/state-spaces/mamba/mamba_ssm/ops/triton/mamba3/mamba3_siso_fwd.py),
28//! - [`mamba3_mimo_fwd.py`](https://github.com/state-spaces/mamba/mamba_ssm/ops/tilelang/mamba3/mamba3_mimo_fwd.py).
29//!
30//! See also: [`crate::mamba3::mamba3`] and [`crate::mamba3::double_ssd::double_ssd`].
31

41impl Mamba3 {
42    /// Process a full input sequence with the **single-ssd (single-pass)**
43    /// pathway.
44    ///
45    /// Functionally equivalent to [`Self::forward_double_ssd`], with about half
46    /// the SSD memory in training. The cache is a separate type
47    /// ([`Mamba3SingleSsdCache`]), because this pathway carries the
48    /// accumulator, which differs from the state mid-sequence.
49    ///
50    /// "Equivalent" covers **everything that a caller can observe**: the
51    /// returned output and every field of the returned cache. It does not cover
52    /// the intermediate `y` of the SSD scan at each *write* position. Under a
53    /// lag-`u` pattern such as [`TapPattern::Vertical`], the correction band
54    /// applies only at the last micro-step of each token (the readout). So the
55    /// `u−1` per-token partial sums that this pathway discards do not match the
56    /// double-SSD ones. Nothing reads them, and the *state* is exact at every
57    /// position in both pathways. See [`crate::mamba3::single_ssd::token_band`],
58    /// which also tells why that band never has to enter the kernel.
59    ///
60    /// Under [`TapPattern::None`] there is no second pass to fuse. The composite
61    /// key scale is `γ`, and the γ-correction is the whole diagonal. So
62    /// this *is* the double-SSD form (`h' ≡ h` at every position, not only at
63    /// boundaries), and the call delegates to `forward_double_ssd`. It does not
64    /// run a strict-mask kernel plus a correction that puts back what the mask
65    /// removed.
66    ///
67    /// `pad_bt` marks a right-padded batch of tokens, as in [`Self::forward`].
68    /// The last tap installment of a real position, which a padded position
69    /// would pay, then stays in the tap slots, exactly as at a call boundary.
70    /// So the accumulator `h'` equals the state from the end of each row on.
71    ///
72    /// # Shapes
73    /// - `input_bsm`: `[batch, sequence, d_model]`
74    /// - `pad_bt`: `[batch, sequence]`
75    /// - output: `[batch, sequence, d_model]`
76    pub fn forward_single_ssd(
77        &self,
78        input_bsm: Tensor<3>,
79        cache: Option<Mamba3SingleSsdCache>,
80        ssd_path: &Mamba3SsdPath,
81        pad_bt: Option<Tensor<2, Bool>>,
82    ) -> (Tensor<3>, Mamba3SingleSsdCache) {
83        self.forward_single_ssd_packed(input_bsm, cache, None, ssd_path, pad_bt, None)
84    }
85
86    /// [`Self::forward_single_ssd`] over a packed batch: `reset_bt` marks the
87    /// first token of each segment, which starts from `origin`, as in
88    /// [`Self::forward_packed`].
89    ///
90    /// A reset is a call boundary inside the row: the positions before it pay
91    /// no `ν` forward (the taps of the segment start read the slots of the
92    /// origin), and the chunk restarts from the `h'` of the origin plus its
93    /// own boundary seed, as a call from the origin does.
94    #[allow(non_snake_case)]
95    pub fn forward_single_ssd_packed(
96        &self,
97        input_bsm: Tensor<3>,
98        cache: Option<Mamba3SingleSsdCache>,
99        origin: Option<Mamba3SingleSsdCache>,
100        ssd_path: &Mamba3SsdPath,
101        pad_bt: Option<Tensor<2, Bool>>,
102        reset_bt: Option<Tensor<2, Bool>>,
103    ) -> (Tensor<3>, Mamba3SingleSsdCache) {
104        if !self.tap_pattern.has_beta_tap() {
105            let (out_bsm, cache) = self.forward_double_ssd_packed(
106                input_bsm,
107                cache.map(Into::into),
108                origin.map(Into::into),
109                ssd_path,
110                pad_bt,
111                reset_bt,
112            );
113            return (out_bsm, cache.into());
114        }
115        let [batch, tokens, _d_model] = input_bsm.dims();
116        let nheads = self.nheads();
117        let per_head_dim = self.per_head_dim();
118        let state_rank = self.state_rank;
119        let mimo_rank = self.mimo_rank;
120        let u = self.micro_steps;
121        let device = input_bsm.device();
122        let sequence = tokens * u;
123
124        // ── The entry of the plant frame ──────────────────────────────────────
125        // At a call boundary, the accumulator `h'` equals the state, so the
126        // caches convert field by field.
127        let chunk_len = ssd_path.chunk_len_or_optimal(self);
128        let PlantEntry {
129            mut cache,
130            packed,
131            end_b,
132            z_bti,
133            x_bshp,
134            b_bsmhr,
135            c_btmhr,
136            da_bsh,
137            alpha_bsh: _,
138            nu_bsh,
139            nu_interior_bsh,
140            gamma_bsh,
141            log_precision_bsh,
142            tropical_ab_bsh,
143            new_rotation,
144            b_last_bumhr,
145            x_last_buhp,
146        } = self.plant_entry(
147            input_bsm,
148            cache.map(Into::into),
149            pad_bt,
150            reset_bt.map(|reset_bt| PlantResets {
151                reset_bt,
152                origin: origin.map(Into::into),
153                chunk_len,
154            }),
155        );
156        let nu_bsh = nu_bsh.expect("a β tap has a mass");
157
158        // ── Compute scaleₜ = γₜ + νₜ₊ₗₐ₉ (+ νⁱⁿᵗₜ₊₁) ─────────────────────────
159        //
160        // Each shifted term is zero for the last positions that its own lag
161        // reaches past: the taps that pay them belong to the *next* call, from
162        // the tap slots. This also makes the accumulator `h'` equal to the
163        // state at a cache boundary, hence the field-identity `From` impls.
164        //
165        // `t+lag` is a later *write* position: lag 1 is
166        // [`TapPattern::HorizontalCarryOver`], lag `u` is [`TapPattern::Vertical`].
167        // This is the `Δ̃` collapse (`info/mamba-3/trapezoid-as-integration.md`
168        // §5). The collapse theorem of §9 is why it holds unchanged for the
169        // wider lag and for a second tap: still one scalar per sample, so one
170        // pass. Only the γ-correction widens from the diagonal to a
171        // `lag`-wide band (see [`crate::mamba3::single_ssd::token_band`]).
172        let lag = self.tap_lag();
173        // νₜ₊ₗ, the mass that a later position pays to this one. Zero past the
174        // end of the call, where the tap belongs to the next call. Zero also
175        // across a reset of a packed row: the first positions of a segment pay
176        // the cache slots, through the seed of their chunk (below). (No output
177        // would see that mass: the restart drops the state it enters, and the
178        // band correction takes it out of the reads of its token. The zero
179        // keeps `h'` at a reset equal to that at a call boundary.)
180        let pay_forward = |nu_bsh: Tensor<3>, l: usize| {
181            let nu_bsh = match &packed {
182                Some(p) => crate::padding::fill_padded(nu_bsh, &p.segments.heads(l), 0.0),
183                None => nu_bsh,
184            };
185            let zero_bLh = Tensor::zeros([batch, l, nheads], &device);
186            if sequence == l {
187                zero_bLh
188            } else {
189                Tensor::cat(vec![nu_bsh.narrow(1, l, sequence - l), zero_bLh], 1)
190            }
191        };
192        let far_shifted_bsh = pay_forward(nu_bsh.clone(), lag);
193        // A two-tap pattern also collapses to one scalar (§9), with a second
194        // shift in it: `scaleₜ = γₜ + νⁱⁿᵗₜ₊₁ + νᶠᵃʳₜ₊ₗₐ₉`.
195        let interior_shifted_bsh = nu_interior_bsh
196            .clone()
197            .map(|nu_bsh| pay_forward(nu_bsh, 1));
198        let scale_bsh = gamma_bsh.clone() + far_shifted_bsh.clone();
199        let scale_bsh = match &interior_shifted_bsh {
200            Some(interior_bsh) => scale_bsh + interior_bsh.clone(),
201            None => scale_bsh,
202        };
203        san(&scale_bsh);
204
205        // ── Boundary β seed for the initial accumulator ─────────────────────
206        // Add Σⱼ νⱼ · Σₘ K_prev[j, m] ⊗ (x_prev[j] ⊙ mimo_xₘ) to the carried
207        // accumulator. Cache slot `j` (oldest first) is the position
208        // whose tap position `j` of this call pays, so it takes `νⱼ` from the
209        // first `lag` positions of this call. `x_prev` already carries the
210        // decay from its own position to the boundary, and `K_prev` its own
211        // rotation, so the pair *is* the transported write.
212        //
213        // Under [`TapPattern::VerticalPlusHorizontalCarryOver`], one more
214        // installment crosses. The newest slot is also the lag-1 tap of the
215        // *first* position of this call, with the empty decay product. So it is
216        // the same term, with `νⁱⁿᵗ₀` added to the own weight of that slot.
217        // Every other pattern has no such term. Either it has no lag-1 tap
218        // beside the tap at `tap_lag`, or that lag-1 tap is closed exactly
219        // there.
220        let mimo_x_hmp = self.mimo_x_hmp.as_ref().map(|p| p.val());
221        // The masses that the first `width` positions of each chunk pay,
222        // `[batch, nchunks, width, nheads]`. Only chunk 0 is the start of the
223        // call. In a packed row, every chunk that starts a segment is the
224        // start of one from the origin.
225        let nchunks = sequence.div_ceil(chunk_len);
226        let heads_of = |t_bsh: Tensor<3>, width: usize| {
227            let pad = nchunks * chunk_len - sequence;
228            let t_bSh = if pad == 0 {
229                t_bsh
230            } else {
231                Tensor::cat(vec![t_bsh, Tensor::zeros([batch, pad, nheads], &device)], 1)
232            };
233            t_bSh
234                .reshape([batch, nchunks, chunk_len, nheads])
235                .narrow(2, 0, width)
236        };
237        let nu_heads_bnuh = heads_of(nu_bsh.clone(), lag);
238        let nu_heads_bnuh = match nu_interior_bsh
239            .filter(|_| self.tap_pattern.two_tap_lag1_crosses_tokens())
240        {
241            // `lag ≥ 2` here: a two-tap pattern folds at `u = 1`, so the lag-1
242            // tap of a two-tap pattern and a one-slot FIFO never coexist.
243            Some(nu_interior_bsh) => Tensor::cat(
244                vec![
245                    nu_heads_bnuh.clone().narrow(2, 0, lag - 1),
246                    nu_heads_bnuh.narrow(2, lag - 1, 1) + heads_of(nu_interior_bsh, 1),
247                ],
248                2,
249            ),
250            None => nu_heads_bnuh,
251        };
252        // The slots of a FIFO, `v` with MIMO_V applied:
253        // `([batch, lag, mimo_rank, nheads, per_head_dim], [batch, lag, mimo_rank, nheads, state_rank])`.
254        let slots = "a β tap keeps its (B, x) cache slots";
255        let prev_slots = |v_state_buhp: Option<Tensor<4>>, k_state_bumhr: Option<Tensor<5>>| {
256            let v_prev_mimo_bumhp =
257                helpers::build_v_with_mimo::<4, 5>(v_state_buhp.expect(slots), mimo_x_hmp.as_ref(), 2);
258            (v_prev_mimo_bumhp, k_state_bumhr.expect(slots))
259        };
260        // The seed of the call, from the slots of the cache and the masses of
261        // chunk 0.
262        let (v_prev_mimo_bumhp, k_prev_bumhr) =
263            prev_slots(cache.v_state_buhp.clone(), cache.k_state_bumhr.clone());
264        let nu_head_buh = nu_heads_bnuh.clone().narrow(1, 0, 1).squeeze_dim::<3>(1);
265        let v_prev_mimo_bumhp = v_prev_mimo_bumhp * nu_head_buh.unsqueeze_dims::<5>(&[2, 4]);
266        // The seed contracts over the slots *and* the ranks (both are outer
267        // products into one accumulator). So, fused, it is the same one
268        // `mimo_outer_sum` that the per-token write uses.
269        let boundary_seed_bhpr = helpers::mimo_outer_sum(
270            v_prev_mimo_bumhp.reshape([batch, lag * mimo_rank, nheads, per_head_dim]),
271            k_prev_bumhr.reshape([batch, lag * mimo_rank, nheads, state_rank]),
272            self.use_siso_decode_kernels(),
273        );
274        let initial_state_bhpr = cache.ssm_bhpr.clone() + boundary_seed_bhpr;
275        san(&initial_state_bhpr);
276        // A packed row: each chunk gets a seed from the slots of the origin,
277        // in case it restarts. Each slot writes one outer product (over the
278        // ranks). Each chunk weighs the slots by its own head masses.
279        let seed_bnhpr = packed.as_ref().map(|p| {
280            let (v_origin_mimo_bumhp, k_origin_bumhr) =
281                prev_slots(p.origin.v_state_buhp.clone(), p.origin.k_state_bumhr.clone());
282            let slot_writes_bhuP = helpers::mimo_outer_sum(
283                v_origin_mimo_bumhp.reshape([batch * lag, mimo_rank, nheads, per_head_dim]),
284                k_origin_bumhr.reshape([batch * lag, mimo_rank, nheads, state_rank]),
285                self.use_siso_decode_kernels(),
286            )
287            .reshape([batch, lag, nheads, per_head_dim * state_rank])
288            .swap_dims(1, 2);
289            nu_heads_bnuh
290                .permute([0, 3, 1, 2]) // _bhnu
291                .matmul(slot_writes_bhuP) // _bhnP
292                .reshape([batch, nheads, nchunks, per_head_dim, state_rank])
293                .swap_dims(1, 2)
294        });
295
296        // ── Pad sequence to multiple of chunk_len ─────────────────────────────
297        let sequence_padded = sequence.next_multiple_of(chunk_len);
298        let pad = sequence_padded - sequence;
299
300        // V passed to SSD is raw x with MIMO_V applied (not γ-scaled).
301        let v_bshmp = helpers::build_v_with_mimo::<4, 5>(x_bshp.clone(), mimo_x_hmp.as_ref(), 2);
302        // v_bshmp has axis order [b, s, m, h, p] (insert_dim=2 onto [b,s,h,p]).
303
304        // ── The lag-`u` correction band ───────────────────────────────────────
305        // At lag 1 the key scale is wrong only on the diagonal, and
306        // `ssd/diag.rs` corrects it inside the kernel. At lag `u` the exception
307        // is `u` wide, and at the positions that the readout keeps it is
308        // exactly the token. So it is one contraction here, not a wider mask.
309        // See [`crate::mamba3::single_ssd::token_band`].
310        //
311        // The band is the **lag-`u`** installment only, also for a two-tap
312        // pattern. At the surviving read `t = τu + u−1`, the lag-1 installment
313        // of a sample has already landed for every `j < u−1`. At
314        // `j = u−1`, the kernel replaces the whole weight with `γ` anyway. Only
315        // the lag-`u` installment is still unpaid there.
316        let band_correction_btmhp = (lag > 1)
317            .then(|| {
318                crate::mamba3::single_ssd::token_band::token_band_correction(
319                    v_bshmp.clone(),
320                    b_bsmhr.clone(),
321                    c_btmhr.clone(),
322                    far_shifted_bsh,
323                    da_bsh.clone(),
324                    u,
325                )
326            })
327            .flatten();
328
329        // `C` and `γ` are on the read axis of the chunk, so they pad by whole
330        // tokens. `chunk_len` is a multiple of `u`, so `pad` is too.
331        let gamma_bth = helpers::read_positions::<3, 4>(gamma_bsh, 1, u);
332        let tokens_padded = sequence_padded / u;
333        let pad_tokens = tokens_padded - tokens;
334
335        #[rustfmt::skip]
336        let (v_bShmp, da_bSh, gamma_bTh, scale_bSh, b_bSmhr, c_bTmhr) = if pad == 0 {
337            (v_bshmp, da_bsh, gamma_bth, scale_bsh, b_bsmhr, c_btmhr)
338        } else {
339            let pad_bShmp = Tensor::zeros([batch, pad, mimo_rank, nheads, per_head_dim], &device);
340            let pad_bSh = Tensor::zeros([batch, pad, nheads], &device);
341            let pad_bTh = Tensor::zeros([batch, pad_tokens, nheads], &device);
342            let pad_bSmhr = Tensor::zeros([batch, pad, mimo_rank, nheads, state_rank], &device);
343            let pad_bTmhr =
344                Tensor::zeros([batch, pad_tokens, mimo_rank, nheads, state_rank], &device);
345            (
346                Tensor::cat(vec![v_bshmp, pad_bShmp], 1),
347                Tensor::cat(vec![da_bsh, pad_bSh.clone()], 1),
348                Tensor::cat(vec![gamma_bth, pad_bTh], 1),
349                Tensor::cat(vec![scale_bsh, pad_bSh], 1),
350                Tensor::cat(vec![b_bsmhr, pad_bSmhr], 1),
351                Tensor::cat(vec![c_btmhr, pad_bTmhr], 1),
352            )
353        };
354
355        // ── Reshape into chunks ───────────────────────────────────────────────
356        let nchunks = sequence_padded / chunk_len;
357        let chunk_tokens = standard_ssd::chunk_tokens(chunk_len, u);
358        let v_bnlmhp =
359            v_bShmp.reshape([batch, nchunks, chunk_len, mimo_rank, nheads, per_head_dim]);
360        let da_bnlh = da_bSh.reshape([batch, nchunks, chunk_len, nheads]);
361        let gamma_bnth = gamma_bTh.reshape([batch, nchunks, chunk_tokens, nheads]);
362        let scale_bnlh = scale_bSh.reshape([batch, nchunks, chunk_len, nheads]);
363        let b_bnlmhr = b_bSmhr.reshape([batch, nchunks, chunk_len, mimo_rank, nheads, state_rank]);
364        let c_bntmhr =
365            c_bTmhr.reshape([batch, nchunks, chunk_tokens, mimo_rank, nheads, state_rank]);
366
367        // ── Run the single-pass form of the SSD ───────────────────────────────
368        let ssd_input = Mamba3SingleSsdInput {
369            v_bnlmhp,
370            b_bnlmhr,
371            c_bntmhr,
372            da_bnlh,
373            gamma_bnth,
374            scale_bnlh,
375            initial_state_bhpr,
376            init_state_hpr: self.init_state_hpr.as_ref().map(|s| s.val()),
377            read_stride: u,
378            siso_specialization: self.siso_specialization,
379            restarts: packed.as_ref().map(|p| crate::packing::Restarts {
380                reset_bn: p.segments.reset_bn.clone(),
381                origin_bhpr: p.origin.ssm_bhpr.clone(),
382            }),
383            seed_bnhpr,
384        };
385        let (y_bntmhp, final_state_bhpr) = ssd_input.run(ssd_path);
386
387        san(&y_bntmhp);
388        san(&final_state_bhpr);
389        cache.ssm_bhpr = final_state_bhpr;
390
391        // ── Unpad ─────────────────────────────────────────────────────────────
392        // The SSD returns token resolution: the readout is after all `u`
393        // writes, and the kernel computed only that position.
394        let y_bTmhp = y_bntmhp.reshape([batch, tokens_padded, mimo_rank, nheads, per_head_dim]);
395        let y_btmhp = if pad == 0 {
396            y_bTmhp
397        } else {
398            y_bTmhp.narrow(1, 0, tokens)
399        };
400
401        let y_btmhp = match band_correction_btmhp {
402            Some(correction_btmhp) => y_btmhp - correction_btmhp,
403            None => y_btmhp,
404        };
405
406        // ── The tail of the plant frame ───────────────────────────────────────
407        let (out_btd, cache) = self.plant_tail(
408            PlantTail {
409                y_btmhp,
410                z_bti,
411                x_bshp,
412                log_precision_bsh,
413                tropical_ab_bsh,
414                end_b,
415                packed,
416                new_rotation,
417                b_last_bumhr,
418                x_last_buhp,
419            },
420            cache,
421        );
422        (out_btd, cache.into())
423    }
424}

426// ---------------------------------------------------------------------------
427// Mamba3::step  (recurrent SSM — token-by-token decoding)
428// ---------------------------------------------------------------------------
429
430mod step {
431    use super::*;
432
433    impl Mamba3 {
434        /// Process a **single token** with the pure recurrent form (the
435        /// formulas are in [`Mamba3::step`]). It converts the cache to the
436        /// double-ssd form, runs [`Mamba3::step_double_ssd`], and converts back.
437        ///
438        /// # Shapes
439        /// - `input_bd` : `[batch, d_model]`
440        /// - output     : `[batch, d_model]`
441        #[allow(non_snake_case)]
442        pub fn step_single_ssd(
443            &self,
444            input_bd: Tensor<2>,
445            cache: Option<Mamba3SingleSsdCache>,
446        ) -> (Tensor<2>, Mamba3SingleSsdCache) {
447            // Token-by-token decoding always uses the recurrent (double-ssd)
448            // form. A single-ssd cache holds the accumulator at a call
449            // boundary, where the accumulator equals the state (see the `From`
450            // impls in `crate::mamba3::cache`). So the
451            // conversion in and back out is lossless. One `step` call is
452            // itself a boundary-to-boundary transition, so the round trip stays
453            // exact.
454            let cache = cache.map(Mamba3DoubleSsdCache::from);
455            let (out_bd, cache) = self.step_double_ssd(input_bd, cache);
456            (out_bd, cache.into())
457        }
458    }
459}
