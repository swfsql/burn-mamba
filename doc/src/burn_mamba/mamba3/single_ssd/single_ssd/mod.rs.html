1//! # Mamba-3 — Single-Pass SSD Forward
2//!
3//! This module provides the `forward_single_ssd` method on [`Mamba3`](crate::mamba3::mamba3::Mamba3):
4//! The burn-mamba implementation of the **official Mamba-3 algorithm**
5//! as shipped in Triton (SISO) and Tilelang (MIMO):
6//!
7//! ```text
8//!   scaleₜ = γₜ + νₜ₊ₗₐ₉ (+ νⁱⁿᵗₜ₊₁)
9//!
10//!   forward_single_ssd:    h' = SSD(V_raw, K_scaled = scaleₜ B) with:
11//!                               * strict lower-triangular intra-chunk mask
12//!                               * additive γ-weighted same-step correction
13//!                               * boundary β seed Σⱼ νⱼ Kⱼ ⊗ xⱼ over the
14//!                                 cache's `lag` tap slots
15//!                               * at lag > 1, the rest of the correction band
16//!                                 ([`crate::mamba3::single_ssd::token_band`])
17//! ```
18//!
19//! `lag` is [`Trapezoid::tap_lag`](crate::mamba3::trapezoid::Trapezoid::tap_lag):
20//! `1` for the default tap pattern, `u` for
21//! [`Trapezoid::Vertical`](crate::mamba3::trapezoid::Trapezoid::Vertical). A
22//! two-tap pattern adds the parenthesised term — a second shift of the same
23//! shape, since §9's collapse leaves one scalar per sample however many taps
24//! there are. That is the whole of its cost here: one pass either way.
25//!
26//! References:
27//! - [`mamba3_siso_fwd.py`](https://github.com/state-spaces/mamba/mamba_ssm/ops/triton/mamba3/mamba3_siso_fwd.py),
28//! - [`mamba3_mimo_fwd.py`](https://github.com/state-spaces/mamba/mamba_ssm/ops/tilelang/mamba3/mamba3_mimo_fwd.py).
29//!
30//! See also: [`crate::mamba3::mamba3`] and [`crate::mamba3::double_ssd::double_ssd`].
31
32use crate::mamba3::double_ssd::prelude::Mamba3DoubleSsdCache;
33use crate::mamba3::helpers;
34use crate::mamba3::prelude::*;
35use crate::mamba3::rotation::rotate_bc_forward;
36use crate::mamba3::single_ssd::prelude::*;
37use burn_stack::modules::Silu;
38use burn_stack::modules::sanity as san;
39use burn::prelude::*;

41impl Mamba3 {
42    /// Process a full input sequence using the **single-ssd form (single-pass)**
43    /// trapezoidal algorithm.
44    ///
45    /// Functionally equivalent to [`Self::forward`] but uses approximately half
46    /// the SSD memory during training. Cache is a separate type
47    /// ([`Mamba3SingleSsdCache`]) because the stored hidden state has different
48    /// semantics than the original-form cache used by [`Self::forward`].
49    ///
50    /// "Equivalent" is over **everything a caller can observe** — the returned
51    /// output and every field of the returned cache — which is the whole of what
52    /// this method is. It is not a claim about the intermediate `y` the SSD core
53    /// produces at each *folded* position: under
54    /// [`Trapezoid::Vertical`] the
55    /// correction band is only applied at each token's last micro-step, the one
56    /// the readout happens at, so the `u−1` per-token partial sums this pathway
57    /// discards do not match the double-SSD ones. Nothing reads them, and the
58    /// *state* is exact at every position in both pathways — see
59    /// [`crate::mamba3::single_ssd::token_band`], which is also why that band
60    /// never has to enter the kernel.
61    ///
62    /// Except under [`Trapezoid::None`], where there is no second pass to fuse:
63    /// the composite key scale is `γ` and the same-step correction is the whole
64    /// diagonal, so this *is* the double-SSD form — `h' ≡ h` at every position,
65    /// not merely at boundaries — and the call delegates rather than run a
66    /// strict-mask kernel plus a correction that reassembles what it masked.
67    ///
68    /// # Shapes
69    /// - `input_bsm`: `[batch, sequence, d_model]`
70    /// - output: `[batch, sequence, d_model]`
71    #[allow(non_snake_case)]
72    pub fn forward_single_ssd(
73        &self,
74        input_bsm: Tensor<3>,
75        cache: Option<Mamba3SingleSsdCache>,
76        ssd_path: &Mamba3SsdPath,
77    ) -> (Tensor<3>, Mamba3SingleSsdCache) {
78        if !self.trapezoid.has_beta_tap() {
79            let (out_bsm, cache) =
80                self.forward_double_ssd(input_bsm, cache.map(Into::into), ssd_path);
81            return (out_bsm, cache.into());
82        }
83        let [batch, tokens, _d_model] = input_bsm.dims();
84        let d_inner = self.d_inner();
85        let nheads = self.nheads();
86        let ngroups = self.ngroups;
87        let per_head_dim = self.per_head_dim();
88        let state_rank = self.state_rank;
89        let mimo_rank = self.mimo_rank;
90        let micro_steps = self.micro_steps;
91        let device = input_bsm.device();
92
93        // MambaProduct: from the split below down to the readout, one sequence
94        // position *is* one micro-step — the `s` in every shape suffix counts
95        // micro-steps, and `tokens` is the only name still at token resolution.
96        // See [`crate::mamba3::product`].
97        let sequence = tokens * micro_steps;
98
99        assert!(tokens > 0, "sequence length must be at least 1");
100        assert_eq!(nheads % ngroups, 0);
101        san(&input_bsm);
102
103        // ── Initialise cache if not provided ──────────────────────────────────
104        let mut cache = cache.unwrap_or_else(|| {
105            let ssm_bhpr = Tensor::zeros([batch, nheads, per_head_dim, state_rank], &device);
106            // Reached only with a β tap (`Trapezoid::None` delegated above).
107            let (k_state_bumhr, v_state_buhp) = self.zero_tap_slots(batch, &device);
108            let rotation = self.zero_rotation_state(batch, &device);
109            Mamba3SingleSsdCache {
110                ssm_bhpr,
111                k_state_bumhr,
112                v_state_buhp,
113                rotation,
114            }
115        });
116
117        // ── Step 1: In-projection ─────────────────────────────────────────────
118        let proj_bsd = self.in_proj.forward(input_bsm);
119        let bc_size = ngroups * state_rank * mimo_rank;
120
121        // `u` = micro_steps widens every per-micro-step segment, and `unfold`
122        // reinterprets each of them as `u` consecutive sequence positions. `z`
123        // (per-token gate) and `C` (per-token read) do not widen — `C` is
124        // instead broadcast across the group so its *last* copy carries the
125        // right cumulative rotation. See [`crate::mamba3::product`].
126        // The optional segments come off the tail first, in layout order: the
127        // rotation (`Real1D` projects none), then `μ` (only a two-tap pattern
128        // projects it), then `λ` — which is always present here,
129        // `Trapezoid::None` having delegated above — since a zero-width segment
130        // would silently vanish from the split below.
131        let u = micro_steps;
132        let (proj_bsd, rot_btA) =
133            helpers::split_trailing(proj_bsd, self.rotation_channels_total(), 2);
134        let (proj_bsd, mu_raw_btH) = helpers::split_trailing(proj_bsd, self.mu_channels_total(), 2);
135        let (proj_bsd, lambda_raw_btH) =
136            helpers::split_trailing(proj_bsd, self.lambda_channels_total(), 2);
137        let lambda_raw_btH = lambda_raw_btH.expect("a β tap projects λ");
138        #[rustfmt::skip]
139        let [
140                z_bsi, x_btI,
141                b_raw_btMGRU, c_raw_btMGR,
142                dd_dt_btH, dd_A_raw_btH,
143        ] = burn_stack::modules::split_into(
144            proj_bsd,
145            [
146                d_inner, u * d_inner,
147                u * bc_size, bc_size,
148                u * nheads, u * nheads,
149            ],
150            2,
151        );
152
153        use crate::mamba3::product::unfold_micro_bs;
154        let x_bsi = unfold_micro_bs(x_btI, u);
155        let b_raw_bsMGR = unfold_micro_bs(b_raw_btMGRU, u);
156        // `C` stays at token resolution all the way into the SSD: the readout
157        // happens once per token, at its last micro-step, and that is the only
158        // position the kernel is ever asked for a `y` at. See
159        // [`Mamba3SingleSsdInput::c_bntmhr`].
160        let dd_dt_bsh = unfold_micro_bs(dd_dt_btH, u);
161        let dd_A_raw_bsh = unfold_micro_bs(dd_A_raw_btH, u);
162        let lambda_raw_bsh = unfold_micro_bs(lambda_raw_btH, u);
163        let mu_raw_bsh = mu_raw_btH.map(|t| unfold_micro_bs(t, u));
164        let rot_bsa = rot_btA.map(|t| unfold_micro_bs(t, u));
165
166        san(&z_bsi);
167        san(&x_bsi);
168        san(&dd_dt_bsh);
169
170        // ── Step 2: Discretisation + trapezoidal coefficients ─────────────────
171        let helpers::TrapezoidCoeffs {
172            dt: dt_bsh,
173            da: da_bsh,
174            alpha: _alpha_bsh,
175            nu: nu_bsh,
176            nu_interior: nu_interior_bsh,
177            gamma: gamma_bsh,
178        } = helpers::trapezoidal_coefficients(
179            dd_dt_bsh,
180            dd_A_raw_bsh,
181            Some(lambda_raw_bsh),
182            mu_raw_bsh,
183            self.dt_bias_h.val(),
184            self.trapezoid_spec(),
185        );
186        let nu_bsh = nu_bsh.expect("a β tap has a mass");
187        san(&dt_bsh);
188        san(&da_bsh);
189        san(&gamma_bsh);
190
191        // ── Compute scaleₜ = γₜ + νₜ₊ₗₐ₉ (+ νⁱⁿᵗₜ₊₁) ─────────────────────────
192        //
193        // Each shifted term is zero for the last positions its own lag reaches
194        // past (the taps that pay them belong to the *next* call, out of the tap
195        // slots) — which is also what makes `h'` coincide with the double-SSD
196        // state at a cache boundary, hence the field-identity `From` impls.
197        //
198        // `t+lag` is a later *folded* position: lag 1 is
199        // [`Trapezoid::HorizontalCarryOver`], lag `u` is [`Trapezoid::Vertical`].
200        // This is the `Δ̃` collapse (`info/trapezoid-as-integration.md` §5), and
201        // §9's collapse theorem is why it survives the wider lag — and a second
202        // tap — unchanged: still one scalar per sample, hence one pass. Only the
203        // same-step correction widens from the diagonal to a `lag`-wide band
204        // (see [`crate::mamba3::single_ssd::token_band`]).
205        let lag = self.tap_lag();
206        // νₜ₊ₗ, the mass a later position pays this one — zero past the end of
207        // the call, where the tap belongs to the next one.
208        let pay_forward = |nu_bsh: Tensor<3>, l: usize| {
209            let zero_bLh = Tensor::zeros([batch, l, nheads], &device);
210            if sequence == l {
211                zero_bLh
212            } else {
213                Tensor::cat(vec![nu_bsh.narrow(1, l, sequence - l), zero_bLh], 1)
214            }
215        };
216        let far_shifted_bsh = pay_forward(nu_bsh.clone(), lag);
217        // A two-tap pattern collapses to one scalar all the same (§9), just with
218        // a second shift in it: `scaleₜ = γₜ + νⁱⁿᵗₜ₊₁ + νᶠᵃʳₜ₊ₗₐ₉`.
219        let interior_shifted_bsh = nu_interior_bsh
220            .clone()
221            .map(|nu_bsh| pay_forward(nu_bsh, 1));
222        let scale_bsh = gamma_bsh.clone() + far_shifted_bsh.clone();
223        let scale_bsh = match &interior_shifted_bsh {
224            Some(interior_bsh) => scale_bsh + interior_bsh.clone(),
225            None => scale_bsh,
226        };
227        san(&scale_bsh);
228
229        // ── Step 3: Reshape x ─────────────────────────────────────────────────
230        let x_bshp = x_bsi.reshape([batch, sequence, nheads, per_head_dim]);
231
232        // ── Step 4: QK-Norm on B and C ────────────────────────────────────────
233        let b_bsmhr = helpers::qk_norm_expand_bias::<5, 6>(
234            b_raw_bsMGR.reshape([batch, sequence, mimo_rank, ngroups, state_rank]),
235            &self.b_norm,
236            self.b_bias_hmr.val(),
237            3,
238            nheads,
239        );
240        let c_btmhr = helpers::qk_norm_expand_bias::<5, 6>(
241            c_raw_btMGR.reshape([batch, tokens, mimo_rank, ngroups, state_rank]),
242            &self.c_norm,
243            self.c_bias_hmr.val(),
244            3,
245            nheads,
246        );
247
248        // ── Step 5: Data-dependent transition rotation of B and C ─────────────
249        // Complex2D: abelian RoPE (cumulative angle). Quaternion4D: cumulative
250        // unit quaternion. Shared with the double-ssd pathway via
251        // [`rotate_bc_forward`]; the single-pass SSD core below is
252        // rotation-agnostic — it only ever consumes the rotated B̄/C̄ (the RoPE
253        // factoring `C̄ₜᵀB̄ᵢ = Cₜᵀ·Rel(t,i)·Bᵢ` holds for either algebra).
254        // `C` is rotated at token resolution too: it needs the cumulative
255        // rotation of the micro-step it is read at, which is a stride slice of
256        // the one `B` is rotated by.
257        let (b_bsmhr, c_btmhr, new_rotation) = rotate_bc_forward(
258            rot_bsa,
259            dt_bsh.clone(),
260            cache.rotation.clone(),
261            b_bsmhr,
262            c_btmhr,
263            u,
264            self.rotation_spec(),
265        );
266        san(&b_bsmhr);
267        san(&c_btmhr);
268
269        // ── Save the last `lag` positions' B and x (raw, no MIMO_V) ───────────
270        // The positions whose second installment the next call pays; at
271        // `lag = u` that window is precisely the last token.
272        let (b_last_bumhr, x_last_buhp) =
273            self.save_tap_slots(&b_bsmhr, &x_bshp, &da_bsh, lag);
274
275        // ── Boundary β seed for initial state ─────────────────────────────────
276        // Add Σⱼ νⱼ · Σₘ K_prev[j, m] ⊗ (x_prev[j] ⊙ mimo_xₘ) to the carried
277        // single-ssd SSM state: cache slot `j` (oldest first) is the position
278        // whose tap is paid by this call's position `j`, so it takes `νⱼ` from
279        // the current call's first `lag` positions. `x_prev` already carries the
280        // decay from its own position to the boundary, and `K_prev` its own
281        // rotation, so the pair *is* the transported write.
282        //
283        // Under [`Trapezoid::VerticalPlusHorizontalCarryOver`] one more
284        // installment crosses: the newest slot is also the lag-1 tap of this
285        // call's *first* position, and it carries the empty decay product, so it
286        // is the same term with `νⁱⁿᵗ₀` on top of that slot's own weight. Every
287        // other pattern's interior tap is closed exactly there.
288        let mimo_x_hmp = self.mimo_x_hmp.as_ref().map(|p| p.val());
289        let nu_head_buh = nu_bsh.clone().narrow(1, 0, lag);
290        let nu_head_buh = match nu_interior_bsh
291            .filter(|_| self.trapezoid.interior_tap_crosses_tokens())
292        {
293            // `lag ≥ 2` here: a two-tap pattern folds at `u = 1`, so an interior
294            // tap and a one-slot FIFO never coexist.
295            Some(nu_interior_bsh) => Tensor::cat(
296                vec![
297                    nu_head_buh.clone().narrow(1, 0, lag - 1),
298                    nu_head_buh.narrow(1, lag - 1, 1) + nu_interior_bsh.narrow(1, 0, 1),
299                ],
300                1,
301            ),
302            None => nu_head_buh,
303        };
304        let v_prev_mimo_bumhp = helpers::build_v_with_mimo::<4, 5>(
305            cache
306                .v_state_buhp
307                .clone()
308                .expect("a β tap keeps its (B, x) cache slots"),
309            mimo_x_hmp.as_ref(),
310            2,
311        ); // [batch, lag, mimo_rank, nheads, per_head_dim]
312        let v_prev_mimo_bumhp = v_prev_mimo_bumhp * nu_head_buh.unsqueeze_dims::<5>(&[2, 4]);
313        // The seed contracts over the slots *and* the ranks — both are just
314        // outer products sharing one state — so fusing them makes it the one
315        // `mimo_outer_sum` the per-token write already uses.
316        let k_prev_bumhr = cache
317            .k_state_bumhr
318            .clone()
319            .expect("a β tap keeps its (B, x) cache slots");
320        let boundary_seed_bhpr = helpers::mimo_outer_sum(
321            v_prev_mimo_bumhp.reshape([batch, lag * mimo_rank, nheads, per_head_dim]),
322            k_prev_bumhr.reshape([batch, lag * mimo_rank, nheads, state_rank]),
323            self.use_siso_decode_kernels(),
324        );
325        let initial_state_bhpr = cache.ssm_bhpr.clone() + boundary_seed_bhpr;
326        san(&initial_state_bhpr);
327
328        // ── Step 6: Pad sequence to multiple of chunk_len ─────────────────────
329        let chunk_len = ssd_path.chunk_len_or_optimal(self);
330        let sequence_padded = sequence.next_multiple_of(chunk_len);
331        let pad = sequence_padded - sequence;
332
333        // V passed to SSD is raw x with MIMO_V applied (not γ-scaled).
334        let v_bshmp = helpers::build_v_with_mimo::<4, 5>(x_bshp.clone(), mimo_x_hmp.as_ref(), 2);
335        // v_bshmp has axis order [b, s, m, h, p] (insert_dim=2 onto [b,s,h,p]).
336
337        // ── The lag-`u` correction band ───────────────────────────────────────
338        // At lag 1 the key scale is wrong only on the diagonal and `ssd/diag.rs`
339        // fixes it inside the kernel. At lag `u` the exception is `u` wide — and
340        // is exactly the token at the positions the readout keeps, so it is one
341        // contraction here rather than a wider mask. See
342        // [`crate::mamba3::single_ssd::token_band`].
343        //
344        // The band is the **far** installment alone, even for a two-tap pattern:
345        // at the surviving read `t = τu + u−1` a sample's interior installment
346        // (lag 1) has already landed for every `j < u−1`, and at `j = u−1` the
347        // kernel replaces the whole weight with `γ` anyway. Only the lag-`u` one
348        // is still unpaid there.
349        let band_correction_btmhp = (lag > 1)
350            .then(|| {
351                crate::mamba3::single_ssd::token_band::token_band_correction(
352                    v_bshmp.clone(),
353                    b_bsmhr.clone(),
354                    c_btmhr.clone(),
355                    far_shifted_bsh,
356                    da_bsh.clone(),
357                    micro_steps,
358                )
359            })
360            .flatten();
361
362        // `C` and `γ` ride the chunk's read axis, so they pad by whole tokens.
363        // `chunk_len` is a multiple of `u`, hence so is `pad`.
364        let gamma_bth = helpers::read_rows::<3, 4>(gamma_bsh, 1, u);
365        let tokens_padded = sequence_padded / u;
366        let pad_tokens = tokens_padded - tokens;
367
368        #[rustfmt::skip]
369        let (v_bShmp, da_bSh, gamma_bTh, scale_bSh, b_bSmhr, c_bTmhr) = if pad == 0 {
370            (v_bshmp, da_bsh, gamma_bth, scale_bsh, b_bsmhr, c_btmhr)
371        } else {
372            let pad_bShmp = Tensor::zeros([batch, pad, mimo_rank, nheads, per_head_dim], &device);
373            let pad_bSh = Tensor::zeros([batch, pad, nheads], &device);
374            let pad_bTh = Tensor::zeros([batch, pad_tokens, nheads], &device);
375            let pad_bSmhr = Tensor::zeros([batch, pad, mimo_rank, nheads, state_rank], &device);
376            let pad_bTmhr =
377                Tensor::zeros([batch, pad_tokens, mimo_rank, nheads, state_rank], &device);
378            (
379                Tensor::cat(vec![v_bshmp, pad_bShmp], 1),
380                Tensor::cat(vec![da_bsh, pad_bSh.clone()], 1),
381                Tensor::cat(vec![gamma_bth, pad_bTh], 1),
382                Tensor::cat(vec![scale_bsh, pad_bSh], 1),
383                Tensor::cat(vec![b_bsmhr, pad_bSmhr], 1),
384                Tensor::cat(vec![c_btmhr, pad_bTmhr], 1),
385            )
386        };
387
388        // ── Reshape into chunks ───────────────────────────────────────────────
389        let nchunks = sequence_padded / chunk_len;
390        let chunk_tokens = Mamba3SsdPath::chunk_tokens(chunk_len, u);
391        let v_bnlmhp =
392            v_bShmp.reshape([batch, nchunks, chunk_len, mimo_rank, nheads, per_head_dim]);
393        let da_bnlh = da_bSh.reshape([batch, nchunks, chunk_len, nheads]);
394        let gamma_bnth = gamma_bTh.reshape([batch, nchunks, chunk_tokens, nheads]);
395        let scale_bnlh = scale_bSh.reshape([batch, nchunks, chunk_len, nheads]);
396        let b_bnlmhr = b_bSmhr.reshape([batch, nchunks, chunk_len, mimo_rank, nheads, state_rank]);
397        let c_bntmhr =
398            c_bTmhr.reshape([batch, nchunks, chunk_tokens, mimo_rank, nheads, state_rank]);
399
400        // ── Step 7: Run single-pass form SSD ───────────────────────────────────────
401        let ssd_input = Mamba3SingleSsdInput {
402            v_bnlmhp,
403            b_bnlmhr,
404            c_bntmhr,
405            da_bnlh,
406            gamma_bnth,
407            scale_bnlh,
408            initial_state_bhpr,
409            init_state_hpr: self.init_state_hpr.as_ref().map(|s| s.val()),
410            read_stride: u,
411            siso_specialization: self.siso_specialization,
412        };
413        let (y_bntmhp, final_state_bhpr) = ssd_input.run(ssd_path);
414
415        san(&y_bntmhp);
416        san(&final_state_bhpr);
417        cache.ssm_bhpr = final_state_bhpr;
418
419        // ── Step 8: Unpad ─────────────────────────────────────────────────────
420        // The SSD returns token resolution: the readout happens after all `u`
421        // writes, and the kernel only ever computed the row it happens at.
422        let y_bTmhp = y_bntmhp.reshape([batch, tokens_padded, mimo_rank, nheads, per_head_dim]);
423        let y_bsmhp = if pad == 0 {
424            y_bTmhp
425        } else {
426            y_bTmhp.narrow(1, 0, tokens)
427        };
428
429        let y_bsmhp = match band_correction_btmhp {
430            Some(correction_btmhp) => y_bsmhp - correction_btmhp,
431            None => y_bsmhp,
432        };
433        let x_bthp = crate::mamba3::product::last_micro4(x_bshp.clone(), micro_steps);
434        let sequence = tokens;
435
436        // ── Step 9: D skip + gate + MIMO_O down-projection ────────────────────
437        // D skip uses raw x ⊙ mimo_x (not γ-scaled, matching forward), at the
438        // micro-step the readout is contemporaneous with.
439        let v_raw_bsmhp = helpers::build_v_with_mimo::<4, 5>(x_bthp, mimo_x_hmp.as_ref(), 2);
440        let d_111h1 = self.d_h.val().unsqueeze_dims::<5>(&[0, 1, 2, 4]);
441        let y_bsmhp = y_bsmhp + d_111h1 * v_raw_bsmhp;
442
443        let y_bsi = if mimo_rank > 1 {
444            let mimo_z_hmp = self.mimo_z_hmp.as_ref().map(|p| p.val()).unwrap();
445            let mimo_o_hmp = self.mimo_o_hmp.as_ref().map(|p| p.val()).unwrap();
446
447            let z_bshp = z_bsi
448                .clone()
449                .reshape([batch, sequence, nheads, per_head_dim]);
450            let z_bsmhp = {
451                let z_bsmhp = z_bshp.unsqueeze_dim::<5>(2).expand([
452                    batch,
453                    sequence,
454                    mimo_rank,
455                    nheads,
456                    per_head_dim,
457                ]);
458                let mimo_z_bsmhp = mimo_z_hmp
459                    .swap_dims(0, 1)
460                    .unsqueeze_dims::<5>(&[0, 1])
461                    .expand([batch, sequence, mimo_rank, nheads, per_head_dim]);
462                z_bsmhp * mimo_z_bsmhp
463            };
464
465            let y_combined_bsmhp = match &self.out_norm {
466                Some(norm) => norm.forward(y_bsmhp, z_bsmhp),
467                None => y_bsmhp * Silu::new().forward(z_bsmhp),
468            };
469
470            let mimo_o_bsmhp = mimo_o_hmp
471                .swap_dims(0, 1)
472                .unsqueeze_dims::<5>(&[0, 1])
473                .expand([batch, sequence, mimo_rank, nheads, per_head_dim]);
474            let y_bshp: Tensor<4> = (y_combined_bsmhp * mimo_o_bsmhp).sum_dim(2).squeeze_dim(2);
475            y_bshp.reshape([batch, sequence, d_inner])
476        } else {
477            let y_bshp: Tensor<4> = y_bsmhp.squeeze_dim(2);
478            let z_bshp = z_bsi.reshape([batch, sequence, nheads, per_head_dim]);
479            let y_combined_bshp = match &self.out_norm {
480                Some(norm) => norm.forward(y_bshp, z_bshp),
481                None => y_bshp * Silu::new().forward(z_bshp),
482            };
483            y_combined_bshp.reshape([batch, sequence, d_inner])
484        };
485        san(&y_bsi);
486
487        // ── Out-projection ────────────────────────────────────────────────────
488        let out_bsm = self.out_proj.forward(y_bsi);
489        san(&out_bsm);
490
491        // ── Update remaining cache fields ─────────────────────────────────────
492        cache.k_state_bumhr = b_last_bumhr;
493        cache.v_state_buhp = x_last_buhp;
494        // The new cumulative rotation (Complex2D: angle wrapped to [−π, π];
495        // Quaternion4D: the cumulative quaternion), from [`rotate_bc_forward`] —
496        // matches the double-ssd cache convention so the two inter-convert.
497        cache.rotation = new_rotation;
498
499        (out_bsm, cache)
500    }
501}

503// ---------------------------------------------------------------------------
504// Mamba3::step  (recurrent SSM — token-by-token decoding)
505// ---------------------------------------------------------------------------
506
507mod step {
508    use super::*;
509
510    impl Mamba3 {
511        /// Process a **single token** using the pure recurrent form.
512        ///
513        /// For SISO (mimo_rank=1):
514        /// ```text
515        ///   hₜ = αₜ hₜ₋₁ + βₜ Bₜ₋₁ ⊗ xₜ₋₁ + γₜ Bₜ ⊗ xₜ
516        ///   yₜ = Cₜᵀ hₜ + D xₜ
517        /// ```
518        ///
519        /// For MIMO (mimo_rank>1):
520        /// ```text
521        ///   hₜ = αₜ hₜ₋₁ + Σₘ βₜ Bₜ₋₁[m] ⊗ (xₜ₋₁ ⊙ mimo_x_hmp[m]) + Σₘ γₜ Bₜ[m] ⊗ (xₜ ⊙ mimo_x_hmp[m])
522        ///   yₜ[r] = Cₜ[r]ᵀ hₜ + D xₜ ⊙ mimo_x_hmp[r]
523        ///   outₜ = Σₘ mimo_o_hmp[m] ⊙ silu(zₜ ⊙ mimo_z_hmp[m]) ⊙ yₜ[m]
524        /// ```
525        ///
526        /// # Shapes
527        /// - `input_bd` : `[batch, d_model]`
528        /// - output     : `[batch, d_model]`
529        #[allow(non_snake_case)]
530        pub fn step_single_ssd(
531            &self,
532            input_bd: Tensor<2>,
533            cache: Option<Mamba3SingleSsdCache>,
534        ) -> (Tensor<2>, Mamba3SingleSsdCache) {
535            // Token-by-token decoding always uses the recurrent (double-ssd)
536            // form. A single-ssd cache holds the trapezoid state at a sequence
537            // boundary, where the single- and double-ssd accumulators coincide
538            // (see the `From` impls in `crate::mamba3::cache`), so converting in
539            // and back out is lossless. The single recurrence step is itself a
540            // boundary-to-boundary transition, so the round-trip stays exact.
541            let cache = cache.map(Mamba3DoubleSsdCache::from);
542            let (out_bd, cache) = self.step_double_ssd(input_bd, cache);
543            (out_bd, cache.into())
544        }
545    }
546}
