1//! # Mamba-3 — Single-Pass SSD Forward
2//!
3//! This module provides the `forward_single_ssd` method on [`Mamba3`](crate::mamba3::mamba3::Mamba3):
4//! The burn-mamba implementation of the **official Mamba-3 algorithm**
5//! as shipped in Triton (SISO) and Tilelang (MIMO):
6//!
7//! ```text
8//!   scaleₜ = γₜ + νₜ₊ₗₐ₉ (+ νⁱⁿᵗₜ₊₁)
9//!
10//!   forward_single_ssd:    h' = SSD(V_raw, K_scaled = scaleₜ B) with:
11//!                               * strict lower-triangular intra-chunk mask
12//!                               * additive γ-weighted same-step correction
13//!                               * boundary β seed Σⱼ νⱼ Kⱼ ⊗ xⱼ over the
14//!                                 cache's `lag` tap slots
15//!                               * at lag > 1, the rest of the correction band
16//!                                 ([`crate::mamba3::single_ssd::token_band`])
17//! ```
18//!
19//! `lag` is [`Trapezoid::tap_lag`](crate::mamba3::trapezoid::Trapezoid::tap_lag):
20//! `1` for the default tap pattern, `u` for
21//! [`Trapezoid::Vertical`](crate::mamba3::trapezoid::Trapezoid::Vertical). A
22//! two-tap pattern adds the parenthesised term — a second shift of the same
23//! shape, since §9's collapse leaves one scalar per sample however many taps
24//! there are. That is the whole of its cost here: one pass either way.
25//!
26//! References:
27//! - [`mamba3_siso_fwd.py`](https://github.com/state-spaces/mamba/mamba_ssm/ops/triton/mamba3/mamba3_siso_fwd.py),
28//! - [`mamba3_mimo_fwd.py`](https://github.com/state-spaces/mamba/mamba_ssm/ops/tilelang/mamba3/mamba3_mimo_fwd.py).
29//!
30//! See also: [`crate::mamba3::mamba3`] and [`crate::mamba3::double_ssd::double_ssd`].
31
32use crate::mamba3::double_ssd::prelude::Mamba3DoubleSsdCache;
33use crate::mamba3::helpers;
34use crate::mamba3::prelude::*;
35use crate::mamba3::rotation::rotate_bc_forward;
36use crate::mamba3::single_ssd::prelude::*;
37use burn_stack::modules::Silu;
38use burn_stack::modules::sanity as san;
39use burn::prelude::*;

41impl Mamba3 {
42    /// Process a full input sequence using the **single-ssd form (single-pass)**
43    /// trapezoidal algorithm.
44    ///
45    /// Functionally equivalent to [`Self::forward`] but uses approximately half
46    /// the SSD memory during training. Cache is a separate type
47    /// ([`Mamba3SingleSsdCache`]) because the stored hidden state has different
48    /// semantics than the original-form cache used by [`Self::forward`].
49    ///
50    /// "Equivalent" is over **everything a caller can observe** — the returned
51    /// output and every field of the returned cache — which is the whole of what
52    /// this method is. It is not a claim about the intermediate `y` the SSD core
53    /// produces at each *folded* position: under
54    /// [`Trapezoid::Vertical`] the
55    /// correction band is only applied at each token's last micro-step, the one
56    /// the readout happens at, so the `u−1` per-token partial sums this pathway
57    /// discards do not match the double-SSD ones. Nothing reads them, and the
58    /// *state* is exact at every position in both pathways — see
59    /// [`crate::mamba3::single_ssd::token_band`], which is also why that band
60    /// never has to enter the kernel.
61    ///
62    /// Except under [`Trapezoid::None`], where there is no second pass to fuse:
63    /// the composite key scale is `γ` and the same-step correction is the whole
64    /// diagonal, so this *is* the double-SSD form — `h' ≡ h` at every position,
65    /// not merely at boundaries — and the call delegates rather than run a
66    /// strict-mask kernel plus a correction that reassembles what it masked.
67    ///
68    /// # Shapes
69    /// - `input_bsm`: `[batch, sequence, d_model]`
70    /// - output: `[batch, sequence, d_model]`
71    #[allow(non_snake_case)]
72    pub fn forward_single_ssd(
73        &self,
74        input_bsm: Tensor<3>,
75        cache: Option<Mamba3SingleSsdCache>,
76        ssd_path: &Mamba3SsdPath,
77    ) -> (Tensor<3>, Mamba3SingleSsdCache) {
78        if !self.trapezoid.has_beta_tap() {
79            let (out_bsm, cache) =
80                self.forward_double_ssd(input_bsm, cache.map(Into::into), ssd_path);
81            return (out_bsm, cache.into());
82        }
83        let [batch, tokens, _d_model] = input_bsm.dims();
84        let d_inner = self.d_inner();
85        let nheads = self.nheads();
86        let ngroups = self.ngroups;
87        let per_head_dim = self.per_head_dim();
88        let state_rank = self.state_rank;
89        let mimo_rank = self.mimo_rank;
90        let micro_steps = self.micro_steps;
91        let device = input_bsm.device();
92
93        // MambaProduct: from the split below down to the readout, one sequence
94        // position *is* one micro-step — the `s` in every shape suffix counts
95        // micro-steps, and `tokens` is the only name still at token resolution.
96        // See [`crate::mamba3::product`].
97        let sequence = tokens * micro_steps;
98
99        assert!(tokens > 0, "sequence length must be at least 1");
100        assert_eq!(nheads % ngroups, 0);
101        san(&input_bsm);
102
103        // ── Initialise cache if not provided ──────────────────────────────────
104        let mut cache = cache.unwrap_or_else(|| {
105            let ssm_bhpr = Tensor::zeros([batch, nheads, per_head_dim, state_rank], &device);
106            // Reached only with a β tap (`Trapezoid::None` delegated above).
107            let (k_state_bumhr, v_state_buhp) = self.zero_tap_slots(batch, &device);
108            let rotation = self.zero_rotation_state(batch, &device);
109            Mamba3SingleSsdCache {
110                ssm_bhpr,
111                k_state_bumhr,
112                v_state_buhp,
113                rotation,
114            }
115        });
116
117        // ── Step 1: In-projection ─────────────────────────────────────────────
118        let proj_bsd = self.in_proj.forward(input_bsm);
119        let bc_size = ngroups * state_rank * mimo_rank;
120
121        // `u` = micro_steps widens every per-micro-step segment, and `unfold`
122        // reinterprets each of them as `u` consecutive sequence positions. `z`
123        // (per-token gate) and `C` (per-token read) do not widen — `C` is
124        // instead broadcast across the group so its *last* copy carries the
125        // right cumulative rotation. See [`crate::mamba3::product`].
126        // The optional segments come off the tail first, in layout order: the
127        // rotation (`Real1D` projects none), then `μ` (only a two-tap pattern
128        // projects it), then `λ` — which is always present here,
129        // `Trapezoid::None` having delegated above — since a zero-width segment
130        // would silently vanish from the split below.
131        let u = micro_steps;
132        let (proj_bsd, rot_btA) =
133            helpers::split_trailing(proj_bsd, self.rotation_channels_total(), 2);
134        let (proj_bsd, mu_raw_btH) = helpers::split_trailing(proj_bsd, self.mu_channels_total(), 2);
135        let (proj_bsd, lambda_raw_btH) =
136            helpers::split_trailing(proj_bsd, self.lambda_channels_total(), 2);
137        let lambda_raw_btH = lambda_raw_btH.expect("a β tap projects λ");
138        #[rustfmt::skip]
139        let [
140                z_bsi, x_btI,
141                b_raw_btMGRU, c_raw_btMGR,
142                dd_dt_btH, dd_A_raw_btH,
143        ] = burn_stack::modules::split_into(
144            proj_bsd,
145            [
146                d_inner, u * d_inner,
147                u * bc_size, bc_size,
148                u * nheads, u * nheads,
149            ],
150            2,
151        );
152
153        use crate::mamba3::product::{repeat_micro_bs, unfold_micro_bs};
154        let x_bsi = unfold_micro_bs(x_btI, u);
155        let b_raw_bsMGR = unfold_micro_bs(b_raw_btMGRU, u);
156        let c_raw_bsMGR = repeat_micro_bs(c_raw_btMGR, u);
157        let dd_dt_bsh = unfold_micro_bs(dd_dt_btH, u);
158        let dd_A_raw_bsh = unfold_micro_bs(dd_A_raw_btH, u);
159        let lambda_raw_bsh = unfold_micro_bs(lambda_raw_btH, u);
160        let mu_raw_bsh = mu_raw_btH.map(|t| unfold_micro_bs(t, u));
161        let rot_bsa = rot_btA.map(|t| unfold_micro_bs(t, u));
162
163        san(&z_bsi);
164        san(&x_bsi);
165        san(&dd_dt_bsh);
166
167        // ── Step 2: Discretisation + trapezoidal coefficients ─────────────────
168        let helpers::TrapezoidCoeffs {
169            dt: dt_bsh,
170            da: da_bsh,
171            alpha: _alpha_bsh,
172            nu: nu_bsh,
173            nu_interior: nu_interior_bsh,
174            gamma: gamma_bsh,
175        } = helpers::trapezoidal_coefficients(
176            dd_dt_bsh,
177            dd_A_raw_bsh,
178            Some(lambda_raw_bsh),
179            mu_raw_bsh,
180            self.dt_bias_h.val(),
181            self.trapezoid_spec(),
182        );
183        let nu_bsh = nu_bsh.expect("a β tap has a mass");
184        san(&dt_bsh);
185        san(&da_bsh);
186        san(&gamma_bsh);
187
188        // ── Compute scaleₜ = γₜ + νₜ₊ₗₐ₉ (+ νⁱⁿᵗₜ₊₁) ─────────────────────────
189        //
190        // Each shifted term is zero for the last positions its own lag reaches
191        // past (the taps that pay them belong to the *next* call, out of the tap
192        // slots) — which is also what makes `h'` coincide with the double-SSD
193        // state at a cache boundary, hence the field-identity `From` impls.
194        //
195        // `t+lag` is a later *folded* position: lag 1 is
196        // [`Trapezoid::HorizontalCarryOver`], lag `u` is [`Trapezoid::Vertical`].
197        // This is the `Δ̃` collapse (`info/trapezoid-as-integration.md` §5), and
198        // §9's collapse theorem is why it survives the wider lag — and a second
199        // tap — unchanged: still one scalar per sample, hence one pass. Only the
200        // same-step correction widens from the diagonal to a `lag`-wide band
201        // (see [`crate::mamba3::single_ssd::token_band`]).
202        let lag = self.tap_lag();
203        // νₜ₊ₗ, the mass a later position pays this one — zero past the end of
204        // the call, where the tap belongs to the next one.
205        let pay_forward = |nu_bsh: Tensor<3>, l: usize| {
206            let zero_bLh = Tensor::zeros([batch, l, nheads], &device);
207            if sequence == l {
208                zero_bLh
209            } else {
210                Tensor::cat(vec![nu_bsh.narrow(1, l, sequence - l), zero_bLh], 1)
211            }
212        };
213        let far_shifted_bsh = pay_forward(nu_bsh.clone(), lag);
214        // A two-tap pattern collapses to one scalar all the same (§9), just with
215        // a second shift in it: `scaleₜ = γₜ + νⁱⁿᵗₜ₊₁ + νᶠᵃʳₜ₊ₗₐ₉`.
216        let interior_shifted_bsh = nu_interior_bsh
217            .clone()
218            .map(|nu_bsh| pay_forward(nu_bsh, 1));
219        let scale_bsh = gamma_bsh.clone() + far_shifted_bsh.clone();
220        let scale_bsh = match &interior_shifted_bsh {
221            Some(interior_bsh) => scale_bsh + interior_bsh.clone(),
222            None => scale_bsh,
223        };
224        san(&scale_bsh);
225
226        // ── Step 3: Reshape x ─────────────────────────────────────────────────
227        let x_bshp = x_bsi.reshape([batch, sequence, nheads, per_head_dim]);
228
229        // ── Step 4: QK-Norm on B and C ────────────────────────────────────────
230        let b_bsmhr = helpers::qk_norm_expand_bias::<5, 6>(
231            b_raw_bsMGR.reshape([batch, sequence, mimo_rank, ngroups, state_rank]),
232            &self.b_norm,
233            self.b_bias_hmr.val(),
234            3,
235            nheads,
236        );
237        let c_bsmhr = helpers::qk_norm_expand_bias::<5, 6>(
238            c_raw_bsMGR.reshape([batch, sequence, mimo_rank, ngroups, state_rank]),
239            &self.c_norm,
240            self.c_bias_hmr.val(),
241            3,
242            nheads,
243        );
244
245        // ── Step 5: Data-dependent transition rotation of B and C ─────────────
246        // Complex2D: abelian RoPE (cumulative angle). Quaternion4D: cumulative
247        // unit quaternion. Shared with the double-ssd pathway via
248        // [`rotate_bc_forward`]; the single-pass SSD core below is
249        // rotation-agnostic — it only ever consumes the rotated B̄/C̄ (the RoPE
250        // factoring `C̄ₜᵀB̄ᵢ = Cₜᵀ·Rel(t,i)·Bᵢ` holds for either algebra).
251        let (b_bsmhr, c_bsmhr, new_rotation) = rotate_bc_forward(
252            rot_bsa,
253            dt_bsh.clone(),
254            cache.rotation.clone(),
255            b_bsmhr,
256            c_bsmhr,
257            self.rotation_spec(),
258        );
259        san(&b_bsmhr);
260        san(&c_bsmhr);
261
262        // ── Save the last `lag` positions' B and x (raw, no MIMO_V) ───────────
263        // The positions whose second installment the next call pays; at
264        // `lag = u` that window is precisely the last token.
265        let (b_last_bumhr, x_last_buhp) =
266            self.save_tap_slots(&b_bsmhr, &x_bshp, &da_bsh, lag);
267
268        // ── Boundary β seed for initial state ─────────────────────────────────
269        // Add Σⱼ νⱼ · Σₘ K_prev[j, m] ⊗ (x_prev[j] ⊙ mimo_xₘ) to the carried
270        // single-ssd SSM state: cache slot `j` (oldest first) is the position
271        // whose tap is paid by this call's position `j`, so it takes `νⱼ` from
272        // the current call's first `lag` positions. `x_prev` already carries the
273        // decay from its own position to the boundary, and `K_prev` its own
274        // rotation, so the pair *is* the transported write.
275        //
276        // Under [`Trapezoid::VerticalPlusHorizontalCarryOver`] one more
277        // installment crosses: the newest slot is also the lag-1 tap of this
278        // call's *first* position, and it carries the empty decay product, so it
279        // is the same term with `νⁱⁿᵗ₀` on top of that slot's own weight. Every
280        // other pattern's interior tap is closed exactly there.
281        let mimo_x_hmp = self.mimo_x_hmp.as_ref().map(|p| p.val());
282        let nu_head_buh = nu_bsh.clone().narrow(1, 0, lag);
283        let nu_head_buh = match nu_interior_bsh
284            .filter(|_| self.trapezoid.interior_tap_crosses_tokens())
285        {
286            // `lag ≥ 2` here: a two-tap pattern folds at `u = 1`, so an interior
287            // tap and a one-slot FIFO never coexist.
288            Some(nu_interior_bsh) => Tensor::cat(
289                vec![
290                    nu_head_buh.clone().narrow(1, 0, lag - 1),
291                    nu_head_buh.narrow(1, lag - 1, 1) + nu_interior_bsh.narrow(1, 0, 1),
292                ],
293                1,
294            ),
295            None => nu_head_buh,
296        };
297        let v_prev_mimo_bumhp = helpers::build_v_with_mimo::<4, 5>(
298            cache
299                .v_state_buhp
300                .clone()
301                .expect("a β tap keeps its (B, x) cache slots"),
302            mimo_x_hmp.as_ref(),
303            2,
304        ); // [batch, lag, mimo_rank, nheads, per_head_dim]
305        let v_prev_mimo_bumhp = v_prev_mimo_bumhp * nu_head_buh.unsqueeze_dims::<5>(&[2, 4]);
306        // The seed contracts over the slots *and* the ranks — both are just
307        // outer products sharing one state — so fusing them makes it the one
308        // `mimo_outer_sum` the per-token write already uses.
309        let k_prev_bumhr = cache
310            .k_state_bumhr
311            .clone()
312            .expect("a β tap keeps its (B, x) cache slots");
313        let boundary_seed_bhpr = helpers::mimo_outer_sum(
314            v_prev_mimo_bumhp.reshape([batch, lag * mimo_rank, nheads, per_head_dim]),
315            k_prev_bumhr.reshape([batch, lag * mimo_rank, nheads, state_rank]),
316            self.use_siso_decode_kernels(),
317        );
318        let initial_state_bhpr = cache.ssm_bhpr.clone() + boundary_seed_bhpr;
319        san(&initial_state_bhpr);
320
321        // ── Step 6: Pad sequence to multiple of chunk_len ─────────────────────
322        let chunk_len = ssd_path.chunk_len_or_optimal(self);
323        let sequence_padded = sequence.next_multiple_of(chunk_len);
324        let pad = sequence_padded - sequence;
325
326        // V passed to SSD is raw x with MIMO_V applied (not γ-scaled).
327        let v_bshmp = helpers::build_v_with_mimo::<4, 5>(x_bshp.clone(), mimo_x_hmp.as_ref(), 2);
328        // v_bshmp has axis order [b, s, m, h, p] (insert_dim=2 onto [b,s,h,p]).
329
330        // ── The lag-`u` correction band ───────────────────────────────────────
331        // At lag 1 the key scale is wrong only on the diagonal and `ssd/diag.rs`
332        // fixes it inside the kernel. At lag `u` the exception is `u` wide — and
333        // is exactly the token at the positions the readout keeps, so it is one
334        // contraction here rather than a wider mask. See
335        // [`crate::mamba3::single_ssd::token_band`].
336        //
337        // The band is the **far** installment alone, even for a two-tap pattern:
338        // at the surviving read `t = τu + u−1` a sample's interior installment
339        // (lag 1) has already landed for every `j < u−1`, and at `j = u−1` the
340        // kernel replaces the whole weight with `γ` anyway. Only the lag-`u` one
341        // is still unpaid there.
342        let band_correction_btmhp = (lag > 1)
343            .then(|| {
344                crate::mamba3::single_ssd::token_band::token_band_correction(
345                    v_bshmp.clone(),
346                    b_bsmhr.clone(),
347                    c_bsmhr.clone(),
348                    far_shifted_bsh,
349                    da_bsh.clone(),
350                    micro_steps,
351                )
352            })
353            .flatten();
354
355        #[rustfmt::skip]
356        let (v_bShmp, da_bSh, gamma_bSh, scale_bSh, b_bSmhr, c_bSmhr) = if pad == 0 {
357            (v_bshmp, da_bsh, gamma_bsh, scale_bsh, b_bsmhr, c_bsmhr)
358        } else {
359            let pad_bShmp = Tensor::zeros([batch, pad, mimo_rank, nheads, per_head_dim], &device);
360            let pad_bSh = Tensor::zeros([batch, pad, nheads], &device);
361            let pad_bSmhr = Tensor::zeros([batch, pad, mimo_rank, nheads, state_rank], &device);
362            (
363                Tensor::cat(vec![v_bshmp, pad_bShmp], 1),
364                Tensor::cat(vec![da_bsh, pad_bSh.clone()], 1),
365                Tensor::cat(vec![gamma_bsh, pad_bSh.clone()], 1),
366                Tensor::cat(vec![scale_bsh, pad_bSh], 1),
367                Tensor::cat(vec![b_bsmhr, pad_bSmhr.clone()], 1),
368                Tensor::cat(vec![c_bsmhr, pad_bSmhr], 1),
369            )
370        };
371
372        // ── Reshape into chunks ───────────────────────────────────────────────
373        let nchunks = sequence_padded / chunk_len;
374        let v_bnlmhp =
375            v_bShmp.reshape([batch, nchunks, chunk_len, mimo_rank, nheads, per_head_dim]);
376        let da_bnlh = da_bSh.reshape([batch, nchunks, chunk_len, nheads]);
377        let gamma_bnlh = gamma_bSh.reshape([batch, nchunks, chunk_len, nheads]);
378        let scale_bnlh = scale_bSh.reshape([batch, nchunks, chunk_len, nheads]);
379        let b_bnlmhr = b_bSmhr.reshape([batch, nchunks, chunk_len, mimo_rank, nheads, state_rank]);
380        let c_bnlmhr = c_bSmhr.reshape([batch, nchunks, chunk_len, mimo_rank, nheads, state_rank]);
381
382        // ── Step 7: Run single-pass form SSD ───────────────────────────────────────
383        let ssd_input = Mamba3SingleSsdInput {
384            v_bnlmhp,
385            b_bnlmhr,
386            c_bnlmhr,
387            da_bnlh,
388            gamma_bnlh,
389            scale_bnlh,
390            initial_state_bhpr,
391            init_state_hpr: self.init_state_hpr.as_ref().map(|s| s.val()),
392            siso_specialization: self.siso_specialization,
393        };
394        let (y_bnlmhp, final_state_bhpr) = ssd_input.run(ssd_path);
395
396        san(&y_bnlmhp);
397        san(&final_state_bhpr);
398        cache.ssm_bhpr = final_state_bhpr;
399
400        // ── Step 8: Unpad ─────────────────────────────────────────────────────
401        let y_bSmhp = y_bnlmhp.reshape([batch, sequence_padded, mimo_rank, nheads, per_head_dim]);
402        let y_bsmhp = if pad == 0 {
403            y_bSmhp
404        } else {
405            y_bSmhp.narrow(1, 0, sequence)
406        };
407
408        // ── Step 8b: back to token resolution (MambaProduct) ──────────────────
409        // The readout happens after all `u` writes, so only each token's last
410        // micro-step survives; the intervening positions computed a `y` from a
411        // repeated `C` and it is dropped here.
412        let y_bsmhp = crate::mamba3::product::last_micro5(y_bsmhp, micro_steps);
413        let y_bsmhp = match band_correction_btmhp {
414            Some(correction_btmhp) => y_bsmhp - correction_btmhp,
415            None => y_bsmhp,
416        };
417        let x_bthp = crate::mamba3::product::last_micro4(x_bshp.clone(), micro_steps);
418        let sequence = tokens;
419
420        // ── Step 9: D skip + gate + MIMO_O down-projection ────────────────────
421        // D skip uses raw x ⊙ mimo_x (not γ-scaled, matching forward), at the
422        // micro-step the readout is contemporaneous with.
423        let v_raw_bsmhp = helpers::build_v_with_mimo::<4, 5>(x_bthp, mimo_x_hmp.as_ref(), 2);
424        let d_111h1 = self.d_h.val().unsqueeze_dims::<5>(&[0, 1, 2, 4]);
425        let y_bsmhp = y_bsmhp + d_111h1 * v_raw_bsmhp;
426
427        let y_bsi = if mimo_rank > 1 {
428            let mimo_z_hmp = self.mimo_z_hmp.as_ref().map(|p| p.val()).unwrap();
429            let mimo_o_hmp = self.mimo_o_hmp.as_ref().map(|p| p.val()).unwrap();
430
431            let z_bshp = z_bsi
432                .clone()
433                .reshape([batch, sequence, nheads, per_head_dim]);
434            let z_bsmhp = {
435                let z_bsmhp = z_bshp.unsqueeze_dim::<5>(2).expand([
436                    batch,
437                    sequence,
438                    mimo_rank,
439                    nheads,
440                    per_head_dim,
441                ]);
442                let mimo_z_bsmhp = mimo_z_hmp
443                    .swap_dims(0, 1)
444                    .unsqueeze_dims::<5>(&[0, 1])
445                    .expand([batch, sequence, mimo_rank, nheads, per_head_dim]);
446                z_bsmhp * mimo_z_bsmhp
447            };
448
449            let y_combined_bsmhp = match &self.out_norm {
450                Some(norm) => norm.forward(y_bsmhp, z_bsmhp),
451                None => y_bsmhp * Silu::new().forward(z_bsmhp),
452            };
453
454            let mimo_o_bsmhp = mimo_o_hmp
455                .swap_dims(0, 1)
456                .unsqueeze_dims::<5>(&[0, 1])
457                .expand([batch, sequence, mimo_rank, nheads, per_head_dim]);
458            let y_bshp: Tensor<4> = (y_combined_bsmhp * mimo_o_bsmhp).sum_dim(2).squeeze_dim(2);
459            y_bshp.reshape([batch, sequence, d_inner])
460        } else {
461            let y_bshp: Tensor<4> = y_bsmhp.squeeze_dim(2);
462            let z_bshp = z_bsi.reshape([batch, sequence, nheads, per_head_dim]);
463            let y_combined_bshp = match &self.out_norm {
464                Some(norm) => norm.forward(y_bshp, z_bshp),
465                None => y_bshp * Silu::new().forward(z_bshp),
466            };
467            y_combined_bshp.reshape([batch, sequence, d_inner])
468        };
469        san(&y_bsi);
470
471        // ── Out-projection ────────────────────────────────────────────────────
472        let out_bsm = self.out_proj.forward(y_bsi);
473        san(&out_bsm);
474
475        // ── Update remaining cache fields ─────────────────────────────────────
476        cache.k_state_bumhr = b_last_bumhr;
477        cache.v_state_buhp = x_last_buhp;
478        // The new cumulative rotation (Complex2D: angle wrapped to [−π, π];
479        // Quaternion4D: the cumulative quaternion), from [`rotate_bc_forward`] —
480        // matches the double-ssd cache convention so the two inter-convert.
481        cache.rotation = new_rotation;
482
483        (out_bsm, cache)
484    }
485}

487// ---------------------------------------------------------------------------
488// Mamba3::step  (recurrent SSM — token-by-token decoding)
489// ---------------------------------------------------------------------------
490
491mod step {
492    use super::*;
493
494    impl Mamba3 {
495        /// Process a **single token** using the pure recurrent form.
496        ///
497        /// For SISO (mimo_rank=1):
498        /// ```text
499        ///   hₜ = αₜ hₜ₋₁ + βₜ Bₜ₋₁ ⊗ xₜ₋₁ + γₜ Bₜ ⊗ xₜ
500        ///   yₜ = Cₜᵀ hₜ + D xₜ
501        /// ```
502        ///
503        /// For MIMO (mimo_rank>1):
504        /// ```text
505        ///   hₜ = αₜ hₜ₋₁ + Σₘ βₜ Bₜ₋₁[m] ⊗ (xₜ₋₁ ⊙ mimo_x_hmp[m]) + Σₘ γₜ Bₜ[m] ⊗ (xₜ ⊙ mimo_x_hmp[m])
506        ///   yₜ[r] = Cₜ[r]ᵀ hₜ + D xₜ ⊙ mimo_x_hmp[r]
507        ///   outₜ = Σₘ mimo_o_hmp[m] ⊙ silu(zₜ ⊙ mimo_z_hmp[m]) ⊙ yₜ[m]
508        /// ```
509        ///
510        /// # Shapes
511        /// - `input_bd` : `[batch, d_model]`
512        /// - output     : `[batch, d_model]`
513        #[allow(non_snake_case)]
514        pub fn step_single_ssd(
515            &self,
516            input_bd: Tensor<2>,
517            cache: Option<Mamba3SingleSsdCache>,
518        ) -> (Tensor<2>, Mamba3SingleSsdCache) {
519            // Token-by-token decoding always uses the recurrent (double-ssd)
520            // form. A single-ssd cache holds the trapezoid state at a sequence
521            // boundary, where the single- and double-ssd accumulators coincide
522            // (see the `From` impls in `crate::mamba3::cache`), so converting in
523            // and back out is lossless. The single recurrence step is itself a
524            // boundary-to-boundary transition, so the round-trip stays exact.
525            let cache = cache.map(Mamba3DoubleSsdCache::from);
526            let (out_bd, cache) = self.step_double_ssd(input_bd, cache);
527            (out_bd, cache.into())
528        }
529    }
530}
