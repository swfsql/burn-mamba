1//! # Mamba-3 — Single-Pass SSD Forward
2//!
3//! `forward_single_ssd` and `step_single_ssd` on
4//! [`Mamba3`](crate::mamba3::mamba3::Mamba3): the **official Mamba-3
5//! algorithm**, as in its Triton (SISO) and Tilelang (MIMO) kernels:
6//!
7//! ```text
8//!   scaleₜ = γₜ + νₜ₊ₗₐ₉ (+ νⁱⁿᵗₜ₊₁)
9//!
10//!   forward_single_ssd:    h' = SSD(V_raw, K_scaled = scaleₜ B) with:
11//!                               * strict lower-triangular intra-chunk mask
12//!                               * additive γ-weighted same-step correction
13//!                               * boundary β seed Σⱼ νⱼ Kⱼ ⊗ xⱼ over the
14//!                                 cache's `lag` tap slots
15//!                               * at lag > 1, the rest of the correction band
16//!                                 ([`crate::mamba3::single_ssd::token_band`])
17//! ```
18//!
19//! `lag` is [`Trapezoid::tap_lag`](crate::mamba3::trapezoid::Trapezoid::tap_lag):
20//! `1` for the default tap pattern, `u` for
21//! [`Trapezoid::Vertical`](crate::mamba3::trapezoid::Trapezoid::Vertical). A
22//! two-tap pattern adds the term in parentheses: a second shift of the same
23//! shape, because the collapse of §9 leaves one scalar per sample for any
24//! number of taps. That is all it costs here: one pass either way.
25//!
26//! References:
27//! - [`mamba3_siso_fwd.py`](https://github.com/state-spaces/mamba/mamba_ssm/ops/triton/mamba3/mamba3_siso_fwd.py),
28//! - [`mamba3_mimo_fwd.py`](https://github.com/state-spaces/mamba/mamba_ssm/ops/tilelang/mamba3/mamba3_mimo_fwd.py).
29//!
30//! See also: [`crate::mamba3::mamba3`] and [`crate::mamba3::double_ssd::double_ssd`].
31

42impl Mamba3 {
43    /// Process a full input sequence with the **single-ssd (single-pass)**
44    /// trapezoidal algorithm.
45    ///
46    /// Functionally equivalent to [`Self::forward_double_ssd`], with about half
47    /// the SSD memory in training. The cache is a separate type
48    /// ([`Mamba3SingleSsdCache`]), because its hidden state has different
49    /// semantics mid-sequence.
50    ///
51    /// "Equivalent" covers **everything that a caller can observe**: the
52    /// returned output and every field of the returned cache. It does not cover
53    /// the intermediate `y` of the SSD core at each *folded* position. Under a
54    /// lag-`u` pattern such as [`Trapezoid::Vertical`], the correction band
55    /// applies only at the last micro-step of each token (the readout). So the
56    /// `u−1` per-token partial sums that this pathway discards do not match the
57    /// double-SSD ones. Nothing reads them, and the *state* is exact at every
58    /// position in both pathways. See [`crate::mamba3::single_ssd::token_band`],
59    /// which also tells why that band never has to enter the kernel.
60    ///
61    /// Under [`Trapezoid::None`] there is no second pass to fuse. The composite
62    /// key scale is `γ`, and the same-step correction is the whole diagonal. So
63    /// this *is* the double-SSD form (`h' ≡ h` at every position, not only at
64    /// boundaries), and the call delegates to `forward_double_ssd`. It does not
65    /// run a strict-mask kernel plus a correction that puts back what the mask
66    /// removed.
67    ///
68    /// `pad_bt` marks a right-padded batch of tokens, as in [`Self::forward`].
69    /// The last tap installment of a real position, which a padded position
70    /// would pay, then stays in the tap slots, exactly as at a call boundary.
71    /// So `h'` holds the double-SSD state from the end of each slot on.
72    ///
73    /// # Shapes
74    /// - `input_bsm`: `[batch, sequence, d_model]`
75    /// - `pad_bt`: `[batch, sequence]`
76    /// - output: `[batch, sequence, d_model]`
77    pub fn forward_single_ssd(
78        &self,
79        input_bsm: Tensor<3>,
80        cache: Option<Mamba3SingleSsdCache>,
81        ssd_path: &Mamba3SsdPath,
82        pad_bt: Option<Tensor<2, Bool>>,
83    ) -> (Tensor<3>, Mamba3SingleSsdCache) {
84        self.forward_single_ssd_packed(input_bsm, cache, None, ssd_path, pad_bt, None)
85    }
86
87    /// [`Self::forward_single_ssd`] over a packed batch: `reset_bt` marks the
88    /// first token of each segment, which starts from `origin`, as in
89    /// [`Self::forward_packed`].
90    ///
91    /// A reset is a call boundary inside the row: the positions before it pay
92    /// no `ν` forward (the taps of the segment start read the slots of the
93    /// origin), and the chunk restarts from the `h'` of the origin plus its
94    /// own boundary seed, as a call from the origin does.
95    #[allow(non_snake_case)]
96    pub fn forward_single_ssd_packed(
97        &self,
98        input_bsm: Tensor<3>,
99        cache: Option<Mamba3SingleSsdCache>,
100        origin: Option<Mamba3SingleSsdCache>,
101        ssd_path: &Mamba3SsdPath,
102        pad_bt: Option<Tensor<2, Bool>>,
103        reset_bt: Option<Tensor<2, Bool>>,
104    ) -> (Tensor<3>, Mamba3SingleSsdCache) {
105        if !self.trapezoid.has_beta_tap() {
106            let (out_bsm, cache) = self.forward_double_ssd_packed(
107                input_bsm,
108                cache.map(Into::into),
109                origin.map(Into::into),
110                ssd_path,
111                pad_bt,
112                reset_bt,
113            );
114            return (out_bsm, cache.into());
115        }
116        let [batch, tokens, _d_model] = input_bsm.dims();
117        let d_inner = self.d_inner();
118        let nheads = self.nheads();
119        let ngroups = self.ngroups;
120        let per_head_dim = self.per_head_dim();
121        let state_rank = self.state_rank;
122        let mimo_rank = self.mimo_rank;
123        let micro_steps = self.micro_steps;
124        let device = input_bsm.device();
125
126        // MambaProduct: from the split below down to the readout, one sequence
127        // position *is* one micro-step. The `s` in every shape suffix counts
128        // micro-steps, and `tokens` (`t`) is at token resolution. See
129        // [`crate::mamba3::product`].
130        let sequence = tokens * micro_steps;
131
132        assert!(tokens > 0, "sequence length must be at least 1");
133        assert_eq!(nheads % ngroups, 0);
134        san(&input_bsm);
135
136        // A padded token pads all of its micro-steps. `end_b` is one past the
137        // last real position of each slot on the folded axis, where its cache
138        // fields are read.
139        let pad_bs = pad_bt.map(|pad_bt| crate::padding::repeat_rows(pad_bt, micro_steps));
140        let end_b = pad_bs.as_ref().map(crate::padding::real_end_b);
141
142        // A reset is at the first token of a chunk, which is its first
143        // micro-step on the folded axis. See [`crate::packing`].
144        let chunk_len = ssd_path.chunk_len_or_optimal(self);
145        let segments = reset_bt.map(|reset_bt| {
146            let chunk_tokens = Mamba3SsdPath::chunk_tokens(chunk_len, micro_steps);
147            let reset_bn = crate::packing::chunk_resets(reset_bt, chunk_tokens);
148            crate::packing::Segments::new(reset_bn, chunk_len, sequence)
149        });
150
151        // ── Initialise cache if not provided ──────────────────────────────────
152        let fresh = || {
153            let ssm_bhpr = Tensor::zeros([batch, nheads, per_head_dim, state_rank], &device);
154            // Reached only with a β tap (`Trapezoid::None` delegated above).
155            let (k_state_bumhr, v_state_buhp) = self.zero_tap_slots(batch, &device);
156            let rotation = self.zero_rotation_state(batch, &device);
157            let (log_precision_bh, tropical_bh) = self.zero_positive_state(batch, &device);
158            Mamba3SingleSsdCache {
159                ssm_bhpr,
160                k_state_bumhr,
161                v_state_buhp,
162                rotation,
163                log_precision_bh,
164                tropical_bh,
165            }
166        };
167        let mut cache = cache.unwrap_or_else(fresh);
168        // A packed row: the cache that each segment starts from. At a call
169        // boundary, `h'` is the double-SSD state, so the fields convert as
170        // they are.
171        let packed = segments.map(|segments| Packed {
172            segments,
173            origin: origin.unwrap_or_else(fresh).into(),
174        });
175
176        // ── Step 1: In-projection ─────────────────────────────────────────────
177        let proj_bsd = self.project_in(input_bsm);
178        // The segments of the positive systems are the outermost tail
179        // ([`crate::mamba3::positive`]). The rest is the stock layout.
180        let (proj_bsd, noise_btH, tropical_btH) = self.split_positive(proj_bsd, 2);
181        let bc_size = ngroups * state_rank * mimo_rank;
182
183        // `u` = micro_steps widens every per-micro-step segment, and `unfold`
184        // reads each of them as `u` consecutive sequence positions. `z`
185        // (per-token gate) and `C` (per-token read) do not widen. See
186        // [`crate::mamba3::product`].
187        // The optional segments come off the tail first, in reverse layout
188        // order: the rotation (`Real1D` projects none), then `μ` (only a
189        // two-tap pattern projects it), then `λ`. `λ` is always present here,
190        // because `Trapezoid::None` delegated above. A zero-width segment would
191        // silently vanish from the split below.
192        let u = micro_steps;
193        let (proj_bsd, rot_btA) =
194            helpers::split_trailing(proj_bsd, self.rotation_channels_total(), 2);
195        let (proj_bsd, mu_raw_btH) = helpers::split_trailing(proj_bsd, self.mu_channels_total(), 2);
196        let (proj_bsd, lambda_raw_btH) =
197            helpers::split_trailing(proj_bsd, self.lambda_channels_total(), 2);
198        let lambda_raw_btH = lambda_raw_btH.expect("a β tap projects λ");
199        #[rustfmt::skip]
200        let [
201                z_bsi, x_btI,
202                b_raw_btMGRU, c_raw_btMGR,
203                dd_dt_btH, dd_A_raw_btH,
204        ] = burn_stack::modules::split_into(
205            proj_bsd,
206            [
207                d_inner, u * d_inner,
208                u * bc_size, bc_size,
209                u * nheads, u * nheads,
210            ],
211            2,
212        );
213
214        use crate::mamba3::product::unfold_micro_bs;
215        let x_bsi = unfold_micro_bs(x_btI, u);
216        let b_raw_bsMGR = unfold_micro_bs(b_raw_btMGRU, u);
217        // `C` stays at token resolution into the SSD. The readout is once per
218        // token, at its last micro-step, and that is the only position where
219        // the kernel computes a `y`. See [`Mamba3SingleSsdInput::c_bntmhr`].
220        let dd_dt_bsh = unfold_micro_bs(dd_dt_btH, u);
221        let dd_A_raw_bsh = unfold_micro_bs(dd_A_raw_btH, u);
222        let lambda_raw_bsh = unfold_micro_bs(lambda_raw_btH, u);
223        let mu_raw_bsh = mu_raw_btH.map(|t| unfold_micro_bs(t, u));
224        let rot_bsa = rot_btA.map(|t| unfold_micro_bs(t, u));
225        let noise_bsh = noise_btH.map(|t| unfold_micro_bs(t, u));
226
227        san(&z_bsi);
228        san(&x_bsi);
229        san(&dd_dt_bsh);
230
231        // ── Step 2: Discretisation + trapezoidal coefficients ─────────────────
232        // Under a Kalman gain, this call computes the decay before the key
233        // scale, the band and the tap slots below read it.
234        let helpers::TrapezoidCoeffs {
235            dt: dt_bsh,
236            da: da_bsh,
237            alpha: _alpha_bsh,
238            nu: nu_bsh,
239            nu_interior: nu_interior_bsh,
240            gamma: gamma_bsh,
241            log_precision: log_precision_bsh,
242        } = helpers::trapezoidal_coefficients(
243            dd_dt_bsh,
244            dd_A_raw_bsh,
245            Some(lambda_raw_bsh),
246            mu_raw_bsh,
247            self.dt_bias_h.val(),
248            self.trapezoid_spec(),
249            self.gain_input(noise_bsh, cache.log_precision_bh.clone(), packed.as_ref()),
250        )
251        .padded(pad_bs.as_ref());
252        // The tropical register's inputs, on the same folded axis.
253        let tropical_ab_bsh = tropical_btH.map(|(a_btH, b_btH)| {
254            (unfold_micro_bs(a_btH, u), unfold_micro_bs(b_btH, u))
255        });
256        let nu_bsh = nu_bsh.expect("a β tap has a mass");
257        san(&dt_bsh);
258        san(&da_bsh);
259        san(&gamma_bsh);
260
261        // ── Compute scaleₜ = γₜ + νₜ₊ₗₐ₉ (+ νⁱⁿᵗₜ₊₁) ─────────────────────────
262        //
263        // Each shifted term is zero for the last positions that its own lag
264        // reaches past: the taps that pay them belong to the *next* call, from
265        // the tap slots. This also makes `h'` equal to the double-SSD state at
266        // a cache boundary, hence the field-identity `From` impls.
267        //
268        // `t+lag` is a later *folded* position: lag 1 is
269        // [`Trapezoid::HorizontalCarryOver`], lag `u` is [`Trapezoid::Vertical`].
270        // This is the `Δ̃` collapse (`info/mamba-3/trapezoid-as-integration.md`
271        // §5). The collapse theorem of §9 is why it holds unchanged for the
272        // wider lag and for a second tap: still one scalar per sample, so one
273        // pass. Only the same-step correction widens from the diagonal to a
274        // `lag`-wide band (see [`crate::mamba3::single_ssd::token_band`]).
275        let lag = self.tap_lag();
276        // νₜ₊ₗ, the mass that a later position pays to this one. Zero past the
277        // end of the call, where the tap belongs to the next call. Zero also
278        // across a reset of a packed row: the first positions of a segment pay
279        // the cache slots, through the seed of their chunk (below). (No output
280        // would see that mass: the restart drops the state it enters, and the
281        // band correction takes it out of the reads of its token. The zero
282        // keeps `h'` at a reset equal to that at a call boundary.)
283        let pay_forward = |nu_bsh: Tensor<3>, l: usize| {
284            let nu_bsh = match &packed {
285                Some(p) => crate::padding::fill_padded(nu_bsh, &p.segments.heads(l), 0.0),
286                None => nu_bsh,
287            };
288            let zero_bLh = Tensor::zeros([batch, l, nheads], &device);
289            if sequence == l {
290                zero_bLh
291            } else {
292                Tensor::cat(vec![nu_bsh.narrow(1, l, sequence - l), zero_bLh], 1)
293            }
294        };
295        let far_shifted_bsh = pay_forward(nu_bsh.clone(), lag);
296        // A two-tap pattern also collapses to one scalar (§9), with a second
297        // shift in it: `scaleₜ = γₜ + νⁱⁿᵗₜ₊₁ + νᶠᵃʳₜ₊ₗₐ₉`.
298        let interior_shifted_bsh = nu_interior_bsh
299            .clone()
300            .map(|nu_bsh| pay_forward(nu_bsh, 1));
301        let scale_bsh = gamma_bsh.clone() + far_shifted_bsh.clone();
302        let scale_bsh = match &interior_shifted_bsh {
303            Some(interior_bsh) => scale_bsh + interior_bsh.clone(),
304            None => scale_bsh,
305        };
306        san(&scale_bsh);
307
308        // ── Step 3: Reshape x ─────────────────────────────────────────────────
309        let x_bshp = x_bsi.reshape([batch, sequence, nheads, per_head_dim]);
310
311        // ── Step 4: QK-Norm on B and C ────────────────────────────────────────
312        let b_bsmhr = helpers::qk_norm_expand_bias::<5, 6>(
313            b_raw_bsMGR.reshape([batch, sequence, mimo_rank, ngroups, state_rank]),
314            &self.b_norm,
315            self.b_bias_hmr.val(),
316            3,
317            nheads,
318        );
319        let c_btmhr = helpers::qk_norm_expand_bias::<5, 6>(
320            c_raw_btMGR.reshape([batch, tokens, mimo_rank, ngroups, state_rank]),
321            &self.c_norm,
322            self.c_bias_hmr.val(),
323            3,
324            nheads,
325        );
326
327        // ── Step 5: Data-dependent transition rotation of B and C ─────────────
328        // Shared with the double-ssd pathway through [`rotate_bc_forward`]. The
329        // single-pass SSD core below does not depend on the rotation: it only
330        // uses the rotated B̄/C̄ (the RoPE factoring `C̄ₜᵀB̄ᵢ = Cₜᵀ·Rel(t,i)·Bᵢ`
331        // holds for every kind). `C` turns at token resolution: it needs the
332        // cumulative rotation of the micro-step where it is read, a stride slice
333        // of the rotation of `B`.
334        let (b_bsmhr, c_btmhr, new_rotation) = rotate_bc_forward(
335            rot_bsa,
336            dt_bsh.clone(),
337            cache.rotation.clone(),
338            b_bsmhr,
339            c_btmhr,
340            u,
341            self.rotation_spec(),
342            packed.as_ref().map(|p| (&p.segments.start_bs, p.origin.rotation.clone())),
343        );
344        san(&b_bsmhr);
345        san(&c_btmhr);
346
347        // ── Save the B and x of the last `lag` positions (raw, no MIMO_V) ─────
348        // The next call pays their second installment. At `lag = u` that
349        // window is exactly the last token.
350        let (b_last_bumhr, x_last_buhp) = self.save_tap_slots(
351            &b_bsmhr,
352            &x_bshp,
353            &da_bsh,
354            lag,
355            end_b.clone().map(|end_b| {
356                (end_b, cache.k_state_bumhr.clone(), cache.v_state_buhp.clone())
357            }),
358            packed.as_ref(),
359        );
360
361        // ── Boundary β seed for initial state ─────────────────────────────────
362        // Add Σⱼ νⱼ · Σₘ K_prev[j, m] ⊗ (x_prev[j] ⊙ mimo_xₘ) to the carried
363        // single-ssd SSM state. Cache slot `j` (oldest first) is the position
364        // whose tap position `j` of this call pays, so it takes `νⱼ` from the
365        // first `lag` positions of this call. `x_prev` already carries the
366        // decay from its own position to the boundary, and `K_prev` its own
367        // rotation, so the pair *is* the transported write.
368        //
369        // Under [`Trapezoid::VerticalPlusHorizontalCarryOver`], one more
370        // installment crosses. The newest slot is also the lag-1 tap of the
371        // *first* position of this call, with the empty decay product. So it is
372        // the same term, with `νⁱⁿᵗ₀` added to the own weight of that slot. For
373        // every other pattern, the interior tap is closed exactly there.
374        let mimo_x_hmp = self.mimo_x_hmp.as_ref().map(|p| p.val());
375        // The masses that the first `width` positions of each chunk pay,
376        // `[batch, nchunks, width, nheads]`. Only chunk 0 is the start of the
377        // call. In a packed row, every chunk that starts a segment is the
378        // start of one from the origin.
379        let nchunks = sequence.div_ceil(chunk_len);
380        let heads_of = |t_bsh: Tensor<3>, width: usize| {
381            let pad = nchunks * chunk_len - sequence;
382            let t_bSh = if pad == 0 {
383                t_bsh
384            } else {
385                Tensor::cat(vec![t_bsh, Tensor::zeros([batch, pad, nheads], &device)], 1)
386            };
387            t_bSh
388                .reshape([batch, nchunks, chunk_len, nheads])
389                .narrow(2, 0, width)
390        };
391        let nu_heads_bnuh = heads_of(nu_bsh.clone(), lag);
392        let nu_heads_bnuh = match nu_interior_bsh
393            .filter(|_| self.trapezoid.interior_tap_crosses_tokens())
394        {
395            // `lag ≥ 2` here: a two-tap pattern folds at `u = 1`, so an interior
396            // tap and a one-slot FIFO never coexist.
397            Some(nu_interior_bsh) => Tensor::cat(
398                vec![
399                    nu_heads_bnuh.clone().narrow(2, 0, lag - 1),
400                    nu_heads_bnuh.narrow(2, lag - 1, 1) + heads_of(nu_interior_bsh, 1),
401                ],
402                2,
403            ),
404            None => nu_heads_bnuh,
405        };
406        // The slots of a FIFO, `v` with MIMO_V applied:
407        // `([batch, lag, mimo_rank, nheads, per_head_dim], [batch, lag, mimo_rank, nheads, state_rank])`.
408        let slots = "a β tap keeps its (B, x) cache slots";
409        let prev_slots = |v_state_buhp: Option<Tensor<4>>, k_state_bumhr: Option<Tensor<5>>| {
410            let v_prev_mimo_bumhp =
411                helpers::build_v_with_mimo::<4, 5>(v_state_buhp.expect(slots), mimo_x_hmp.as_ref(), 2);
412            (v_prev_mimo_bumhp, k_state_bumhr.expect(slots))
413        };
414        // The seed of the call, from the slots of the cache and the masses of
415        // chunk 0.
416        let (v_prev_mimo_bumhp, k_prev_bumhr) =
417            prev_slots(cache.v_state_buhp.clone(), cache.k_state_bumhr.clone());
418        let nu_head_buh = nu_heads_bnuh.clone().narrow(1, 0, 1).squeeze_dim::<3>(1);
419        let v_prev_mimo_bumhp = v_prev_mimo_bumhp * nu_head_buh.unsqueeze_dims::<5>(&[2, 4]);
420        // The seed contracts over the slots *and* the ranks (both are outer
421        // products into one state). So, fused, it is the same one
422        // `mimo_outer_sum` that the per-token write uses.
423        let boundary_seed_bhpr = helpers::mimo_outer_sum(
424            v_prev_mimo_bumhp.reshape([batch, lag * mimo_rank, nheads, per_head_dim]),
425            k_prev_bumhr.reshape([batch, lag * mimo_rank, nheads, state_rank]),
426            self.use_siso_decode_kernels(),
427        );
428        let initial_state_bhpr = cache.ssm_bhpr.clone() + boundary_seed_bhpr;
429        san(&initial_state_bhpr);
430        // A packed row: each chunk gets a seed from the slots of the origin,
431        // in case it restarts. Each slot writes one outer product (over the
432        // ranks). Each chunk weighs the slots by its own head masses.
433        let seed_bnhpr = packed.as_ref().map(|p| {
434            let (v_origin_mimo_bumhp, k_origin_bumhr) =
435                prev_slots(p.origin.v_state_buhp.clone(), p.origin.k_state_bumhr.clone());
436            let slot_writes_bhuP = helpers::mimo_outer_sum(
437                v_origin_mimo_bumhp.reshape([batch * lag, mimo_rank, nheads, per_head_dim]),
438                k_origin_bumhr.reshape([batch * lag, mimo_rank, nheads, state_rank]),
439                self.use_siso_decode_kernels(),
440            )
441            .reshape([batch, lag, nheads, per_head_dim * state_rank])
442            .swap_dims(1, 2);
443            nu_heads_bnuh
444                .permute([0, 3, 1, 2]) // _bhnu
445                .matmul(slot_writes_bhuP) // _bhnP
446                .reshape([batch, nheads, nchunks, per_head_dim, state_rank])
447                .swap_dims(1, 2)
448        });
449
450        // ── Step 6: Pad sequence to multiple of chunk_len ─────────────────────
451        let sequence_padded = sequence.next_multiple_of(chunk_len);
452        let pad = sequence_padded - sequence;
453
454        // V passed to SSD is raw x with MIMO_V applied (not γ-scaled).
455        let v_bshmp = helpers::build_v_with_mimo::<4, 5>(x_bshp.clone(), mimo_x_hmp.as_ref(), 2);
456        // v_bshmp has axis order [b, s, m, h, p] (insert_dim=2 onto [b,s,h,p]).
457
458        // ── The lag-`u` correction band ───────────────────────────────────────
459        // At lag 1 the key scale is wrong only on the diagonal, and
460        // `ssd/diag.rs` corrects it inside the kernel. At lag `u` the exception
461        // is `u` wide, and at the positions that the readout keeps it is
462        // exactly the token. So it is one contraction here, not a wider mask.
463        // See [`crate::mamba3::single_ssd::token_band`].
464        //
465        // The band is the **far** installment only, also for a two-tap
466        // pattern. At the surviving read `t = τu + u−1`, the interior (lag-1)
467        // installment of a sample has already landed for every `j < u−1`. At
468        // `j = u−1`, the kernel replaces the whole weight with `γ` anyway. Only
469        // the lag-`u` installment is still unpaid there.
470        let band_correction_btmhp = (lag > 1)
471            .then(|| {
472                crate::mamba3::single_ssd::token_band::token_band_correction(
473                    v_bshmp.clone(),
474                    b_bsmhr.clone(),
475                    c_btmhr.clone(),
476                    far_shifted_bsh,
477                    da_bsh.clone(),
478                    micro_steps,
479                )
480            })
481            .flatten();
482
483        // `C` and `γ` are on the read axis of the chunk, so they pad by whole
484        // tokens. `chunk_len` is a multiple of `u`, so `pad` is too.
485        let gamma_bth = helpers::read_rows::<3, 4>(gamma_bsh, 1, u);
486        let tokens_padded = sequence_padded / u;
487        let pad_tokens = tokens_padded - tokens;
488
489        #[rustfmt::skip]
490        let (v_bShmp, da_bSh, gamma_bTh, scale_bSh, b_bSmhr, c_bTmhr) = if pad == 0 {
491            (v_bshmp, da_bsh, gamma_bth, scale_bsh, b_bsmhr, c_btmhr)
492        } else {
493            let pad_bShmp = Tensor::zeros([batch, pad, mimo_rank, nheads, per_head_dim], &device);
494            let pad_bSh = Tensor::zeros([batch, pad, nheads], &device);
495            let pad_bTh = Tensor::zeros([batch, pad_tokens, nheads], &device);
496            let pad_bSmhr = Tensor::zeros([batch, pad, mimo_rank, nheads, state_rank], &device);
497            let pad_bTmhr =
498                Tensor::zeros([batch, pad_tokens, mimo_rank, nheads, state_rank], &device);
499            (
500                Tensor::cat(vec![v_bshmp, pad_bShmp], 1),
501                Tensor::cat(vec![da_bsh, pad_bSh.clone()], 1),
502                Tensor::cat(vec![gamma_bth, pad_bTh], 1),
503                Tensor::cat(vec![scale_bsh, pad_bSh], 1),
504                Tensor::cat(vec![b_bsmhr, pad_bSmhr], 1),
505                Tensor::cat(vec![c_btmhr, pad_bTmhr], 1),
506            )
507        };
508
509        // ── Reshape into chunks ───────────────────────────────────────────────
510        let nchunks = sequence_padded / chunk_len;
511        let chunk_tokens = Mamba3SsdPath::chunk_tokens(chunk_len, u);
512        let v_bnlmhp =
513            v_bShmp.reshape([batch, nchunks, chunk_len, mimo_rank, nheads, per_head_dim]);
514        let da_bnlh = da_bSh.reshape([batch, nchunks, chunk_len, nheads]);
515        let gamma_bnth = gamma_bTh.reshape([batch, nchunks, chunk_tokens, nheads]);
516        let scale_bnlh = scale_bSh.reshape([batch, nchunks, chunk_len, nheads]);
517        let b_bnlmhr = b_bSmhr.reshape([batch, nchunks, chunk_len, mimo_rank, nheads, state_rank]);
518        let c_bntmhr =
519            c_bTmhr.reshape([batch, nchunks, chunk_tokens, mimo_rank, nheads, state_rank]);
520
521        // ── Step 7: Run single-pass form SSD ───────────────────────────────────────
522        let ssd_input = Mamba3SingleSsdInput {
523            v_bnlmhp,
524            b_bnlmhr,
525            c_bntmhr,
526            da_bnlh,
527            gamma_bnth,
528            scale_bnlh,
529            initial_state_bhpr,
530            init_state_hpr: self.init_state_hpr.as_ref().map(|s| s.val()),
531            read_stride: u,
532            siso_specialization: self.siso_specialization,
533            restarts: packed.as_ref().map(|p| crate::packing::Restarts {
534                reset_bn: p.segments.reset_bn.clone(),
535                origin_bhpr: p.origin.ssm_bhpr.clone(),
536            }),
537            seed_bnhpr,
538        };
539        let (y_bntmhp, final_state_bhpr) = ssd_input.run(ssd_path);
540
541        san(&y_bntmhp);
542        san(&final_state_bhpr);
543        cache.ssm_bhpr = final_state_bhpr;
544
545        // ── Step 8: Unpad ─────────────────────────────────────────────────────
546        // The SSD returns token resolution: the readout is after all `u`
547        // writes, and the kernel computed only that row.
548        let y_bTmhp = y_bntmhp.reshape([batch, tokens_padded, mimo_rank, nheads, per_head_dim]);
549        let y_bsmhp = if pad == 0 {
550            y_bTmhp
551        } else {
552            y_bTmhp.narrow(1, 0, tokens)
553        };
554
555        let y_bsmhp = match band_correction_btmhp {
556            Some(correction_btmhp) => y_bsmhp - correction_btmhp,
557            None => y_bsmhp,
558        };
559        let (y_bsmhp, log_precision_bh, tropical_bh) = self.positive_tail(
560            y_bsmhp,
561            log_precision_bsh,
562            tropical_ab_bsh,
563            cache.tropical_bh.clone(),
564            end_b.map(|end_b| (end_b, cache.log_precision_bh.clone())),
565            packed.as_ref(),
566        );
567        cache.log_precision_bh = log_precision_bh;
568        cache.tropical_bh = tropical_bh;
569        let x_bthp = crate::mamba3::product::last_micro4(x_bshp.clone(), micro_steps);
570        let sequence = tokens;
571
572        // ── Step 9: D skip + gate + MIMO_O down-projection ────────────────────
573        // The D skip uses the raw x ⊙ mimo_x (not γ-scaled, as in the double
574        // pathway), at the micro-step of the readout.
575        let v_raw_bsmhp = helpers::build_v_with_mimo::<4, 5>(x_bthp, mimo_x_hmp.as_ref(), 2);
576        let d_111h1 = self.d_h.val().unsqueeze_dims::<5>(&[0, 1, 2, 4]);
577        let y_bsmhp = y_bsmhp + d_111h1 * v_raw_bsmhp;
578
579        let y_bsi = if mimo_rank > 1 {
580            let mimo_z_hmp = self.mimo_z_hmp.as_ref().map(|p| p.val()).unwrap();
581            let mimo_o_hmp = self.mimo_o_hmp.as_ref().map(|p| p.val()).unwrap();
582
583            let z_bshp = z_bsi
584                .clone()
585                .reshape([batch, sequence, nheads, per_head_dim]);
586            let z_bsmhp = {
587                let z_bsmhp = z_bshp.unsqueeze_dim::<5>(2).expand([
588                    batch,
589                    sequence,
590                    mimo_rank,
591                    nheads,
592                    per_head_dim,
593                ]);
594                let mimo_z_bsmhp = mimo_z_hmp
595                    .swap_dims(0, 1)
596                    .unsqueeze_dims::<5>(&[0, 1])
597                    .expand([batch, sequence, mimo_rank, nheads, per_head_dim]);
598                z_bsmhp * mimo_z_bsmhp
599            };
600
601            let y_combined_bsmhp = match &self.out_norm {
602                Some(norm) => norm.forward(y_bsmhp, z_bsmhp),
603                None => y_bsmhp * Silu::new().forward(z_bsmhp),
604            };
605
606            let mimo_o_bsmhp = mimo_o_hmp
607                .swap_dims(0, 1)
608                .unsqueeze_dims::<5>(&[0, 1])
609                .expand([batch, sequence, mimo_rank, nheads, per_head_dim]);
610            let y_bshp: Tensor<4> = (y_combined_bsmhp * mimo_o_bsmhp).sum_dim(2).squeeze_dim(2);
611            y_bshp.reshape([batch, sequence, d_inner])
612        } else {
613            let y_bshp: Tensor<4> = y_bsmhp.squeeze_dim(2);
614            let z_bshp = z_bsi.reshape([batch, sequence, nheads, per_head_dim]);
615            let y_combined_bshp = match &self.out_norm {
616                Some(norm) => norm.forward(y_bshp, z_bshp),
617                None => y_bshp * Silu::new().forward(z_bshp),
618            };
619            y_combined_bshp.reshape([batch, sequence, d_inner])
620        };
621        san(&y_bsi);
622
623        // ── Out-projection ────────────────────────────────────────────────────
624        let out_bsm = self.out_proj.forward(y_bsi);
625        san(&out_bsm);
626
627        // ── Update remaining cache fields ─────────────────────────────────────
628        cache.k_state_bumhr = b_last_bumhr;
629        cache.v_state_buhp = x_last_buhp;
630        // The new cumulative rotation from [`rotate_bc_forward`]. It has the
631        // convention of the double-ssd cache, so the two caches convert.
632        cache.rotation = new_rotation;
633
634        (out_bsm, cache)
635    }
636}

638// ---------------------------------------------------------------------------
639// Mamba3::step  (recurrent SSM — token-by-token decoding)
640// ---------------------------------------------------------------------------
641
642mod step {
643    use super::*;
644
645    impl Mamba3 {
646        /// Process a **single token** with the pure recurrent form (the
647        /// formulas are in [`Mamba3::step`]). It converts the cache to the
648        /// double-ssd form, runs [`Mamba3::step_double_ssd`], and converts back.
649        ///
650        /// # Shapes
651        /// - `input_bd` : `[batch, d_model]`
652        /// - output     : `[batch, d_model]`
653        #[allow(non_snake_case)]
654        pub fn step_single_ssd(
655            &self,
656            input_bd: Tensor<2>,
657            cache: Option<Mamba3SingleSsdCache>,
658        ) -> (Tensor<2>, Mamba3SingleSsdCache) {
659            // Token-by-token decoding always uses the recurrent (double-ssd)
660            // form. A single-ssd cache holds the trapezoid state at a call
661            // boundary, where the single- and double-ssd accumulators are equal
662            // (see the `From` impls in `crate::mamba3::cache`). So the
663            // conversion in and back out is lossless. One recurrence step is
664            // itself a boundary-to-boundary transition, so the round trip stays
665            // exact.
666            let cache = cache.map(Mamba3DoubleSsdCache::from);
667            let (out_bd, cache) = self.step_double_ssd(input_bd, cache);
668            (out_bd, cache.into())
669        }
670    }
671}
