1//! # Mamba-3 Single-pass SSD Inference Cache
2//!
3//! The cache used by [`crate::mamba3::mamba3::Mamba3::forward_single_ssd`]
4//! (the single-pass SSD algorithm — see the Triton SISO and Tilelang MIMO
5//! reference kernels).
6//! The four tensor fields mirror those of [`Mamba3Cache`](crate::mamba3::cache::Mamba3Cache) but their
7//! **SSM accumulator carries different semantics**:
8//!
9//! - [`Mamba3Cache`](crate::mamba3::cache::Mamba3Cache): `ssm_bhpr` holds the double-ssd trapezoidal hidden state
10//!   `hₜ = αₜ hₜ₋₁ + βₜ Bₜ₋₁ ⊗ xₜ₋₁ + γₜ Bₜ ⊗ xₜ`.
11//! - [`Mamba3SingleSsdCache`](crate::mamba3::single_ssd::cache::Mamba3SingleSsdCache): `ssm_bhpr` holds the **trapezoid accumulator** `h'ₜ`
12//!   defined by `h'ₜ = αₜ h'ₜ₋₁ + scaleₜ Bₜ ⊗ xₜ`, where
13//!   `scaleₜ = γₜ + (1 − λₜ₊₁) · Δₜ₊₁`. The single-ssd form gives the correct output
14//!   `yₜ = Cₜᵀ h'ₜ` for all positions except the diagonal (s = t), which is
15//!   patched by an explicit `γₜ · (Cₜᵀ Bₜ) · xₜ` correction term in the kernel.
16//!
17//! Under [`Trapezoid::None`](crate::mamba3::trapezoid::Trapezoid::None) that
18//! difference disappears — `scaleₜ = γₜ` everywhere, so `h' ≡ h` at every
19//! position and the tap slots are absent from both caches.
20//!
21//! Because the two accumulators differ, the two caches are not interchangeable.
22//! The distinct type prevents accidentally feeding a `forward_double_ssd` cache into
23//! `forward_single_ssd` (or vice versa) mid-sequence — that would silently corrupt state.
24

30// ---------------------------------------------------------------------------
31// Mamba3SingleSsdCaches  (one cache entry per layer)
32// ---------------------------------------------------------------------------
33
34/// A collection of per-layer single-ssd form caches for a complete Mamba-3 network.
35#[derive(Module, Debug)]
36pub struct Mamba3SingleSsdCaches {
37    /// Per-layer caches. Length equals the number of virtual layers.
38    pub caches: Vec<Mamba3SingleSsdCache>,
39}
40
41/// Configuration / factory for [`Mamba3SingleSsdCaches`].
42#[derive(Config, Debug)]
43pub struct Mamba3SingleSsdCachesConfig {
44    /// Number of cache slots (= number of virtual layers).
45    pub n_real_caches: usize,
46
47    /// Shared configuration that determines the shape of each cache.
48    pub cache: Mamba3SingleSsdCacheConfig,
49}
50
51impl Mamba3SingleSsdCachesConfig {
52    /// Convenience constructor from a block config.
53    pub fn new_from_block_config(
54        n_real_caches: usize,
55        batch: usize,
56        block_config: Mamba3Config,
57    ) -> Self {
58        Self {
59            n_real_caches,
60            cache: Mamba3SingleSsdCacheConfig::new_from_block_config(batch, block_config),
61        }
62    }
63
64    /// Allocate all cache tensors (zero-initialised) on `device`.
65    pub fn init(&self, device: &Device) -> Mamba3SingleSsdCaches {
66        let caches = (0..self.n_real_caches)
67            .map(|_| self.cache.clone().init(device))
68            .collect();
69        Mamba3SingleSsdCaches { caches }
70    }
71}

73// ---------------------------------------------------------------------------
74// Mamba3SingleSsdCache  (state for a single layer)
75// ---------------------------------------------------------------------------
76
77/// Mutable state for a single Mamba-3 layer running the single-ssd form algorithm.
78///
79/// Tensor shapes match [`Mamba3Cache`]. The semantic difference lives entirely
80/// in `ssm_bhpr` (see the module-level documentation).
81#[derive(Module, Debug)]
82pub struct Mamba3SingleSsdCache {
83    /// **SingleSsd-form SSM accumulator** `h'ₜ`.
84    ///
85    /// Update rule: `h'ₜ = αₜ h'ₜ₋₁ + scaleₜ · sumₘ Bₜ[m] ⊗ (xₜ ⊙ mimo_xₘ)`.
86    /// Different from `Mamba3Cache::ssm_bhpr`.
87    ///
88    /// Shape: `[batch, nheads, per_head_dim, state_rank]`
89    pub ssm_bhpr: Tensor<4>,
90
91    /// **The tap FIFO's K** (post-RoPE, post-bias `B`), one slot per lagged
92    /// position, **oldest first**.
93    ///
94    /// Used at the start of the next forward_single_ssd call to seed the deferred
95    /// boundary β contributions `Σⱼ (1 − λⱼ) · Δⱼ · Bₚ₋ₗₐ₉₊ⱼ ⊗ xₚ₋ₗₐ₉₊ⱼ` (which the
96    /// previous call could not yet add because it did not know the `λ, Δ` of the
97    /// `lag` positions that pay them).
98    ///
99    /// `None` under [`Trapezoid::None`], which has no boundary β term to defer.
100    ///
101    /// Shape: `[batch, tap_slots, mimo_rank, nheads, state_rank]`
102    pub k_state_bumhr: Option<Tensor<5>>,
103
104    /// **The tap FIFO's x**, matching [`Self::k_state_bumhr`] slot for slot, and
105    /// `None` with it. Pre-scaled by the decay accumulated since its own
106    /// position, exactly as in the double-SSD cache (whose field this is).
107    ///
108    /// Shape: `[batch, tap_slots, nheads, per_head_dim]`
109    pub v_state_buhp: Option<Tensor<4>>,
110
111    /// **Cumulative data-dependent rotation** up to the current position
112    /// ([`RotationState`]).
113    ///
114    /// Same role as in [`Mamba3Cache`]: continued across calls for streaming.
115    /// Carries the same value as the double-ssd cache's field (the `From` impls
116    /// move it across), so the two caches still inter-convert by field identity.
117    pub rotation: RotationState,
118}
119
120impl Mamba3SingleSsdCache {
121    /// Run the [`NaN`/`Inf` guards](burn_stack::modules::misc::sanity) on every cached tensor.
122    pub fn sanity(&self) {
123        san(&self.ssm_bhpr);
124        assert_eq!(
125            self.k_state_bumhr.is_some(),
126            self.v_state_buhp.is_some(),
127            "the trapezoid's tap slots are present or absent together"
128        );
129        if let Some(k_state_bumhr) = &self.k_state_bumhr {
130            san(k_state_bumhr);
131        }
132        if let Some(v_state_buhp) = &self.v_state_buhp {
133            san(v_state_buhp);
134        }
135        self.rotation.sanity();
136    }
137}

139/// Configuration / factory for a single [`Mamba3SingleSsdCache`].
140#[derive(Config, Debug)]
141pub struct Mamba3SingleSsdCacheConfig {
142    /// Batch size.
143    pub batch: usize,
144
145    /// State rank.
146    #[config(default = 128)]
147    pub state_rank: usize,
148
149    /// Head dimension per_head_dim.
150    #[config(default = 64)]
151    pub per_head_dim: usize,
152
153    /// Number of SSM heads.
154    pub nheads: usize,
155
156    /// MIMO rank. 1 = SISO.
157    #[config(default = 1)]
158    pub mimo_rank: usize,
159
160    /// Number of RoPE angle pairs
161    /// (see [`crate::mamba3::double_ssd::cache::Mamba3DoubleSsdCacheConfig::num_rope_angles`]).
162    pub num_rope_angles: usize,
163
164    /// Which transition rotation the block uses (see
165    /// [`crate::mamba3::double_ssd::cache::Mamba3DoubleSsdCacheConfig::rotation`]).
166    #[config(default = "crate::mamba3::rotation::RotationKind::Complex2D")]
167    pub rotation: RotationKind,
168
169    /// Number of quaternion blocks (`rope_dim / 4`); only used for
170    /// [`RotationKind::Quaternion4D`] / [`RotationKind::Rotor4D`].
171    #[config(default = 1)]
172    pub num_quat_blocks: usize,
173
174    /// The block's tap pattern (see
175    /// [`Mamba3DoubleSsdCacheConfig::trapezoid`](crate::mamba3::double_ssd::cache::Mamba3DoubleSsdCacheConfig::trapezoid)).
176    #[config(default = "crate::mamba3::trapezoid::Trapezoid::HorizontalCarryOver")]
177    pub trapezoid: Trapezoid,
178
179    /// Recurrence micro-steps per token (`u`); with [`Self::trapezoid`] it fixes
180    /// the tap FIFO's depth (see [`Trapezoid::tap_lag`]).
181    #[config(default = 1)]
182    pub micro_steps: usize,
183}
184
185impl Mamba3SingleSsdCacheConfig {
186    /// Derive cache shapes from a Mamba-3 block configuration plus a batch size.
187    pub fn new_from_block_config(batch: usize, block_config: Mamba3Config) -> Self {
188        Self {
189            batch,
190            state_rank: block_config.state_rank,
191            per_head_dim: block_config.per_head_dim,
192            nheads: block_config.nheads(),
193            mimo_rank: block_config.mimo_rank,
194            num_rope_angles: block_config.num_rope_angles(),
195            rotation: block_config.rotation,
196            num_quat_blocks: block_config.num_quat_blocks(),
197            trapezoid: block_config.trapezoid,
198            micro_steps: block_config.micro_steps,
199        }
200    }
201
202    /// Allocate zero/identity-initialised cache tensors on `device`.
203    pub fn init(&self, device: &Device) -> Mamba3SingleSsdCache {
204        let ssm_bhpr = Tensor::zeros(
205            [self.batch, self.nheads, self.per_head_dim, self.state_rank],
206            device,
207        );
208        let slots = self.trapezoid.tap_slots(self.micro_steps);
209        let tap = slots > 0;
210        let k_state_bumhr = tap.then(|| {
211            Tensor::zeros(
212                [
213                    self.batch,
214                    slots,
215                    self.mimo_rank,
216                    self.nheads,
217                    self.state_rank,
218                ],
219                device,
220            )
221        });
222        let v_state_buhp = tap.then(|| {
223            Tensor::zeros(
224                [self.batch, slots, self.nheads, self.per_head_dim],
225                device,
226            )
227        });
228        let rotation = RotationState::identity(
229            self.rotation,
230            self.batch,
231            self.nheads,
232            self.num_rope_angles,
233            self.num_quat_blocks,
234            device,
235        );
236        Mamba3SingleSsdCache {
237            ssm_bhpr,
238            k_state_bumhr,
239            v_state_buhp,
240            rotation,
241        }
242    }
