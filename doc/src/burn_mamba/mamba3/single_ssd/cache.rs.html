1//! # Mamba-3 Caches (single-SSD pathway)
2//!
3//! The cache of [`crate::mamba3::mamba3::Mamba3::forward_single_ssd`] (the
4//! single-pass SSD pathway of the Triton SISO and Tilelang MIMO reference
5//! kernels). Its fields mirror those of
6//! [`Mamba3DoubleSsdCache`](crate::mamba3::double_ssd::cache::Mamba3DoubleSsdCache),
7//! but `ssm_bhpr` holds the **accumulator**, which has different semantics
8//! mid-sequence:
9//!
10//! - double-SSD: `ssm_bhpr` holds the trapezoidal state
11//!   `hₜ = αₜ hₜ₋₁ + βₜ Bₜ₋₁ ⊗ xₜ₋₁ + γₜ Bₜ ⊗ xₜ` (at the default lag 1).
12//! - single-SSD: `ssm_bhpr` holds the **accumulator**
13//!   `h'ₜ = αₜ h'ₜ₋₁ + scaleₜ Bₜ ⊗ xₜ`, with
14//!   `scaleₜ = γₜ + νₜ₊ₗₐ₉ (+ νⁱⁿᵗₜ₊₁)`. With it, `yₜ = Cₜᵀ h'ₜ` is correct at
15//!   all positions except the `lag`-wide band before a tap is paid. The kernel
16//!   patches the diagonal (lag 1) with an explicit `γₜ · (Cₜᵀ Bₜ) · xₜ` term,
17//!   and `token_band` patches the wider band (lag `u`).
18//!
19//! The accumulator and the state are equal at call boundaries, where caches are
20//! produced and consumed (see [`crate::mamba3::cache`]). Under
21//! [`TapPattern::None`](crate::mamba3::tap_pattern::TapPattern::None) the difference
22//! disappears everywhere: `scaleₜ = γₜ`, so `h' ≡ h` at every position, and
23//! neither cache has a tap FIFO.
24//!
25//! The distinct type prevents a `forward_double_ssd` cache from going into
26//! `forward_single_ssd` (or the reverse) mid-sequence, which would silently
27//! corrupt the state.
28

38/// A collection of per-layer single-ssd caches for a complete Mamba-3 network.

45// ---------------------------------------------------------------------------
46// Mamba3SingleSsdCache  (cache for a single block)
47// ---------------------------------------------------------------------------
48
49/// The cache of one Mamba-3 block that runs the single-ssd pathway.
50///
51/// The fields and shapes match
52/// [`Mamba3DoubleSsdCache`](crate::mamba3::double_ssd::cache::Mamba3DoubleSsdCache):
53/// the `ssm_bhpr` of the pathway and the frame cache. The semantic difference
54/// is only in `ssm_bhpr` (see the module header).
55#[derive(Module, Debug)]
56pub struct Mamba3SingleSsdCache {
57    /// **Accumulator** `h'ₜ`.
58    ///
59    /// Update rule: `h'ₜ = αₜ h'ₜ₋₁ + scaleₜ · sumₘ Bₜ[m] ⊗ (xₜ ⊙ mimo_xₘ)`.
60    /// Mid-sequence, it is different from the double-ssd `ssm_bhpr`.
61    ///
62    /// Shape: `[batch, nheads, per_head_dim, state_rank]`
63    pub ssm_bhpr: Tensor<4>,
64
65    /// **The frame cache** ([`Mamba3FrameCache`]), the same value as in the
66    /// double-SSD cache. The `From` impls move it across.
67    pub frame: Mamba3FrameCache,
68}
69
70impl Mamba3SingleSsdCache {
71    /// Run the [`NaN`/`Inf` guards](burn_stack::modules::misc::sanity) on every cached tensor.
72    pub fn sanity(&self) {
73        san(&self.ssm_bhpr);
74        self.frame.sanity();
75    }
76}
