1//! # Single-Pass SSD (Minimal / segsum variant)
2//!
3//! This is the MIMO-first, single SSD pass implementation of the
4//! Mamba-3 trapezoid recurrence. It is the Burn analogue of the official
5//! Tilelang MIMO kernel and Triton SISO kernel; SISO is the `mimo_rank = 1`
6//! degenerate case.
7//!
8//! ## Background — the single-ssd recurrence
9//!
10//! The double-ssd trapezoid hidden state is
11//!
12//! ```text
13//!   hₜ = αₜ hₜ₋₁ + βₜ (Bₜ₋₁ ⊗ xₜ₋₁) + γₜ (Bₜ ⊗ xₜ)
14//! ```
15//!
16//! Expanding the recurrence and grouping by `(Bₛ ⊗ xₛ)` gives the coefficient
17//! `(Πᵣ₌ₛ₊₁ᵗ αᵣ) · [γₛ + (1−λₛ₊₁)·Δₛ₊₁]` for the contribution of step `s` to
18//! state `t` (for `s < t`). At `s = t` the coefficient is just `γₜ`.
19//!
20//! Define `scaleₜ = γₜ + (1−λₜ₊₁)·Δₜ₊₁` (with `scaleₜ = γₜ` at the last
21//! position). The single-SSD
22//!
23//! ```text
24//!   h'ₜ = αₜ h'ₜ₋₁ + scaleₜ (Bₜ ⊗ xₜ)
25//! ```
26//!
27//! produces the same outputs `yₜ = Cₜᵀ h'ₜ` as the double-ssd one **except**
28//! at the same-step diagonal (`s = t`), where the single-ssd form has `scaleₜ`
29//! instead of `γₜ`. We compensate by:
30//!
31//! 1. Using a **strict** lower-triangular mask in the intra-chunk path (the
32//!    `s = t` block is excluded from the trapezoid sum).
33//! 2. Adding a separate γ-weighted same-step term `γₜ · (Cₜᵀ Bₜ) · xₜ`.
34//!
35//! ## Algorithm (per chunk, MIMO-first)
36//!
37//! ```text
38//!   K_scaled[t, m, h, n] = scaleₜ · B[t, m, h, n]     // K scaled inside the SSD
39//!
40//!   y_lower  = (C ⊗ K_scaledᵀ ⊙ L_strict) · PsiV     // strict lower-tri
41//!   y_diag   = γₜ · (C ⊗ Bᵀ at same step) · PsiV      // diagonal correction
42//!   y_off    = C · h'_chunk_in · exp(da_cs)           // state-to-output
43//!
44//!   y        = y_lower + y_diag + y_off
45//!
46//!   h'_chunk_out = exp(da_cs_last) · h'_chunk_in
47//!                + K_scaled · exp(da_cs_rev)ᵀ · PsiV   // standard state update
48//! ```
49//!
50//! The MIMO causal mask is identical to [`crate::mamba3::double_ssd::ssd::minimal`] but
51//! with a stricter inequality (`i_time > j_time` rather than `i_time ≥ j_time`).
52//!
53//! Reference implementations:
54//! - SISO: `refs/state-spaces/mamba/mamba_ssm/ops/triton/mamba3/mamba3_siso_fwd.py`
55//! - MIMO: `refs/state-spaces/mamba/mamba_ssm/ops/tilelang/mamba3/mamba3_mimo_fwd.py`
56
57use crate::mamba3::helpers;
58use crate::mamba3::prelude::Mamba3SsdPath;
59use crate::mamba3::single_ssd::prelude::*;
60use crate::mamba3::single_ssd::ssd::diag::y_diag_correction;
61use burn_stack::modules::segsum;
62use burn::prelude::*;
63
64impl Mamba3SingleSsdInput {
65    /// MIMO-first single-SSD — segsum variant.
66    ///
67    /// See module documentation for the algorithm. Returns the chunked outputs
68    /// and the final single-ssd accumulator.
69    ///
70    /// # Shapes
71    /// `T = chunk_tokens = chunk_len / read_stride` is the chunk's read axis
72    /// (see [`Mamba3SingleSsdInput::c_bntmhr`]); at `read_stride = 1` it is
73    /// `chunk_len` and every shape here is the stock one.
74    /// - input: see [`Mamba3SingleSsdInput`]
75    /// - output `(y_bntmhp, final_state_bhpr)`:
76    ///   - `y_bntmhp`:           `[batch, nchunks, T, mimo_rank, nheads, per_head_dim]`
77    ///   - `final_state_bhpr`:   `[batch, nheads, per_head_dim, state_rank]`
78    #[allow(non_snake_case)]
79    pub fn single_ssd_minimal(self) -> (Tensor<6>, Tensor<4>) {
80        let input = self;
81        input.sanity();
82        let [batch, nchunks, chunk_len, mimo_rank, nheads, per_head_dim] = input.v_bnlmhp.dims();
83        let [.., state_rank] = input.b_bnlmhr.dims();
84        let device = &input.v_bnlmhp.device();
85        let read_stride = input.read_stride;
86        let chunk_tokens = Mamba3SsdPath::chunk_tokens(chunk_len, read_stride);
87        let fused = chunk_len * mimo_rank; // write (source) axis
88        let read = chunk_tokens * mimo_rank; // read (target) axis
89
90        assert!(nchunks >= 1, "sequence must be non-empty");
91        assert!(chunk_len > 0, "chunk_len must be positive");
92        assert_eq!(
93            [batch, nchunks, chunk_tokens, nheads],
94            input.gamma_bnth.dims(),
95            "gamma must align with the read axis"
96        );
97        assert_eq!(
98            [batch, nchunks, chunk_len, nheads],
99            input.scale_bnlh.dims(),
100            "scale must align with da"
101        );
102
103        // ── Fuse mimo_rank into the chunk axes (matches `ssd_minimal`) ────────
104        let c_bnTMhr =
105            input
106                .c_bntmhr
107                .clone()
108                .reshape([batch, nchunks, read, nheads, state_rank]);
109        let v_bnLMhp =
110            input
111                .v_bnlmhp
112                .clone()
113                .reshape([batch, nchunks, fused, nheads, per_head_dim]);
114
115        // Per-time-step cumulative log-decay (used for L_strict, decay_states, y_off)
116        let a_bhnl = input.da_bnlh.permute([0, 3, 1, 2]);
117        let a_cumsum_bhnl = a_bhnl.clone().cumsum(3);
118
119        // K scaled for lower-triangular and state recurrence paths
120        // (the diagonal correction reuses the unscaled `b_bnlmhr`).
121        // scale_bnlh broadcast over (mimo_rank, state_rank):
122        let scale_bnlh11 = input
123            .scale_bnlh
124            .clone()
125            .unsqueeze_dims::<6>(&[3, 5]) // scale_bnlh -> scale_bnl1h1
126            ;
127        let k_scaled_bnlmhr = input.b_bnlmhr.clone() * scale_bnlh11;
128        let k_scaled_bnLMhr =
129            k_scaled_bnlmhr.reshape([batch, nchunks, fused, nheads, state_rank]);
130
131        // =============================================================
132        // STEP 1a: Strict lower-triangular intra-chunk output (y_lower)
133        //
134        // y_lower[t1] = Σ_{t2 < t1}  (C[t1] · K_scaled[t2]^T)
135        //              · exp(cumA[t1] - cumA[t2])  · PsiV[t2]
136        // (block-diagonal in time t1 = t2 is excluded — handled by y_diag.)
137        // =============================================================
138        let y_lower_bnTMhp = {
139            let c_bnhTMr = c_bnTMhr.clone().swap_dims(2, 3);
140            let k_bnhrLM = k_scaled_bnLMhr.clone().permute([0, 1, 3, 4, 2]);
141            // [batch, nchunks, nheads, T*mimo_rank, chunk_len*mimo_rank]
142            let cb_bnhTMLM = c_bnhTMr.matmul(k_bnhrLM);
143
144            // L_strict_base[i, j] = exp(cumA[row(i)] - cumA[j]) for row(i) > j,
145            // else 0 — rectangular, the read rows against every source.
146            //
147            // Like `segsum` but with -inf on the diagonal as well (so exp = 0
148            // there), i.e. `triu(0)` rather than `triu(1)`.
149            let l_strict_base_bhntl = {
150                let x_cumsum = a_bhnl.clone().cumsum(3);
151                let row: Tensor<5> =
152                    helpers::read_rows::<4, 5>(x_cumsum.clone(), 3, read_stride).unsqueeze_dim(4); // [..., t, 1]
153                let col: Tensor<5> = x_cumsum.unsqueeze_dim(3); // [..., 1, l]
154                let diff = row - col; // [..., t, l]
155                let neg_inf_strict =
156                    helpers::read_causal_mask(chunk_len, read_stride, 0, device).unsqueeze::<5>();
157                (diff + neg_inf_strict).exp()
158            };
159
160            // Interleave-expand both axes (L_strict[i,j] = L_strict_base[i//m, j//m]):
161            let l_strict_bhnTMLM = l_strict_base_bhntl
162                .unsqueeze_dim::<6>(4)
163                .expand([batch, nheads, nchunks, chunk_tokens, mimo_rank, chunk_len])
164                .reshape([batch, nheads, nchunks, read, chunk_len])
165                .unsqueeze_dim::<6>(5)
166                .expand([batch, nheads, nchunks, read, chunk_len, mimo_rank])
167                .reshape([batch, nheads, nchunks, read, fused]);
168
169            // (CB ⊙ L_strict) · V    (back in MIMO-fused layout)
170            let cb_bnTMhLM = cb_bnhTMLM.swap_dims(2, 3);
171            let l_bnTMhLM = l_strict_bhnTMLM.permute([0, 2, 3, 1, 4]);
172            let masked_cb_bnhTMLM = (cb_bnTMhLM * l_bnTMhLM).swap_dims(2, 3);
173
174            let v_bnhLMp = v_bnLMhp.clone().swap_dims(2, 3);
175            let y_lower_bnhTMp = masked_cb_bnhTMLM.matmul(v_bnhLMp);
176
177            y_lower_bnhTMp.swap_dims(2, 3) // y_lower_bnTMhp
178        };
179
180        // =============================================================
181        // STEP 1b: γ-weighted same-step diagonal correction (y_diag)
182        //
183        // y_diag[t, m_out, h, p] = γₜ · Σ_{m_in} (C[t, m_out, h, ·] · B[t, m_in, h, ·]) · PsiV[t, m_in, h, p]
184        // =============================================================
185        // Same step means the *read* row, so V and B enter at those positions.
186        let y_diag_bntmhp = y_diag_correction(
187            helpers::read_rows::<6, 7>(input.v_bnlmhp, 2, read_stride),
188            helpers::read_rows::<6, 7>(input.b_bnlmhr, 2, read_stride),
189            input.c_bntmhr,
190            input.gamma_bnth,
191            input.siso_specialization,
192        );
193        // Reshape to fused layout for combination with y_lower / y_off.
194        let y_diag_bnTMhp =
195            y_diag_bntmhp.reshape([batch, nchunks, read, nheads, per_head_dim]);
196
197        // =============================================================
198        // STEP 2: Per-chunk single-ssd state (standard SSD with K_scaled)
199        //
200        // s[n] = Σ_{t,m} exp(cumA[n,-1] - cumA[n,t]) · V[n,t*M+m] · K_scaled[n,t*M+m]^T
201        // =============================================================
202        let state_bnhpr = {
203            let a_cumsum_last_bhn1 = a_cumsum_bhnl.clone().slice(s![.., .., .., -1]);
204            let a_cumsum_bhnLM = a_cumsum_bhnl
205                .clone()
206                .unsqueeze_dim::<5>(4)
207                .expand([batch, nheads, nchunks, chunk_len, mimo_rank])
208                .reshape([batch, nheads, nchunks, chunk_len * mimo_rank]);
209            let decay_bhnLM = (a_cumsum_last_bhn1 - a_cumsum_bhnLM).exp();
210
211            let decay_bnLMh1 = decay_bhnLM.permute([0, 2, 3, 1]).unsqueeze_dim(4);
212            let decayed_v_bnLMhp = decay_bnLMh1 * v_bnLMhp.clone();
213
214            let decayed_v_bnhpLM = decayed_v_bnLMhp.permute([0, 1, 3, 4, 2]);
215            let k_scaled_bnhLMr = k_scaled_bnLMhr.swap_dims(2, 3);
216            decayed_v_bnhpLM.matmul(k_scaled_bnhLMr) // state_bnhpr
217        };
218
219        // =============================================================
220        // STEP 3: Inter-chunk state scan (segsum-based state passing)
221        //
222        // h'[n] = Ā_chunk[n] · h'[n-1] + s[n]
223        // =============================================================
224        let (state_bnhpr, final_state_bhpr) = {
225            let initial_state_b1hpr = input.initial_state_bhpr.unsqueeze_dim(1);
226            let initial_state_b1hpr = if let Some(init_hpr) = input.init_state_hpr {
227                let init_b1hpr = init_hpr.unsqueeze_dim::<4>(0).expand([
228                    batch,
229                    1,
230                    nheads,
231                    per_head_dim,
232                    state_rank,
233                ]);
234                initial_state_b1hpr + init_b1hpr
235            } else {
236                initial_state_b1hpr
237            };
238
239            let state_bNhpr = Tensor::cat(vec![initial_state_b1hpr, state_bnhpr], 1);
240
241            let a_cumsum_last_bhn: Tensor<3> = a_cumsum_bhnl
242                .clone()
243                .slice(s![.., .., .., -1])
244                .squeeze_dim(3);
245            let a_chunk_pad_bhN = Tensor::cat(
246                vec![Tensor::zeros([batch, nheads, 1], device), a_cumsum_last_bhn],
247                2,
248            );
249            let decay_chunk_bhNN = segsum::<3, 4>(a_chunk_pad_bhN).exp();
250
251            let flat = per_head_dim * state_rank;
252            let state_bhNPR =
253                state_bNhpr
254                    .clone()
255                    .swap_dims(1, 2)
256                    .reshape([batch, nheads, 1 + nchunks, flat]);
257
258            let new_state_bhNPR = decay_chunk_bhNN.matmul(state_bhNPR);
259            let new_state_bhNpr =
260                new_state_bhNPR.reshape([batch, nheads, 1 + nchunks, per_head_dim, state_rank]);
261
262            let new_state_bnhpr = new_state_bhNpr
263                .clone()
264                .slice(s![.., .., 0..nchunks, .., ..])
265                .swap_dims(1, 2);
266            let last_state_bhpr: Tensor<4> = new_state_bhNpr
267                .slice(s![.., .., nchunks, .., ..])
268                .squeeze_dim(2);
269
270            (new_state_bnhpr, last_state_bhpr)
271        };
272
273        // =============================================================
274        // STEP 4: State-to-output (y_off)
275        //
276        // y_off[n, t*M+m] = C[t*M+m]^T · exp(cumA[t]) · h'[n-1]
277        // =============================================================
278        let y_off_bnTMhp = {
279            let state_decay_bhnTM = helpers::read_rows::<4, 5>(a_cumsum_bhnl, 3, read_stride)
280                .unsqueeze_dim::<5>(4)
281                .expand([batch, nheads, nchunks, chunk_tokens, mimo_rank])
282                .reshape([batch, nheads, nchunks, read])
283                .exp();
284
285            let c_bnhTMr = c_bnTMhr.swap_dims(2, 3);
286            let state_bnhrp = state_bnhpr.transpose();
287            let ch_bnhTMp = c_bnhTMr.matmul(state_bnhrp);
288
289            let decay_bnhTM1 = state_decay_bhnTM.swap_dims(1, 2).unsqueeze_dim(4);
290            let y_off_bnhTMp = ch_bnhTMp * decay_bnhTM1;
291            y_off_bnhTMp.swap_dims(2, 3)
292        };
293
294        // ── Combine and reshape ───────────────────────────────────────────────
295        let y_bnTMhp = y_lower_bnTMhp + y_diag_bnTMhp + y_off_bnTMhp;
296        let y_bntmhp =
297            y_bnTMhp.reshape([batch, nchunks, chunk_tokens, mimo_rank, nheads, per_head_dim]);
298
299        (y_bntmhp, final_state_bhpr)
300    }
