1//! # Single-Pass SSD — Input Bundle
2//!
3//! Sibling of [`crate::mamba3::double_ssd::ssd::ssd_path`]. The double-ssd
4//! pathway runs the *standard* SSD once per term (γ-term and β-terms). The
5//! [`Mamba3SingleSsdInput`] of this module runs **one** merged SSD pass that
6//! absorbs all terms: it scales `K` by `scaleₜ = γₜ + νₜ₊ₗₐ₉ (+ νⁱⁿᵗₜ₊₁)`. The
7//! same-step diagonal contribution is different (it must use `γₜ`, not
8//! `scaleₜ`), and an explicit correction term inside each variant patches it.
9//!
10//! Reference kernels, in `state-spaces/mamba`:
11//! - `mamba_ssm/ops/triton/mamba3/mamba3_siso_fwd.py`
12//! - `mamba_ssm/ops/tilelang/mamba3/mamba3_mimo_fwd.py`
13//!
14//! The interface is MIMO-first (as the other burn-mamba SSD inputs), and
15//! `mimo_rank = 1` is the SISO case.
16//! [`Mamba3SsdPath`](crate::mamba3::ssd_path::Mamba3SsdPath), shared with the
17//! double-ssd pathway, selects the algorithm.
18
19use crate::mamba3::prelude::*;
20use burn::prelude::*;

22/// MIMO-first input bundle for the merged-form SSD.
23///
24/// The caller (`Mamba3::forward_single_ssd`) pre-processes all tensors:
25///
26/// - B/C are QK-normed, bias-added, rotated, and expanded to per-head,
27/// - V is the raw, *unscaled* MIMO-expanded value,
28/// - the log-decay `da` (`Δ·A`, or the Kalman decay) is pre-computed,
29/// - the two trapezoidal coefficients `gammaₜ` and `scaleₜ` are separate
30///   inputs, because the SSD does the K-scaling and the γ-weighted diagonal
31///   correction itself.
32///
33/// The caller adds the D skip and the Z gate.
34pub struct Mamba3SingleSsdInput {
35    /// Value tensor, MIMO-expanded but **not** trapezoidally scaled.
36    ///
37    /// # Shape
38    /// - `[batch, nchunks, chunk_len, mimo_rank, nheads, per_head_dim]`
39    pub v_bnlmhp: Tensor<6>,
40
41    /// K/B tensor: QK-normed, bias-added, rotated, expanded to per-head. Not
42    /// pre-scaled: the SSD multiplies it by `scaleₜ` for the lower-triangular
43    /// and state-recurrence paths, and the diagonal correction uses the
44    /// unscaled tensor.
45    ///
46    /// # Shape
47    /// - `[batch, nchunks, chunk_len, mimo_rank, nheads, state_rank]`
48    pub b_bnlmhr: Tensor<6>,
49
50    /// Q/C tensor: the same processing as `b_bnlmhr`, but on the **read** axis
51    /// of the chunk: one row per token, not per folded position.
52    ///
53    /// Every micro-step of a token writes to the state. The token is read
54    /// once, at its last micro-step
55    /// ([`micro_steps`](crate::mamba3::mamba3::Mamba3Config::micro_steps) is
56    /// [`Self::read_stride`]). So `C` is necessary only there, and the `y` that
57    /// this bundle returns is already at token resolution. See
58    /// [the read axis](crate::mamba3::product).
59    ///
60    /// # Shape
61    /// - `[batch, nchunks, chunk_tokens, mimo_rank, nheads, state_rank]`,
62    ///   `chunk_tokens = chunk_len / read_stride`
63    pub c_bntmhr: Tensor<6>,
64
65    /// Pre-computed log-decay (negative): `Δ·A`, or the Kalman decay.
66    ///
67    /// # Shape
68    /// - `[batch, nchunks, chunk_len, nheads]`
69    pub da_bnlh: Tensor<4>,
70
71    /// `γₜ = λₜ · Δₜ`: the per-token diagonal multiplier.
72    ///
73    /// On the **read** axis, like `c_bntmhr`: γ weights only the same-step term
74    /// of the output of a row (the state recurrence runs on `scale_bnlh`). So
75    /// the folded positions between the reads never use it.
76    ///
77    /// # Shape
78    /// - `[batch, nchunks, chunk_tokens, nheads]`
79    pub gamma_bnth: Tensor<4>,
80
81    /// `scaleₜ = γₜ + νₜ₊ₗₐ₉ (+ νⁱⁿᵗₜ₊₁)`: the factor on K for the
82    /// lower-triangular and state-recurrence paths. A shifted term is zero at
83    /// the last positions that its lag reaches past (the next call pays them).
84    ///
85    /// # Shape
86    /// - `[batch, nchunks, chunk_len, nheads]`
87    pub scale_bnlh: Tensor<4>,
88
89    /// Initial SSM hidden state (merged-form accumulator).
90    ///
91    /// When the call continues a prior call, this must already hold the
92    /// boundary β seed `Σⱼ νⱼ · Σₘ K_prev[j, m] ⊗ (x_prev[j] ⊙ mimo_xₘ)`. The
93    /// previous call could not add it, because it did not know the `ν` of the
94    /// first `lag` positions of this call.
95    ///
96    /// # Shape
97    /// - `[batch, nheads, per_head_dim, state_rank]`
98    pub initial_state_bhpr: Tensor<4>,
99
100    /// Optional learnable initial state (broadcast over batch). Only the
101    /// `Minimal` path supports it: the serial paths panic when it is `Some`.
102    ///
103    /// # Shape
104    /// - `[nheads, per_head_dim, state_rank]`
105    pub init_state_hpr: Option<Tensor<3>>,
106
107    /// The read stride of the chunk: `micro_steps`, the number of folded
108    /// positions (writes) in a token. `1` for stock Mamba-3, where the read
109    /// axis *is* the chunk and every shape below reduces to `chunk_len`.
110    ///
111    /// `chunk_len` is a multiple of it. So a chunk holds a whole number of
112    /// tokens, and their rows are the contiguous run
113    /// `[chunk · chunk_tokens, (chunk+1) · chunk_tokens)`.
114    pub read_stride: usize,
115
116    /// Whether the specialized `mimo_rank == 1` γ-correction can be used
117    /// ([`Mamba3Config::siso_specialization`](crate::mamba3::mamba3::Mamba3Config::siso_specialization),
118    /// the chunkwise flag; the per-token sites have their own).
119    ///
120    /// Performance-only: both branches of
121    /// [`y_diag_correction`](crate::mamba3::single_ssd::ssd::diag::y_diag_correction)
122    /// compute the same values and gradients. No effect at `mimo_rank > 1`.
123    pub siso_specialization: bool,
124}
125
126impl Mamba3SingleSsdInput {
127    /// Run the [`NaN`/`Inf` guards](burn_stack::modules::misc::sanity) on every input tensor.
128    pub fn sanity(&self) {
129        use burn_stack::modules::sanity as san;
130        san(&self.v_bnlmhp);
131        san(&self.b_bnlmhr);
132        san(&self.c_bntmhr);
133        san(&self.da_bnlh);
134        san(&self.gamma_bnth);
135        san(&self.scale_bnlh);
136        san(&self.initial_state_bhpr);
137        if let Some(ref init_state_hpr) = self.init_state_hpr {
138            san(init_state_hpr);
139        }
140    }
141}
142
143impl Mamba3SingleSsdInput {
144    /// Run the selected merged-form (single-ssd) algorithm on this MIMO-first input.
145    ///
146    /// Dispatches by [`Mamba3SsdPath`] variant to `single_ssd_minimal`,
147    /// `single_ssd_serial`, or `single_ssd_serial_recalculated`.
148    ///
149    /// # Returns
150    /// - `y_bntmhp`: `[batch, nchunks, chunk_tokens, mimo_rank, nheads, per_head_dim]`
151    ///   — token resolution, the readout's own (see [`Self::c_bntmhr`])
152    /// - `final_state_bhpr`: `[batch, nheads, per_head_dim, state_rank]` —
153    ///   the merged-form accumulator at the last token (to be stored in the
154    ///   cache for streaming).
155    pub fn run(self, path: &Mamba3SsdPath) -> (Tensor<6>, Tensor<4>) {
156        match path {
157            Mamba3SsdPath::Minimal(_) => self.single_ssd_minimal(),
158            Mamba3SsdPath::Serial(_) => self.single_ssd_serial(),
159            Mamba3SsdPath::SerialRecalculated(_) => self.single_ssd_serial_recalculated(),
160        }
161    }
162}
