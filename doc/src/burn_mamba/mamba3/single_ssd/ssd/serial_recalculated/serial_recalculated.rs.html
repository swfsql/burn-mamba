1//! # Serial SSD with a custom, memory-efficient backward (Mamba-3 single-SSD)
2//!
3//! The `SerialRecalculated` path of the single-SSD pathway. The forward is the
4//! same serial scan as [`super::super::serial`], but it goes through the
5//! [`Mamba3SingleSsdBackendExt`] trait. So an `Autodiff` backend can use a
6//! custom backward that recomputes per-chunk intermediates instead of storing
7//! them (see [`super::backward`] / [`super::combined_backward`]). Unlike the
8//! double-SSD form, the kernels here apply the trapezoid `gamma`/`scale`
9//! themselves, so the backward also returns `d_gamma`/`d_scale`. (The caller
10//! folds the boundary-β seed into the initial accumulator.)
11//!
12//! The default body runs under a generic backend `B`, where the high-level
13//! `Tensor` (pinned to `Dispatch`) is not available. So the K1–K5 math uses
14//! the rank-tagged [`F`] primitive wrapper. K1–K4 do not depend on the form,
15//! and come from the [standard SSD scan](crate::standard_ssd). Only the
16//! forward K5 of single-SSD (strict-lower intra-chunk + γ-correction) is
17//! defined here. The backward of its LOWER and BLUE terms is in the
18//! [kernels of that scan](crate::standard_ssd::kernels). The backward of its
19//! DIAG term is in [`super::diag`].
20

38impl Mamba3SingleSsdInput {
39    /// MIMO-first single-ssd form Serial SSD with recalculated backward.
40    ///
41    /// Sends the full K1–K5 (single-ssd) computation to
42    /// [`Mamba3SingleSsdBackendExt::single_ssd_serial_recalculated`]. An
43    /// autodiff backend adds the memory-efficient custom backward. A plain
44    /// backend runs the standard K1–K5 forward.
45    ///
46    /// # Returns
47    /// - `y_bntmhp`:         `[batch, nchunks, chunk_tokens, mimo_rank, nheads, per_head_dim]`
48    /// - `final_state_bhpr`: `[batch, nheads, per_head_dim, state_rank]`
49    pub fn single_ssd_serial_recalculated(self) -> (Tensor<6>, Tensor<4>) {
50        let input = self;
51        input.sanity();
52        assert!(
53            input.init_state_hpr.is_none(),
54            "init_state_hpr not yet implemented for single_ssd_serial_recalculated"
55        );
56
57        let resets = crate::packing::SsdResets::op_inputs(input.resets, &input.v_bnlmhp);
58        let seeds = match input.seed_bnhpr {
59            Some(seed_bnhpr) => OpSeeds::Some { seed_bnhpr: seed_bnhpr.into_dispatch() },
60            None => OpSeeds::None,
61        };
62        let (y_bntmhp, final_state_bhpr) =
63            <Dispatch as Mamba3SingleSsdBackendExt>::single_ssd_serial_recalculated(
64                input.v_bnlmhp.into_dispatch(),
65                input.da_bnlh.into_dispatch(),
66                input.b_bnlmhr.into_dispatch(),
67                input.c_bntmhr.into_dispatch(),
68                input.gamma_bnth.into_dispatch(),
69                input.scale_bnlh.into_dispatch(),
70                input.initial_state_bhpr.into_dispatch(),
71                input.read_stride,
72                input.siso_specialization,
73                resets,
74                seeds,
75            );
76        let y_bntmhp = Tensor::from_dispatch(y_bntmhp);
77        let final_state_bhpr = Tensor::from_dispatch(final_state_bhpr);
78        (y_bntmhp, final_state_bhpr)
79    }
80}

82/// Extends the backend for the memory-efficient single-ssd form serial SSD.
83///
84/// The default implementation runs K1–K5 with primitive tensor operations:
85/// K1/K2/K3/K4 of the double-SSD forward (they do not depend on the form) and
86/// the single-ssd K5 below. The Autodiff wrapper overrides it: its backward
87/// recomputes the forward intermediates instead of saving them.
88#[backend_extension(
89    // Every `#[backend_extension]` site of this crate has the same list.
90    // Change them together. `build.rs` sets `cube_backend`.
91    Cube: cfg(cube_backend),
92    Flex:  cfg(feature = "backend-flex"),
93    NdArray:  cfg(feature = "backend-ndarray"),
94    LibTorch:  cfg(any(feature = "backend-tch-cpu", feature = "backend-tch-gpu")),
95    Autodiff:  cfg(feature = "autodiff"),
96)]
97pub trait Mamba3SingleSsdBackendExt: Backend {
98    /// Memory-efficient MIMO single-ssd form serial SSD.
99    ///
100    /// # Arguments
101    /// - `v_bnlmhp`:           `[batch, nchunks, chunk_len, mimo_rank, nheads, per_head_dim]`
102    /// - `da_bnlh`:            `[batch, nchunks, chunk_len, nheads]` — the log-decay
103    /// - `b_bnlmhr`:           `[batch, nchunks, chunk_len, mimo_rank, nheads, state_rank]`
104    /// - `c_bntmhr`:           `[batch, nchunks, chunk_tokens, mimo_rank, nheads, state_rank]`
105    /// - `gamma_bnth`:         `[batch, nchunks, chunk_tokens, nheads]` — `γₜ = λₜ Δₜ`
106    /// - `scale_bnlh`:         `[batch, nchunks, chunk_len, nheads]` — `scaleₜ` (see
107    ///   [`Mamba3SingleSsdInput::scale_bnlh`])
108    /// - `initial_state_bhpr`: `[batch, nheads, per_head_dim, state_rank]`
109    /// - `read_stride`:        `micro_steps` — write positions per read position, so
110    ///   `chunk_tokens = chunk_len / read_stride` (see
111    ///   [`Mamba3SingleSsdInput::read_stride`])
112    /// - `siso_specialization`: allow the specialized `mimo_rank == 1`
113    ///   γ-correction (performance-only; see
114    ///   [`Mamba3Config::siso_specialization`](crate::mamba3::mamba3::Mamba3Config::siso_specialization))
115    /// - `resets`:             the chunks that start a new segment of a packed
116    ///   row, and the accumulator that they start from, before their seed
117    ///   ([`OpResets`])
118    /// - `seeds`:              the boundary seed of each chunk ([`OpSeeds`])
119    ///
120    /// # Returns
121    /// - `y_bntmhp`:         `[batch, nchunks, chunk_tokens, mimo_rank, nheads, per_head_dim]`
122    /// - `final_state_bhpr`: `[batch, nheads, per_head_dim, state_rank]`
123    #[allow(clippy::too_many_arguments)]
124    fn single_ssd_serial_recalculated(
125        v_bnlmhp: FloatTensor<Self>,
126        da_bnlh: FloatTensor<Self>,
127        b_bnlmhr: FloatTensor<Self>,
128        c_bntmhr: FloatTensor<Self>,
129        gamma_bnth: FloatTensor<Self>,
130        scale_bnlh: FloatTensor<Self>,
131        initial_state_bhpr: FloatTensor<Self>,
132        read_stride: usize,
133        siso_specialization: bool,
134        #[extension_type] resets: OpResets<Self>,
135        #[extension_type] seeds: OpSeeds<Self>,
136    ) -> (FloatTensor<Self>, FloatTensor<Self>) {
137        // Default impl: replicate the single-ssd form K1–K5 on primitives.
138        let v_bnlmhp = F::<Self, 6>::new(v_bnlmhp);
139        let da_bnlh = F::<Self, 4>::new(da_bnlh);
140        let b_bnlmhr = F::<Self, 6>::new(b_bnlmhr);
141        let c_bntmhr = F::<Self, 6>::new(c_bntmhr);
142        let gamma_bnth = F::<Self, 4>::new(gamma_bnth);
143        let scale_bnlh = F::<Self, 4>::new(scale_bnlh);
144        let initial_state_bhpr = F::<Self, 4>::new(initial_state_bhpr);
145        let resets = resets.into_option();
146        let seed_bnhpr = seeds.into_option();
147
148        // K1 — chunk cumulative decay.
149        let (da_cumsum_bhnl, da_chunk_end_bhn) = k1_ssd_chunk_cumsum::<Self>(da_bnlh);
150        san(&da_cumsum_bhnl);
151
152        // K2 — CB matrix on unscaled B/C.
153        let cb_bnhTMLM = k2_ssd_bmm::<Self>(c_bntmhr.clone(), b_bnlmhr.clone());
154        san(&cb_bnhTMLM);
155
156        // K3 — chunk state on K_scaled = scaleₜ · B.
157        let scale_bnlh11 = scale_bnlh.clone().unsqueeze_dims::<6>(&[3, 5]);
158        let k_scaled_bnlmhr = b_bnlmhr.clone() * scale_bnlh11;
159        let intra_chunk_state_bnhpr =
160            k3_ssd_chunk_state::<Self>(v_bnlmhp.clone(), k_scaled_bnlmhr, da_cumsum_bhnl.clone());
161        san(&intra_chunk_state_bnhpr);
162
163        // K4 — sequential state passing.
164        let (chunk_input_state_bnhpr, final_state_bhpr) = k4_ssd_state_passing::<Self>(
165            intra_chunk_state_bnhpr,
166            da_chunk_end_bhn,
167            initial_state_bhpr,
168            resets,
169            seed_bnhpr,
170        );
171        san(&chunk_input_state_bnhpr);
172        san(&final_state_bhpr);
173
174        // K5 — single-ssd form chunk scan.
175        let y_bntmhp = k5_single_ssd_chunk_scan::<Self>(
176            da_cumsum_bhnl,
177            v_bnlmhp,
178            c_bntmhr,
179            b_bnlmhr,
180            cb_bnhTMLM,
181            gamma_bnth,
182            scale_bnlh,
183            chunk_input_state_bnhpr,
184            read_stride,
185            siso_specialization,
186        );
187        san(&y_bntmhp);
188
189        (y_bntmhp.inner(), final_state_bhpr.inner())
190    }
191}

193/// The boundary seeds of a single-SSD op, on the primitives of `B`. An op
194/// argument of this type has `#[extension_type]`. This enum takes the place of
195/// an `Option`, for the reason of [`OpResets`].
196#[derive(ExtensionType)]
197pub enum OpSeeds<B: Backend> {
198    /// No seeds: the call has no resets.
199    None,
200    /// The seeds of a call with resets.
201    Some {
202        /// `[batch, nchunks, nheads, per_head_dim, state_rank]`: the boundary
203        /// seed of each chunk (see [`Mamba3SingleSsdInput::seed_bnhpr`]).
204        seed_bnhpr: FloatTensor<B>,
205    },
206}
207
208impl<B: Backend> OpSeeds<B> {
209    /// The seeds that K4 takes.
210    pub(crate) fn into_option(self) -> Option<F<B, 5>> {
211        match self {
212            OpSeeds::None => None,
213            OpSeeds::Some { seed_bnhpr } => Some(F::new(seed_bnhpr)),
214        }
215    }
216
217    /// The seeds from the two parts that a recompute node keeps (see
218    /// [`OpSeeds::into_recompute`]). `seed_bnhpr` stays unread when `seeded`
219    /// is `false`.
220    #[cfg(feature = "autodiff")]
221    pub(crate) fn from_recompute(seeded: bool, seed_bnhpr: FloatTensor<B>) -> Self {
222        if seeded { OpSeeds::Some { seed_bnhpr } } else { OpSeeds::None }
223    }
224}

226#[cfg(feature = "autodiff")]
227impl<B: Backend, C: CheckpointStrategy> OpSeeds<Autodiff<B, C>> {
228    /// The seeds as the `Autodiff` impl gives them to its recompute node:
229    ///
230    /// - `seeded`: `false` for `None`.
231    /// - `seed_bnhpr`: a tracked input. For `None`, it is a
232    ///   [`placeholder`](crate::packing::placeholder).
233    pub(crate) fn into_recompute(
234        self,
235        like: &FloatTensor<Autodiff<B, C>>,
236    ) -> (bool, FloatTensor<Autodiff<B, C>>) {
237        match self {
238            OpSeeds::Some { seed_bnhpr } => (true, seed_bnhpr),
239            OpSeeds::None => (false, crate::packing::placeholder::<B, C, 5>(like)),
240        }
241    }
242}

250// ─── Primitive forward K5 (single-ssd) ───────────────────────────────────────
251// Primitive port of [`super::super::serial::k5_single_ssd_chunk_scan`]. Combines
252// the strict-lower intra-chunk (LOWER), the γ-weighted diagonal (DIAG), and the
253// state-to-output (BLUE/Y_off) contributions. The backward of LOWER and BLUE is
254// `k5_ssd_chunk_scan_backward` of the standard SSD scan. The backward of DIAG is
255// in `super::diag`. K1–K4 come from the standard SSD scan (they do not depend on
256// the pathway).
257
258/// SingleSsd chunk scan, on primitives.
259///
260/// - **Strict lower triangular intra-chunk** (`t1 > t2`):
261///   `(cb[i,j] · scale[t2] · exp(cumA[t1] − cumA[t2])) · V[t2]`
262/// - **Diagonal (`t1 == t2`) γ-correction**:
263///   `γ[t] · (Σₙ C[t,r_out,n] · B[t,r_in,n]) · V[t,r_in,p]`
264/// - **State-to-output (Y_off)**: `exp(cumA[t]) · C[t] · h'[n-1]`
265///
266/// `cb_bnhTMLM` is the unscaled `C · Bᵀ` from K2; `b_bnlmhr` is the unscaled
267/// K/B tensor (γ-correction matmul). `T = chunk_tokens = chunk_len /
268/// read_stride` is the chunk's read axis.
269#[allow(clippy::too_many_arguments)]
270fn k5_single_ssd_chunk_scan<B: Backend>(
271    da_cumsum_bhnl: F<B, 4>,
272    v_bnlmhp: F<B, 6>,
273    c_bntmhr: F<B, 6>,
274    b_bnlmhr: F<B, 6>,
275    cb_bnhTMLM: F<B, 5>,
276    gamma_bnth: F<B, 4>,
277    scale_bnlh: F<B, 4>,
278    chunk_input_state_bnhpr: F<B, 5>,
279    read_stride: usize,
280    siso_specialization: bool,
281) -> F<B, 6> {
282    let [batch, nchunks, chunk_len, mimo_rank, nheads, per_head_dim] = v_bnlmhp.dims();
283    let [.., chunk_tokens, _, _, state_rank] = c_bntmhr.dims();
284    let device = v_bnlmhp.device();
285    let dtype = v_bnlmhp.dtype();
286    let fused = chunk_len * mimo_rank; // write (source) axis
287    let read = chunk_tokens * mimo_rank; // read (target) axis
288
289    // Fuse mimo_rank into the chunk axes for the SSM-style matmuls.
290    let v_bnLMhp = v_bnlmhp
291        .clone()
292        .reshape([batch, nchunks, fused, nheads, per_head_dim]);
293    let c_bnTMhr = c_bntmhr
294        .clone()
295        .reshape([batch, nchunks, read, nheads, state_rank]);
296
297    let (da_cumsum_bhnLM, da_cumsum_bhnTM) =
298        k5_da_cumsum_on_chunk_axes::<B>(da_cumsum_bhnl, mimo_rank, read_stride);
299
300    // ── Y_off: exp(cumA[t]) · C[t] · h'[n-1] ────────────────────────────────
301    let exp_da_bnhTMp = da_cumsum_bhnTM
302        .clone()
303        .exp()
304        .swap_dims(1, 2) // bnhTM
305        .unsqueeze_dim::<5>(4) // bnhTM1
306        .expand([batch, nchunks, nheads, read, per_head_dim]);
307    let c_bnhTMr = c_bnTMhr.swap_dims(2, 3);
308    let chunk_input_state_bnhrp = chunk_input_state_bnhpr.transpose();
309    let ch_bnhTMp = c_bnhTMr.matmul(chunk_input_state_bnhrp);
310    let y_off_bnhTMp = ch_bnhTMp * exp_da_bnhTMp;
311
312    // ── Y_lower: strict lower-tri intra-chunk with scale and decay ──────────
313    let target_da_cumsum_bnhTMLM = da_cumsum_bhnTM
314        .swap_dims(1, 2) // bnhTM
315        .unsqueeze_dim::<5>(4) // bnhTM1
316        .expand([batch, nchunks, nheads, read, fused]);
317    let source_da_cumsum_bnhTMLM = da_cumsum_bhnLM
318        .swap_dims(1, 2) // bnhLM
319        .unsqueeze_dim::<5>(3) // bnh1LM
320        .expand([batch, nchunks, nheads, read, fused]);
321    let diff_bnhTMLM = target_da_cumsum_bnhTMLM - source_da_cumsum_bnhTMLM;
322
323    // Strict-upper -inf mask over (read position, write source) on the base time
324    // grid, then interleave-expand both axes so MIMO same-time blocks are
325    // zeroed.
326    let inf_upper_bnhTMLM = expand_causal_mask(
327        read_causal_mask::<B>(chunk_len, read_stride, 0, &device, dtype),
328        batch,
329        nchunks,
330        nheads,
331        mimo_rank,
332    );
333    let decay_strict_bnhTMLM = (diff_bnhTMLM + inf_upper_bnhTMLM).exp();
334
335    // Per-column scale: `scale[t2]` lives on the source axis (column).
336    let scale_bnhLM = scale_bnlh
337        .transpose() // bnhl
338        .unsqueeze_dim::<5>(4) // bnhl1
339        .expand([batch, nchunks, nheads, chunk_len, mimo_rank])
340        .reshape([batch, nchunks, nheads, fused]);
341    let scale_col_bnhTMLM = scale_bnhLM
342        .unsqueeze_dim::<5>(3) // bnh1LM
343        .expand([batch, nchunks, nheads, read, fused]);
344
345    let kernel_bnhTMLM = decay_strict_bnhTMLM * scale_col_bnhTMLM;
346    let masked_cb_bnhTMLM = cb_bnhTMLM * kernel_bnhTMLM;
347    let v_bnhLMp = v_bnLMhp.swap_dims(2, 3);
348    let y_lower_bnhTMp = masked_cb_bnhTMLM.matmul(v_bnhLMp);
349
350    // ── Y_diag: γ-weighted diagonal correction, at the read positions ────────────
351    let y_diag_bntmhp = y_diag_correction::<B>(
352        read_positions::<B, 6, 7>(v_bnlmhp, 2, read_stride),
353        read_positions::<B, 6, 7>(b_bnlmhr, 2, read_stride),
354        c_bntmhr,
355        gamma_bnth,
356        siso_specialization,
357    );
358    let y_diag_bnTMhp = y_diag_bntmhp.reshape([batch, nchunks, read, nheads, per_head_dim]);
359    let y_diag_bnhTMp = y_diag_bnTMhp.swap_dims(2, 3);
360
361    // ── Combine and reshape ─────────────────────────────────────────────────
362    let y_bnhTMp = y_off_bnhTMp + y_lower_bnhTMp + y_diag_bnhTMp;
363    let y_bnTMhp = y_bnhTMp.swap_dims(2, 3);
364    y_bnTMhp.reshape([batch, nchunks, chunk_tokens, mimo_rank, nheads, per_head_dim])
365}
