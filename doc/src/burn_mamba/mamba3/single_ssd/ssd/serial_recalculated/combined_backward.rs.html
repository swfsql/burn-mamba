1//! # Recompute-based gradient math for the Mamba-3 single-SSD
2//!
3//! The analytic backward of the single-pass MIMO-first scan.  Forward
4//! intermediates (K1–K4) are recomputed from the saved leaf inputs, then every
5//! chunk-**local** gradient — the K5 state-to-output (BLUE), the strict
6//! lower-triangular intra-chunk (LOWER) and the γ-weighted same-step (DIAG)
7//! terms — is computed batched, the first two a chunk group at a time
8//! ([`Mamba3SsdPath::backward_chunk_group`](crate::mamba3::ssd_path::Mamba3SsdPath::backward_chunk_group)).  Only the K4 state-passing
9//! backward is a walk, being the one recurrence.  Because this pathway applies
10//! the trapezoid weights internally, it additionally returns `d_gamma` and
11//! `d_scale`.  The shared K3 extended helper (and K1/K2/K4, forward and
12//! backward) are reused from the double-SSD module.
13//!
14//! Everything operates on backend **primitives** through the rank-tagged `F`
15//! wrapper: the custom [`Backward`](burn::backend::autodiff::ops::Backward) node
16//! runs with a generic backend `B`, so the high-level `Tensor` is unavailable
17//! and the math uses `B`'s `float_*` ops.
18

34/// Per-input gradients produced by [`combined_backward`] for the Single-SSD.
35/// Adds `d_gamma_bnlh` and `d_scale_bnlh` over the double-ssd form
36/// [`crate::mamba3::double_ssd::ssd::serial_recalculated::combined_backward::CombinedGrads`].
37#[non_exhaustive]
38pub struct CombinedSingleSsdGrads<B: Backend> {
39    /// Gradient of the raw input `v`.
40    pub d_v_bnlmhp: F<B, 6>,
41    /// Gradient of `Δ·A` (`da`).
42    pub d_da_bnlh: F<B, 4>,
43    /// Gradient of the input projection `B`.
44    pub d_b_bnlmhr: F<B, 6>,
45    /// Gradient of the output projection `C`, on the chunk's read axis.
46    pub d_c_bntmhr: F<B, 6>,
47    /// Gradient of the same-step trapezoid weight `γ`, on the read axis.
48    pub d_gamma_bnth: F<B, 4>,
49    /// Gradient of the key scale `scale = γ + (1−λ₊₁)·Δ₊₁`.
50    pub d_scale_bnlh: F<B, 4>,
51    /// Gradient of the initial SSM state.
52    pub d_initial_state_bhpr: F<B, 4>,
53}
54
55/// Memory-efficient backward for the Mamba-3 MIMO-first chunkwise Single-SSD.
56///
57/// Recomputes the forward intermediates (K1–K4) from the saved inputs, then:
58/// - computes the K5 BLUE (state-to-output) and strict lower-triangular LOWER
59///   (intra-chunk) backwards a chunk group at a time (the score is what that
60///   group prices — see [`Mamba3SsdPath::backward_chunk_group`]),
61/// - computes the γ-weighted same-step DIAG backward batched (its `m × m`
62///   working tensors are tiny), and
63/// - walks the K4 state-passing backward, the one recurrence, over the
64///   chunk-input-state gradient the first step produced.
65///
66/// K3/K2/K1 backwards then run as single batched ops.
67///
68/// # Arguments
69/// - `d_y_bntmhp` — upstream gradient of the SSD output (read axis)
70/// - `d_final_bhpr` — upstream gradient of the final SSM state
71/// - `v_bnlmhp`, `da_bnlh`, `b_bnlmhr`, `c_bntmhr`, `gamma_bnth`, `scale_bnlh`,
72///   `initial_state_bhpr` — the seven saved forward inputs
73/// - `read_stride` — folded positions per read row (see
74///   [`Mamba3SingleSsdInput::read_stride`](crate::mamba3::single_ssd::ssd::Mamba3SingleSsdInput))
75/// - `siso_specialization` — the forward's γ-correction branch choice, replayed
76///   here so the backward matches it (performance-only; both agree)
77///
78/// # Returns
79/// One [`CombinedSingleSsdGrads`] with gradients for all 7 inputs.
80#[allow(clippy::too_many_arguments)]
81pub fn combined_backward<B: Backend>(
82    d_y_bntmhp: F<B, 6>,
83    d_final_bhpr: F<B, 4>,
84    //
85    v_bnlmhp: F<B, 6>,
86    da_bnlh: F<B, 4>,
87    b_bnlmhr: F<B, 6>,
88    c_bntmhr: F<B, 6>,
89    gamma_bnth: F<B, 4>,
90    scale_bnlh: F<B, 4>,
91    initial_state_bhpr: F<B, 4>,
92    read_stride: usize,
93    siso_specialization: bool,
94) -> CombinedSingleSsdGrads<B> {
95    let [batch, nchunks, chunk_len, mimo_rank, nheads, per_head_dim] = v_bnlmhp.dims();
96    let [.., state_rank] = b_bnlmhr.dims();
97    let device = v_bnlmhp.device();
98    let dtype = v_bnlmhp.dtype();
99    // The chunk's two axes: `fused` is what it writes, `read` what it is read
100    // at. They coincide unless `micro_steps > 1`.
101    let chunk_tokens = Mamba3SsdPath::chunk_tokens(chunk_len, read_stride);
102    let fused = chunk_len * mimo_rank;
103    let read = chunk_tokens * mimo_rank;
104
105    san(&d_y_bntmhp);
106    san(&d_final_bhpr);
107    san(&v_bnlmhp);
108    san(&da_bnlh);
109    san(&b_bnlmhr);
110    san(&c_bntmhr);
111    san(&gamma_bnth);
112    san(&scale_bnlh);
113    san(&initial_state_bhpr);
114
115    // ═══════════════════════════════════════════════════════════════════════
116    // RECOMPUTE FORWARD INTERMEDIATES (K1–K4, single-ssd form)
117    // ═══════════════════════════════════════════════════════════════════════
118
119    // K1
120    let (da_cumsum_bhnl, da_chunk_end_bhn) = k1_ssd_chunk_cumsum(da_bnlh.clone());
121    san(&da_cumsum_bhnl);
122
123    // K2 — CB matrix (unscaled), used by LOWER.
124    let cb_bnhTMLM = k2_ssd_bmm(c_bntmhr.clone(), b_bnlmhr.clone());
125    san(&cb_bnhTMLM);
126
127    // K3 — chunk state from K_scaled = scaleₜ·B.
128    let scale_bnlh11 = scale_bnlh.clone().unsqueeze_dims::<6>(&[3, 5]);
129    let k_scaled_bnlmhr = b_bnlmhr.clone() * scale_bnlh11.clone();
130    let (intra_chunk_state_bnhpr, k3_decay_bhnLM, k3_decayed_v_bnLMhp) =
131        k3_ssd_chunk_state_extended(
132            v_bnlmhp.clone(),
133            k_scaled_bnlmhr.clone(),
134            da_cumsum_bhnl.clone(),
135        );
136
137    // K4 — chunk-input state stream consumed by BLUE.
138    let (chunk_input_state_bnhpr, _final_state_bhpr) = k4_ssd_state_passing(
139        intra_chunk_state_bnhpr,
140        da_chunk_end_bhn.clone(),
141        initial_state_bhpr,
142    );
143
144    // Fused-position cumulative decay, on the write axis and on the read one.
145    let da_cumsum_bhnLM = da_cumsum_bhnl
146        .clone()
147        .unsqueeze_dim::<5>(4)
148        .expand([batch, nheads, nchunks, chunk_len, mimo_rank])
149        .reshape([batch, nheads, nchunks, fused]);
150    let da_cumsum_bhnTM = read_rows::<B, 4, 5>(da_cumsum_bhnl.clone(), 3, read_stride)
151        .unsqueeze_dim::<5>(4)
152        .expand([batch, nheads, nchunks, chunk_tokens, mimo_rank])
153        .reshape([batch, nheads, nchunks, read]);
154
155    // d_y in (batch, nchunks, nheads, chunk_tokens·mimo_rank, per_head_dim) ordering.
156    let d_y_bnhTMp = d_y_bntmhp
157        .clone()
158        .reshape([batch, nchunks, read, nheads, per_head_dim])
159        .swap_dims(2, 3);
160    san(&d_y_bnhTMp);
161
162    // ═══════════════════════════════════════════════════════════════════════
163    // DIAG BACKWARD (batched — no recurrence; m × m working set is tiny)
164    //
165    // Forward (per (b,n,l,h)):
166    //   qk_dot[m_out, m_in] = Σ_r C[m_out, r] · B[m_in, r]
167    //   y_diag[m_out, p]    = γ · Σ_{m_in} qk_dot[m_out, m_in] · V[m_in, p]
168    // ═══════════════════════════════════════════════════════════════════════
169    // The same step is the *read* row, so V and B enter at those positions and
170    // their gradients are scattered back onto the folded axis.
171    let DiagGrads {
172        d_v_bnlmhp: d_v_diag_bntmhp,
173        d_c_bnlmhr: d_c_diag_bntmhr,
174        d_b_bnlmhr: d_b_diag_bntmhr,
175        d_gamma_bnlh: d_gamma_bnth,
176    } = y_diag_correction_backward(
177        d_y_bntmhp.clone(),
178        read_rows::<B, 6, 7>(v_bnlmhp.clone(), 2, read_stride),
179        read_rows::<B, 6, 7>(b_bnlmhr.clone(), 2, read_stride),
180        c_bntmhr.clone(),
181        gamma_bnth.clone(),
182        siso_specialization,
183    );
184    let d_v_diag_bnlmhp = scatter_read_rows::<B, 6, 7>(d_v_diag_bntmhp, 2, read_stride);
185    let d_b_diag_bnlmhr = scatter_read_rows::<B, 6, 7>(d_b_diag_bntmhr, 2, read_stride);
186    san(&d_gamma_bnth);
187
188    // Reusable [chunk_tokens, chunk_len] -inf strict-upper mask (triu(0) on the
189    // folded grid: on+above diagonal → -inf) for the LOWER (strict
190    // lower-triangular) path.
191    let neg_inf_strict_tl: F<B, 2> =
192        read_causal_mask::<B>(chunk_len, read_stride, 0, &device, dtype);
193
194    // ═══════════════════════════════════════════════════════════════════════
195    // CHUNK-LOCAL BACKWARD — K5 (BLUE + LOWER), batched over a chunk group
196    //
197    // Neither term is recurrent: both need chunk `i`'s own slices plus
198    // `chunk_input_state[i]`, which the recomputed K4 already produced batched.
199    // So this side mirrors the forward's own batched K5, `group` chunks at a
200    // time — the group being what prices the score `[b, group, h, TM, LM]`
201    // against that forward's peak (see [`Mamba3SsdPath::backward_chunk_group`]).
202    // The state gradient, the one part that *is* a scan, follows below.
203    // ═══════════════════════════════════════════════════════════════════════
204    let group = Mamba3SsdPath::backward_chunk_group(nchunks);
205    let ngroups = nchunks.div_ceil(group);
206    let mut vec_lower_d_v_bnhLMp: Vec<F<B, 5>> = Vec::with_capacity(ngroups);
207    let mut vec_blue_d_c_bnhTMr: Vec<F<B, 5>> = Vec::with_capacity(ngroups);
208    let mut vec_d_cb_bnhTMLM: Vec<F<B, 5>> = Vec::with_capacity(ngroups);
209    let mut vec_blue_d_da_bnhl: Vec<F<B, 4>> = Vec::with_capacity(ngroups);
210    let mut vec_lower_d_da_bnhl: Vec<F<B, 4>> = Vec::with_capacity(ngroups);
211    let mut vec_lower_d_scale_bnhl: Vec<F<B, 4>> = Vec::with_capacity(ngroups);
212    let mut vec_d_chunk_input_state_bnhpr: Vec<F<B, 5>> = Vec::with_capacity(ngroups);
213
214    for start in (0..nchunks).step_by(group) {
215        // The group's own chunk count — `group`, short at the tail. It is the
216        // same `n` axis throughout, just narrowed, so the names do not change.
217        let n = group.min(nchunks - start);
218
219        // ── Group slices (fused chunk_len · mimo_rank) ─────────────────
220        let v_bnhLMp: F<B, 5> = v_bnlmhp
221            .clone()
222            .narrow(1, start, n)
223            .reshape([batch, n, fused, nheads, per_head_dim])
224            .swap_dims(2, 3);
225
226        let c_bnhTMr: F<B, 5> = c_bntmhr
227            .clone()
228            .narrow(1, start, n)
229            .reshape([batch, n, read, nheads, state_rank])
230            .swap_dims(2, 3);
231
232        let cb_bnhTMLM: F<B, 5> = cb_bnhTMLM.clone().narrow(1, start, n);
233
234        let da_cumsum_bnhLM: F<B, 4> = da_cumsum_bhnLM.clone().narrow(2, start, n).swap_dims(1, 2);
235        let da_cumsum_bnhTM: F<B, 4> = da_cumsum_bhnTM.clone().narrow(2, start, n).swap_dims(1, 2);
236
237        // scaleₜ per fused source position: scale[s_time] broadcast over s_m.
238        let scale_bnhLM: F<B, 4> = scale_bnlh
239            .clone()
240            .narrow(1, start, n) // [b, n, l, h]
241            .swap_dims(2, 3) // [b, n, h, l]
242            .unsqueeze_dim::<5>(4) // [b, n, h, l, 1]
243            .expand([batch, n, nheads, chunk_len, mimo_rank])
244            .reshape([batch, n, nheads, fused]);
245
246        let chunk_input_state_bnhpr: F<B, 5> = chunk_input_state_bnhpr.clone().narrow(1, start, n);
247        san(&chunk_input_state_bnhpr);
248
249        let d_y_bnhTMp: F<B, 5> = d_y_bnhTMp.clone().narrow(1, start, n);
250
251        // ── BLUE backward (identical to double-ssd form) ─────────────────
252        let exp_da_cumsum_bnhTMp: F<B, 5> = da_cumsum_bnhTM
253            .clone()
254            .exp()
255            .unsqueeze_dim::<5>(4)
256            .expand([batch, n, nheads, read, per_head_dim]);
257        let d_ch_bnhTMp: F<B, 5> = d_y_bnhTMp.clone() * exp_da_cumsum_bnhTMp.clone();
258        san(&d_ch_bnhTMp);
259
260        let d_chunk_input_state_bnhpr: F<B, 5> = c_bnhTMr
261            .clone()
262            .transpose() // c_bnhrTM
263            .matmul(d_ch_bnhTMp.clone()) // bnhrp
264            .transpose(); // bnhpr
265        san(&d_chunk_input_state_bnhpr);
266        vec_d_chunk_input_state_bnhpr.push(d_chunk_input_state_bnhpr);
267
268        let d_c_blue_bnhTMr: F<B, 5> = d_ch_bnhTMp.matmul(chunk_input_state_bnhpr.clone());
269        vec_blue_d_c_bnhTMr.push(d_c_blue_bnhTMr);
270
271        let ch_bnhTMp: F<B, 5> = c_bnhTMr
272            .clone()
273            .matmul(chunk_input_state_bnhpr.transpose());
274        let d_da_blue_bnhTM: F<B, 4> = (d_y_bnhTMp.clone() * ch_bnhTMp * exp_da_cumsum_bnhTMp)
275            .sum_dim(4)
276            .squeeze_dim::<4>(4);
277        // The row's own `da` is one of the folded positions, so its gradient
278        // goes back to that one and nowhere else.
279        let d_da_blue_bnhl: F<B, 4> = scatter_read_rows::<B, 4, 5>(
280            d_da_blue_bnhTM
281                .reshape([batch, n, nheads, chunk_tokens, mimo_rank])
282                .sum_dim(4)
283                .squeeze_dim::<4>(4),
284            3,
285            read_stride,
286        );
287        vec_blue_d_da_bnhl.push(d_da_blue_bnhl);
288
289        // ── LOWER backward (strict lower-tri + per-column scale) ────────
290        let da_target_bnhTMLM: F<B, 5> = da_cumsum_bnhTM
291            .unsqueeze_dim::<5>(4)
292            .expand([batch, n, nheads, read, fused]);
293        let da_source_bnhTMLM: F<B, 5> = da_cumsum_bnhLM
294            .unsqueeze_dim::<5>(3)
295            .expand([batch, n, nheads, read, fused]);
296        let diff_bnhTMLM = da_target_bnhTMLM - da_source_bnhTMLM;
297
298        // Strict-lower MIMO mask: -inf where s_time ≥ t_time — interleaved
299        // expansion of the [l, l] strict-upper (triu(0)) base mask.
300        let neg_inf_mimo_bnhTMLM: F<B, 5> = neg_inf_strict_tl
301            .clone()
302            .unsqueeze_dims::<5>(&[0, 1, 2])
303            .expand([batch, n, nheads, chunk_tokens, chunk_len])
304            .unsqueeze_dim::<6>(4)
305            .expand([batch, n, nheads, chunk_tokens, mimo_rank, chunk_len])
306            .reshape([batch, n, nheads, read, chunk_len])
307            .unsqueeze_dim::<6>(5)
308            .expand([batch, n, nheads, read, chunk_len, mimo_rank])
309            .reshape([batch, n, nheads, read, fused]);
310        let decay_strict_bnhTMLM = (diff_bnhTMLM + neg_inf_mimo_bnhTMLM).exp();
311        san(&decay_strict_bnhTMLM);
312
313        let scale_col_bnhTMLM: F<B, 5> = scale_bnhLM
314            .unsqueeze_dim::<5>(3) // [b,n,h,1,LMs]
315            .expand([batch, n, nheads, read, fused]);
316
317        // w = cb · decay_strict · scale_col
318        let prod_bnhTMLM = cb_bnhTMLM.clone() * decay_strict_bnhTMLM.clone();
319        let w_bnhTMLM = prod_bnhTMLM.clone() * scale_col_bnhTMLM.clone();
320
321        // d_w = d_y · vᵀ
322        let d_w_bnhTMLM: F<B, 5> = d_y_bnhTMp.clone().matmul(v_bnhLMp.transpose());
323        san(&d_w_bnhTMLM);
324
325        // d_v_lower = wᵀ · d_y
326        let d_v_lower_bnhLMp: F<B, 5> = w_bnhTMLM.transpose().matmul(d_y_bnhTMp);
327        san(&d_v_lower_bnhLMp);
328        vec_lower_d_v_bnhLMp.push(d_v_lower_bnhLMp);
329
330        // d_prod = d_w · scale_col ; d_scale_at = d_w · prod
331        let d_prod_bnhTMLM = d_w_bnhTMLM.clone() * scale_col_bnhTMLM;
332        let d_scale_at_bnhTMLM = d_w_bnhTMLM * prod_bnhTMLM;
333
334        // d_cb_lower = d_prod · decay_strict
335        let d_cb_lower_bnhTMLM = d_prod_bnhTMLM.clone() * decay_strict_bnhTMLM.clone();
336        vec_d_cb_bnhTMLM.push(d_cb_lower_bnhTMLM);
337
338        // d_decay_strict = d_prod · cb ; d_diff = d_decay_strict · decay_strict
339        let d_decay_strict_bnhTMLM = d_prod_bnhTMLM * cb_bnhTMLM;
340        let d_diff_bnhTMLM = d_decay_strict_bnhTMLM * decay_strict_bnhTMLM;
341
342        // The target term lands on the read rows (scattered back onto the
343        // folded axis), the source term on every folded position.
344        let d_da_target_bnhTM: F<B, 4> = d_diff_bnhTMLM.clone().sum_dim(4).squeeze_dim::<4>(4);
345        let d_da_source_bnhLM: F<B, 4> = d_diff_bnhTMLM.sum_dim(3).squeeze_dim::<4>(3);
346        let d_da_target_bnhl: F<B, 4> = scatter_read_rows::<B, 4, 5>(
347            d_da_target_bnhTM
348                .reshape([batch, n, nheads, chunk_tokens, mimo_rank])
349                .sum_dim(4)
350                .squeeze_dim::<4>(4),
351            3,
352            read_stride,
353        );
354        let d_da_source_bnhl: F<B, 4> = d_da_source_bnhLM
355            .reshape([batch, n, nheads, chunk_len, mimo_rank])
356            .sum_dim(4)
357            .squeeze_dim::<4>(4);
358        vec_lower_d_da_bnhl.push(d_da_target_bnhl - d_da_source_bnhl);
359
360        // d_scale[s_time] = Σ_{TMt, s_m} d_scale_at[TMt, LMs]
361        let d_scale_lower_bnhl: F<B, 4> = d_scale_at_bnhTMLM
362            .sum_dim(3) // sum over target TMt → [b,n,h,1,LMs]
363            .squeeze_dim::<4>(3) // [b,n,h,LMs]
364            .reshape([batch, n, nheads, chunk_len, mimo_rank])
365            .sum_dim(4) // sum over source mimo → [b,n,h,l,1]
366            .squeeze_dim::<4>(4); // [b,n,h,l]
367        vec_lower_d_scale_bnhl.push(d_scale_lower_bnhl);
368    }
369
370    // ── Rejoin the groups along the chunk axis ────────────────────────────
371    let d_v_lower_bnhLMp: F<B, 5> = cat_chunk_groups(vec_lower_d_v_bnhLMp, 1);
372    let d_c_blue_bnhTMr: F<B, 5> = cat_chunk_groups(vec_blue_d_c_bnhTMr, 1);
373    let d_cb_bnhTMLM: F<B, 5> = cat_chunk_groups(vec_d_cb_bnhTMLM, 1);
374    let d_da_blue_bhnl: F<B, 4> = cat_chunk_groups(vec_blue_d_da_bnhl, 1).swap_dims(1, 2);
375    let d_da_lower_bhnl: F<B, 4> = cat_chunk_groups(vec_lower_d_da_bnhl, 1).swap_dims(1, 2);
376    let d_scale_lower_bhnl: F<B, 4> = cat_chunk_groups(vec_lower_d_scale_bnhl, 1).swap_dims(1, 2);
377    let d_chunk_input_state_bnhpr: F<B, 5> = cat_chunk_groups(vec_d_chunk_input_state_bnhpr, 1);
378
379    // ── K4 backward — the reverse of the state-passing scan ───────────────
380    let (d_intra_chunk_state_bnhpr, d_da_end_bhn, d_initial_state_bhpr) =
381        k4_ssd_state_passing_backward(
382            d_chunk_input_state_bnhpr,
383            chunk_input_state_bnhpr,
384            da_chunk_end_bhn,
385            d_final_bhpr,
386        );
387    let d_da_cumsum_k4_bhnl: F<B, 4> = {
388        let zeros = F::<B, 4>::zeros([batch, nheads, nchunks, chunk_len - 1], &device, dtype);
389        let d_da_end_bhn1 = d_da_end_bhn.unsqueeze_dim::<4>(3);
390        F::cat(vec![zeros, d_da_end_bhn1], 3)
391    };
392
393    // ═══════════════════════════════════════════════════════════════════════
394    // K3 BACKWARD (batched) — K_scaled = scaleₜ·B
395    //
396    // intra_state = decayed_vᵀ @ K_scaled, with decayed_v = decay·V.
397    // d_K_scaled = decayed_vᵀ-contraction ; then split into d_b_k3 (·scale) and
398    // d_scale_k3 (Σ_{m,r} ·B).
399    // ═══════════════════════════════════════════════════════════════════════
400    let v_bnLMhp =
401        v_bnlmhp
402            .clone()
403            .reshape([batch, nchunks, chunk_len * mimo_rank, nheads, per_head_dim]);
404    let k_scaled_bnLMhr =
405        k_scaled_bnlmhr.reshape([batch, nchunks, chunk_len * mimo_rank, nheads, state_rank]);
406    let k_scaled_bnhLMr = k_scaled_bnLMhr.swap_dims(2, 3);
407    let decayed_v_bnhpLM = k3_decayed_v_bnLMhp.permute([0, 1, 3, 4, 2]);
408
409    let d_decayed_v_bnhpLM: F<B, 5> = d_intra_chunk_state_bnhpr
410        .clone()
411        .matmul(k_scaled_bnhLMr.clone().transpose()); // k_scaled_bnhrLM
412    let d_k_scaled_bnhLMr: F<B, 5> = decayed_v_bnhpLM
413        .transpose() // decayed_v_bnhLMp
414        .matmul(d_intra_chunk_state_bnhpr);
415
416    let d_decayed_v_bnLMhp = d_decayed_v_bnhpLM.permute([0, 1, 4, 2, 3]);
417    let d_decay_bhnLM: F<B, 4> = (d_decayed_v_bnLMhp.clone() * v_bnLMhp)
418        .sum_dim(4)
419        .squeeze_dim::<4>(4)
420        .permute([0, 3, 1, 2]);
421
422    let k3_decay_bnLMh1 = k3_decay_bhnLM
423        .clone()
424        .permute([0, 2, 3, 1])
425        .unsqueeze_dim::<5>(4);
426    let d_v_k3_bnLMhp: F<B, 5> = d_decayed_v_bnLMhp * k3_decay_bnLMh1;
427    let d_v_k3_bnlmhp: F<B, 6> =
428        d_v_k3_bnLMhp.reshape([batch, nchunks, chunk_len, mimo_rank, nheads, per_head_dim]);
429
430    // d(cumA_last − cumA) = d_decay · decay
431    let d_decay_times_decay_bhnLM = d_decay_bhnLM * k3_decay_bhnLM;
432    let d_a_cumsum_last_bhn: F<B, 3> = d_decay_times_decay_bhnLM
433        .clone()
434        .sum_dim(3)
435        .squeeze_dim::<3>(3);
436    let d_da_cumsum_bhnLM = -d_decay_times_decay_bhnLM;
437
438    let d_da_cumsum_k3_from_fused_bhnl: F<B, 4> = d_da_cumsum_bhnLM
439        .reshape([batch, nheads, nchunks, chunk_len, mimo_rank])
440        .sum_dim(4)
441        .squeeze_dim::<4>(4);
442    let d_da_cumsum_k3_from_last_bhnl: F<B, 4> = {
443        let zeros = F::<B, 4>::zeros([batch, nheads, nchunks, chunk_len - 1], &device, dtype);
444        let d_last = d_a_cumsum_last_bhn.unsqueeze_dim::<4>(3);
445        F::cat(vec![zeros, d_last], 3)
446    };
447    let d_da_cumsum_k3_bhnl = d_da_cumsum_k3_from_fused_bhnl + d_da_cumsum_k3_from_last_bhnl;
448
449    // d_K_scaled → bnlmhr, then split into d_b_k3 (·scale) and d_scale_k3 (Σ·B).
450    let d_k_scaled_bnlmhr: F<B, 6> = d_k_scaled_bnhLMr
451        .swap_dims(2, 3) // bnLMhr
452        .reshape([batch, nchunks, chunk_len, mimo_rank, nheads, state_rank]);
453    let d_b_k3_bnlmhr: F<B, 6> = d_k_scaled_bnlmhr.clone() * scale_bnlh11;
454    let d_scale_k3_bnlh: F<B, 4> = (d_k_scaled_bnlmhr * b_bnlmhr.clone())
455        .sum_dim(5) // sum over state_rank → [b,n,l,m,h,1]
456        .squeeze_dim::<5>(5) // [b,n,l,m,h]
457        .sum_dim(3) // sum over mimo_rank → [b,n,l,1,h]
458        .squeeze_dim::<4>(3); // [b,n,l,h]
459
460    // ═══════════════════════════════════════════════════════════════════════
461    // K2 BACKWARD (batched) — cb = C @ Bᵀ
462    // ═══════════════════════════════════════════════════════════════════════
463    let b_bnLMhr = b_bnlmhr
464        .clone()
465        .reshape([batch, nchunks, fused, nheads, state_rank]);
466    let c_bnhTMr = c_bntmhr
467        .clone()
468        .reshape([batch, nchunks, read, nheads, state_rank])
469        .swap_dims(2, 3);
470    let b_for_k2_bnhLMr = b_bnLMhr.swap_dims(2, 3);
471
472    let d_c_k2_bnhTMr: F<B, 5> = d_cb_bnhTMLM.clone().matmul(b_for_k2_bnhLMr);
473    let d_b_k2_bnhrLM: F<B, 5> = c_bnhTMr.transpose().matmul(d_cb_bnhTMLM);
474
475    let d_c_k2_bntmhr: F<B, 6> = d_c_k2_bnhTMr
476        .swap_dims(2, 3)
477        .reshape([batch, nchunks, chunk_tokens, mimo_rank, nheads, state_rank]);
478    let d_b_k2_bnlmhr: F<B, 6> = d_b_k2_bnhrLM
479        .permute([0, 1, 4, 2, 3])
480        .reshape([batch, nchunks, chunk_len, mimo_rank, nheads, state_rank]);
481
482    // ── Unstack d_c_blue / d_v_lower and reshape back ─────────────────────
483    let d_c_blue_bntmhr: F<B, 6> = d_c_blue_bnhTMr
484        .swap_dims(2, 3)
485        .reshape([batch, nchunks, chunk_tokens, mimo_rank, nheads, state_rank]);
486    let d_v_lower_bnlmhp: F<B, 6> = d_v_lower_bnhLMp.swap_dims(2, 3).reshape([
487        batch,
488        nchunks,
489        chunk_len,
490        mimo_rank,
491        nheads,
492        per_head_dim,
493    ]);
494
495    // ═══════════════════════════════════════════════════════════════════════
496    // K1 BACKWARD + SUM CONTRIBUTIONS
497    // ═══════════════════════════════════════════════════════════════════════
498    let d_da_cumsum_bhnl =
499        d_da_blue_bhnl + d_da_lower_bhnl + d_da_cumsum_k3_bhnl + d_da_cumsum_k4_bhnl;
500    san(&d_da_cumsum_bhnl);
501
502    // K1 inverse: suffix sum.
503    let d_da_bhnl = {
504        let d_total_bhnl = d_da_cumsum_bhnl
505            .clone()
506            .sum_dim(3)
507            .expand([batch, nheads, nchunks, chunk_len]);
508        let prefix_bhnl = d_da_cumsum_bhnl.cumsum(3);
509        let zeros_bhn1 = F::<B, 4>::zeros([batch, nheads, nchunks, 1], &device, dtype);
510        let prefix_shifted_bhnl =
511            F::cat(vec![zeros_bhn1, prefix_bhnl.narrow(3, 0, chunk_len - 1)], 3);
512        d_total_bhnl - prefix_shifted_bhnl
513    };
514    let d_da_bnlh = d_da_bhnl.permute([0, 2, 3, 1]);
515
516    // ── Combine per-input gradient contributions ──────────────────────────
517    let d_v_bnlmhp = d_v_k3_bnlmhp + d_v_lower_bnlmhp + d_v_diag_bnlmhp;
518    let d_b_bnlmhr = d_b_k2_bnlmhr + d_b_k3_bnlmhr + d_b_diag_bnlmhr;
519    let d_c_bntmhr = d_c_k2_bntmhr + d_c_blue_bntmhr + d_c_diag_bntmhr;
520    let d_scale_bnlh = d_scale_lower_bhnl.permute([0, 2, 3, 1]) + d_scale_k3_bnlh;
521
522    san(&d_v_bnlmhp);
523    san(&d_da_bnlh);
524    san(&d_b_bnlmhr);
525    san(&d_c_bntmhr);
526    san(&d_gamma_bnth);
527    san(&d_scale_bnlh);
528    san(&d_initial_state_bhpr);
529
530    CombinedSingleSsdGrads {
531        d_v_bnlmhp,
532        d_da_bnlh,
533        d_b_bnlmhr,
534        d_c_bntmhr,
535        d_gamma_bnth,
536        d_scale_bnlh,
537        d_initial_state_bhpr,
538    }
539}
