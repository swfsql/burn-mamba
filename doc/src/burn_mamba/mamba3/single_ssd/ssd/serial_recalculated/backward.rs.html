1//! # `RecomputeOp` of the Mamba-3 single-SSD
2//!
3//! Implements [`Mamba3SingleSsdBackendExt`] for `Autodiff<B>` with one
4//! [recompute node](burn_stack::utils::recompute_node). The node saves the
5//! inputs of the forward and not its large intermediates. The backward replays
6//! the serial kernels and the gradient math in
7//! [`super::combined_backward`](crate::mamba3::single_ssd::ssd::serial_recalculated::combined_backward).
8//!
9//! Without resets, `origin_bhpr` and `seed_bnhpr` are placeholders with one
10//! element (`packing::placeholder`), which the node saves at no real cost.
11
12use crate::mamba3::single_ssd::ssd::serial_recalculated::{
13    Mamba3SingleSsdBackendExt, OpSeeds, combined_backward::combined_backward,
14};
15use crate::packing::OpRestarts;
16use burn::backend::Backend;
17use burn::backend::autodiff::{Autodiff, checkpoint::strategy::CheckpointStrategy};
18use burn::backend::tensor::FloatTensor;
19use burn_stack::utils::fprim::F;
20use burn_stack::utils::recompute_node::{Prim, RecomputeOp, recompute_pair};
21
22burn_stack::recompute_inputs!(Inputs {

34/// The values that the op takes with no gradient.
35#[derive(Clone, Debug)]
36struct Extra<B: Backend> {
37    read_stride: usize,
38    siso_specialization: bool,
39    /// `None` without resets.
40    keep_bn: Option<Prim<B>>,
41    /// `false` without seeds.
42    seeded: bool,
43}
44
45/// The [`RecomputeOp`] of the Mamba-3 single-SSD.
46struct SsdRecompute;
47
48impl<B: Backend + Mamba3SingleSsdBackendExt> RecomputeOp<B, 9> for SsdRecompute {
49    type Inputs<T> = Inputs<T>;
50    type Extra = Extra<B>;
51
52    fn forward(inputs: Inputs<Prim<B>>, extra: &Extra<B>) -> (Prim<B>, Prim<B>) {
53        B::single_ssd_serial_recalculated(
54            inputs.v_bnlmhp,
55            inputs.da_bnlh,
56            inputs.b_bnlmhr,
57            inputs.c_bntmhr,
58            inputs.gamma_bnth,
59            inputs.scale_bnlh,
60            inputs.initial_state_bhpr,
61            extra.read_stride,
62            extra.siso_specialization,
63            OpRestarts::from_recompute(extra.keep_bn.clone(), inputs.origin_bhpr),
64            OpSeeds::from_recompute(extra.seeded, inputs.seed_bnhpr),
65        )
66    }
67
68    fn backward(
69        d_y_bntmhp: Prim<B>,
70        d_final_state_bhpr: Prim<B>,
71        inputs: Inputs<Prim<B>>,
72        extra: Extra<B>,
73    ) -> Inputs<Option<Prim<B>>> {
74        let grads = combined_backward::<B>(
75            F::new(d_y_bntmhp),
76            F::new(d_final_state_bhpr),
77            F::new(inputs.v_bnlmhp),
78            F::new(inputs.da_bnlh),
79            F::new(inputs.b_bnlmhr),
80            F::new(inputs.c_bntmhr),
81            F::new(inputs.gamma_bnth),
82            F::new(inputs.scale_bnlh),
83            F::new(inputs.initial_state_bhpr),
84            extra.read_stride,
85            extra.siso_specialization,
86            OpRestarts::from_recompute(extra.keep_bn, inputs.origin_bhpr).into_option(),
87            OpSeeds::from_recompute(extra.seeded, inputs.seed_bnhpr).into_option(),
88        );
89        Inputs {
90            v_bnlmhp: Some(grads.d_v_bnlmhp.inner()),
91            da_bnlh: Some(grads.d_da_bnlh.inner()),
92            b_bnlmhr: Some(grads.d_b_bnlmhr.inner()),
93            c_bntmhr: Some(grads.d_c_bntmhr.inner()),
94            gamma_bnth: Some(grads.d_gamma_bnth.inner()),
95            scale_bnlh: Some(grads.d_scale_bnlh.inner()),
96            initial_state_bhpr: Some(grads.d_initial_state_bhpr.inner()),
97            seed_bnhpr: grads.d_seed_bnhpr.map(F::inner),
98            origin_bhpr: grads.d_origin_bhpr.map(F::inner),
99        }
100    }
101}
102
103impl<B: Backend + Mamba3SingleSsdBackendExt, C: CheckpointStrategy> Mamba3SingleSsdBackendExt
104    for Autodiff<B, C>
105{
106    fn single_ssd_serial_recalculated(
107        v_bnlmhp: FloatTensor<Self>,
108        da_bnlh: FloatTensor<Self>,
109        b_bnlmhr: FloatTensor<Self>,
110        c_bntmhr: FloatTensor<Self>,
111        gamma_bnth: FloatTensor<Self>,
112        scale_bnlh: FloatTensor<Self>,
113        initial_state_bhpr: FloatTensor<Self>,
114        read_stride: usize,
115        siso_specialization: bool,
116        restarts: OpRestarts<Self>,
117        seeds: OpSeeds<Self>,
118    ) -> (FloatTensor<Self>, FloatTensor<Self>) {
119        let (keep_bn, origin_bhpr) = restarts.into_recompute(&v_bnlmhp);
120        let (seeded, seed_bnhpr) = seeds.into_recompute(&v_bnlmhp);
121        recompute_pair::<B, C, SsdRecompute, _>(
122            Inputs {
123                v_bnlmhp,
124                da_bnlh,
125                b_bnlmhr,
126                c_bntmhr,
127                gamma_bnth,
128                scale_bnlh,
129                initial_state_bhpr,
130                seed_bnhpr,
131                origin_bhpr,
132            },
133            Extra { read_stride, siso_specialization, keep_bn, seeded },
134        )
135    }
136}
