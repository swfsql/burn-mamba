1//! Shared helpers used by both [`Mamba3::forward`](super::mamba3::Mamba3::forward)
2//! and [`Mamba3::step`](super::mamba3::Mamba3::step). They isolate three blocks
3//! that previously appeared in both methods at different ranks:
4//!
5//! 1. Trapezoidal discretisation: `dt`, `α`, `β`, `γ`, `da`.
6//! 2. QK-norm + GQA expansion + per-(head, mimo-rank) bias on B / C.
7//! 3. MIMO `V` construction: broadcast-multiply `x` by `mimo_x_hmp`.
8//! 4. The rank-summed outer product `Σₘ v[m] ⊗ k[m]` feeding the SSM state
9//!    (SISO-branched).
10//! 5. Peeling the rotation channels off the in-projection.
11//! 6. The trapezoid tap's lag arithmetic: the shift, the gap transport, the
12//!    decay a cached slot carries across a call boundary, and the per-position
13//!    gate a tap pattern admits its taps by.
14//! 7. [`prefix_sum`]: the log-depth inclusive scan every sequence-length
15//!    cumulative sum goes through instead of `Tensor::cumsum`.
16//!
17//! Most helpers are generic over the rank `D` of the data tensors so a single
18//! definition serves both the sequence-aware (`forward`) and single-token
19//! (`step`) code paths. The discretisation is the exception: MambaProduct gives
20//! `step` a `u` axis of its own, so both paths reach it at rank 3.
21

28/// Peel a **trailing** in-projection segment of `width` channels off `proj`,
29/// yielding `None` for it when the block projects none.
30///
31/// Used for the two segments a block may omit entirely — the rotation channels
32/// ([`RotationKind::Real1D`](crate::mamba3::rotation::RotationKind::Real1D)
33/// projects none) and then `λ`
34/// ([`Trapezoid::None`](crate::mamba3::trapezoid::Trapezoid::None) projects
35/// none) — which is why the in-projection lays them out last, in that order.
36///
37/// An optional slice cannot simply be one more entry in the main `split_into`:
38/// Burn has no zero-width tensors, and `split_with_sizes` *drops* a zero-length
39/// segment rather than returning an empty one, so the destructuring would come
40/// up one part short.
41///
42/// # Shapes
43/// - `proj` : `[..., w]` along `dim`
44/// - out    : `[..., w − width]` and, if any, `[..., width]`
45pub fn split_trailing<const D: usize>(
46    proj: Tensor<D>,
47    width: usize,
48    dim: usize,
49) -> (Tensor<D>, Option<Tensor<D>>) {
50    if width == 0 {
51        return (proj, None);
52    }
53    let rest = proj.dims()[dim] - width;
54    let tail = proj.clone().narrow(dim, rest, width);
55    (proj.narrow(dim, 0, rest), Some(tail))
56}

58/// Shift a per-position stream back by `lag`, seeding the first `lag` positions
59/// from the cache's tap slots: `out[p] = stream[p − lag]`, and `prefix[p]` where
60/// that index is before the call.
61///
62/// This is the "shift-before-chunking" of the double-SSD pathway generalised
63/// from lag 1 to [`Trapezoid::tap_lag`](crate::mamba3::trapezoid::Trapezoid::tap_lag).
64/// A folded sequence is `tokens · u` long and `lag ∈ {1, u}`, so `sequence ≥
65/// lag` always holds; at equality the whole call is prefix.
66///
67/// # Shapes
68/// - `stream` : `[batch, sequence, …]`
69/// - `prefix` : `[batch, lag, …]`
70/// - out      : `[batch, sequence, …]`
71pub fn shift_stream<const D: usize>(
72    stream: Tensor<D>,
73    prefix: Tensor<D>,
74    lag: usize,
75) -> Tensor<D> {
76    let sequence = stream.dims()[1];
77    assert_eq!(prefix.dims()[1], lag, "one prefix slot per lagged position");
78    assert!(sequence >= lag, "a call is at least one token, i.e. `lag` long");
79    if sequence == lag {
80        prefix
81    } else {
82        Tensor::cat(vec![prefix, stream.narrow(1, 0, sequence - lag)], 1)
83    }
84}

86/// The in-block length [`prefix_sum`] scans directly.
87///
88/// A `cumsum` of length `L` over `N` elements costs `∝ N·L` (below), so a
89/// blocked scan costs `∝ N·(block + len/block²)` — one pass inside each block
90/// plus one over the `len/block` block totals. That is least at
91/// `block = ∛(2·len)`, which this walks up to on a power-of-two grid. The floor
92/// of 16 keeps the fixed cost (a reshape, a subtract, a broadcast add) amortised
93/// on short axes; the ceiling bounds the in-block quadratic.
94///
95/// `scan_block(len) < len` is exactly "this length takes the blocked branch",
96/// which the parity tests assert so that re-tuning the rule cannot silently stop
97/// covering it.
98pub(crate) fn scan_block(len: usize) -> usize {
99    let mut block = 16;
100    while block < 256 && block * block * block < 2 * len {
101        block *= 2;
102    }
103    block
104}

106/// Inclusive prefix sum along `dim`, continued from `init`:
107/// `out[i] = init + Σ_{j ≤ i} t[j]`.
108///
109/// The values [`Tensor::cumsum`] computes, but **blocked**: the scanned axis is
110/// split into runs of [`scan_block`], each run scanned on its own, then offset
111/// by the exclusive prefix of the run totals. Both of those scans are short and
112/// bounded, and the whole thing is a handful of ops whatever `len` is.
113///
114/// `init` is the scan's carry-in — for the cumulative rotation angle, the one
115/// the cache brings from the previous call, exactly as `quat_cumprod` takes its
116/// `init` quaternion. It rides the block offset rather than costing a pass of
117/// its own, which is what keeps this at parity with a plain `cumsum` **plus the
118/// caller's add** on a backend whose `cumsum` is already linear.
119///
120/// It exists because Burn's `cumsum` is **quadratic in the scanned length** on
121/// the cubecl backends: measured on CUDA over `[2, len, 32, 32]` along `dim 1`,
122/// 0.27 ms at `len = 256` rising to 143 ms at 4096, ≈4× per doubling — and the
123/// same scan on the trailing, contiguous axis is only 2.6× cheaper and still
124/// ≈4× per doubling, so it is the scan and not the stride. Every other scan in
125/// this crate runs over `chunk_len`, `lag` or `micro_steps`, all bounded, which
126/// is the regime this restores the angle scan to: the cumulative rotation angle
127/// is the one that runs over the whole **folded** sequence, and `micro_steps`
128/// multiplies that axis, so `u > 1` reaches the quadratic `u`× sooner. Blocked,
129/// the same case is 1.9× / 6.7× / 23× / 46× faster at `len = 256 / 512 / 1024 /
130/// 2048`, and the forward is near-flat in `len` (0.14 → 0.51 ms over that span).
131///
132/// The non-abelian sibling ([`quat_scan`](crate::mamba3::quat_scan)) solves the
133/// same problem with a Hillis–Steele doubling instead, having no `cumprod` to
134/// block with. Doubling works here too, and blocking beats it at **every** length
135/// measured (256 … 8192): 4.8× → 6.6× on CUDA forward, 2.3× → 3.3× with the
136/// backward, and 15× → 41× on the CPU backend. `O(len·log len)` work in
137/// `3·⌈log₂ len⌉` full-tensor kernels loses to `O(len·∛len)` in six, on hardware
138/// that is bandwidth- and launch-bound rather than FLOP-bound. So there is
139/// nothing here for a runtime knob to pick between.
140///
141/// The additions associate differently from a sequential scan, so the two agree
142/// to rounding rather than bit-for-bit.
143///
144/// # Shapes
145/// - `t`    : any rank, scanned along `dim`; `DP1` is `D + 1`
146/// - `init` : `t`'s shape with `1` along `dim` (it broadcasts along it)
147/// - out    : `t`'s shape
148pub fn prefix_sum<const D: usize, const DP1: usize>(
149    t: Tensor<D>,
150    dim: usize,
151    init: Option<Tensor<D>>,
152) -> Tensor<D> {
153    let len = t.dims()[dim];
154    let block = scan_block(len);
155    if len <= block {
156        let scanned = t.cumsum(dim);
157        return match init {
158            Some(init) => scanned + init,
159            None => scanned,
160        };
161    }
162    let device = t.device();
163
164    // Pad up to a whole number of blocks. Zero is the additive identity, so the
165    // padding contributes to no prefix, and it is narrowed off at the end.
166    let nblocks = len.div_ceil(block);
167    let pad = nblocks * block - len;
168    let t = if pad == 0 {
169        t
170    } else {
171        let mut pad_dims = t.dims();
172        pad_dims[dim] = pad;
173        Tensor::cat(vec![t, Tensor::zeros(pad_dims, &device)], dim)
174    };
175
176    // Split the scanned axis into (which block, where in it). Row-major, so the
177    // reshape moves nothing.
178    let dims = t.dims();
179    let mut split = [0usize; DP1];
180    split[..dim].copy_from_slice(&dims[..dim]);
181    split[dim] = nblocks;
182    split[dim + 1] = block;
183    split[dim + 2..].copy_from_slice(&dims[dim + 1..]);
184
185    let inner = t.reshape(split).cumsum(dim + 1);
186    // Each block's total is its last in-block prefix; kept at width 1 on that
187    // axis so the carry broadcasts back over the block. `init` joins the carry
188    // here, for free — one add serves both.
189    let totals = inner.clone().narrow(dim + 1, block - 1, 1);
190    let carry = totals.clone().cumsum(dim) - totals;
191    let carry = match init {
192        Some(init) => carry + init.unsqueeze_dim::<DP1>(dim + 1),
193        None => carry,
194    };
195
196    let joined = (inner + carry).reshape(dims);
197    if pad == 0 {
198        joined
199    } else {
200        joined.narrow(dim, 0, len)
201    }
202}

204// ---------------------------------------------------------------------------
205// The read axis
206// ---------------------------------------------------------------------------
207//
208// A chunk has two axes, and `micro_steps` widens only one of them. Every one of
209// a token's `u` micro-steps **writes** to the state, so the source axis is the
210// folded `chunk_len`; the token is **read** once, at its last micro-step, so
211// the target axis is `chunk_len / u` — the rows whose `y` survives, everything
212// else being multiplied by a zero gradient. `read_stride` is that `u` (and `1`
213// wherever there is no fold: Mamba-2, `micro_steps = 1`), which makes both
214// helpers below the identity there. `step` reads the axis too, its token being
215// a block of `u` positions with one read row.
216//
217// The alignment is what keeps this a reshape rather than a gather:
218// `chunk_len` is a multiple of `read_stride`
219// ([`Mamba3SsdPath::chunk_len_or_optimal`](crate::mamba3::ssd_path::Mamba3SsdPath::chunk_len_or_optimal)),
220// so chunk `c` covers folded positions `[c·L, (c+1)·L)` = tokens
221// `[c·T, (c+1)·T)` and its read rows are a **contiguous run** of them.
222
223/// The read rows of a folded axis: index `i·stride + (stride − 1)` for each
224/// `i`, i.e. the last position of every run of `stride`.
225///
226/// The identity at `stride = 1`.
227///
228/// # Shapes
229/// - `t`   : `[…, len, …]` at `dim`, `len` a multiple of `stride`
230/// - out   : `[…, len / stride, …]`
231///
232/// `DP1 = D + 1`.
233pub fn read_rows<const D: usize, const DP1: usize>(
234    t: Tensor<D>,
235    dim: usize,
236    stride: usize,
237) -> Tensor<D> {
238    if stride <= 1 {
239        return t;
240    }
241    let dims = t.dims();
242    let len = dims[dim];
243    assert_eq!(
244        len % stride,
245        0,
246        "a folded axis of {len} is not a whole number of {stride}-step tokens",
247    );
248    let mut split = [0usize; DP1];
249    split[..dim].copy_from_slice(&dims[..dim]);
250    split[dim] = len / stride;
251    split[dim + 1] = stride;
252    split[dim + 2..].copy_from_slice(&dims[dim + 1..]);
253
254    let mut out = dims;
255    out[dim] = len / stride;
256    // Row-major, so the split moves nothing and the narrow is a strided view.
257    t.reshape(split).narrow(dim + 1, stride - 1, 1).reshape(out)
258}

260/// The additive causal mask over `(read row, folded source)`:
261/// `0` where the source is early enough to contribute, `−∞` where it is not.
262///
263/// `diagonal` is passed to [`Tensor::triu`] on the **folded** grid, so it says
264/// the same thing it says today: `0` excludes the same position (the single-SSD
265/// strict mask, whose diagonal `ssd::diag` adds back with `γ`), `1` keeps it
266/// (the double-SSD inclusive mask).
267///
268/// `−∞` rather than a `0/1` multiply because the mask is added *before* the
269/// `exp`: for a source after the row the decay difference is positive and would
270/// overflow.
271///
272/// # Shapes
273/// - out : `[chunk_len / stride, chunk_len]`
274pub fn read_causal_mask(
275    chunk_len: usize,
276    stride: usize,
277    diagonal: i64,
278    device: &Device,
279) -> Tensor<2> {
280    // Rows of the full folded mask, at the positions the readout happens at —
281    // which is the same statement as `read_rows`, so it is the same op.
282    let full = Tensor::<2>::full([chunk_len, chunk_len], f32::NEG_INFINITY, device).triu(diagonal);
283    read_rows::<2, 3>(full, 0, stride)
284}

286/// The read axis on primitives, for the recompute-backward math.
287///
288/// Same definitions as [`read_rows`] / [`read_causal_mask`] one module up, on
289/// [`F`] instead of `Tensor` — the split every kernel in this crate already has
290/// between its `Tensor` body and the `F<B, _>` one a
291/// [`Backward`](burn::backend::autodiff::ops::Backward) node can run (see
292/// [`fprim`](burn_stack::utils::fprim)). Both are reshape-and-narrow, so
293/// neither needs an op `F` does not carry.

296    use burn::backend::Backend;
297    use burn::backend::tensor::Device;
298    use burn::backend::FloatDType;

300    /// [`super::read_rows`] on primitives.
301    pub fn read_rows<B: Backend, const D: usize, const DP1: usize>(
302        t: F<B, D>,
303        dim: usize,
304        stride: usize,
305    ) -> F<B, D> {
306        if stride <= 1 {
307            return t;
308        }
309        let dims = t.dims();
310        let len = dims[dim];
311        assert_eq!(
312            len % stride,
313            0,
314            "a folded axis of {len} is not a whole number of {stride}-step tokens",
315        );
316        let mut split = [0usize; DP1];
317        split[..dim].copy_from_slice(&dims[..dim]);
318        split[dim] = len / stride;
319        split[dim + 1] = stride;
320        split[dim + 2..].copy_from_slice(&dims[dim + 1..]);
321
322        let mut out = dims;
323        out[dim] = len / stride;
324        t.reshape::<DP1>(split)
325            .narrow(dim + 1, stride - 1, 1)
326            .reshape::<D>(out)
327    }

329    /// The transpose of [`read_rows`]: scatter a read-row tensor back onto the
330    /// folded axis, zero at the positions no readout happens at.
331    ///
332    /// The gradient of `read_rows` — the hand-written backward's counterpart to
333    /// what autodiff does for the `Minimal` / `Serial` paths. The identity at
334    /// `stride = 1`.
335    ///
336    /// # Shapes
337    /// - `t`  : `[…, len / stride, …]` at `dim`
338    /// - out  : `[…, len, …]`
339    pub fn scatter_read_rows<B: Backend, const D: usize, const DP1: usize>(
340        t: F<B, D>,
341        dim: usize,
342        stride: usize,
343    ) -> F<B, D> {
344        if stride <= 1 {
345            return t;
346        }
347        let dims = t.dims();
348        let device = t.device();
349        let dtype = t.dtype();
350
351        let mut split = [0usize; DP1];
352        split[..dim].copy_from_slice(&dims[..dim]);
353        split[dim] = dims[dim];
354        split[dim + 1] = 1;
355        split[dim + 2..].copy_from_slice(&dims[dim + 1..]);
356        let mut pad = split;
357        pad[dim + 1] = stride - 1;
358
359        let mut out = dims;
360        out[dim] = dims[dim] * stride;
361        // The row is the *last* of its run, so the zeros go in front of it.
362        F::cat(
363            vec![F::<B, DP1>::zeros(pad, &device, dtype), t.reshape::<DP1>(split)],
364            dim + 1,
365        )
366        .reshape::<D>(out)
367    }

369    /// [`super::read_causal_mask`] on primitives.
370    pub fn read_causal_mask<B: Backend>(
371        chunk_len: usize,
372        stride: usize,
373        diagonal: i64,
374        device: &Device<B>,
375        dtype: FloatDType,
376    ) -> F<B, 2> {
377        let full = F::<B, 2>::full(
378            [chunk_len, chunk_len],
379            f32::NEG_INFINITY,
380            device,
381            dtype,
382        )
383        .triu(diagonal);
384        read_rows::<B, 2, 3>(full, 0, stride)
385    }

388/// The **interior** of a lag-`lag` tap's gap: `Πᵈ⁼¹..ˡᵃᵍ⁻¹ αₚ₋ᵈ`, or `None` at
389/// `lag = 1` (where the gap has no interior).
390///
391/// A tap at lag `L` is transported across its own gap, `Πᵈ⁼⁰..ᴸ⁻¹ αₚ₋ᵈ`
392/// (`info/trapezoid-as-integration.md` §9). `β = (1−λ)Δα` already carries the
393/// `d = 0` factor, so this is the rest of it.
394///
395/// The front is **zero-padded**, not clamped to what the call happens to hold:
396/// for `p < L` the missing factors are exactly the ones the cache's `v` slots
397/// were already scaled by when they were stored (see [`tail_decay`]).
398///
399/// # Shapes
400/// - `da_bsh` : `[batch, sequence, nheads]` (`Δ·A`, the log-decay)
401/// - out      : `[batch, sequence, nheads]`
402pub fn interior_gap_decay(da_bsh: Tensor<3>, lag: usize) -> Option<Tensor<3>> {
403    if lag <= 1 {
404        return None;
405    }
406    let [batch, sequence, nheads] = da_bsh.dims();
407    let device = da_bsh.device();
408    let mut window_bsh = Tensor::zeros([batch, sequence, nheads], &device);
409    for d in 1..lag {
410        let zeros_bdh = Tensor::zeros([batch, d, nheads], &device);
411        let shifted = Tensor::cat(vec![zeros_bdh, da_bsh.clone().narrow(1, 0, sequence - d)], 1);
412        window_bsh = window_bsh + shifted;
413    }
414    Some(window_bsh.exp())
415}

417/// The decay each cached tap slot has already accumulated: `Πᵣ₌q₊₁^{S−1} αᵣ` for
418/// the last `lag` positions `q` (oldest first), or `None` at `lag = 1` (the one
419/// slot *is* the last position, so the product is empty).
420///
421/// Storing the tap slots pre-scaled by this is what lets a lag-`L` tap span a
422/// call boundary: the next call supplies the in-call part of the gap
423/// ([`interior_gap_decay`], front-zero-padded) and the slot carries the rest.
424///
425/// # Shapes
426/// - `da_bsh` : `[batch, sequence, nheads]`
427/// - out      : `[batch, lag, nheads]`
428pub fn tail_decay(da_bsh: Tensor<3>, lag: usize) -> Option<Tensor<3>> {
429    if lag <= 1 {
430        return None;
431    }
432    let sequence = da_bsh.dims()[1];
433    // Only the tail window matters, so the cumulative sum is over `lag` terms
434    // and never accumulates the whole sequence's decay.
435    let tail_blh = da_bsh.narrow(1, sequence - lag, lag);
436    let cumulative_blh = tail_blh.cumsum(1);
437    let total_b1h = cumulative_blh.clone().narrow(1, lag - 1, 1);
438    Some((total_b1h - cumulative_blh).exp())
439}

441/// A `[1, len, 1]` gate over a folded axis: `0` at each token's **first**
442/// micro-step (`p ≡ 0 mod u`) and `1` elsewhere.
443///
444/// Broadcasts against any `[batch, len, nheads]` stream, in `forward` (`len` is
445/// the folded sequence, which starts at a token boundary) and in `step` (`len`
446/// is `u` itself, so the gate is its first entry). At `u = 1` every position is
447/// a token start and the gate is all zeros — which is the whole of why
448/// [`Trapezoid::HorizontalReset`](crate::mamba3::trapezoid::Trapezoid::HorizontalReset)
449/// degenerates to [`None`](crate::mamba3::trapezoid::Trapezoid::None) there.
450///
451/// This is where a pattern's admissibility rule becomes a tensor; *which* mass
452/// it applies to is the caller's — `λ` for
453/// [`Trapezoid::far_tap_crosses_tokens`](crate::mamba3::trapezoid::Trapezoid::far_tap_crosses_tokens),
454/// `μ` for
455/// [`interior_tap_crosses_tokens`](crate::mamba3::trapezoid::Trapezoid::interior_tap_crosses_tokens).
456pub fn token_start_gate(len: usize, micro_steps: usize, device: &Device) -> Tensor<3> {
457    let open: Vec<f32> = (0..len)
458        .map(|p| if p.is_multiple_of(micro_steps) { 0.0 } else { 1.0 })
459        .collect();
460    Tensor::<1>::from_floats(open.as_slice(), device).reshape([1, len, 1])
461}

463/// Output of [`trapezoidal_coefficients`] — the step's mass `Δₜ`, split.
464///
465/// The masses are **untransported**: `α` and any wider gap decay are the
466/// consumer's to apply, since which one a tap needs depends on the pathway (the
467/// double-SSD pass multiplies them in, the single-SSD key scale never does).
468pub struct TrapezoidCoeffs {
469    /// `Δₜ = softplus(dd_dt + dt_bias)`, clamped.
470    pub dt: Tensor<3>,
471    /// `Δₜ · Aₜ` (negative; the log-decay).
472    pub da: Tensor<3>,
473    /// `αₜ = exp(Δₜ · Aₜ) ∈ (0, 1]` — decay.
474    pub alpha: Tensor<3>,
475    /// `νₜ` — the mass of the tap at
476    /// [`tap_lag`](crate::mamba3::trapezoid::Trapezoid::tap_lag), before its
477    /// transport. `None` under
478    /// [`Trapezoid::None`](crate::mamba3::trapezoid::Trapezoid::None), which has
479    /// no left endpoint: the tensor is not formed rather than formed as zeros.
480    pub nu: Option<Tensor<3>>,
481    /// `νⁱⁿᵗₜ` — the mass of the extra lag-1 tap, `None` unless
482    /// [`has_interior_tap`](crate::mamba3::trapezoid::Trapezoid::has_interior_tap).
483    pub nu_interior: Option<Tensor<3>>,
484    /// `γₜ = λₜ · Δₜ` — right-endpoint weight, and **`Δₜ` itself** when there is
485    /// no `λ` (the whole step is paid at the right endpoint).
486    pub gamma: Tensor<3>,
487}

489/// Compute the trapezoidal discretisation coefficients from the raw
490/// (data-dependent) projections. See the top-of-`mamba3.rs` docs for the
491/// formulas and `trapezoid.rs`'s header for how the masses divide.
492///
493/// All data tensors are `[batch, len, nheads]`, `len` being the folded sequence
494/// (`forward`) or the `u` micro-steps of one token (`step`); `dt_bias_h` is
495/// broadcast to match. Axis 1 is what the patterns' gates are periodic in, in
496/// both cases.
497///
498/// `lambda_raw` is `None` under
499/// [`Trapezoid::None`](crate::mamba3::trapezoid::Trapezoid::None) — the block
500/// projects no `λ` channels — and the coefficients then take their `λ ≡ 1`
501/// values *by construction*: no mass is formed and `γ` **is** `Δ`, sharing its
502/// tensor rather than recomputing `1 · Δ`. `mu_raw` is `None` unless the pattern
503/// [`has_interior_tap`](crate::mamba3::trapezoid::Trapezoid::has_interior_tap),
504/// and the same holds one level down: no `μ`, no second mass.
505///
506/// A gate closes by handing its mass back, not by dropping it: `λ` is set to `1`
507/// where the far tap is inadmissible (so `γ` takes the step whole) and `μ` to
508/// `0` where the interior one is (so the far tap takes the left endpoint whole).
509/// Both are exact at the ends, so the degenerate members are their targets bit
510/// for bit.
511pub fn trapezoidal_coefficients(
512    dd_dt: Tensor<3>,
513    dd_a_raw: Tensor<3>,
514    lambda_raw: Option<Tensor<3>>,
515    mu_raw: Option<Tensor<3>>,
516    dt_bias_h: Tensor<1>,
517    spec: TrapezoidSpec,
518) -> TrapezoidCoeffs {
519    let (dt_limit, a_floor) = (spec.dt_limit, spec.a_floor);
520    // Broadcast dt_bias_h [nheads] → [1, 1, nheads] so the addition aligns on
521    // the last dim.
522    let dt_bias_broadcast = dt_bias_h.unsqueeze::<3>();
523    let dt = softplus(dd_dt + dt_bias_broadcast).clamp(dt_limit.0, dt_limit.1);
524    // `A = −max(softplus(·), a_floor) ∈ (−∞, −a_floor]`. The floor must be
525    // applied to the (positive) softplus *before* negating: a method call
526    // binds tighter than unary minus, so `-softplus(x).clamp(NEG_INFINITY,
527    // -a_floor)` would collapse the positive softplus to the constant
528    // `-a_floor` and yield `A ≡ +a_floor` — a *growing* state (`α > 1`) with a
529    // dead `dd_A` projection.
530    let a = -softplus(dd_a_raw).clamp(a_floor, f64::INFINITY);
531    let da = dt.clone() * a;
532    let alpha = da.clone().exp();
533    let gate = |crosses: bool| {
534        (!crosses).then(|| token_start_gate(dt.dims()[1], spec.micro_steps, &dt.device()))
535    };
536    let (nu, nu_interior, gamma) = match lambda_raw {
537        Some(lambda_raw) => {
538            let lambda = burn::tensor::activation::sigmoid(lambda_raw);
539            // A closed far tap means λ = 1 there: the whole step is paid at the
540            // right endpoint, which is what makes the gated pattern a submodel
541            // of the ungated one rather than a lossy version of it.
542            let lambda = match gate(spec.pattern.far_tap_crosses_tokens()) {
543                Some(open_1s1) => lambda * open_1s1.clone() + (-open_1s1 + 1.0),
544                None => lambda,
545            };
546            let nu = (-lambda.clone() + 1.0) * dt.clone();
547            let gamma = lambda * dt.clone();
548            // …and a closed interior tap means μ = 0 there: its share of the
549            // left endpoint returns to the far tap, one level up.
550            let (nu, nu_interior) = match mu_raw {
551                Some(mu_raw) => {
552                    let mu = burn::tensor::activation::sigmoid(mu_raw);
553                    let mu = match gate(spec.pattern.interior_tap_crosses_tokens()) {
554                        Some(open_1s1) => mu * open_1s1,
555                        None => mu,
556                    };
557                    let nu_interior = nu.clone() * mu.clone();
558                    (nu * (-mu + 1.0), Some(nu_interior))
559                }
560                None => (nu, None),
561            };
562            (Some(nu), nu_interior, gamma)
563        }
564        // λ ≡ 1: the left endpoint is unweighted and the right one takes Δ whole.
565        None => (None, None, dt.clone()),
566    };
567    TrapezoidCoeffs {
568        dt,
569        da,
570        alpha,
571        nu,
572        nu_interior,
573        gamma,
574    }
575}

577/// QK-Norm → GQA-expand groups→heads → add per-(head, mimo-rank) bias.
578///
579/// The input is the raw B/C projection already reshaped to expose the group
580/// dim, with last dim = `state_rank`. The output replaces the group dim with
581/// the head dim, leaving the last dim untouched.
582///
583/// `DP1 = D + 1` (required by [`gqa_expand_to_heads`]'s intermediate rank).
584pub fn qk_norm_expand_bias<const D: usize, const DP1: usize>(
585    raw_mgr: Tensor<D>,
586    norm: &RmsNorm,
587    bias_hmr: Tensor<3>,
588    group_dim: usize,
589    nheads: usize,
590) -> Tensor<D> {
591    // RmsNorm operates on the last dim only, so the leading shape passes through.
592    let normed = norm.forward(raw_mgr);
593    let expanded = gqa_expand_to_heads::<D, DP1>(normed, group_dim, nheads);
594    // Broadcast bias [nheads, mimo_rank, state_rank] → [1, ..., 1, mimo_rank, nheads, state_rank].
595    let bias = bias_hmr.swap_dims(0, 1).unsqueeze::<D>();
596    expanded + bias
597}

599/// Rank-summed outer product `state[b, h, p, r] = Σₘ v[b, m, h, p] · k[b, m, h, r]`
600/// (`einsum('bmhp,bmhr->bhpr')`).
601///
602/// This is the per-token state contribution: each MIMO rank contributes an
603/// outer product `v[m] ⊗ k[m]` and the shared state accumulates their sum.
604///
605/// At `mimo_rank == 1` the contracted dimension is 1, so the GEMM is rank-1 and
606/// the sum is a single outer product — [`mimo_outer_sum_siso`] writes it as a
607/// broadcast multiply instead. Which form wins is backend-dependent, and by a
608/// wide margin at these (tiny, decode-sized) shapes: the broadcast is the
609/// faster one on CUDA and *much* slower on the portable CPU backends, whose
610/// broadcast-elementwise path trails their matmul by more than an order of
611/// magnitude here. The choice is therefore
612/// [`Mamba3Config::siso_specialization_decode`](crate::mamba3::mamba3::Mamba3Config::siso_specialization_decode)'s
613/// — the per-token flag, *not* the chunkwise one, whose verdict is the opposite;
614/// both branches compute the same values and gradients.
615pub fn mimo_outer_sum(
616    v_bmhp: Tensor<4>,
617    k_bmhr: Tensor<4>,
618    siso_specialization: bool,
619) -> Tensor<4> {
620    let [_batch, mimo_rank, _nheads, _per_head_dim] = v_bmhp.dims();
621    if mimo_rank == 1 && siso_specialization {
622        mimo_outer_sum_siso(v_bmhp, k_bmhr)
623    } else {
624        mimo_outer_sum_mimo(v_bmhp, k_bmhr)
625    }
626}
627
628/// SISO (`mimo_rank == 1`) rank-summed outer product: the single `v ⊗ k` as a
629/// broadcast multiply of `[b, h, p, 1]` by `[b, h, 1, r]`.
630pub fn mimo_outer_sum_siso(v_bmhp: Tensor<4>, k_bmhr: Tensor<4>) -> Tensor<4> {
631    let v_bhp1: Tensor<4> = v_bmhp.squeeze_dim::<3>(1).unsqueeze_dim(3);
632    let k_bh1r: Tensor<4> = k_bmhr.squeeze_dim::<3>(1).unsqueeze_dim(2);
633    v_bhp1 * k_bh1r
634}
635
636/// General MIMO rank-summed outer product: a matmul contracting `mimo_rank`.
637pub fn mimo_outer_sum_mimo(v_bmhp: Tensor<4>, k_bmhr: Tensor<4>) -> Tensor<4> {
638    // v_bmhp [b, m, h, p] -> v_bhpm [b, h, p, m]; k_bmhr [b, m, h, r] -> k_bhmr.
639    let v_bhpm = v_bmhp.permute([0, 2, 3, 1]);
640    let k_bhmr = k_bmhr.swap_dims(1, 2);
641    v_bhpm.matmul(k_bhmr)
642}

647/// Build the MIMO value tensor `v = x ⊙ mimo_x` with broadcasting.
648///
649/// Inserts a `mimo_rank` axis at `insert_dim`. When `mimo_x_hmp` is `None`
650/// (SISO), the inserted axis has size 1 and `x` is passed through; otherwise
651/// broadcasting fills the inserted axis to size `mimo_rank`.
652///
653/// `DP1 = D + 1`.
654pub fn build_v_with_mimo<const D: usize, const DP1: usize>(
655    x: Tensor<D>,
656    mimo_x_hmp: Option<&Tensor<3>>,
657    insert_dim: usize,
658) -> Tensor<DP1> {
659    let x_with_rank_axis = x.unsqueeze_dim::<DP1>(insert_dim);
660    match mimo_x_hmp {
661        None => x_with_rank_axis,
662        Some(mimo_x_hmp) => {
663            // mimo_x_hmp [nheads, mimo_rank, per_head_dim] → swap_dims to
664            // [mimo_rank, nheads, per_head_dim] → unsqueeze leading 1s. The
665            // result broadcasts against `x_with_rank_axis` over (batch, seq, …).
666            let mimo_x_broadcast = mimo_x_hmp.clone().swap_dims(0, 1).unsqueeze::<DP1>();
667            x_with_rank_axis * mimo_x_broadcast
668        }
669    }
670}
