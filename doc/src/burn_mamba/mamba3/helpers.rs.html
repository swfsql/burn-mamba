1//! Shared helpers used by both [`Mamba3::forward`](super::mamba3::Mamba3::forward)
2//! and [`Mamba3::step`](super::mamba3::Mamba3::step). They isolate three blocks
3//! that previously appeared in both methods at different ranks:
4//!
5//! 1. Trapezoidal discretisation: `dt`, `α`, `β`, `γ`, `da`.
6//! 2. QK-norm + GQA expansion + per-(head, mimo-rank) bias on B / C.
7//! 3. MIMO `V` construction: broadcast-multiply `x` by `mimo_x_hmp`.
8//! 4. The rank-summed outer product `Σₘ v[m] ⊗ k[m]` feeding the SSM state
9//!    (SISO-branched).
10//!
11//! Most helpers are generic over the rank `D` of the data tensors so a single
12//! definition serves both the sequence-aware (`forward`) and single-token
13//! (`step`) code paths.
14
15use crate::modules::RmsNorm;
16use crate::modules::gqa_expand_to_heads;

18use burn::prelude::*;

20/// Output of [`trapezoidal_coefficients`].
21///
22/// All tensors share the rank `D` of the inputs.
23pub struct TrapezoidCoeffs<const D: usize> {
24    /// `Δₜ = softplus(dd_dt + dt_bias)`, clamped.
25    pub dt: Tensor<D>,
26    /// `Δₜ · Aₜ` (negative; the log-decay).
27    pub da: Tensor<D>,
28    /// `αₜ = exp(Δₜ · Aₜ) ∈ (0, 1]` — decay.
29    pub alpha: Tensor<D>,
30    /// `βₜ = (1 − λₜ) · Δₜ · αₜ` — left-endpoint weight.
31    pub beta: Tensor<D>,
32    /// `γₜ = λₜ · Δₜ` — right-endpoint weight.
33    pub gamma: Tensor<D>,
34}
35
36/// Compute the trapezoidal discretisation coefficients from the raw
37/// (data-dependent) projections. See the top-of-`mamba3.rs` docs for the
38/// formulas.
39///
40/// All four data tensors share rank `D` and have `nheads` as the last dim.
41/// `dt_bias_h` is broadcast to match.
42pub fn trapezoidal_coefficients<const D: usize>(
43    dd_dt: Tensor<D>,
44    dd_a_raw: Tensor<D>,
45    lambda_raw: Tensor<D>,
46    dt_bias_h: Tensor<1>,
47    dt_limit: (f64, f64),
48    a_floor: f64,
49) -> TrapezoidCoeffs<D> {
50    // Broadcast dt_bias_h [nheads] → [1, ..., 1, nheads] so the addition aligns
51    // on the last dim regardless of leading shape.
52    let dt_bias_broadcast = dt_bias_h.unsqueeze::<D>();
53    let dt = softplus(dd_dt + dt_bias_broadcast).clamp(dt_limit.0, dt_limit.1);
54    // `A = −max(softplus(·), a_floor) ∈ (−∞, −a_floor]`. The floor must be
55    // applied to the (positive) softplus *before* negating: a method call
56    // binds tighter than unary minus, so `-softplus(x).clamp(NEG_INFINITY,
57    // -a_floor)` would collapse the positive softplus to the constant
58    // `-a_floor` and yield `A ≡ +a_floor` — a *growing* state (`α > 1`) with a
59    // dead `dd_A` projection.
60    let a = -softplus(dd_a_raw).clamp(a_floor, f64::INFINITY);
61    let da = dt.clone() * a;
62    let lambda = burn::tensor::activation::sigmoid(lambda_raw);
63    let alpha = da.clone().exp();
64    let beta = (-lambda.clone() + 1.0) * dt.clone() * alpha.clone();
65    let gamma = lambda * dt.clone();
66    TrapezoidCoeffs {
67        dt,
68        da,
69        alpha,
70        beta,
71        gamma,
72    }
73}

75/// QK-Norm → GQA-expand groups→heads → add per-(head, mimo-rank) bias.
76///
77/// The input is the raw B/C projection already reshaped to expose the group
78/// dim, with last dim = `state_rank`. The output replaces the group dim with
79/// the head dim, leaving the last dim untouched.
80///
81/// `DP1 = D + 1` (required by [`gqa_expand_to_heads`]'s intermediate rank).
82pub fn qk_norm_expand_bias<const D: usize, const DP1: usize>(
83    raw_mgr: Tensor<D>,
84    norm: &RmsNorm,
85    bias_hmr: Tensor<3>,
86    group_dim: usize,
87    nheads: usize,
88) -> Tensor<D> {
89    // RmsNorm operates on the last dim only, so the leading shape passes through.
90    let normed = norm.forward(raw_mgr);
91    let expanded = gqa_expand_to_heads::<D, DP1>(normed, group_dim, nheads);
92    // Broadcast bias [nheads, mimo_rank, state_rank] → [1, ..., 1, mimo_rank, nheads, state_rank].
93    let bias = bias_hmr.swap_dims(0, 1).unsqueeze::<D>();
94    expanded + bias
95}

97/// Rank-summed outer product `state[b, h, p, r] = Σₘ v[b, m, h, p] · k[b, m, h, r]`
98/// (`einsum('bmhp,bmhr->bhpr')`).
99///
100/// This is the per-token state contribution: each MIMO rank contributes an
101/// outer product `v[m] ⊗ k[m]` and the shared state accumulates their sum.
102///
103/// At `mimo_rank == 1` the contracted dimension is 1, so the GEMM is rank-1 and
104/// the sum is a single outer product — [`mimo_outer_sum_siso`] writes it as a
105/// broadcast multiply instead. Which form wins is backend-dependent, and by a
106/// wide margin at these (tiny, decode-sized) shapes: the broadcast is the
107/// faster one on CUDA and *much* slower on the portable CPU backends, whose
108/// broadcast-elementwise path trails their matmul by more than an order of
109/// magnitude here. The choice is therefore
110/// [`Mamba3Config::siso_specialization_decode`](crate::mamba3::mamba3::Mamba3Config::siso_specialization_decode)'s
111/// — the per-token flag, *not* the chunkwise one, whose verdict is the opposite;
112/// both branches compute the same values and gradients.
113pub fn mimo_outer_sum(
114    v_bmhp: Tensor<4>,
115    k_bmhr: Tensor<4>,
116    siso_specialization: bool,
117) -> Tensor<4> {
118    let [_batch, mimo_rank, _nheads, _per_head_dim] = v_bmhp.dims();
119    if mimo_rank == 1 && siso_specialization {
120        mimo_outer_sum_siso(v_bmhp, k_bmhr)
121    } else {
122        mimo_outer_sum_mimo(v_bmhp, k_bmhr)
123    }
124}
125
126/// SISO (`mimo_rank == 1`) rank-summed outer product: the single `v ⊗ k` as a
127/// broadcast multiply of `[b, h, p, 1]` by `[b, h, 1, r]`.
128pub fn mimo_outer_sum_siso(v_bmhp: Tensor<4>, k_bmhr: Tensor<4>) -> Tensor<4> {
129    let v_bhp1: Tensor<4> = v_bmhp.squeeze_dim::<3>(1).unsqueeze_dim(3);
130    let k_bh1r: Tensor<4> = k_bmhr.squeeze_dim::<3>(1).unsqueeze_dim(2);
131    v_bhp1 * k_bh1r
132}
133
134/// General MIMO rank-summed outer product: a matmul contracting `mimo_rank`.
135pub fn mimo_outer_sum_mimo(v_bmhp: Tensor<4>, k_bmhr: Tensor<4>) -> Tensor<4> {
136    // v_bmhp [b, m, h, p] -> v_bhpm [b, h, p, m]; k_bmhr [b, m, h, r] -> k_bhmr.
137    let v_bhpm = v_bmhp.permute([0, 2, 3, 1]);
138    let k_bhmr = k_bmhr.swap_dims(1, 2);
139    v_bhpm.matmul(k_bhmr)
140}

145/// Build the MIMO value tensor `v = x ⊙ mimo_x` with broadcasting.
146///
147/// Inserts a `mimo_rank` axis at `insert_dim`. When `mimo_x_hmp` is `None`
148/// (SISO), the inserted axis has size 1 and `x` is passed through; otherwise
149/// broadcasting fills the inserted axis to size `mimo_rank`.
150///
151/// `DP1 = D + 1`.
152pub fn build_v_with_mimo<const D: usize, const DP1: usize>(
153    x: Tensor<D>,
154    mimo_x_hmp: Option<&Tensor<3>>,
155    insert_dim: usize,
156) -> Tensor<DP1> {
157    let x_with_rank_axis = x.unsqueeze_dim::<DP1>(insert_dim);
158    match mimo_x_hmp {
159        None => x_with_rank_axis,
160        Some(mimo_x_hmp) => {
161            // mimo_x_hmp [nheads, mimo_rank, per_head_dim] → swap_dims to
162            // [mimo_rank, nheads, per_head_dim] → unsqueeze leading 1s. The
163            // result broadcasts against `x_with_rank_axis` over (batch, seq, …).
164            let mimo_x_broadcast = mimo_x_hmp.clone().swap_dims(0, 1).unsqueeze::<DP1>();
165            x_with_rank_axis * mimo_x_broadcast
166        }
167    }
168}
