1//! Helpers that [`Mamba3::forward`](super::mamba3::Mamba3::forward) (both
2//! pathways) and [`Mamba3::step`](super::mamba3::Mamba3::step) share, so the
3//! two modes cannot drift:
4//!
5//! 1. Trapezoidal discretisation ([`trapezoidal_coefficients`]): `Δ`, `da`,
6//!    `α`, the tap masses `ν`, and `γ`.
7//! 2. QK-norm + GQA expansion + per-(head, mimo-rank) bias on B / C
8//!    ([`qk_norm_expand_bias`]).
9//! 3. MIMO `V` construction: broadcast-multiply `x` by `mimo_x_hmp`
10//!    ([`build_v_with_mimo`]).
11//! 4. The rank-summed outer product `Σₘ v[m] ⊗ k[m]` that writes the SSM state
12//!    ([`mimo_outer_sum`], SISO-branched).
13//! 5. [`split_trailing`]: removes an optional trailing segment from the
14//!    in-projection.
15//! 6. The lag arithmetic of the trapezoid tap: the shift, the gap transport,
16//!    the decay that a cached slot carries across a call boundary, and the
17//!    per-position gate of a tap pattern.
18//! 7. [`prefix_sum`]: the blocked inclusive scan that replaces
19//!    `Tensor::cumsum` for every sequence-length cumulative sum.
20//! 8. The **read axis** ([`read_rows`], [`read_causal_mask`], and their
21//!    primitive twins in [`prim`]).
22//!
23//! Most helpers are generic over the rank `D` of the data tensors, so one
24//! definition serves both the sequence (`forward`) and single-token (`step`)
25//! code paths. The discretisation is the exception: MambaProduct gives `step`
26//! its own `u` axis, so both paths call it at rank 3.
27

34/// Remove a **trailing** in-projection segment of `width` channels from
35/// `proj`. The segment is `None` when the block projects none.
36///
37/// It serves the segments that a block can omit: the tropical `(a, b)` and the
38/// Kalman noise ([`Mamba3::split_positive`](crate::mamba3::mamba3::Mamba3)),
39/// then the rotation channels
40/// ([`RotationKind::Real1D`](crate::mamba3::rotation::RotationKind::Real1D)
41/// projects none), `μ` (a one-tap pattern projects none) and `λ`
42/// ([`Trapezoid::None`](crate::mamba3::trapezoid::Trapezoid::None) projects
43/// none). That is why the in-projection puts them last.
44///
45/// An optional slice cannot be one more entry in the main `split_into`. Burn
46/// has no zero-width tensors, and `split_with_sizes` *drops* a zero-length
47/// segment instead of returning an empty one. The destructuring would then
48/// have one part too few.
49///
50/// # Shapes
51/// - `proj` : `[..., w]` along `dim`
52/// - out    : `[..., w − width]` and, if any, `[..., width]`
53pub fn split_trailing<const D: usize>(
54    proj: Tensor<D>,
55    width: usize,
56    dim: usize,
57) -> (Tensor<D>, Option<Tensor<D>>) {
58    if width == 0 {
59        return (proj, None);
60    }
61    let rest = proj.dims()[dim] - width;
62    let tail = proj.clone().narrow(dim, rest, width);
63    (proj.narrow(dim, 0, rest), Some(tail))
64}

66/// Shift a per-position stream back by `lag`. The tap slots of the cache fill
67/// the first `lag` positions: `out[p] = stream[p − lag]`, or `prefix[p]` where
68/// that index is before the call.
69///
70/// This is the "shift-before-chunking" of the double-SSD pathway, generalised
71/// from lag 1 to [`Trapezoid::tap_lag`](crate::mamba3::trapezoid::Trapezoid::tap_lag).
72/// A folded sequence is `tokens · u` long and `lag ∈ {1, u}`, so
73/// `sequence ≥ lag` always. At equality, the whole call is prefix.
74///
75/// # Shapes
76/// - `stream` : `[batch, sequence, …]`
77/// - `prefix` : `[batch, lag, …]`
78/// - out      : `[batch, sequence, …]`
79pub fn shift_stream<const D: usize>(
80    stream: Tensor<D>,
81    prefix: Tensor<D>,
82    lag: usize,
83) -> Tensor<D> {
84    let sequence = stream.dims()[1];
85    assert_eq!(prefix.dims()[1], lag, "one prefix slot per lagged position");
86    assert!(sequence >= lag, "a call is at least one token, i.e. `lag` long");
87    if sequence == lag {
88        prefix
89    } else {
90        Tensor::cat(vec![prefix, stream.narrow(1, 0, sequence - lag)], 1)
91    }
92}

94/// The in-block length [`prefix_sum`] scans directly.
95///
96/// A `cumsum` of length `L` over `N` elements costs `∝ N·L` (below). So a
97/// blocked scan costs `∝ N·(block + len/block²)`: one pass inside each block,
98/// plus one over the `len/block` block totals. The minimum is at
99/// `block = ∛(2·len)`, which this function approaches on a power-of-two grid.
100/// The floor of 16 amortises the fixed cost (a reshape, a subtract, a
101/// broadcast add) on short axes. The ceiling bounds the in-block quadratic.
102///
103/// `scan_block(len) < len` means exactly "this length takes the blocked
104/// branch". The parity tests assert it, so a new tuning of the rule cannot
105/// silently stop covering that branch.
106pub(crate) fn scan_block(len: usize) -> usize {
107    let mut block = 16;
108    while block < 256 && block * block * block < 2 * len {
109        block *= 2;
110    }
111    block
112}

114/// Inclusive prefix sum along `dim`, continued from `init`:
115/// `out[i] = init + Σ_{j ≤ i} t[j]`.
116///
117/// The values of [`Tensor::cumsum`], but **blocked**:
118///
119/// 1. split the scanned axis into runs of [`scan_block`],
120/// 2. scan each run alone,
121/// 3. add the exclusive prefix of the run totals.
122///
123/// Both scans are short and bounded, and the whole is a few ops for any `len`.
124///
125/// `init` is the carry-in of the scan. For the cumulative rotation angle, it
126/// is the angle that the cache brings from the previous call, as
127/// `quat_cumprod` takes its `init` quaternion. It joins the block offset and
128/// needs no pass of its own. So on a backend whose `cumsum` is already linear,
129/// this costs the same as a plain `cumsum` **plus the add of the caller**.
130///
131/// It exists because Burn's `cumsum` is **quadratic in the scanned length** on
132/// the cubecl backends. Measured on CUDA over `[2, len, 32, 32]` along `dim 1`:
133/// 0.27 ms at `len = 256`, 143 ms at 4096, ≈4× per doubling. The same scan on
134/// the trailing, contiguous axis is only 2.6× cheaper and still ≈4× per
135/// doubling, so the cause is the scan, not the stride.
136///
137/// Every other scan in this crate runs over `chunk_len`, `lag` or
138/// `micro_steps`, all bounded. The cumulative rotation angle is the one scan
139/// over the whole **folded** sequence, and `micro_steps` multiplies that axis,
140/// so `u > 1` reaches the quadratic `u`× sooner. Blocked, the same case is
141/// 1.9× / 6.7× / 23× / 46× faster at `len = 256 / 512 / 1024 / 2048`, and the
142/// forward is almost flat in `len` (0.14 → 0.51 ms over that span).
143///
144/// The non-abelian sibling ([`quat_scan`](crate::mamba3::quat_scan)) solves
145/// the same problem with a Hillis–Steele doubling, because it has no
146/// `cumprod` to block with. Doubling also works here, but blocking is faster
147/// at **every** measured length (256 … 8192): 4.8× → 6.6× on CUDA forward,
148/// 2.3× → 3.3× with the backward, and 15× → 41× on the CPU backend. On
149/// bandwidth- and launch-bound hardware, `O(len·log len)` work in
150/// `3·⌈log₂ len⌉` full-tensor kernels loses to `O(len·∛len)` in six. So a
151/// runtime knob would have nothing to choose.
152///
153/// The additions associate differently from a sequential scan, so the two
154/// agree to rounding, not bit-for-bit.
155///
156/// # Shapes
157/// - `t`    : any rank, scanned along `dim`; `DP1` is `D + 1`
158/// - `init` : `t`'s shape with `1` along `dim` (it broadcasts along it)
159/// - out    : `t`'s shape
160pub fn prefix_sum<const D: usize, const DP1: usize>(
161    t: Tensor<D>,
162    dim: usize,
163    init: Option<Tensor<D>>,
164) -> Tensor<D> {
165    let len = t.dims()[dim];
166    let block = scan_block(len);
167    if len <= block {
168        let scanned = t.cumsum(dim);
169        return match init {
170            Some(init) => scanned + init,
171            None => scanned,
172        };
173    }
174    let device = t.device();
175
176    // Pad up to a whole number of blocks. Zero is the additive identity, so the
177    // padding contributes to no prefix, and it is narrowed off at the end.
178    let nblocks = len.div_ceil(block);
179    let pad = nblocks * block - len;
180    let t = if pad == 0 {
181        t
182    } else {
183        let mut pad_dims = t.dims();
184        pad_dims[dim] = pad;
185        Tensor::cat(vec![t, Tensor::zeros(pad_dims, &device)], dim)
186    };
187
188    // Split the scanned axis into (which block, where in it). Row-major, so the
189    // reshape moves nothing.
190    let dims = t.dims();
191    let mut split = [0usize; DP1];
192    split[..dim].copy_from_slice(&dims[..dim]);
193    split[dim] = nblocks;
194    split[dim + 1] = block;
195    split[dim + 2..].copy_from_slice(&dims[dim + 1..]);
196
197    let inner = t.reshape(split).cumsum(dim + 1);
198    // Each block's total is its last in-block prefix; kept at width 1 on that
199    // axis so the carry broadcasts back over the block. `init` joins the carry
200    // here, for free — one add serves both.
201    let totals = inner.clone().narrow(dim + 1, block - 1, 1);
202    let carry = totals.clone().cumsum(dim) - totals;
203    let carry = match init {
204        Some(init) => carry + init.unsqueeze_dim::<DP1>(dim + 1),
205        None => carry,
206    };
207
208    let joined = (inner + carry).reshape(dims);
209    if pad == 0 {
210        joined
211    } else {
212        joined.narrow(dim, 0, len)
213    }
214}

216// ---------------------------------------------------------------------------
217// The read axis
218// ---------------------------------------------------------------------------
219//
220// A chunk has two axes, and `micro_steps` widens only one of them. Each of the
221// `u` micro-steps of a token **writes** to the state, so the source axis is
222// the folded `chunk_len`. The block **reads** the token once, at its last
223// micro-step, so the target axis is `chunk_len / u`: the rows whose `y`
224// survives (a zero gradient multiplies all the others). `read_stride` is that
225// `u`, or `1` where there is no fold (Mamba-2, `micro_steps = 1`), where both
226// helpers below are the identity. `step` also reads this axis: its token is a
227// block of `u` positions with one read row.
228//
229// The alignment keeps this a reshape, not a gather. `chunk_len` is a multiple
230// of `read_stride`
231// ([`Mamba3SsdPath::chunk_len_or_optimal`](crate::mamba3::ssd_path::Mamba3SsdPath::chunk_len_or_optimal)),
232// so chunk `c` covers folded positions `[c·L, (c+1)·L)` = tokens
233// `[c·T, (c+1)·T)`, and its read rows are a **contiguous run** of them.
234
235/// The read rows of a folded axis: index `i·stride + (stride − 1)` for each
236/// `i`, i.e. the last position of every run of `stride`.
237///
238/// The identity at `stride = 1`.
239///
240/// # Shapes
241/// - `t`   : `[…, len, …]` at `dim`, `len` a multiple of `stride`
242/// - out   : `[…, len / stride, …]`
243///
244/// `DP1 = D + 1`.
245pub fn read_rows<const D: usize, const DP1: usize>(
246    t: Tensor<D>,
247    dim: usize,
248    stride: usize,
249) -> Tensor<D> {
250    if stride <= 1 {
251        return t;
252    }
253    let dims = t.dims();
254    let len = dims[dim];
255    assert_eq!(
256        len % stride,
257        0,
258        "a folded axis of {len} is not a whole number of {stride}-step tokens",
259    );
260    let mut split = [0usize; DP1];
261    split[..dim].copy_from_slice(&dims[..dim]);
262    split[dim] = len / stride;
263    split[dim + 1] = stride;
264    split[dim + 2..].copy_from_slice(&dims[dim + 1..]);
265
266    let mut out = dims;
267    out[dim] = len / stride;
268    // Row-major, so the split moves nothing and the narrow is a strided view.
269    t.reshape(split).narrow(dim + 1, stride - 1, 1).reshape(out)
270}

272/// The additive causal mask over `(read row, folded source)`:
273/// `0` where the source is early enough to contribute, `−∞` where it is not.
274///
275/// `diagonal` goes to [`Tensor::triu`] on the **folded** grid, so it has the
276/// usual meaning. `0` excludes the same position (the single-SSD strict mask,
277/// whose diagonal `ssd::diag` adds back with `γ`). `1` keeps it (the
278/// double-SSD inclusive mask).
279///
280/// The mask is `−∞`, not a `0/1` multiply, because it is added *before* the
281/// `exp`. For a source after the row, the decay difference is positive and the
282/// `exp` would overflow.
283///
284/// # Shapes
285/// - out : `[chunk_len / stride, chunk_len]`
286pub fn read_causal_mask(
287    chunk_len: usize,
288    stride: usize,
289    diagonal: i64,
290    device: &Device,
291) -> Tensor<2> {
292    // The rows of the full folded mask at the readout positions. That is what
293    // `read_rows` selects, so it is the same op.
294    let full = Tensor::<2>::full([chunk_len, chunk_len], f32::NEG_INFINITY, device).triu(diagonal);
295    read_rows::<2, 3>(full, 0, stride)
296}

298/// The read axis on primitives, for the recompute-backward math.
299///
300/// The same definitions as [`read_rows`] / [`read_causal_mask`] one module up,
301/// on [`F`] instead of `Tensor`. Every kernel in this crate has this split: a
302/// `Tensor` body, and an `F<B, _>` body that a
303/// [`Backward`](burn::backend::autodiff::ops::Backward) node can run (see
304/// [`fprim`](burn_stack::utils::fprim)). Both are reshape-and-narrow, so
305/// neither needs an op that `F` does not have.

307    use burn_stack::utils::fprim::F;
308    use burn::backend::Backend;
309    use burn::backend::tensor::Device;

312    /// [`super::read_rows`] on primitives.
313    pub fn read_rows<B: Backend, const D: usize, const DP1: usize>(
314        t: F<B, D>,
315        dim: usize,
316        stride: usize,
317    ) -> F<B, D> {
318        if stride <= 1 {
319            return t;
320        }
321        let dims = t.dims();
322        let len = dims[dim];
323        assert_eq!(
324            len % stride,
325            0,
326            "a folded axis of {len} is not a whole number of {stride}-step tokens",
327        );
328        let mut split = [0usize; DP1];
329        split[..dim].copy_from_slice(&dims[..dim]);
330        split[dim] = len / stride;
331        split[dim + 1] = stride;
332        split[dim + 2..].copy_from_slice(&dims[dim + 1..]);
333
334        let mut out = dims;
335        out[dim] = len / stride;
336        t.reshape::<DP1>(split)
337            .narrow(dim + 1, stride - 1, 1)
338            .reshape::<D>(out)
339    }

341    /// The transpose of [`read_rows`]: scatter a read-row tensor back onto the
342    /// folded axis, with zeros at the positions that have no readout.
343    ///
344    /// This is the gradient of `read_rows`. The hand-written backward uses it
345    /// where autodiff does the same for the `Minimal` / `Serial` paths. The
346    /// identity at `stride = 1`.
347    ///
348    /// # Shapes
349    /// - `t`  : `[…, len / stride, …]` at `dim`
350    /// - out  : `[…, len, …]`
351    pub fn scatter_read_rows<B: Backend, const D: usize, const DP1: usize>(
352        t: F<B, D>,
353        dim: usize,
354        stride: usize,
355    ) -> F<B, D> {
356        if stride <= 1 {
357            return t;
358        }
359        let dims = t.dims();
360        let device = t.device();
361        let dtype = t.dtype();
362
363        let mut split = [0usize; DP1];
364        split[..dim].copy_from_slice(&dims[..dim]);
365        split[dim] = dims[dim];
366        split[dim + 1] = 1;
367        split[dim + 2..].copy_from_slice(&dims[dim + 1..]);
368        let mut pad = split;
369        pad[dim + 1] = stride - 1;
370
371        let mut out = dims;
372        out[dim] = dims[dim] * stride;
373        // The row is the *last* of its run, so the zeros go in front of it.
374        F::cat(
375            vec![F::<B, DP1>::zeros(pad, &device, dtype), t.reshape::<DP1>(split)],
376            dim + 1,
377        )
378        .reshape::<D>(out)
379    }

381    /// [`super::read_causal_mask`] on primitives.
382    pub fn read_causal_mask<B: Backend>(
383        chunk_len: usize,
384        stride: usize,
385        diagonal: i64,
386        device: &Device<B>,
387        dtype: FloatDType,
388    ) -> F<B, 2> {
389        let full = F::<B, 2>::full(
390            [chunk_len, chunk_len],
391            f32::NEG_INFINITY,
392            device,
393            dtype,
394        )
395        .triu(diagonal);
396        read_rows::<B, 2, 3>(full, 0, stride)
397    }

400/// The **interior** of the gap of a lag-`lag` tap: `Πᵈ⁼¹..ˡᵃᵍ⁻¹ αₚ₋ᵈ`, or
401/// `None` at `lag = 1` (where the gap has no interior).
402///
403/// A tap at lag `L` is transported across its own gap, `Πᵈ⁼⁰..ᴸ⁻¹ αₚ₋ᵈ`
404/// (`info/mamba-3/trapezoid-as-integration.md` §9). `β = (1−λ)Δα` already
405/// holds the `d = 0` factor, so this is the rest.
406///
407/// The front is **zero-padded** (in log space), not clamped to what the call
408/// holds. For `p < L`, the missing factors are exactly those that already
409/// scaled the `v` slots of the cache when they were stored (see
410/// [`tail_decay`]).
411///
412/// # Shapes
413/// - `da_bsh` : `[batch, sequence, nheads]` (`Δ·A`, the log-decay)
414/// - out      : `[batch, sequence, nheads]`
415pub fn interior_gap_decay(da_bsh: Tensor<3>, lag: usize) -> Option<Tensor<3>> {
416    if lag <= 1 {
417        return None;
418    }
419    let [batch, sequence, nheads] = da_bsh.dims();
420    let device = da_bsh.device();
421    let mut window_bsh = Tensor::zeros([batch, sequence, nheads], &device);
422    for d in 1..lag {
423        let zeros_bdh = Tensor::zeros([batch, d, nheads], &device);
424        let shifted = Tensor::cat(vec![zeros_bdh, da_bsh.clone().narrow(1, 0, sequence - d)], 1);
425        window_bsh = window_bsh + shifted;
426    }
427    Some(window_bsh.exp())
428}

430/// The decay that each cached tap slot already accumulated:
431/// `Πᵣ₌q₊₁^{S−1} αᵣ` for the last `lag` positions `q` (oldest first). `None` at
432/// `lag = 1`: the one slot *is* the last position, so the product is empty.
433///
434/// The tap slots are stored pre-scaled by this, so a lag-`L` tap can cross a
435/// call boundary. The next call supplies the in-call part of the gap
436/// ([`interior_gap_decay`], front-zero-padded), and the slot carries the rest.
437///
438/// # Shapes
439/// - `da_bsh` : `[batch, sequence, nheads]`
440/// - out      : `[batch, lag, nheads]`
441pub fn tail_decay(da_bsh: Tensor<3>, lag: usize) -> Option<Tensor<3>> {
442    if lag <= 1 {
443        return None;
444    }
445    let sequence = da_bsh.dims()[1];
446    // Only the tail window matters, so the cumulative sum is over `lag` terms,
447    // never over the decay of the whole sequence.
448    let tail_blh = da_bsh.narrow(1, sequence - lag, lag);
449    let cumulative_blh = tail_blh.cumsum(1);
450    let total_b1h = cumulative_blh.clone().narrow(1, lag - 1, 1);
451    Some((total_b1h - cumulative_blh).exp())
452}

454/// A `[1, len, 1]` gate over a folded axis: `0` at the **first** micro-step of
455/// each token (`p ≡ 0 mod u`) and `1` elsewhere.
456///
457/// It broadcasts against any `[batch, len, nheads]` stream, in `forward`
458/// (`len` is the folded sequence, which starts at a token boundary) and in
459/// `step` (`len` is `u`, so the gate is its first entry). At `u = 1` every
460/// position is a token start, and the gate is all zeros. That alone is why
461/// [`Trapezoid::HorizontalReset`](crate::mamba3::trapezoid::Trapezoid::HorizontalReset)
462/// becomes [`None`](crate::mamba3::trapezoid::Trapezoid::None) there.
463///
464/// Here, the admissibility rule of a pattern becomes a tensor. The caller
465/// decides *which* mass it applies to: `λ` for
466/// [`Trapezoid::far_tap_crosses_tokens`](crate::mamba3::trapezoid::Trapezoid::far_tap_crosses_tokens),
467/// `μ` for
468/// [`interior_tap_crosses_tokens`](crate::mamba3::trapezoid::Trapezoid::interior_tap_crosses_tokens).
469pub fn token_start_gate(len: usize, micro_steps: usize, device: &Device) -> Tensor<3> {
470    let open: Vec<f32> = (0..len)
471        .map(|p| if p.is_multiple_of(micro_steps) { 0.0 } else { 1.0 })
472        .collect();
473    Tensor::<1>::from_floats(open.as_slice(), device).reshape([1, len, 1])
474}

476/// Output of [`trapezoidal_coefficients`]: the mass `Δₜ` of the step, split.
477///
478/// The masses are **untransported**: the consumer applies `α` and any wider
479/// gap decay, because the pathway decides which one a tap needs. The
480/// double-SSD pass multiplies them in. The single-SSD key scale never does.
481pub struct TrapezoidCoeffs {
482    /// `Δₜ = softplus(dd_dt + dt_bias)`, clamped.
483    pub dt: Tensor<3>,
484    /// The log-decay: `Δₜ · Aₜ`, or, under a
485    /// [`Gain::Kalman`](crate::mamba3::positive::Gain::Kalman) member, the
486    /// decay that the gate computes from it. Every consumer reads this one.
487    pub da: Tensor<3>,
488    /// `αₜ = exp(da) ∈ (0, 1]`: the decay, projected or computed (as `da`).
489    pub alpha: Tensor<3>,
490    /// `νₜ`: the mass of the tap at
491    /// [`tap_lag`](crate::mamba3::trapezoid::Trapezoid::tap_lag), before its
492    /// transport. `None` under
493    /// [`Trapezoid::None`](crate::mamba3::trapezoid::Trapezoid::None), which
494    /// has no left endpoint: the tensor is not formed (not formed as zeros).
495    pub nu: Option<Tensor<3>>,
496    /// `νⁱⁿᵗₜ`: the mass of the extra lag-1 tap. `None` unless
497    /// [`has_interior_tap`](crate::mamba3::trapezoid::Trapezoid::has_interior_tap).
498    pub nu_interior: Option<Tensor<3>>,
499    /// `γₜ = λₜ · Δₜ`: the right-endpoint weight. It is **`Δₜ` itself** when
500    /// there is no `λ` (all of the step is paid at the right endpoint).
501    pub gamma: Tensor<3>,
502    /// `ℓₜ = ln Λₜ` at every position, when the decay comes from a Kalman gate
503    /// ([`crate::mamba3::positive::kalman`]). Its last entry is the next carry.
504    pub log_precision: Option<Tensor<3>>,
505}

507impl TrapezoidCoeffs {
508    /// Makes every padded position (`pad_bs`, `[batch, len]`, `true` at
509    /// padding, `None` ⇒ no padding) the identity step: no decay (`da = 0`,
510    /// `α = 1`) and no mass (`Δ = ν = νⁱⁿᵗ = γ = 0`). Each consumer follows:
511    ///
512    /// - the SSD core carries the state through unchanged,
513    /// - a tap mass that a real position pays forward to a padded one is
514    ///   dropped, exactly as at a call boundary,
515    /// - the rotation (which steps by `Δ`) does not move.
516    ///
517    /// `log_precision` stays as is. Only causal sites read it, and each slot
518    /// takes its carry at its own end.
519    pub fn padded(self, pad_bs: Option<&Tensor<2, Bool>>) -> Self {
520        let Some(pad_bs) = pad_bs else {
521            return self;
522        };
523        let zero = |t: Tensor<3>| crate::padding::fill_padded(t, pad_bs, 0.0);
524        let da = zero(self.da);
525        Self {
526            dt: zero(self.dt),
527            alpha: da.clone().exp(),
528            da,
529            nu: self.nu.map(zero),
530            nu_interior: self.nu_interior.map(zero),
531            gamma: zero(self.gamma),
532            log_precision: self.log_precision,
533        }
534    }
535}

537/// Compute the trapezoidal discretisation coefficients from the raw
538/// (data-dependent) projections. See the header of `mamba3.rs` for the
539/// formulas and the header of `trapezoid.rs` for how the masses divide.
540///
541/// All data tensors are `[batch, len, nheads]`, where `len` is the folded
542/// sequence (`forward`) or the `u` micro-steps of one token (`step`).
543/// `dt_bias_h` broadcasts to match. In both cases, the gates of the patterns
544/// are periodic along axis 1.
545///
546/// `lambda_raw` is `None` under
547/// [`Trapezoid::None`](crate::mamba3::trapezoid::Trapezoid::None), whose block
548/// projects no `λ` channels. The coefficients then have their `λ ≡ 1` values
549/// *by construction*: no mass is formed, and `γ` **is** `Δ` (the same tensor,
550/// not a new `1 · Δ`). `mu_raw` is `None` unless the pattern
551/// [`has_interior_tap`](crate::mamba3::trapezoid::Trapezoid::has_interior_tap).
552/// The same holds one level down: no `μ`, no second mass.
553///
554/// A gate closes by giving its mass back, not by dropping it. `λ` is `1` where
555/// the far tap is not admissible (so `γ` takes all of the step). `μ` is `0`
556/// where the interior tap is not admissible (so the far tap takes all of the
557/// left endpoint). Both are exact at the ends, so the degenerate members equal
558/// their targets bit for bit.
559///
560/// `gain` is `Some` under a Kalman [`Gain`](crate::mamba3::positive::Gain).
561/// This function then computes the decay
562/// ([`crate::mamba3::positive::kalman::gate`]) **before** it forms `α`. So no
563/// consumer (the tap transports, the tap slots, the single-SSD key scale) can
564/// read the projected decay by mistake. The masses keep the projected `Δ`. The
565/// gate reads them in logs, formed from the same pre-activations
566/// ([`LogMasses`](crate::mamba3::positive::kalman::LogMasses)).
567pub fn trapezoidal_coefficients(
568    dd_dt: Tensor<3>,
569    dd_a_raw: Tensor<3>,
570    lambda_raw: Option<Tensor<3>>,
571    mu_raw: Option<Tensor<3>>,
572    dt_bias_h: Tensor<1>,
573    spec: TrapezoidSpec,
574    gain: Option<crate::mamba3::positive::kalman::GainInput>,
575) -> TrapezoidCoeffs {
576    let (dt_limit, a_floor) = (spec.dt_limit, spec.a_floor);
577    // Broadcast dt_bias_h [nheads] → [1, 1, nheads] so the addition aligns on
578    // the last dim.
579    let dt_bias_broadcast = dt_bias_h.unsqueeze::<3>();
580    let dt_raw = dd_dt + dt_bias_broadcast;
581    let dt = softplus(dt_raw.clone()).clamp(dt_limit.0, dt_limit.1);
582    // `A = −max(softplus(·), a_floor) ∈ (−∞, −a_floor]`. Apply the floor to
583    // the (positive) softplus *before* the negation. A method call binds
584    // tighter than unary minus, so `-softplus(x).clamp(NEG_INFINITY,
585    // -a_floor)` would clamp the positive softplus to the constant `-a_floor`
586    // and give `A ≡ +a_floor`: a *growing* state (`α > 1`) with a dead `dd_A`
587    // projection.
588    let a = -softplus(dd_a_raw).clamp(a_floor, f64::INFINITY);
589    let da = dt.clone() * a;
590    let gate = |crosses: bool| {
591        (!crosses).then(|| token_start_gate(dt.dims()[1], spec.micro_steps, &dt.device()))
592    };
593    // Where the far tap is closed — read by `λ` below and by the gate's masses.
594    let far_open_1s1 = lambda_raw
595        .as_ref()
596        .and_then(|_| gate(spec.pattern.far_tap_crosses_tokens()));
597    let (da, log_precision) = match gain {
598        Some(gain) => {
599            use crate::mamba3::positive::kalman;
600            let masses =
601                kalman::LogMasses::new(dt_raw, dt_limit, lambda_raw.clone(), far_open_1s1.clone());
602            let out = kalman::gate(da, masses, gain);
603            (out.da_bsh, Some(out.log_precision_bsh))
604        }
605        None => (da, None),
606    };
607    let alpha = da.clone().exp();
608    let (nu, nu_interior, gamma) = match lambda_raw {
609        Some(lambda_raw) => {
610            let lambda = burn::tensor::activation::sigmoid(lambda_raw);
611            // A closed far tap means λ = 1 there: all of the step is paid at
612            // the right endpoint. This makes the gated pattern a submodel of
613            // the ungated one, not a lossy version of it.
614            let lambda = match far_open_1s1 {
615                Some(open_1s1) => lambda * open_1s1.clone() + (-open_1s1 + 1.0),
616                None => lambda,
617            };
618            let nu = (-lambda.clone() + 1.0) * dt.clone();
619            let gamma = lambda * dt.clone();
620            // …and a closed interior tap means μ = 0 there: its share of the
621            // left endpoint returns to the far tap, one level up.
622            let (nu, nu_interior) = match mu_raw {
623                Some(mu_raw) => {
624                    let mu = burn::tensor::activation::sigmoid(mu_raw);
625                    let mu = match gate(spec.pattern.interior_tap_crosses_tokens()) {
626                        Some(open_1s1) => mu * open_1s1,
627                        None => mu,
628                    };
629                    let nu_interior = nu.clone() * mu.clone();
630                    (nu * (-mu + 1.0), Some(nu_interior))
631                }
632                None => (nu, None),
633            };
634            (Some(nu), nu_interior, gamma)
635        }
636        // λ ≡ 1: the left endpoint is unweighted and the right one takes Δ whole.
637        None => (None, None, dt.clone()),
638    };
639    TrapezoidCoeffs {
640        dt,
641        da,
642        alpha,
643        nu,
644        nu_interior,
645        gamma,
646        log_precision,
647    }
648}

650/// QK-Norm → GQA-expand groups→heads → add per-(head, mimo-rank) bias.
651///
652/// The input is the raw B/C projection already reshaped to expose the group
653/// dim, with last dim = `state_rank`. The output replaces the group dim with
654/// the head dim, leaving the last dim untouched.
655///
656/// `DP1 = D + 1` (required by [`gqa_expand_to_heads`]'s intermediate rank).
657pub fn qk_norm_expand_bias<const D: usize, const DP1: usize>(
658    raw_mgr: Tensor<D>,
659    norm: &RmsNorm,
660    bias_hmr: Tensor<3>,
661    group_dim: usize,
662    nheads: usize,
663) -> Tensor<D> {
664    // RmsNorm operates on the last dim only, so the leading shape passes through.
665    let normed = norm.forward(raw_mgr);
666    let expanded = gqa_expand_to_heads::<D, DP1>(normed, group_dim, nheads);
667    // Broadcast bias [nheads, mimo_rank, state_rank] → [1, ..., 1, mimo_rank, nheads, state_rank].
668    let bias = bias_hmr.swap_dims(0, 1).unsqueeze::<D>();
669    expanded + bias
670}

672/// Rank-summed outer product `state[b, h, p, r] = Σₘ v[b, m, h, p] · k[b, m, h, r]`
673/// (`einsum('bmhp,bmhr->bhpr')`).
674///
675/// This is the per-token state contribution: each MIMO rank adds an outer
676/// product `v[m] ⊗ k[m]`, and the shared state accumulates their sum.
677///
678/// At `mimo_rank == 1` the contracted dimension is 1, so the GEMM is rank-1
679/// and the sum is one outer product. [`mimo_outer_sum_siso`] writes it as a
680/// broadcast multiply instead. The faster form depends on the backend, by a
681/// wide margin at these (tiny, decode-sized) shapes. The broadcast is faster
682/// on CUDA and *much* slower on the portable CPU backends, whose
683/// broadcast-elementwise path is more than 10× slower than their matmul here.
684/// So the per-token flag
685/// [`Mamba3Config::siso_specialization_decode`](crate::mamba3::mamba3::Mamba3Config::siso_specialization_decode)
686/// selects it, not the chunkwise flag (whose trade-off is the opposite). Both
687/// branches compute the same values and gradients.
688pub fn mimo_outer_sum(
689    v_bmhp: Tensor<4>,
690    k_bmhr: Tensor<4>,
691    siso_specialization: bool,
692) -> Tensor<4> {
693    let [_batch, mimo_rank, _nheads, _per_head_dim] = v_bmhp.dims();
694    if mimo_rank == 1 && siso_specialization {
695        mimo_outer_sum_siso(v_bmhp, k_bmhr)
696    } else {
697        mimo_outer_sum_mimo(v_bmhp, k_bmhr)
698    }
699}
700
701/// SISO (`mimo_rank == 1`) rank-summed outer product: the single `v ⊗ k` as a
702/// broadcast multiply of `[b, h, p, 1]` by `[b, h, 1, r]`.
703pub fn mimo_outer_sum_siso(v_bmhp: Tensor<4>, k_bmhr: Tensor<4>) -> Tensor<4> {
704    let v_bhp1: Tensor<4> = v_bmhp.squeeze_dim::<3>(1).unsqueeze_dim(3);
705    let k_bh1r: Tensor<4> = k_bmhr.squeeze_dim::<3>(1).unsqueeze_dim(2);
706    v_bhp1 * k_bh1r
707}
708
709/// General MIMO rank-summed outer product: a matmul contracting `mimo_rank`.
710pub fn mimo_outer_sum_mimo(v_bmhp: Tensor<4>, k_bmhr: Tensor<4>) -> Tensor<4> {
711    // v_bmhp [b, m, h, p] -> v_bhpm [b, h, p, m]; k_bmhr [b, m, h, r] -> k_bhmr.
712    let v_bhpm = v_bmhp.permute([0, 2, 3, 1]);
713    let k_bhmr = k_bmhr.swap_dims(1, 2);
714    v_bhpm.matmul(k_bhmr)
715}

717/// Build the MIMO value tensor `v = x ⊙ mimo_x` with broadcasting.
718///
719/// Inserts a `mimo_rank` axis at `insert_dim`. When `mimo_x_hmp` is `None`
720/// (SISO), the new axis has size 1 and `x` goes through unchanged. Otherwise,
721/// broadcasting fills the new axis to size `mimo_rank`.
722///
723/// `DP1 = D + 1`.
724pub fn build_v_with_mimo<const D: usize, const DP1: usize>(
725    x: Tensor<D>,
726    mimo_x_hmp: Option<&Tensor<3>>,
727    insert_dim: usize,
728) -> Tensor<DP1> {
729    let x_with_rank_axis = x.unsqueeze_dim::<DP1>(insert_dim);
730    match mimo_x_hmp {
731        None => x_with_rank_axis,
732        Some(mimo_x_hmp) => {
733            // mimo_x_hmp [nheads, mimo_rank, per_head_dim] → swap_dims to
734            // [mimo_rank, nheads, per_head_dim] → unsqueeze leading 1s. The
735            // result broadcasts against `x_with_rank_axis` over (batch, seq, …).
736            let mimo_x_broadcast = mimo_x_hmp.clone().swap_dims(0, 1).unsqueeze::<DP1>();
737            x_with_rank_axis * mimo_x_broadcast
738        }
739    }
740}
