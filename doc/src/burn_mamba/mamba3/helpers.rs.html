1//! Shared helpers used by both [`Mamba3::forward`](super::mamba3::Mamba3::forward)
2//! and [`Mamba3::step`](super::mamba3::Mamba3::step). They isolate three blocks
3//! that previously appeared in both methods at different ranks:
4//!
5//! 1. Trapezoidal discretisation: `dt`, `α`, `β`, `γ`, `da`.
6//! 2. QK-norm + GQA expansion + per-(head, mimo-rank) bias on B / C.
7//! 3. MIMO `V` construction: broadcast-multiply `x` by `mimo_x_hmp`.
8//! 4. The rank-summed outer product `Σₘ v[m] ⊗ k[m]` feeding the SSM state
9//!    (SISO-branched).
10//! 5. Peeling the rotation channels off the in-projection.
11//! 6. The trapezoid tap's lag arithmetic: the shift, the gap transport, the
12//!    decay a cached slot carries across a call boundary, and the per-position
13//!    gate a tap pattern admits its taps by.
14//!
15//! Most helpers are generic over the rank `D` of the data tensors so a single
16//! definition serves both the sequence-aware (`forward`) and single-token
17//! (`step`) code paths. The discretisation is the exception: MambaProduct gives
18//! `step` a `u` axis of its own, so both paths reach it at rank 3.
19

26/// Peel a **trailing** in-projection segment of `width` channels off `proj`,
27/// yielding `None` for it when the block projects none.
28///
29/// Used for the two segments a block may omit entirely — the rotation channels
30/// ([`RotationKind::Real1D`](crate::mamba3::rotation::RotationKind::Real1D)
31/// projects none) and then `λ`
32/// ([`Trapezoid::None`](crate::mamba3::trapezoid::Trapezoid::None) projects
33/// none) — which is why the in-projection lays them out last, in that order.
34///
35/// An optional slice cannot simply be one more entry in the main `split_into`:
36/// Burn has no zero-width tensors, and `split_with_sizes` *drops* a zero-length
37/// segment rather than returning an empty one, so the destructuring would come
38/// up one part short.
39///
40/// # Shapes
41/// - `proj` : `[..., w]` along `dim`
42/// - out    : `[..., w − width]` and, if any, `[..., width]`
43pub fn split_trailing<const D: usize>(
44    proj: Tensor<D>,
45    width: usize,
46    dim: usize,
47) -> (Tensor<D>, Option<Tensor<D>>) {
48    if width == 0 {
49        return (proj, None);
50    }
51    let rest = proj.dims()[dim] - width;
52    let tail = proj.clone().narrow(dim, rest, width);
53    (proj.narrow(dim, 0, rest), Some(tail))
54}

56/// Shift a per-position stream back by `lag`, seeding the first `lag` positions
57/// from the cache's tap slots: `out[p] = stream[p − lag]`, and `prefix[p]` where
58/// that index is before the call.
59///
60/// This is the "shift-before-chunking" of the double-SSD pathway generalised
61/// from lag 1 to [`Trapezoid::tap_lag`](crate::mamba3::trapezoid::Trapezoid::tap_lag).
62/// A folded sequence is `tokens · u` long and `lag ∈ {1, u}`, so `sequence ≥
63/// lag` always holds; at equality the whole call is prefix.
64///
65/// # Shapes
66/// - `stream` : `[batch, sequence, …]`
67/// - `prefix` : `[batch, lag, …]`
68/// - out      : `[batch, sequence, …]`
69pub fn shift_stream<const D: usize>(
70    stream: Tensor<D>,
71    prefix: Tensor<D>,
72    lag: usize,
73) -> Tensor<D> {
74    let sequence = stream.dims()[1];
75    assert_eq!(prefix.dims()[1], lag, "one prefix slot per lagged position");
76    assert!(sequence >= lag, "a call is at least one token, i.e. `lag` long");
77    if sequence == lag {
78        prefix
79    } else {
80        Tensor::cat(vec![prefix, stream.narrow(1, 0, sequence - lag)], 1)
81    }
82}

84/// The **interior** of a lag-`lag` tap's gap: `Πᵈ⁼¹..ˡᵃᵍ⁻¹ αₚ₋ᵈ`, or `None` at
85/// `lag = 1` (where the gap has no interior).
86///
87/// A tap at lag `L` is transported across its own gap, `Πᵈ⁼⁰..ᴸ⁻¹ αₚ₋ᵈ`
88/// (`info/trapezoid-as-integration.md` §9). `β = (1−λ)Δα` already carries the
89/// `d = 0` factor, so this is the rest of it.
90///
91/// The front is **zero-padded**, not clamped to what the call happens to hold:
92/// for `p < L` the missing factors are exactly the ones the cache's `v` slots
93/// were already scaled by when they were stored (see [`tail_decay`]).
94///
95/// # Shapes
96/// - `da_bsh` : `[batch, sequence, nheads]` (`Δ·A`, the log-decay)
97/// - out      : `[batch, sequence, nheads]`
98pub fn interior_gap_decay(da_bsh: Tensor<3>, lag: usize) -> Option<Tensor<3>> {
99    if lag <= 1 {
100        return None;
101    }
102    let [batch, sequence, nheads] = da_bsh.dims();
103    let device = da_bsh.device();
104    let mut window_bsh = Tensor::zeros([batch, sequence, nheads], &device);
105    for d in 1..lag {
106        let zeros_bdh = Tensor::zeros([batch, d, nheads], &device);
107        let shifted = Tensor::cat(vec![zeros_bdh, da_bsh.clone().narrow(1, 0, sequence - d)], 1);
108        window_bsh = window_bsh + shifted;
109    }
110    Some(window_bsh.exp())
111}

113/// The decay each cached tap slot has already accumulated: `Πᵣ₌q₊₁^{S−1} αᵣ` for
114/// the last `lag` positions `q` (oldest first), or `None` at `lag = 1` (the one
115/// slot *is* the last position, so the product is empty).
116///
117/// Storing the tap slots pre-scaled by this is what lets a lag-`L` tap span a
118/// call boundary: the next call supplies the in-call part of the gap
119/// ([`interior_gap_decay`], front-zero-padded) and the slot carries the rest.
120///
121/// # Shapes
122/// - `da_bsh` : `[batch, sequence, nheads]`
123/// - out      : `[batch, lag, nheads]`
124pub fn tail_decay(da_bsh: Tensor<3>, lag: usize) -> Option<Tensor<3>> {
125    if lag <= 1 {
126        return None;
127    }
128    let sequence = da_bsh.dims()[1];
129    // Only the tail window matters, so the cumulative sum is over `lag` terms
130    // and never accumulates the whole sequence's decay.
131    let tail_blh = da_bsh.narrow(1, sequence - lag, lag);
132    let cumulative_blh = tail_blh.cumsum(1);
133    let total_b1h = cumulative_blh.clone().narrow(1, lag - 1, 1);
134    Some((total_b1h - cumulative_blh).exp())
135}

137/// A `[1, len, 1]` gate over a folded axis: `0` at each token's **first**
138/// micro-step (`p ≡ 0 mod u`) and `1` elsewhere.
139///
140/// Broadcasts against any `[batch, len, nheads]` stream, in `forward` (`len` is
141/// the folded sequence, which starts at a token boundary) and in `step` (`len`
142/// is `u` itself, so the gate is its first entry). At `u = 1` every position is
143/// a token start and the gate is all zeros — which is the whole of why
144/// [`Trapezoid::HorizontalReset`](crate::mamba3::trapezoid::Trapezoid::HorizontalReset)
145/// degenerates to [`None`](crate::mamba3::trapezoid::Trapezoid::None) there.
146///
147/// This is where a pattern's admissibility rule becomes a tensor; *which* mass
148/// it applies to is the caller's — `λ` for
149/// [`Trapezoid::far_tap_crosses_tokens`](crate::mamba3::trapezoid::Trapezoid::far_tap_crosses_tokens),
150/// `μ` for
151/// [`interior_tap_crosses_tokens`](crate::mamba3::trapezoid::Trapezoid::interior_tap_crosses_tokens).
152pub fn token_start_gate(len: usize, micro_steps: usize, device: &Device) -> Tensor<3> {
153    let open: Vec<f32> = (0..len)
154        .map(|p| if p.is_multiple_of(micro_steps) { 0.0 } else { 1.0 })
155        .collect();
156    Tensor::<1>::from_floats(open.as_slice(), device).reshape([1, len, 1])
157}

159/// Output of [`trapezoidal_coefficients`] — the step's mass `Δₜ`, split.
160///
161/// The masses are **untransported**: `α` and any wider gap decay are the
162/// consumer's to apply, since which one a tap needs depends on the pathway (the
163/// double-SSD pass multiplies them in, the single-SSD key scale never does).
164pub struct TrapezoidCoeffs {
165    /// `Δₜ = softplus(dd_dt + dt_bias)`, clamped.
166    pub dt: Tensor<3>,
167    /// `Δₜ · Aₜ` (negative; the log-decay).
168    pub da: Tensor<3>,
169    /// `αₜ = exp(Δₜ · Aₜ) ∈ (0, 1]` — decay.
170    pub alpha: Tensor<3>,
171    /// `νₜ` — the mass of the tap at
172    /// [`tap_lag`](crate::mamba3::trapezoid::Trapezoid::tap_lag), before its
173    /// transport. `None` under
174    /// [`Trapezoid::None`](crate::mamba3::trapezoid::Trapezoid::None), which has
175    /// no left endpoint: the tensor is not formed rather than formed as zeros.
176    pub nu: Option<Tensor<3>>,
177    /// `νⁱⁿᵗₜ` — the mass of the extra lag-1 tap, `None` unless
178    /// [`has_interior_tap`](crate::mamba3::trapezoid::Trapezoid::has_interior_tap).
179    pub nu_interior: Option<Tensor<3>>,
180    /// `γₜ = λₜ · Δₜ` — right-endpoint weight, and **`Δₜ` itself** when there is
181    /// no `λ` (the whole step is paid at the right endpoint).
182    pub gamma: Tensor<3>,
183}

185/// Compute the trapezoidal discretisation coefficients from the raw
186/// (data-dependent) projections. See the top-of-`mamba3.rs` docs for the
187/// formulas and `trapezoid.rs`'s header for how the masses divide.
188///
189/// All data tensors are `[batch, len, nheads]`, `len` being the folded sequence
190/// (`forward`) or the `u` micro-steps of one token (`step`); `dt_bias_h` is
191/// broadcast to match. Axis 1 is what the patterns' gates are periodic in, in
192/// both cases.
193///
194/// `lambda_raw` is `None` under
195/// [`Trapezoid::None`](crate::mamba3::trapezoid::Trapezoid::None) — the block
196/// projects no `λ` channels — and the coefficients then take their `λ ≡ 1`
197/// values *by construction*: no mass is formed and `γ` **is** `Δ`, sharing its
198/// tensor rather than recomputing `1 · Δ`. `mu_raw` is `None` unless the pattern
199/// [`has_interior_tap`](crate::mamba3::trapezoid::Trapezoid::has_interior_tap),
200/// and the same holds one level down: no `μ`, no second mass.
201///
202/// A gate closes by handing its mass back, not by dropping it: `λ` is set to `1`
203/// where the far tap is inadmissible (so `γ` takes the step whole) and `μ` to
204/// `0` where the interior one is (so the far tap takes the left endpoint whole).
205/// Both are exact at the ends, so the degenerate members are their targets bit
206/// for bit.
207pub fn trapezoidal_coefficients(
208    dd_dt: Tensor<3>,
209    dd_a_raw: Tensor<3>,
210    lambda_raw: Option<Tensor<3>>,
211    mu_raw: Option<Tensor<3>>,
212    dt_bias_h: Tensor<1>,
213    spec: TrapezoidSpec,
214) -> TrapezoidCoeffs {
215    let (dt_limit, a_floor) = (spec.dt_limit, spec.a_floor);
216    // Broadcast dt_bias_h [nheads] → [1, 1, nheads] so the addition aligns on
217    // the last dim.
218    let dt_bias_broadcast = dt_bias_h.unsqueeze::<3>();
219    let dt = softplus(dd_dt + dt_bias_broadcast).clamp(dt_limit.0, dt_limit.1);
220    // `A = −max(softplus(·), a_floor) ∈ (−∞, −a_floor]`. The floor must be
221    // applied to the (positive) softplus *before* negating: a method call
222    // binds tighter than unary minus, so `-softplus(x).clamp(NEG_INFINITY,
223    // -a_floor)` would collapse the positive softplus to the constant
224    // `-a_floor` and yield `A ≡ +a_floor` — a *growing* state (`α > 1`) with a
225    // dead `dd_A` projection.
226    let a = -softplus(dd_a_raw).clamp(a_floor, f64::INFINITY);
227    let da = dt.clone() * a;
228    let alpha = da.clone().exp();
229    let gate = |crosses: bool| {
230        (!crosses).then(|| token_start_gate(dt.dims()[1], spec.micro_steps, &dt.device()))
231    };
232    let (nu, nu_interior, gamma) = match lambda_raw {
233        Some(lambda_raw) => {
234            let lambda = burn::tensor::activation::sigmoid(lambda_raw);
235            // A closed far tap means λ = 1 there: the whole step is paid at the
236            // right endpoint, which is what makes the gated pattern a submodel
237            // of the ungated one rather than a lossy version of it.
238            let lambda = match gate(spec.pattern.far_tap_crosses_tokens()) {
239                Some(open_1s1) => lambda * open_1s1.clone() + (-open_1s1 + 1.0),
240                None => lambda,
241            };
242            let nu = (-lambda.clone() + 1.0) * dt.clone();
243            let gamma = lambda * dt.clone();
244            // …and a closed interior tap means μ = 0 there: its share of the
245            // left endpoint returns to the far tap, one level up.
246            let (nu, nu_interior) = match mu_raw {
247                Some(mu_raw) => {
248                    let mu = burn::tensor::activation::sigmoid(mu_raw);
249                    let mu = match gate(spec.pattern.interior_tap_crosses_tokens()) {
250                        Some(open_1s1) => mu * open_1s1,
251                        None => mu,
252                    };
253                    let nu_interior = nu.clone() * mu.clone();
254                    (nu * (-mu + 1.0), Some(nu_interior))
255                }
256                None => (nu, None),
257            };
258            (Some(nu), nu_interior, gamma)
259        }
260        // λ ≡ 1: the left endpoint is unweighted and the right one takes Δ whole.
261        None => (None, None, dt.clone()),
262    };
263    TrapezoidCoeffs {
264        dt,
265        da,
266        alpha,
267        nu,
268        nu_interior,
269        gamma,
270    }
271}

273/// QK-Norm → GQA-expand groups→heads → add per-(head, mimo-rank) bias.
274///
275/// The input is the raw B/C projection already reshaped to expose the group
276/// dim, with last dim = `state_rank`. The output replaces the group dim with
277/// the head dim, leaving the last dim untouched.
278///
279/// `DP1 = D + 1` (required by [`gqa_expand_to_heads`]'s intermediate rank).
280pub fn qk_norm_expand_bias<const D: usize, const DP1: usize>(
281    raw_mgr: Tensor<D>,
282    norm: &RmsNorm,
283    bias_hmr: Tensor<3>,
284    group_dim: usize,
285    nheads: usize,
286) -> Tensor<D> {
287    // RmsNorm operates on the last dim only, so the leading shape passes through.
288    let normed = norm.forward(raw_mgr);
289    let expanded = gqa_expand_to_heads::<D, DP1>(normed, group_dim, nheads);
290    // Broadcast bias [nheads, mimo_rank, state_rank] → [1, ..., 1, mimo_rank, nheads, state_rank].
291    let bias = bias_hmr.swap_dims(0, 1).unsqueeze::<D>();
292    expanded + bias
293}

295/// Rank-summed outer product `state[b, h, p, r] = Σₘ v[b, m, h, p] · k[b, m, h, r]`
296/// (`einsum('bmhp,bmhr->bhpr')`).
297///
298/// This is the per-token state contribution: each MIMO rank contributes an
299/// outer product `v[m] ⊗ k[m]` and the shared state accumulates their sum.
300///
301/// At `mimo_rank == 1` the contracted dimension is 1, so the GEMM is rank-1 and
302/// the sum is a single outer product — [`mimo_outer_sum_siso`] writes it as a
303/// broadcast multiply instead. Which form wins is backend-dependent, and by a
304/// wide margin at these (tiny, decode-sized) shapes: the broadcast is the
305/// faster one on CUDA and *much* slower on the portable CPU backends, whose
306/// broadcast-elementwise path trails their matmul by more than an order of
307/// magnitude here. The choice is therefore
308/// [`Mamba3Config::siso_specialization_decode`](crate::mamba3::mamba3::Mamba3Config::siso_specialization_decode)'s
309/// — the per-token flag, *not* the chunkwise one, whose verdict is the opposite;
310/// both branches compute the same values and gradients.
311pub fn mimo_outer_sum(
312    v_bmhp: Tensor<4>,
313    k_bmhr: Tensor<4>,
314    siso_specialization: bool,
315) -> Tensor<4> {
316    let [_batch, mimo_rank, _nheads, _per_head_dim] = v_bmhp.dims();
317    if mimo_rank == 1 && siso_specialization {
318        mimo_outer_sum_siso(v_bmhp, k_bmhr)
319    } else {
320        mimo_outer_sum_mimo(v_bmhp, k_bmhr)
321    }
322}

324/// SISO (`mimo_rank == 1`) rank-summed outer product: the single `v ⊗ k` as a
325/// broadcast multiply of `[b, h, p, 1]` by `[b, h, 1, r]`.
326pub fn mimo_outer_sum_siso(v_bmhp: Tensor<4>, k_bmhr: Tensor<4>) -> Tensor<4> {
327    let v_bhp1: Tensor<4> = v_bmhp.squeeze_dim::<3>(1).unsqueeze_dim(3);
328    let k_bh1r: Tensor<4> = k_bmhr.squeeze_dim::<3>(1).unsqueeze_dim(2);
329    v_bhp1 * k_bh1r
330}
331
332/// General MIMO rank-summed outer product: a matmul contracting `mimo_rank`.
333pub fn mimo_outer_sum_mimo(v_bmhp: Tensor<4>, k_bmhr: Tensor<4>) -> Tensor<4> {
334    // v_bmhp [b, m, h, p] -> v_bhpm [b, h, p, m]; k_bmhr [b, m, h, r] -> k_bhmr.
335    let v_bhpm = v_bmhp.permute([0, 2, 3, 1]);
336    let k_bhmr = k_bmhr.swap_dims(1, 2);
337    v_bhpm.matmul(k_bhmr)
338}

343/// Build the MIMO value tensor `v = x ⊙ mimo_x` with broadcasting.
344///
345/// Inserts a `mimo_rank` axis at `insert_dim`. When `mimo_x_hmp` is `None`
346/// (SISO), the inserted axis has size 1 and `x` is passed through; otherwise
347/// broadcasting fills the inserted axis to size `mimo_rank`.
348///
349/// `DP1 = D + 1`.
350pub fn build_v_with_mimo<const D: usize, const DP1: usize>(
351    x: Tensor<D>,
352    mimo_x_hmp: Option<&Tensor<3>>,
353    insert_dim: usize,
354) -> Tensor<DP1> {
355    let x_with_rank_axis = x.unsqueeze_dim::<DP1>(insert_dim);
356    match mimo_x_hmp {
357        None => x_with_rank_axis,
358        Some(mimo_x_hmp) => {
359            // mimo_x_hmp [nheads, mimo_rank, per_head_dim] → swap_dims to
360            // [mimo_rank, nheads, per_head_dim] → unsqueeze leading 1s. The
361            // result broadcasts against `x_with_rank_axis` over (batch, seq, …).
362            let mimo_x_broadcast = mimo_x_hmp.clone().swap_dims(0, 1).unsqueeze::<DP1>();
363            x_with_rank_axis * mimo_x_broadcast
364        }
365    }
