1//! # Constant-input stepping shortcut — `step_infinite`
2//!
3//! When the **same token** is fed to [`Mamba3::step`] over and over, every
4//! data-dependent quantity is the same at each step: the trapezoid
5//! coefficients `α`, `β`, `γ`, the QK-normed `b`/`c`, and the per-step
6//! rotation increment (angle `θ̂` for `Complex2D`, unit quaternion `q` for
7//! `Quaternion4D`). Only the *cumulative* rotation moves — by that same
8//! increment each step — so with `P` the per-step rotation operator and `R₁`
9//! the cumulative rotation at the first constant step,
10//!
11//! ```text
12//!   Bₜ = R₁ Pᵗ⁻¹ b ,   Cₜ = R₁ Pᵗ⁻¹ c ,
13//!   hₜ = α hₜ₋₁ + x ⊗ (β Bₜ₋₁ + γ Bₜ)            (t ≥ 2)
14//! ```
15//!
16//! and the unrolled steps collapse to a **matrix geometric series** in
17//! `α P⁻¹` (spectral radius `α < 1`, since `a_floor > 0` and `Δ > 0`).
18//!
19//! `P` is block-diagonal: per RoPE pair it is the complex scalar `e^{iθ̂}`, per
20//! quaternion block left-multiplication by `w = q*`. In both cases all the
21//! factors live in one **abelian** subalgebra (powers of a single rotation
22//! commute — for quaternions, `span{1, û} ≅ ℂ`), so the series is a handful of
23//! scalar complex / quaternion ops per (head, pair/block): `n` steps cost O(1).
24//!
25//! In the readout `y_n = Cₙᵀ h_n + D x` the unbounded phase cancels
26//! (`⟨R c, R v⟩ = ⟨c, v⟩`), leaving only the *relative* rotation `P⁻¹`. As
27//! `n → ∞` the initial-state term and the partial-sum correction decay like
28//! `αⁿ`, so the **output converges** even though the state `h` orbits forever:
29//!
30//! ```text
31//!   y_∞ = xᵀ · cᵀ (γ + β P⁻¹)(I − α P⁻¹)⁻¹ b  +  D x
32//! ```
33//!
34//! — the block's stationary fixed point, independent of any starting cache.
35//! [`Mamba3::step_infinite`] evaluates exactly this (and therefore takes and
36//! returns no cache).
37//!
38//! Numerics: the rotation angle is reduced mod `2π` with the value-exact
39//! [`wrap_angle`]; the geometric denominators satisfy
40//! `|1 − α e^{−iθ̂}|² ≥ (1 − α)²` and are floored by
41//! [`div_eps`](crate::utils::div_eps). When `α → 1` *and* `θ̂ → 0` the series
42//! value `(β+γ)/(1−α)` stays finite but loses fp32 precision once `1 − α` nears
43//! the epsilon floor — the same regime where the unrolled recurrence itself
44//! accumulates near-undamped terms.
45

62impl Mamba3 {
63    /// Stationary **fixed point** of the block under a constant token: the
64    /// limit of `step(input, …)` outputs as the same `input_bd` is stepped
65    /// forever.
66    ///
67    /// Closed form, O(1) in the horizon (see the module header for the
68    /// derivation). The limit forgets any starting state (`αⁿ → 0`) and the
69    /// SSM state itself never converges (it keeps rotating; only the output
70    /// does), so this takes **no cache and returns none**.
71    ///
72    /// Both rotation kinds and both SSD pathways are covered (the recurrence
73    /// is pathway-agnostic at boundaries). Differentiable; gradients are the
74    /// limit of the unrolled gradients.
75    ///
76    /// # Shapes
77    /// - `input_bd` : `[batch, d_model]`
78    /// - output     : `[batch, d_model]`
79    pub fn step_infinite(&self, input_bd: Tensor<2>) -> Tensor<2> {
80        let StepProjection {
81            z_bi,
82            x_bhp,
83            b_bmhr,
84            c_bmhr,
85            rot_ba,
86            dt_bh,
87            alpha_bh,
88            beta_bh,
89            gamma_bh,
90        } = self.step_project(input_bd);
91        let [batch, mimo_rank, nheads, _state_rank] = b_bmhr.dims();
92        let eps = div_eps(alpha_bh.dtype());
93
94        // Rotation-free channels: (β + γ) / (1 − α).
95        let tail_bh =
96            (beta_bh.clone() + gamma_bh.clone()) / (-alpha_bh.clone() + 1.0).clamp_min(eps);
97
98        // Per-pair/block readout factor  m = (γ + β P⁻¹)(1 − α P⁻¹)⁻¹  applied
99        // to `b`; the cumulative rotation cancels against `Cₙ` (orthogonality).
100        let spec = self.rotation_spec();
101        let b_eff_bmhr = match spec.kind {
102            RotationKind::Complex2D => {
103                // The same per-step increment `forward`/`step` use — one
104                // definition, so the fixed point cannot drift from the
105                // recurrence it is the limit of.
106                let theta_bha = angle_increment::<2, 3>(rot_ba, dt_bh, spec.range); // θ̂
107                let (cos, sin) = cos_sin(theta_bha);
108                let a_bh1 = alpha_bh.unsqueeze_dim::<3>(2);
109                let beta_bh1 = beta_bh.unsqueeze_dim::<3>(2);
110                let gamma_bh1 = gamma_bh.unsqueeze_dim::<3>(2);
111                // (γ + β e^{−iθ̂}) / (1 − α e^{−iθ̂})
112                let num_re = gamma_bh1 + beta_bh1.clone() * cos.clone();
113                let num_im = -beta_bh1 * sin.clone();
114                let den_re = -a_bh1.clone() * cos + 1.0;
115                let den_im = a_bh1 * sin;
116                let (m_re, m_im) = complex_div(num_re, num_im, den_re, den_im);
117                mul_complex_partial(b_bmhr, m_re, m_im, tail_bh, self.rope_dim, mimo_rank == 1)
118            }
119            RotationKind::Rotor4D => {
120                let g_bhk3 = generator_increment::<2, 3, 4>(
121                    rot_ba,
122                    dt_bh,
123                    self.num_rotation_blocks(),
124                    spec.range,
125                );
126                let qp_bhk4 = quat_from_scaled_axis::<4>(g_bhk3);
127                let (q_bhj4, p_bhj4) = split_rotor(qp_bhk4); // M(v) = q ⊗ v ⊗ p̄
128                let rope_width = spec.rope_dim.min(self.num_quat_blocks * 4);
129                rotor_resolvent_partial(
130                    b_bmhr, q_bhj4, p_bhj4, alpha_bh, beta_bh, gamma_bh, tail_bh, rope_width,
131                )
132            }
133            RotationKind::Quaternion4D => {
134                let g_bhj3 =
135                    generator_increment::<2, 3, 4>(rot_ba, dt_bh, self.num_quat_blocks, spec.range);
136                let q_bhj4 = quat_from_scaled_axis::<4>(g_bhj3); // P⁻¹ ↔ q
137                let alpha_bh11 = alpha_bh.unsqueeze_dims::<4>(&[2, 3]);
138                let beta_bh11 = beta_bh.unsqueeze_dims::<4>(&[2, 3]);
139                let gamma_bh11 = gamma_bh.unsqueeze_dims::<4>(&[2, 3]);
140                // (γ + β q) ⊗ (1 − α q)⁻¹  — all in the abelian subalgebra of q.
141                let num = quat_scalar_affine(gamma_bh11, beta_bh11, q_bhj4.clone());
142                let den_inv = quat_inv(quat_one_minus(q_bhj4 * alpha_bh11));
143                let f_bhj4 = quat_mul(num, den_inv);
144                // As in `rotate_bc_step`, the rotated width comes from
145                // `rope_dim`, so `rope_fraction = 0` rotates nothing.
146                let rope_width = spec.rope_dim.min(self.num_quat_blocks * 4);
147                mul_quat_partial(b_bmhr, f_bhj4, tail_bh, rope_width)
148            }
149        };
150        san(&b_eff_bmhr);
151
152        // y_∞[m] = Σ_m' ⟨c[m], m·b[m']⟩ · x_vals[m']   (then D-skip/gate/out-proj).
153        let mimo_x_hmp = self.mimo_x_hmp.as_ref().map(|p| p.val());
154        let x_vals_bmhp = helpers::build_v_with_mimo::<3, 4>(x_bhp, mimo_x_hmp.as_ref(), 1);
155        let out_m_bmhp = if self.use_siso_decode_kernels() {
156            // SISO: the Gram matrix is the scalar ⟨c, m·b⟩ per (batch, nheads),
157            // so both matmuls degenerate — contract `state_rank` with a
158            // reduction and broadcast the result over `per_head_dim`.
159            let gram_bmh1: Tensor<4> = (c_bmhr * b_eff_bmhr).sum_dim(3);
160            x_vals_bmhp.clone() * gram_bmh1
161        } else {
162            let gram_bhmm = {
163                let c_bhmr = c_bmhr.swap_dims(1, 2);
164                let b_bhrm = b_eff_bmhr.permute([0, 2, 3, 1]);
165                c_bhmr.matmul(b_bhrm) // [batch, nheads, mimo_rank, mimo_rank']
166            };
167            let x_bhmp = x_vals_bmhp.clone().swap_dims(1, 2);
168            gram_bhmm.matmul(x_bhmp).swap_dims(1, 2)
169        };
170        assert_eq!(
171            [batch, mimo_rank, nheads, self.per_head_dim()],
172            out_m_bmhp.dims()
173        );
174        san(&out_m_bmhp);
175
176        self.step_finish(out_m_bmhp, x_vals_bmhp, z_bi)
177    }
178}

184/// `(cos, sin)` of an angle tensor, reduced mod `2π` first (value-exact).
185fn cos_sin(angle_bha: Tensor<3>) -> (Tensor<3>, Tensor<3>) {
186    let a = wrap_angle(angle_bha);
187    (a.clone().cos(), a.sin())
188}

190/// Component-wise complex quotient; `|den|²` floored by `div_eps`.
191fn complex_div(
192    nr: Tensor<3>,
193    ni: Tensor<3>,
194    dr: Tensor<3>,
195    di: Tensor<3>,
196) -> (Tensor<3>, Tensor<3>) {
197    let eps = div_eps(dr.dtype());
198    let d2 = (dr.clone() * dr.clone() + di.clone() * di.clone()).clamp_min(eps);
199    (
200        (nr.clone() * dr.clone() + ni.clone() * di.clone()) / d2.clone(),
201        (ni * dr - nr * di) / d2,
202    )
203}

205/// Multiply the rotation-active entries of `x` by the complex scalar
206/// `(re, im)` per pair — same pairing conventions as [`apply_rope_partial`] —
207/// and the pass-through entries by the real scalar `tail`.
208///
209/// `re`/`im` are `[batch, nheads, num_rope_angles]` (broadcast over the
210/// `mimo_rank` axis); `tail` is `[batch, nheads]`.
211fn mul_complex_partial(
212    x_bmhr: Tensor<4>,
213    re_bha: Tensor<3>,
214    im_bha: Tensor<3>,
215    tail_bh: Tensor<2>,
216    rope_dim: usize,
217    rotate_pairwise: bool,
218) -> Tensor<4> {
219    let [batch, mimo_rank, nheads, state_rank] = x_bmhr.dims();
220    let tail_b1h1 = tail_bh.unsqueeze_dims::<4>(&[1, 3]);
221    if rope_dim == 0 {
222        // RoPE disabled: every channel is a plain geometric series.
223        return x_bmhr * tail_b1h1;
224    }
225    let re_b1ha = re_bha.unsqueeze_dim::<4>(1);
226    let im_b1ha = im_bha.unsqueeze_dim::<4>(1);
227
228    if rotate_pairwise {
229        // Interleaved (SISO/NeoX): pairs are local, so the first `rope_dim`
230        // entries can be handled standalone.
231        let n2 = rope_dim / 2;
232        let head = x_bmhr.clone().narrow(3, 0, rope_dim);
233        let head_pairs = head.reshape([batch, mimo_rank, nheads, n2, 2]);
234        let x0 = head_pairs.clone().narrow(4, 0, 1).squeeze_dim::<4>(4);
235        let x1 = head_pairs.narrow(4, 1, 1).squeeze_dim::<4>(4);
236        let x0m = re_b1ha.clone() * x0.clone() - im_b1ha.clone() * x1.clone();
237        let x1m = im_b1ha * x0 + re_b1ha * x1;
238        let head = Tensor::cat(
239            vec![x0m.unsqueeze_dim::<5>(4), x1m.unsqueeze_dim::<5>(4)],
240            4,
241        )
242        .reshape([batch, mimo_rank, nheads, rope_dim]);
243        if rope_dim == state_rank {
244            head
245        } else {
246            let tail = x_bmhr.narrow(3, rope_dim, state_rank - rope_dim) * tail_b1h1;
247            Tensor::cat(vec![head, tail], 3)
248        }
249    } else {
250        // Half-and-half (MIMO/GPT-J): entry `i` pairs with `i + state_rank/2`;
251        // only the first `rope_dim / 2` pairs are active.
252        let half = state_rank / 2;
253        let active = rope_dim / 2;
254        let x_h1 = x_bmhr.clone().narrow(3, 0, half);
255        let x_h2 = x_bmhr.narrow(3, half, half);
256        let x_h1_rope = x_h1.clone().narrow(3, 0, active);
257        let x_h2_rope = x_h2.clone().narrow(3, 0, active);
258        let h1_m = re_b1ha.clone() * x_h1_rope.clone() - im_b1ha.clone() * x_h2_rope.clone();
259        let h2_m = im_b1ha * x_h1_rope + re_b1ha * x_h2_rope;
260        if active == half {
261            Tensor::cat(vec![h1_m, h2_m], 3)
262        } else {
263            let h1_pass = x_h1.narrow(3, active, half - active) * tail_b1h1.clone();
264            let h2_pass = x_h2.narrow(3, active, half - active) * tail_b1h1;
265            Tensor::cat(vec![h1_m, h1_pass, h2_m, h2_pass], 3)
266        }
267    }
268}

274/// `1 − sq` for a (scalar-scaled) quaternion `sq`: negate, add 1 to the real
275/// part.
276fn quat_one_minus(sq_bhj4: Tensor<4>) -> Tensor<4> {
277    let w = -sq_bhj4.clone().narrow(3, 0, 1) + 1.0;
278    let xyz = -sq_bhj4.narrow(3, 1, 3);
279    Tensor::cat(vec![w, xyz], 3)
280}

282/// `a + b·q` for per-head scalars `a`, `b` (`[batch, nheads, 1, 1]`) and a
283/// quaternion tensor `q`.
284fn quat_scalar_affine(a_bh11: Tensor<4>, b_bh11: Tensor<4>, q_bhj4: Tensor<4>) -> Tensor<4> {
285    let w = a_bh11 + b_bh11.clone() * q_bhj4.clone().narrow(3, 0, 1);
286    let xyz = b_bh11 * q_bhj4.narrow(3, 1, 3);
287    Tensor::cat(vec![w, xyz], 3)
288}

290/// Quaternion inverse `q⁻¹ = q* / ‖q‖²`, with `‖q‖²` floored by `div_eps`.
291/// (Used on `1 − α q`, whose norm is bounded below by `1 − α > 0`.)
292fn quat_inv(q_bhj4: Tensor<4>) -> Tensor<4> {
293    let eps = div_eps(q_bhj4.dtype());
294    let n2 = (q_bhj4.clone() * q_bhj4.clone()).sum_dim(3).clamp_min(eps);
295    quat_conj(q_bhj4) / n2
296}

298/// Left-multiply the first `rope_width` state-rank entries of `x` by the (not
299/// necessarily unit) quaternion `f` per block, and scale the pass-through
300/// entries by the real scalar `tail` — the quaternion analogue of
301/// [`mul_complex_partial`].
302fn mul_quat_partial(
303    x_bmhr: Tensor<4>,
304    f_bhj4: Tensor<4>,
305    tail_bh: Tensor<2>,
306    rope_width: usize,
307) -> Tensor<4> {
308    let [batch, mimo_rank, nheads, state_rank] = x_bmhr.dims();
309    let blocks = f_bhj4.dims()[2];
310    let tail_b1h1 = tail_bh.unsqueeze_dims::<4>(&[1, 3]);
311    if rope_width == 0 {
312        return x_bmhr * tail_b1h1;
313    }
314    let f_bmhj4 = f_bhj4
315        .unsqueeze_dim::<5>(1)
316        .expand([batch, mimo_rank, nheads, blocks, 4]);
317    if rope_width == state_rank {
318        rotate_state_rank_blocks::<4, 5>(x_bmhr, f_bmhj4)
319    } else {
320        let head =
321            rotate_state_rank_blocks::<4, 5>(x_bmhr.clone().narrow(3, 0, rope_width), f_bmhj4);
322        let tail = x_bmhr.narrow(3, rope_width, state_rank - rope_width) * tail_b1h1;
323        Tensor::cat(vec![head, tail], 3)
324    }
325}

331/// Apply `(γ + βM)(I − αM)⁻¹` to the rotation-active entries of `x`, where
332/// `M(v) = q ⊗ v ⊗ p̄` is the per-step `SO(4)` rotation of
333/// [`RotationKind::Rotor4D`], and scale the pass-through entries by the real
334/// scalar `tail` — the two-sided analogue of [`mul_quat_partial`].
335///
336/// Two-sided, the geometric series no longer lives in a commutative subalgebra
337/// (that shortcut is what [`quat_inv`] exploits for `Quaternion4D`), so the
338/// resolvent is taken over `ℝ⁴` by **Cayley–Hamilton**. `M` is orthogonal with
339/// characteristic polynomial `λ⁴ − c₁λ³ + c₂λ² − c₁λ + 1`, and its two plane
340/// angles are `a∓b` for the half-angles `a`, `b` of `q` and `p` (the standard
341/// `SO(4) ≅ (SU(2)×SU(2))/±1` decomposition — the axes fix the invariant
342/// *planes*, only the half-angles fix the *angles*), so with
343/// `k₁ = cos(a−b)`, `k₂ = cos(a+b)`:
344///
345/// ```text
346///   c₁ = 2(k₁ + k₂) = tr M ,   c₂ = 2 + 4k₁k₂ ,
347///   det(I − αM) = (1 − 2αk₁ + α²)(1 − 2αk₂ + α²) ,
348///   (I − αM)⁻¹  = (e₀ + e₁M + e₂M² + e₃M³) / det(I − αM)
349/// ```
350///
351/// with `e₃ = α³`, `e₂ = α² − c₁α³`, `e₁ = α − c₁α² + c₂α³`,
352/// `e₀ = 1 − c₁α + c₂α² − c₁α³` (the divided-difference form
353/// `q(λ) = (χ(μ)−χ(λ))/(μ−λ)` at `μ = 1/α`, rescaled). The cubic is evaluated
354/// by Horner **on the vector**, so no `4×4` matrix is ever materialised: each
355/// application of `M` is two quaternion products. `k₁ = k₂ = 1` (no rotation)
356/// collapses the whole thing to `(β+γ)/(1−α)`, i.e. `tail`.
357///
358/// Numerics: the determinant is formed factor-wise as `(1−α)² + 2α(1−kᵢ)`,
359/// which is manifestly `≥ (1−α)²` and free of the cancellation the expanded
360/// quartic would suffer; the numerator carries the same `α → 1` precision
361/// caveat as the abelian branch (see the module header).
362#[allow(clippy::too_many_arguments)]
363fn rotor_resolvent_partial(
364    x_bmhr: Tensor<4>,
365    q_bhj4: Tensor<4>,
366    p_bhj4: Tensor<4>,
367    alpha_bh: Tensor<2>,
368    beta_bh: Tensor<2>,
369    gamma_bh: Tensor<2>,
370    tail_bh: Tensor<2>,
371    rope_width: usize,
372) -> Tensor<4> {
373    let [batch, mimo_rank, nheads, state_rank] = x_bmhr.dims();
374    let blocks = q_bhj4.dims()[2];
375    let tail_b1h1 = tail_bh.unsqueeze_dims::<4>(&[1, 3]);
376    if rope_width == 0 {
377        return x_bmhr * tail_b1h1;
378    }
379    assert_eq!(
380        rope_width,
381        blocks * 4,
382        "the rotated width is a whole number of quaternion blocks"
383    );
384    let eps = div_eps(alpha_bh.dtype());
385
386    // Per-block plane cosines. `Re q = cos a`, `‖Im q‖ = |sin a|`; a sign flip
387    // of `sin a` only swaps k₁ ↔ k₂, and every quantity below is symmetric in
388    // the two, so the absolute value costs nothing.
389    let per_block = |t: Tensor<4>| t.unsqueeze_dim::<5>(1); // [b,h,J,1] → [b,1,h,J,1]
390    let wc = per_block(q_bhj4.clone().narrow(3, 0, 1) * p_bhj4.clone().narrow(3, 0, 1));
391    let ss = per_block(
392        safe_norm(q_bhj4.clone().narrow(3, 1, 3)) * safe_norm(p_bhj4.clone().narrow(3, 1, 3)),
393    );
394    let k1 = wc.clone() + ss.clone();
395    let k2 = wc - ss;
396
397    // Scalars, broadcast over (mimo, block, component).
398    let a = alpha_bh.unsqueeze_dims::<5>(&[1, 3, 4]);
399    let beta = beta_bh.unsqueeze_dims::<5>(&[1, 3, 4]);
400    let gamma = gamma_bh.unsqueeze_dims::<5>(&[1, 3, 4]);
401    let a2 = a.clone() * a.clone();
402    let a3 = a2.clone() * a.clone();
403    let c1 = (k1.clone() + k2.clone()) * 2.0;
404    let c2 = k1.clone() * k2.clone() * 4.0 + 2.0;
405    let one_minus_a2 = (-a.clone() + 1.0) * (-a.clone() + 1.0);
406    let det = (one_minus_a2.clone() + (-k1 + 1.0) * a.clone() * 2.0)
407        * (one_minus_a2 + (-k2 + 1.0) * a.clone() * 2.0);
408    let det = det.clamp_min(eps);
409    let e3 = a3.clone();
410    let e2 = a2.clone() - c1.clone() * a3.clone();
411    let e1 = a.clone() - c1.clone() * a2.clone() + c2.clone() * a3.clone();
412    let e0 = -c1.clone() * a + c2 * a2 - c1 * a3 + 1.0;
413
414    // M(v) = q ⊗ v ⊗ p̄, on the block-split view of the rotated entries.
415    let expand = |q: Tensor<4>| {
416        q.unsqueeze_dim::<5>(1)
417            .expand([batch, mimo_rank, nheads, blocks, 4])
418    };
419    let ql = expand(q_bhj4);
420    let qr = expand(quat_conj(p_bhj4));
421    let m = |v: Tensor<5>| quat_mul(quat_mul(ql.clone(), v), qr.clone());
422
423    let head = x_bmhr
424        .clone()
425        .narrow(3, 0, rope_width)
426        .reshape([batch, mimo_rank, nheads, blocks, 4]);
427    // Horner on the vector: u = (e₀ + e₁M + e₂M² + e₃M³) x.
428    let u = head.clone() * e3;
429    let u = m(u) + head.clone() * e2;
430    let u = m(u) + head.clone() * e1;
431    let u = m(u) + head * e0;
432    let out = (u.clone() * gamma + m(u) * beta) / det;
433    let out = out.reshape([batch, mimo_rank, nheads, rope_width]);
434
435    if rope_width == state_rank {
436        out
437    } else {
438        let tail = x_bmhr.narrow(3, rope_width, state_rank - rope_width) * tail_b1h1;
439        Tensor::cat(vec![out, tail], 3)
440    }
441}
