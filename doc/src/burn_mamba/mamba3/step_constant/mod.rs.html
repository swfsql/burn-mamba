1//! # Constant-input stepping shortcut — `step_infinite`
2//!
3//! When the **same token** is fed to [`Mamba3::step`] over and over, every
4//! data-dependent quantity is the same at each step: the trapezoid
5//! coefficients `α`, `β`, `γ`, the QK-normed `b`/`c`, and the per-step
6//! rotation increment (angle `θ̂` for `Complex2D`, unit quaternion `q` for
7//! `Quaternion4D`). Only the *cumulative* rotation moves — by that same
8//! increment each step — so with `P` the per-step rotation operator and `R₁`
9//! the cumulative rotation at the first constant step,
10//!
11//! ```text
12//!   Bₜ = R₁ Pᵗ⁻¹ b ,   Cₜ = R₁ Pᵗ⁻¹ c ,
13//!   hₜ = α hₜ₋₁ + x ⊗ (β Bₜ₋₁ + γ Bₜ)            (t ≥ 2)
14//! ```
15//!
16//! and the unrolled steps collapse to a **matrix geometric series** in
17//! `α P⁻¹` (spectral radius `α < 1`, since `a_floor > 0` and `Δ > 0`).
18//!
19//! `P` is block-diagonal: per RoPE pair it is the complex scalar `e^{iθ̂}`, per
20//! quaternion block left-multiplication by `w = q*`. In both cases all the
21//! factors live in one **abelian** subalgebra (powers of a single rotation
22//! commute — for quaternions, `span{1, û} ≅ ℂ`), so the series is a handful of
23//! scalar complex / quaternion ops per (head, pair/block): `n` steps cost O(1).
24//!
25//! In the readout `y_n = Cₙᵀ h_n + D x` the unbounded phase cancels
26//! (`⟨R c, R v⟩ = ⟨c, v⟩`), leaving only the *relative* rotation `P⁻¹`. As
27//! `n → ∞` the initial-state term and the partial-sum correction decay like
28//! `αⁿ`, so the **output converges** even though the state `h` orbits forever:
29//!
30//! ```text
31//!   y_∞ = xᵀ · cᵀ (γ + β P⁻¹)(I − α P⁻¹)⁻¹ b  +  D x
32//! ```
33//!
34//! — the block's stationary fixed point, independent of any starting cache.
35//! [`Mamba3::step_infinite`] evaluates exactly this (and therefore takes and
36//! returns no cache).
37//!
38//! Numerics: the rotation angle is reduced mod `2π` with the value-exact
39//! [`wrap_angle`]; the geometric denominators satisfy
40//! `|1 − α e^{−iθ̂}|² ≥ (1 − α)²` and are floored by
41//! [`div_eps`](crate::utils::div_eps). When `α → 1` *and* `θ̂ → 0` the series
42//! value `(β+γ)/(1−α)` stays finite but loses fp32 precision once `1 − α` nears
43//! the epsilon floor — the same regime where the unrolled recurrence itself
44//! accumulates near-undamped terms.
45

63impl Mamba3 {
64    /// Stationary **fixed point** of the block under a constant token: the
65    /// limit of `step(input, …)` outputs as the same `input_bd` is stepped
66    /// forever.
67    ///
68    /// Closed form, O(1) in the horizon (see the [module header](self) for the
69    /// derivation). The limit forgets any starting state (`αⁿ → 0`) and the
70    /// SSM state itself never converges (it keeps rotating; only the output
71    /// does), so this takes **no cache and returns none**.
72    ///
73    /// Both rotation kinds and both SSD pathways are covered (the recurrence
74    /// is pathway-agnostic at boundaries). Differentiable; gradients are the
75    /// limit of the unrolled gradients.
76    ///
77    /// # Shapes
78    /// - `input_bd` : `[batch, d_model]`
79    /// - output     : `[batch, d_model]`
80    pub fn step_infinite(&self, input_bd: Tensor<2>) -> Tensor<2> {
81        let StepProjection {
82            z_bi,
83            x_bhp,
84            b_bmhr,
85            c_bmhr,
86            rot_ba,
87            dt_bh,
88            alpha_bh,
89            beta_bh,
90            gamma_bh,
91        } = self.step_project(input_bd);
92        let [batch, mimo_rank, nheads, _state_rank] = b_bmhr.dims();
93        let eps = div_eps(alpha_bh.dtype());
94
95        // Rotation-free channels: (β + γ) / (1 − α).
96        let tail_bh = (beta_bh.clone() + gamma_bh.clone())
97            / (-alpha_bh.clone() + 1.0).clamp_min(eps);
98
99        // Per-pair/block readout factor  m = (γ + β P⁻¹)(1 − α P⁻¹)⁻¹  applied
100        // to `b`; the cumulative rotation cancels against `Cₙ` (orthogonality).
101        let b_eff_bmhr = match self.rotation_kind() {
102            RotationKind::Complex2D => {
103                let theta_bha = per_step_angle(rot_ba, dt_bh); // θ̂
104                let (cos, sin) = cos_sin(theta_bha);
105                let a_bh1 = alpha_bh.unsqueeze_dim::<3>(2);
106                let beta_bh1 = beta_bh.unsqueeze_dim::<3>(2);
107                let gamma_bh1 = gamma_bh.unsqueeze_dim::<3>(2);
108                // (γ + β e^{−iθ̂}) / (1 − α e^{−iθ̂})
109                let num_re = gamma_bh1 + beta_bh1.clone() * cos.clone();
110                let num_im = -beta_bh1 * sin.clone();
111                let den_re = -a_bh1.clone() * cos + 1.0;
112                let den_im = a_bh1 * sin;
113                let (m_re, m_im) = complex_div(num_re, num_im, den_re, den_im);
114                mul_complex_partial(
115                    b_bmhr,
116                    m_re,
117                    m_im,
118                    tail_bh,
119                    self.rope_dim,
120                    mimo_rank == 1,
121                )
122            }
123            RotationKind::Quaternion4D => {
124                let g_bhj3 = per_step_generator(rot_ba, dt_bh, self.num_quat_blocks);
125                let q_bhj4 = quat_from_scaled_axis::<4>(g_bhj3); // P⁻¹ ↔ q
126                let alpha_bh11 = alpha_bh.unsqueeze_dims::<4>(&[2, 3]);
127                let beta_bh11 = beta_bh.unsqueeze_dims::<4>(&[2, 3]);
128                let gamma_bh11 = gamma_bh.unsqueeze_dims::<4>(&[2, 3]);
129                // (γ + β q) ⊗ (1 − α q)⁻¹  — all in the abelian subalgebra of q.
130                let num = quat_scalar_affine(gamma_bh11, beta_bh11, q_bhj4.clone());
131                let den_inv = quat_inv(quat_one_minus(q_bhj4 * alpha_bh11));
132                let f_bhj4 = quat_mul(num, den_inv);
133                mul_quat_partial(b_bmhr, f_bhj4, tail_bh, self.num_quat_blocks * 4)
134            }
135        };
136        san(&b_eff_bmhr);
137
138        // y_∞[m] = Σ_m' ⟨c[m], m·b[m']⟩ · x_vals[m']   (then D-skip/gate/out-proj).
139        let mimo_x_hmp = self.mimo_x_hmp.as_ref().map(|p| p.val());
140        let x_vals_bmhp = helpers::build_v_with_mimo::<3, 4>(x_bhp, mimo_x_hmp.as_ref(), 1);
141        let gram_bhmm = {
142            let c_bhmr = c_bmhr.swap_dims(1, 2);
143            let b_bhrm = b_eff_bmhr.permute([0, 2, 3, 1]);
144            c_bhmr.matmul(b_bhrm) // [batch, nheads, mimo_rank, mimo_rank']
145        };
146        let out_m_bmhp = {
147            let x_bhmp = x_vals_bmhp.clone().swap_dims(1, 2);
148            gram_bhmm.matmul(x_bhmp).swap_dims(1, 2)
149        };
150        assert_eq!(
151            [batch, mimo_rank, nheads, self.per_head_dim()],
152            out_m_bmhp.dims()
153        );
154        san(&out_m_bmhp);
155
156        self.step_finish(out_m_bmhp, x_vals_bmhp, z_bi)
157    }
158}

164/// `θ̂ = Δ · π·tanh(rot)` — the constant per-step RoPE angle increment.
165fn per_step_angle(rot_ba: Tensor<2>, dt_bh: Tensor<2>) -> Tensor<3> {
166    dt_bh.unsqueeze_dim::<3>(2) * (rot_ba.tanh() * PI).unsqueeze_dim::<3>(1)
167}
168
169/// `g = Δ · π·tanh(rot)` per quaternion block — the constant per-step rotation
170/// generator (`q = exp(g/2)`).
171fn per_step_generator(rot_ba: Tensor<2>, dt_bh: Tensor<2>, blocks: usize) -> Tensor<4> {
172    let [batch, _a] = rot_ba.dims();
173    (rot_ba.tanh() * PI)
174        .reshape([batch, blocks, 3])
175        .unsqueeze_dim::<4>(1)
176        * dt_bh.unsqueeze_dim::<3>(2).unsqueeze_dim::<4>(3)
177}

183/// `(cos, sin)` of an angle tensor, reduced mod `2π` first (value-exact).
184fn cos_sin(angle_bha: Tensor<3>) -> (Tensor<3>, Tensor<3>) {
185    let a = wrap_angle(angle_bha);
186    (a.clone().cos(), a.sin())
187}

189/// Component-wise complex quotient; `|den|²` floored by `div_eps`.
190fn complex_div(
191    nr: Tensor<3>,
192    ni: Tensor<3>,
193    dr: Tensor<3>,
194    di: Tensor<3>,
195) -> (Tensor<3>, Tensor<3>) {
196    let eps = div_eps(dr.dtype());
197    let d2 = (dr.clone() * dr.clone() + di.clone() * di.clone()).clamp_min(eps);
198    (
199        (nr.clone() * dr.clone() + ni.clone() * di.clone()) / d2.clone(),
200        (ni * dr - nr * di) / d2,
201    )
202}

204/// Multiply the rotation-active entries of `x` by the complex scalar
205/// `(re, im)` per pair — same pairing conventions as [`apply_rope_partial`] —
206/// and the pass-through entries by the real scalar `tail`.
207///
208/// `re`/`im` are `[batch, nheads, num_rope_angles]` (broadcast over the
209/// `mimo_rank` axis); `tail` is `[batch, nheads]`.
210fn mul_complex_partial(
211    x_bmhr: Tensor<4>,
212    re_bha: Tensor<3>,
213    im_bha: Tensor<3>,
214    tail_bh: Tensor<2>,
215    rope_dim: usize,
216    rotate_pairwise: bool,
217) -> Tensor<4> {
218    let [batch, mimo_rank, nheads, state_rank] = x_bmhr.dims();
219    let tail_b1h1 = tail_bh.unsqueeze_dims::<4>(&[1, 3]);
220    if rope_dim == 0 {
221        // RoPE disabled: every channel is a plain geometric series.
222        return x_bmhr * tail_b1h1;
223    }
224    let re_b1ha = re_bha.unsqueeze_dim::<4>(1);
225    let im_b1ha = im_bha.unsqueeze_dim::<4>(1);
226
227    if rotate_pairwise {
228        // Interleaved (SISO/NeoX): pairs are local, so the first `rope_dim`
229        // entries can be handled standalone.
230        let n2 = rope_dim / 2;
231        let head = x_bmhr.clone().narrow(3, 0, rope_dim);
232        let head_pairs = head.reshape([batch, mimo_rank, nheads, n2, 2]);
233        let x0 = head_pairs.clone().narrow(4, 0, 1).squeeze_dim::<4>(4);
234        let x1 = head_pairs.narrow(4, 1, 1).squeeze_dim::<4>(4);
235        let x0m = re_b1ha.clone() * x0.clone() - im_b1ha.clone() * x1.clone();
236        let x1m = im_b1ha * x0 + re_b1ha * x1;
237        let head = Tensor::cat(
238            vec![x0m.unsqueeze_dim::<5>(4), x1m.unsqueeze_dim::<5>(4)],
239            4,
240        )
241        .reshape([batch, mimo_rank, nheads, rope_dim]);
242        if rope_dim == state_rank {
243            head
244        } else {
245            let tail = x_bmhr.narrow(3, rope_dim, state_rank - rope_dim) * tail_b1h1;
246            Tensor::cat(vec![head, tail], 3)
247        }
248    } else {
249        // Half-and-half (MIMO/GPT-J): entry `i` pairs with `i + state_rank/2`;
250        // only the first `rope_dim / 2` pairs are active.
251        let half = state_rank / 2;
252        let active = rope_dim / 2;
253        let x_h1 = x_bmhr.clone().narrow(3, 0, half);
254        let x_h2 = x_bmhr.narrow(3, half, half);
255        let x_h1_rope = x_h1.clone().narrow(3, 0, active);
256        let x_h2_rope = x_h2.clone().narrow(3, 0, active);
257        let h1_m = re_b1ha.clone() * x_h1_rope.clone() - im_b1ha.clone() * x_h2_rope.clone();
258        let h2_m = im_b1ha * x_h1_rope + re_b1ha * x_h2_rope;
259        if active == half {
260            Tensor::cat(vec![h1_m, h2_m], 3)
261        } else {
262            let h1_pass = x_h1.narrow(3, active, half - active) * tail_b1h1.clone();
263            let h2_pass = x_h2.narrow(3, active, half - active) * tail_b1h1;
264            Tensor::cat(vec![h1_m, h1_pass, h2_m, h2_pass], 3)
265        }
266    }
267}

273/// `1 − sq` for a (scalar-scaled) quaternion `sq`: negate, add 1 to the real
274/// part.
275fn quat_one_minus(sq_bhj4: Tensor<4>) -> Tensor<4> {
276    let w = -sq_bhj4.clone().narrow(3, 0, 1) + 1.0;
277    let xyz = -sq_bhj4.narrow(3, 1, 3);
278    Tensor::cat(vec![w, xyz], 3)
279}
280
281/// `a + b·q` for per-head scalars `a`, `b` (`[batch, nheads, 1, 1]`) and a
282/// quaternion tensor `q`.
283fn quat_scalar_affine(a_bh11: Tensor<4>, b_bh11: Tensor<4>, q_bhj4: Tensor<4>) -> Tensor<4> {
284    let w = a_bh11 + b_bh11.clone() * q_bhj4.clone().narrow(3, 0, 1);
285    let xyz = b_bh11 * q_bhj4.narrow(3, 1, 3);
286    Tensor::cat(vec![w, xyz], 3)
287}

289/// Quaternion inverse `q⁻¹ = q* / ‖q‖²`, with `‖q‖²` floored by `div_eps`.
290/// (Used on `1 − α q`, whose norm is bounded below by `1 − α > 0`.)
291fn quat_inv(q_bhj4: Tensor<4>) -> Tensor<4> {
292    let eps = div_eps(q_bhj4.dtype());
293    let n2 = (q_bhj4.clone() * q_bhj4.clone()).sum_dim(3).clamp_min(eps);
294    quat_conj(q_bhj4) / n2
295}

297/// Left-multiply the first `rope_width` state-rank entries of `x` by the (not
298/// necessarily unit) quaternion `f` per block, and scale the pass-through
299/// entries by the real scalar `tail` — the quaternion analogue of
300/// [`mul_complex_partial`].
301fn mul_quat_partial(
302    x_bmhr: Tensor<4>,
303    f_bhj4: Tensor<4>,
304    tail_bh: Tensor<2>,
305    rope_width: usize,
306) -> Tensor<4> {
307    let [batch, mimo_rank, nheads, state_rank] = x_bmhr.dims();
308    let blocks = f_bhj4.dims()[2];
309    let tail_b1h1 = tail_bh.unsqueeze_dims::<4>(&[1, 3]);
310    if rope_width == 0 {
311        return x_bmhr * tail_b1h1;
312    }
313    let f_bmhj4 = f_bhj4
314        .unsqueeze_dim::<5>(1)
315        .expand([batch, mimo_rank, nheads, blocks, 4]);
316    if rope_width == state_rank {
317        rotate_state_rank_blocks::<4, 5>(x_bmhr, f_bmhj4)
318    } else {
319        let head = rotate_state_rank_blocks::<4, 5>(
320            x_bmhr.clone().narrow(3, 0, rope_width),
321            f_bmhj4,
322        );
323        let tail = x_bmhr.narrow(3, rope_width, state_rank - rope_width) * tail_b1h1;
324        Tensor::cat(vec![head, tail], 3)
325    }
326}
