14/// Reduce angles modulo `2π` into `[−π, π]`, leaving the autodiff graph intact.
15///
16/// `sin`/`cos` are `2π`-periodic, so a subtraction of an integer multiple of
17/// `2π` does not change the rotation. In f16 and bf16, the dtype cannot hold
18/// `2π`, so the subtraction is in two parts (`helpers::sub_periods`).
19/// Otherwise, each wrap of the decode cache would lose `1.9·10⁻³` per turn.
20/// `|angle| ≤ π` keeps precision in low-bit floats: about
21/// half of the representable `f16` values are in `|x| ≤ 1`, and the periodic
22/// `sin`/`cos` lose accuracy only when the argument drifts to large
23/// magnitudes. The same applies to the cumulative angle accumulator. Without
24/// this, it would grow without bound over a long sequence or many decode
25/// steps.
26///
27/// The wrap does not remove the rounding of the angle itself. In f16, the
28/// spacing at `|angle| ≈ π` is `2⁻⁹ ≈ 2·10⁻³`, and each decode step rounds the
29/// sum again. So a long f16 decode drifts: in a numpy simulation, 4096
30/// micro-steps of 1–2.5 rad drift by 0.18–0.82 rad. Only an f32 angle scan and an f32
31/// angle cache remove this drift.
32///
33/// The integer multiple `k` is `detach`ed, so autodiff sees it as a constant:
34/// `d/dx (x − k·2π) = 1`, and the backward pass is the same as for the
35/// unwrapped angle.
36pub fn wrap_angle<const D: usize>(angles: Tensor<D>) -> Tensor<D> {
37    let two_pi = 2.0 * std::f32::consts::PI;
38    let k = (angles.clone().detach() * (1.0f32 / two_pi)).round();
39    crate::mamba3::helpers::sub_periods(angles, k, two_pi)
40}

42/// Rotate the pairs of `x` along its last dimension (the RoPE trick).
43///
44/// Purely the mechanical pairwise rotation. In Mamba-3 the `angles` are the
45/// cumulative rotation of the *state transition* (see [`crate::mamba3::mamba3`]),
46/// not a position index.
47///
48/// The matrix of a pair is `R(+angle)`: `x₀' = cos·x₀ − sin·x₁` and
49/// `x₁' = sin·x₀ + cos·x₁`. Section 2 of that header compares this direction with
50/// the sources.
51///
52/// Two pairing conventions are supported, selected by `rotate_pairwise`:
53///
54/// - `rotate_pairwise = true` — **interleaved** (NeoX / Triton style): adjacent
55///   pairs `(0,1)`, `(2,3)`, … are rotated together. Used by the SISO Triton
56///   kernel (`mamba3_siso_*.py`).
57/// - `rotate_pairwise = false` — **half-and-half** (GPT-J style): position `n`
58///   is paired with `n + state_rank/2`. Used by the MIMO Tilelang kernel
59///   (`mamba3_mimo_fwd.py`).
60///
61/// Reference: `mamba3.py:335` sets `rotate_pairwise = not self.is_mimo`.
62///
63/// # Shapes
64/// - `x`:      `[..., state_rank]` where `state_rank` is even
65/// - `angles`: `[..., state_rank / 2]`  (one angle per pair)
66/// - output:   same shape as `x`
67pub fn apply_rope<const D: usize>(
68    x: Tensor<D>,
69    angles: Tensor<D>,
70    rotate_pairwise: bool,
71) -> Tensor<D> {
72    let dims = x.dims();
73    let n = dims[D - 1];
74    let n2 = n / 2;
75    let leading: usize = dims[..D - 1].iter().product();
76
77    let angles_flat = wrap_angle(angles.reshape([leading, n2]));
78    let cos = angles_flat.clone().cos();
79    let sin = angles_flat.sin();
80
81    if rotate_pairwise {
82        // Interleaved: reshape to [leading, n2, 2], pairs along last axis.
83        let x_pairs = x.reshape([leading, n2, 2]);
84        let x0 = x_pairs.clone().narrow(2, 0, 1).squeeze_dim(2);
85        let x1 = x_pairs.narrow(2, 1, 1).squeeze_dim(2);
86
87        let x0r = cos.clone() * x0.clone() - sin.clone() * x1.clone();
88        let x1r = sin * x0 + cos * x1;
89
90        Tensor::cat(
91            vec![x0r.unsqueeze_dim::<3>(2), x1r.unsqueeze_dim::<3>(2)],
92            2,
93        )
94        .reshape(dims)
95    } else {
96        // Half-and-half: reshape to [leading, 2, n2], halves along middle axis.
97        let x_halves = x.reshape([leading, 2, n2]);
98        let x0 = x_halves.clone().narrow(1, 0, 1).squeeze_dim(1);
99        let x1 = x_halves.narrow(1, 1, 1).squeeze_dim(1);
100
101        let x0r = cos.clone() * x0.clone() - sin.clone() * x1.clone();
102        let x1r = sin * x0 + cos * x1;
103
104        Tensor::cat(
105            vec![x0r.unsqueeze_dim::<3>(1), x1r.unsqueeze_dim::<3>(1)],
106            1,
107        )
108        .reshape(dims)
109    }
110}

112/// Apply the RoPE trick to only the rotation-active entries of the last
113/// dimension. The rest is unchanged. When `rotation_dim == state_rank` (a full
114/// rotation), this is [`apply_rope`]. `rotation_dim` must be positive: a block
115/// that rotates nothing is
116/// [`RotationKind::Real1D`](crate::mamba3::rotation::RotationKind::Real1D),
117/// which never comes here.
118///
119/// Pairing scheme (it must match the reference kernels: see Section
120/// "Data-Dependent RoPE" in the paper, and `mamba3_siso_fwd.py` /
121/// `mamba3_mimo_fwd.py`):
122///
123/// - `rotate_pairwise = true` (SISO, interleaved/NeoX): pairs `(0,1), (2,3), …`.
124///   Only pairs `0..num_rotation_angles` turn, and the pairs after them are
125///   unchanged. This is the same as a rotation of the first `rotation_dim`
126///   entries.
127/// - `rotate_pairwise = false` (MIMO, half-and-half/GPT-J): the pair distance
128///   is always `state_rank/2`: element `n` pairs with element
129///   `state_rank/2 + n`. With a partial rotation, only the first
130///   `num_rotation_angles` pairs turn. The other elements in both halves are
131///   unchanged.
132pub fn apply_rope_partial<const D: usize>(
133    x: Tensor<D>,
134    angles: Tensor<D>,
135    rotation_dim: usize,
136    rotate_pairwise: bool,
137) -> Tensor<D> {
138    debug_assert!(rotation_dim > 0, "a partial rotation requires rotation_dim > 0");
139
140    let state_rank = x.dims()[D - 1];
141    if rotation_dim == state_rank {
142        return apply_rope::<D>(x, angles, rotate_pairwise);
143    }
144
145    if rotate_pairwise {
146        // Pairs are local. A slice of the first rotation_dim entries gives the
147        // same result as the reference (which rotates the whole headdim, with
148        // identity cos/sin for the tail pairs).
149        let x_rope = x.clone().narrow(D - 1, 0, rotation_dim);
150        let x_rest = x.narrow(D - 1, rotation_dim, state_rank - rotation_dim);
151        let x_rope_rotated = apply_rope::<D>(x_rope, angles, true);
152        return Tensor::cat(vec![x_rope_rotated, x_rest], D - 1);
153    }
154
155    // Half-and-half partial rotation: the pair distance must be `state_rank/2`,
156    // not `rotation_dim/2`. `apply_rope` on a slice of the first `rotation_dim`
157    // entries would pair within the slice and give the wrong rotation.
158    let half = state_rank / 2;
159    let num_rotation_angles = rotation_dim / 2;
160    debug_assert!(
161        num_rotation_angles < half,
162        "a partial rotation requires rotation_dim < state_rank here"
163    );
164
165    // Split x into the two halves, then within each half separate the
166    // rotation-active prefix from the pass-through suffix.
167    let x_h1 = x.clone().narrow(D - 1, 0, half);
168    let x_h2 = x.narrow(D - 1, half, half);
169    let x_h1_rope = x_h1.clone().narrow(D - 1, 0, num_rotation_angles);
170    let x_h1_pass = x_h1.narrow(D - 1, num_rotation_angles, half - num_rotation_angles);
171    let x_h2_rope = x_h2.clone().narrow(D - 1, 0, num_rotation_angles);
172    let x_h2_pass = x_h2.narrow(D - 1, num_rotation_angles, half - num_rotation_angles);
173
174    // angles: [..., num_rotation_angles] — broadcasts element-wise against the
175    // rotation-active slices.
176    let angles = wrap_angle(angles);
177    let cos = angles.clone().cos();
178    let sin = angles.sin();
179    let x_h1_rot = cos.clone() * x_h1_rope.clone() - sin.clone() * x_h2_rope.clone();
180    let x_h2_rot = sin * x_h1_rope + cos * x_h2_rope;
181
182    // Reassemble: [ first-half-rotated | first-half-passthrough | second-half-rotated | second-half-passthrough ]
183    let x_h1_out = Tensor::cat(vec![x_h1_rot, x_h1_pass], D - 1);
184    let x_h2_out = Tensor::cat(vec![x_h2_rot, x_h2_pass], D - 1);
185    Tensor::cat(vec![x_h1_out, x_h2_out], D - 1)
186}
