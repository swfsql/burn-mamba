1//! # Mamba-3 — Plant Frame
2//!
3//! The plant frame is the work of a Mamba-3 block around the plant. Both
4//! forward pathways and `step` call it. Each one keeps only its middle: its
5//! scan and the inputs that only this scan needs. `step` enters at
6//! `tokens = 1`. It adds that read axis to its input and removes it from its
7//! output.
8//!
9//! ```text
10//!   input  ── plant_entry ──▶ rotated B and C, masses, tap FIFO
11//!                                        │
12//!                              the scan of the pathway
13//!                                        │
14//!   output ◀── plant_tail ───          readout
15//! ```
16//!
17//! - [`Mamba3::plant_entry`] holds the padding, the segments, the fresh cache,
18//!   the origin, the in-projection, the split, the trapezoid coefficients,
19//!   QK-Norm, the rotation and the tap FIFO.
20//! - [`Mamba3::plant_tail`] holds the positive systems, the `D` skip, the gate,
21//!   the MIMO merge and `out_proj`. The band correction of single-SSD runs
22//!   before the tail and stays in that pathway.
23//!
24//! The frame has one cache type, `Mamba3DoubleSsdCache`. A forward starts and
25//! ends at a call boundary, where the accumulator of single-SSD equals the
26//! state. So the `From` impls convert the cache of that pathway (ADR-0003).
27//! The frame writes the frame cache ([`Mamba3FrameCache`]) of the cache. Only
28//! the scan of each pathway writes the state.
29//!
30//! A bug in the frame fails both pathways alike, and a parity test cannot show
31//! it. So the tests of this module compare the split and the tail with their
32//! definitions (ADR-0007).
33

46/// The channel groups of the in-projection, in the order that
47/// [`Mamba3::in_proj`] documents. A per-micro-step group is on the write axis.
48/// The gate `z` and the read `C` stay at token resolution.
49#[allow(non_snake_case)]
50pub(crate) struct InProjStreams {
51    pub z_bti: Tensor<3>,
52    pub x_bsi: Tensor<3>,
53    pub b_raw_bsMGR: Tensor<3>,
54    pub c_raw_btMGR: Tensor<3>,
55    pub dd_dt_bsh: Tensor<3>,
56    pub dd_a_raw_bsh: Tensor<3>,
57    /// `None` under `TapPattern::None`.
58    pub lambda_raw_bsh: Option<Tensor<3>>,
59    /// `None` unless the pattern has two taps.
60    pub mu_raw_bsh: Option<Tensor<3>>,
61    /// `None` under `Real1D`.
62    pub rot_bsa: Option<Tensor<3>>,
63    /// `None` unless the gain projects its noise.
64    pub noise_bsh: Option<Tensor<3>>,
65    /// `(a, b)`. `None` without the tropical register.
66    pub tropical_ab_bsh: Option<(Tensor<3>, Tensor<3>)>,
67}

69/// The resets of a packed call. The chunk length comes with them, because only
70/// the resets need it: `step` has no SSD path.
71pub(crate) struct PlantResets {
72    /// The first token of each segment, `[batch, tokens]`.
73    pub reset_bt: Tensor<2, Bool>,
74    /// The cache that each segment starts from. `None` uses a fresh cache.
75    pub origin: Option<Mamba3DoubleSsdCache>,
76    /// The chunk length on the write axis.
77    pub chunk_len: usize,
78}

80/// The result of [`Mamba3::plant_entry`], one field for each stream.
81///
82/// A pathway moves each field out at once. It does not hold this struct across
83/// its scan, so each tensor drops at its last use.
84pub(crate) struct PlantEntry {
85    /// The cache of the call: the one that the caller gave, or a fresh one.
86    pub cache: Mamba3DoubleSsdCache,
87    /// The resets and the origin of a packed call.
88    pub packed: Option<PackedCall>,
89    /// The row-end reader of the call, on the write axis.
90    pub row_end: RowEnd,
91    /// The gate.
92    pub z_bti: Tensor<3>,
93    /// The value stream.
94    pub x_bshp: Tensor<4>,
95    /// The rotated `B`.
96    pub b_bsmhr: Tensor<5>,
97    /// The rotated `C`.
98    pub c_btmhr: Tensor<5>,
99    /// The log-decay.
100    pub da_bsh: Tensor<3>,
101    /// `α = exp(da)`.
102    pub alpha_bsh: Tensor<3>,
103    /// The tap mass `ν`. `None` under `TapPattern::None`.
104    pub nu_bsh: Option<Tensor<3>>,
105    /// The mass of the lag-1 tap of a two-tap pattern.
106    pub nu_interior_bsh: Option<Tensor<3>>,
107    /// `γ = λ·Δ`.
108    pub gamma_bsh: Tensor<3>,
109    /// `ln Λ` at each write position, under a Kalman gain.
110    pub log_precision_bsh: Option<Tensor<3>>,
111    /// The `(a, b)` of the tropical register, on the write axis.
112    pub tropical_ab_bsh: Option<(Tensor<3>, Tensor<3>)>,
113    /// The cumulative rotation for the next call.
114    pub new_rotation: RotationState,
115    /// The tap FIFO for the next call.
116    pub new_tap_fifo: Option<TapFifo>,
117}

119/// The input of [`Mamba3::plant_tail`]. The readout comes from the middle of a
120/// pathway. Each other field is the field of [`PlantEntry`] with the same name.
121pub(crate) struct PlantTail {
122    /// The readout of the scan.
123    pub y_btmhp: Tensor<5>,
124    pub z_bti: Tensor<3>,
125    pub x_bshp: Tensor<4>,
126    pub log_precision_bsh: Option<Tensor<3>>,
127    pub tropical_ab_bsh: Option<(Tensor<3>, Tensor<3>)>,
128    pub row_end: RowEnd,
129    pub packed: Option<PackedCall>,
130    pub new_rotation: RotationState,
131    pub new_tap_fifo: Option<TapFifo>,
132}

134impl Mamba3 {
135    /// The entry of the plant frame: from the input to the rotated `B` and `C`,
136    /// the masses and the tap FIFO.
137    ///
138    /// `cache` and the origin of `resets` start fresh when they are `None`.
139    ///
140    /// # Shapes
141    /// - `input_btd` : `[batch, tokens, d_model]`
142    /// - `pad_bt`    : `[batch, tokens]`
143    #[allow(non_snake_case)]
144    pub(crate) fn plant_entry(
145        &self,
146        input_btd: Tensor<3>,
147        cache: Option<Mamba3DoubleSsdCache>,
148        pad_bt: Option<Tensor<2, Bool>>,
149        resets: Option<PlantResets>,
150    ) -> PlantEntry {
151        let [batch, tokens, _d_model] = input_btd.dims();
152        let nheads = self.nheads();
153        let ngroups = self.ngroups;
154        let per_head_dim = self.per_head_dim();
155        let state_rank = self.state_rank;
156        let mimo_rank = self.mimo_rank;
157        let u = self.micro_steps;
158        let device = input_btd.device();
159
160        // MambaProduct: from the split down to the readout, one sequence
161        // position *is* one micro-step. The `s` in each shape suffix counts
162        // micro-steps, and `t` counts tokens. See [`crate::mamba3::product`].
163        let sequence = tokens * u;
164
165        assert!(tokens > 0, "sequence length must be at least 1");
166        assert_eq!(nheads % ngroups, 0);
167        san(&input_btd);
168
169        // A padded token pads all of its micro-steps.
170        let pad_bs = pad_bt.map(|pad_bt| crate::padding::repeat_positions(pad_bt, u));
171
172        // ── Initialise the caches that the caller did not give ────────────────
173        let fresh = || Mamba3DoubleSsdCache::from(self.zero_cache(batch, &device));
174        let cache = cache.unwrap_or_else(fresh);
175        // A reset is at the first token of a chunk, which is its first
176        // micro-step on the write axis. See [`crate::packing`].
177        let packed = resets.map(|PlantResets { reset_bt, origin, chunk_len }| {
178            let chunk_tokens = standard_ssd::chunk_tokens(chunk_len, u);
179            let reset_bn = crate::packing::chunk_resets(reset_bt, chunk_tokens);
180            PackedCall {
181                segments: crate::packing::Segments::new(reset_bn, chunk_len, sequence),
182                origin: origin.unwrap_or_else(fresh),
183            }
184        });
185        // Each "last samples" cache field is read at the row end of each row,
186        // on the write axis, where one token is `u` positions.
187        let row_end = RowEnd::new(
188            pad_bs.as_ref(),
189            packed.as_ref().map(|p| (&p.segments.reset_bn, p.segments.chunk_len)),
190            u,
191        );
192
193        // ── In-projection ─────────────────────────────────────────────────────
194        let proj_btd = self.project_in(input_btd);
195        san(&proj_btd);
196        let InProjStreams {
197            z_bti,
198            x_bsi,
199            b_raw_bsMGR,
200            c_raw_btMGR,
201            dd_dt_bsh,
202            dd_a_raw_bsh,
203            lambda_raw_bsh,
204            mu_raw_bsh,
205            rot_bsa,
206            noise_bsh,
207            tropical_ab_bsh,
208        } = self.split_in_proj(proj_btd);
209        san(&z_bti);
210        san(&x_bsi);
211        san(&dd_dt_bsh);
212
213        // ── Discretisation + trapezoidal coefficients ─────────────────────────
214        // Under a Kalman gain, this call computes the decay before anything
215        // below reads it.
216        let helpers::TrapezoidCoeffs {
217            dt: dt_bsh,
218            da: da_bsh,
219            alpha: alpha_bsh,
220            nu: nu_bsh,
221            nu_interior: nu_interior_bsh,
222            gamma: gamma_bsh,
223            log_precision: log_precision_bsh,
224        } = helpers::trapezoidal_coefficients(
225            dd_dt_bsh,
226            dd_a_raw_bsh,
227            lambda_raw_bsh,
228            mu_raw_bsh,
229            self.dt_bias_h.val(),
230            self.tap_pattern_spec(),
231            self.gain_input(noise_bsh, cache.frame.log_precision_bh.clone(), packed.as_ref()),
232        )
233        .padded(pad_bs.as_ref());
234        san(&dt_bsh);
235        san(&da_bsh);
236        san(&alpha_bsh);
237        for nu_bsh in [&nu_bsh, &nu_interior_bsh].into_iter().flatten() {
238            san(nu_bsh);
239        }
240        san(&gamma_bsh);
241
242        let x_bshp = x_bsi.reshape([batch, sequence, nheads, per_head_dim]);
243
244        // ── QK-Norm on B and C ────────────────────────────────────────────────
245        // QK-Norm over state_rank, then expand ngroups→nheads, then add a
246        // per-(head, mimo-rank) bias. The group dim is axis 3 of `_btmgr`.
247        // `C` stays at token resolution: the readout is once per token, at its
248        // last micro-step.
249        let b_bsmhr = helpers::qk_norm_expand_bias::<5, 6>(
250            b_raw_bsMGR.reshape([batch, sequence, mimo_rank, ngroups, state_rank]),
251            &self.b_norm,
252            self.b_bias_hmr.val(),
253            3,
254            nheads,
255        );
256        let c_btmhr = helpers::qk_norm_expand_bias::<5, 6>(
257            c_raw_btMGR.reshape([batch, tokens, mimo_rank, ngroups, state_rank]),
258            &self.c_norm,
259            self.c_bias_hmr.val(),
260            3,
261            nheads,
262        );
263        assert_eq!([batch, sequence, mimo_rank, nheads, state_rank], b_bsmhr.dims());
264        assert_eq!([batch, tokens, mimo_rank, nheads, state_rank], c_btmhr.dims());
265
266        // ── Data-dependent transition rotation of B and C ─────────────────────
267        // Per `RotationKind`: nothing (Real1D), the cumulative angle
268        // (Complex2D), or the cumulative quaternion(s) (Quaternion4D, Rotor4D).
269        // `C` turns at token resolution: it needs the cumulative rotation of
270        // the micro-step where it is read, a stride slice of the rotation of
271        // `B`.
272        let (b_bsmhr, c_btmhr, new_rotation) = rotate_bc_forward(
273            rot_bsa,
274            dt_bsh,
275            cache.frame.rotation.clone(),
276            b_bsmhr,
277            c_btmhr,
278            u,
279            self.rotation_spec(),
280            packed.as_ref().map(|p| (&p.segments.start_bs, p.origin.frame.rotation.clone())),
281        );
282        san(&b_bsmhr);
283        san(&c_btmhr);
284        new_rotation.sanity();
285
286        // ── Save the tap FIFO: the B and x of the last `lag` positions ────────
287        // These are exactly the positions whose taps the next call must pay.
288        // So the trapezoid continues unchanged across a split prefill (at
289        // `lag = u`, these positions are exactly the last token). Each row
290        // reads them at its row end. With no β tap there is nothing to
291        // continue, and the tap FIFO stays `None`.
292        let new_tap_fifo = self.save_tap_fifo(
293            &b_bsmhr,
294            &x_bshp,
295            &da_bsh,
296            self.tap_lag(),
297            cache.frame.tap_fifo.clone(),
298            packed.as_ref().and_then(|p| p.origin.frame.tap_fifo.clone()),
299            &row_end,
300        );
301
302        PlantEntry {
303            cache,
304            packed,
305            row_end,
306            z_bti,
307            x_bshp,
308            b_bsmhr,
309            c_btmhr,
310            da_bsh,
311            alpha_bsh,
312            nu_bsh,
313            nu_interior_bsh,
314            gamma_bsh,
315            log_precision_bsh,
316            tropical_ab_bsh,
317            new_rotation,
318            new_tap_fifo,
319        }
320    }
321
322    /// The tail of the plant frame: from the readout of a scan to `out_proj`.
323    ///
324    /// It runs the positive systems, the `D` skip, the gate (or the gated
325    /// RMSNorm), the MIMO merge and `out_proj`. It then writes the frame cache
326    /// of the next call. The caller has already written the state.
327    ///
328    /// # Shapes
329    /// - output : `[batch, tokens, d_model]`
330    pub(crate) fn plant_tail(
331        &self,
332        tail: PlantTail,
333        mut cache: Mamba3DoubleSsdCache,
334    ) -> (Tensor<3>, Mamba3DoubleSsdCache) {
335        let PlantTail {
336            y_btmhp,
337            z_bti,
338            x_bshp,
339            log_precision_bsh,
340            tropical_ab_bsh,
341            row_end,
342            packed,
343            new_rotation,
344            new_tap_fifo,
345        } = tail;
346        let [batch, tokens, mimo_rank, nheads, per_head_dim] = y_btmhp.dims();
347        let btmhp = [batch, tokens, mimo_rank, nheads, per_head_dim];
348        let d_inner = self.d_inner();
349
350        let (y_btmhp, log_precision_bh, tropical_bh) = self.positive_tail(
351            y_btmhp,
352            log_precision_bsh,
353            tropical_ab_bsh,
354            cache.frame.log_precision_bh.clone(),
355            cache.frame.tropical_bh.clone(),
356            &row_end,
357            packed.as_ref(),
358        );
359
360        // ── D skip ────────────────────────────────────────────────────────────
361        // It reads the raw `x ⊙ mimo_x` (not scaled by `γ`), at the micro-step
362        // of the readout.
363        let x_bthp = last_micro4(x_bshp, self.micro_steps);
364        let mimo_x_hmp = self.mimo_x_hmp.as_ref().map(|p| p.val());
365        let v_raw_btmhp = helpers::build_v_with_mimo::<4, 5>(x_bthp, mimo_x_hmp.as_ref(), 2);
366        let d_111h1 = self.d_h.val().unsqueeze_dims::<5>(&[0, 1, 2, 4]);
367        let y_btmhp = y_btmhp + d_111h1 * v_raw_btmhp;
368        san(&y_btmhp);
369
370        // ── Gate (or gated norm) and rank aggregation ─────────────────────────
371        // When `out_norm` is set, a per-head gated RMSNorm replaces the SiLU
372        // gate: `RmsNormGated(y, z) = norm(y) * silu(z)`, with the norm over
373        // `per_head_dim`.
374        let z_bthp = z_bti.reshape([batch, tokens, nheads, per_head_dim]);
375        let y_bti = if mimo_rank > 1 {
376            let mimo_z_hmp = self.mimo_z_hmp.as_ref().map(|p| p.val()).unwrap();
377            let mimo_o_hmp = self.mimo_o_hmp.as_ref().map(|p| p.val()).unwrap();
378
379            // zₘ = z ⊙ mimo_z[m]
380            let z_btmhp = z_bthp
381                .unsqueeze_dim::<5>(2) // z_bt1hp
382                .expand(btmhp);
383            let mimo_z_btmhp = mimo_z_hmp
384                .swap_dims(0, 1) // mimo_z_mhp
385                .unsqueeze_dims::<5>(&[0, 1]) // mimo_z_11mhp
386                .expand(btmhp);
387            let z_btmhp = z_btmhp * mimo_z_btmhp;
388            san(&z_btmhp);
389
390            let combined_btmhp = match &self.out_norm {
391                Some(norm) => norm.forward(y_btmhp, z_btmhp),
392                None => y_btmhp * Silu::new().forward(z_btmhp),
393            };
394            san(&combined_btmhp);
395
396            // out = sumₘ mimo_o[m] ⊙ combined[m]
397            let mimo_o_btmhp = mimo_o_hmp
398                .swap_dims(0, 1) // mimo_o_mhp
399                .unsqueeze_dims::<5>(&[0, 1]) // mimo_o_11mhp
400                .expand(btmhp);
401            let y_bthp: Tensor<4> = (combined_btmhp * mimo_o_btmhp)
402                .sum_dim(2) // y_bt1hp
403                .squeeze_dim(2);
404            y_bthp.reshape([batch, tokens, d_inner])
405        } else {
406            // SISO: one rank, so the gate alone.
407            let y_bthp: Tensor<4> = y_btmhp.squeeze_dim(2);
408            let combined_bthp = match &self.out_norm {
409                Some(norm) => norm.forward(y_bthp, z_bthp),
410                None => y_bthp * Silu::new().forward(z_bthp),
411            };
412            combined_bthp.reshape([batch, tokens, d_inner])
413        };
414        san(&y_bti);
415
416        // ── Out-projection ────────────────────────────────────────────────────
417        let out_btd = self.out_proj.forward(y_bti);
418        san(&out_btd);
419
420        cache.frame = Mamba3FrameCache {
421            tap_fifo: new_tap_fifo,
422            rotation: new_rotation,
423            log_precision_bh,
424            tropical_bh,
425        };
426
427        (out_btd, cache)
428    }
429
430    /// Split the output of the in-projection into its channel groups, in the
431    /// layout of the table in [`crate::mamba3::in_proj_layout`].
432    ///
433    /// # Shapes
434    /// - `proj_btd` : `[batch, tokens, d_in_proj]`
435    pub(crate) fn split_in_proj(&self, proj_btd: Tensor<3>) -> InProjStreams {
436        use GroupKey::*;
437        let mut parts = self.in_proj_layout().split(proj_btd);
438        InProjStreams {
439            z_bti: parts.take_fixed(Z),
440            x_bsi: parts.take_fixed(X),
441            b_raw_bsMGR: parts.take_fixed(BRaw),
442            c_raw_btMGR: parts.take_fixed(CRaw),
443            dd_dt_bsh: parts.take_fixed(DdDt),
444            dd_a_raw_bsh: parts.take_fixed(DdARaw),
445            lambda_raw_bsh: parts.take(LambdaRaw),
446            mu_raw_bsh: parts.take(MuRaw),
447            rot_bsa: parts.take(Rotation),
448            noise_bsh: parts.take(KalmanNoise),
449            tropical_ab_bsh: parts.take(TropicalA).zip(parts.take(TropicalB)),
450        }
451    }
