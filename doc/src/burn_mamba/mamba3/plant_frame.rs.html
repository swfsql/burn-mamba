1//! # Mamba-3 — Plant Frame
2//!
3//! The plant frame is the work of a Mamba-3 block around the plant. Both
4//! forward pathways and `step` call it. Each one keeps only its middle: its
5//! scan and the inputs that only this scan needs. `step` enters at
6//! `tokens = 1`. It adds that read axis to its input and removes it from its
7//! output.
8//!
9//! ```text
10//!   input  ── plant_entry ──▶ rotated B and C, masses, tap slots
11//!                                        │
12//!                              the scan of the pathway
13//!                                        │
14//!   output ◀── plant_tail ───          readout
15//! ```
16//!
17//! - [`Mamba3::plant_entry`] holds the padding, the segments, the fresh cache,
18//!   the origin, the in-projection, the split, the trapezoid coefficients,
19//!   QK-Norm, the rotation and the tap slots.
20//! - [`Mamba3::plant_tail`] holds the positive systems, the `D` skip, the gate,
21//!   the MIMO merge and `out_proj`. The band correction of single-SSD runs
22//!   before the tail and stays in that pathway.
23//!
24//! The frame has one cache type, `Mamba3DoubleSsdCache`. A forward starts and
25//! ends at a call boundary, where the accumulator of single-SSD equals the
26//! state. So the `From` impls convert the cache of that pathway (ADR-0003).
27//!
28//! A bug in the frame fails both pathways alike, and a parity test cannot show
29//! it. So the tests of this module compare the split and the tail with their
30//! definitions (ADR-0007).
31

42/// The channel groups of the in-projection, in the order that
43/// [`Mamba3::in_proj`] documents. A per-micro-step group is on the write axis.
44/// The gate `z` and the read `C` stay at token resolution.
45#[allow(non_snake_case)]
46pub(crate) struct InProjStreams {
47    pub z_bti: Tensor<3>,
48    pub x_bsi: Tensor<3>,
49    pub b_raw_bsMGR: Tensor<3>,
50    pub c_raw_btMGR: Tensor<3>,
51    pub dd_dt_bsh: Tensor<3>,
52    pub dd_a_raw_bsh: Tensor<3>,
53    /// `None` under `TapPattern::None`.
54    pub lambda_raw_bsh: Option<Tensor<3>>,
55    /// `None` unless the pattern has two taps.
56    pub mu_raw_bsh: Option<Tensor<3>>,
57    /// `None` under `Real1D`.
58    pub rot_bsa: Option<Tensor<3>>,
59    /// `None` unless the gain projects its noise.
60    pub noise_bsh: Option<Tensor<3>>,
61    /// `(a, b)`. `None` without the tropical register.
62    pub tropical_ab_bsh: Option<(Tensor<3>, Tensor<3>)>,
63}

65/// The resets of a packed call. The chunk length comes with them, because only
66/// the resets need it: `step` has no SSD path.
67pub(crate) struct PlantResets {
68    /// The first token of each segment, `[batch, tokens]`.
69    pub reset_bt: Tensor<2, Bool>,
70    /// The cache that each segment starts from. `None` uses a fresh cache.
71    pub origin: Option<Mamba3DoubleSsdCache>,
72    /// The chunk length on the write axis.
73    pub chunk_len: usize,
74}

76/// The result of [`Mamba3::plant_entry`], one field for each stream.
77///
78/// A pathway moves each field out at once. It does not hold this struct across
79/// its scan, so each tensor drops at its last use.
80pub(crate) struct PlantEntry {
81    /// The cache of the call: the one that the caller gave, or a fresh one.
82    pub cache: Mamba3DoubleSsdCache,
83    /// The resets and the origin of a packed call.
84    pub packed: Option<Packed>,
85    /// One past the last real write position of each row, for a padded call.
86    pub end_b: Option<Tensor<1, Int>>,
87    /// The gate.
88    pub z_bti: Tensor<3>,
89    /// The value stream.
90    pub x_bshp: Tensor<4>,
91    /// The rotated `B`.
92    pub b_bsmhr: Tensor<5>,
93    /// The rotated `C`.
94    pub c_btmhr: Tensor<5>,
95    /// The log-decay.
96    pub da_bsh: Tensor<3>,
97    /// `α = exp(da)`.
98    pub alpha_bsh: Tensor<3>,
99    /// The tap mass `ν`. `None` under `TapPattern::None`.
100    pub nu_bsh: Option<Tensor<3>>,
101    /// The mass of the lag-1 tap of a two-tap pattern.
102    pub nu_interior_bsh: Option<Tensor<3>>,
103    /// `γ = λ·Δ`.
104    pub gamma_bsh: Tensor<3>,
105    /// `ln Λ` at each write position, under a Kalman gain.
106    pub log_precision_bsh: Option<Tensor<3>>,
107    /// The `(a, b)` of the tropical register, on the write axis.
108    pub tropical_ab_bsh: Option<(Tensor<3>, Tensor<3>)>,
109    /// The cumulative rotation for the next call.
110    pub new_rotation: RotationState,
111    /// The `B` tap slots for the next call.
112    pub b_last_bumhr: Option<Tensor<5>>,
113    /// The `x` tap slots for the next call.
114    pub x_last_buhp: Option<Tensor<4>>,
115}

117/// The input of [`Mamba3::plant_tail`]. The readout comes from the middle of a
118/// pathway. Each other field is the field of [`PlantEntry`] with the same name.
119pub(crate) struct PlantTail {
120    /// The readout of the scan.
121    pub y_btmhp: Tensor<5>,
122    pub z_bti: Tensor<3>,
123    pub x_bshp: Tensor<4>,
124    pub log_precision_bsh: Option<Tensor<3>>,
125    pub tropical_ab_bsh: Option<(Tensor<3>, Tensor<3>)>,
126    pub end_b: Option<Tensor<1, Int>>,
127    pub packed: Option<Packed>,
128    pub new_rotation: RotationState,
129    pub b_last_bumhr: Option<Tensor<5>>,
130    pub x_last_buhp: Option<Tensor<4>>,
131}
132
133impl Mamba3 {
134    /// The entry of the plant frame: from the input to the rotated `B` and `C`,
135    /// the masses and the tap slots.
136    ///
137    /// `cache` and the origin of `resets` start fresh when they are `None`.
138    ///
139    /// # Shapes
140    /// - `input_btd` : `[batch, tokens, d_model]`
141    /// - `pad_bt`    : `[batch, tokens]`
142    #[allow(non_snake_case)]
143    pub(crate) fn plant_entry(
144        &self,
145        input_btd: Tensor<3>,
146        cache: Option<Mamba3DoubleSsdCache>,
147        pad_bt: Option<Tensor<2, Bool>>,
148        resets: Option<PlantResets>,
149    ) -> PlantEntry {
150        let [batch, tokens, _d_model] = input_btd.dims();
151        let nheads = self.nheads();
152        let ngroups = self.ngroups;
153        let per_head_dim = self.per_head_dim();
154        let state_rank = self.state_rank;
155        let mimo_rank = self.mimo_rank;
156        let u = self.micro_steps;
157        let device = input_btd.device();
158
159        // MambaProduct: from the split down to the readout, one sequence
160        // position *is* one micro-step. The `s` in each shape suffix counts
161        // micro-steps, and `t` counts tokens. See [`crate::mamba3::product`].
162        let sequence = tokens * u;
163
164        assert!(tokens > 0, "sequence length must be at least 1");
165        assert_eq!(nheads % ngroups, 0);
166        san(&input_btd);
167
168        // A padded token pads all of its micro-steps. `end_b` is one past the
169        // last real position of each row on the write axis, where its cache
170        // fields are read.
171        let pad_bs = pad_bt.map(|pad_bt| crate::padding::repeat_positions(pad_bt, u));
172        let end_b = pad_bs.as_ref().map(crate::padding::real_end_b);
173
174        // ── Initialise the caches that the caller did not give ────────────────
175        let fresh = || {
176            let ssm_bhpr = Tensor::zeros([batch, nheads, per_head_dim, state_rank], &device);
177            let (k_state_bumhr, v_state_buhp) = self.zero_tap_slots(batch, &device);
178            let rotation = self.zero_rotation_state(batch, &device);
179            let (log_precision_bh, tropical_bh) = self.zero_positive_state(batch, &device);
180            Mamba3DoubleSsdCache {
181                ssm_bhpr,
182                k_state_bumhr,
183                v_state_buhp,
184                rotation,
185                log_precision_bh,
186                tropical_bh,
187            }
188        };
189        let cache = cache.unwrap_or_else(fresh);
190        // A reset is at the first token of a chunk, which is its first
191        // micro-step on the write axis. See [`crate::packing`].
192        let packed = resets.map(|PlantResets { reset_bt, origin, chunk_len }| {
193            let chunk_tokens = standard_ssd::chunk_tokens(chunk_len, u);
194            let reset_bn = crate::packing::chunk_resets(reset_bt, chunk_tokens);
195            Packed {
196                segments: crate::packing::Segments::new(reset_bn, chunk_len, sequence),
197                origin: origin.unwrap_or_else(fresh),
198            }
199        });
200
201        // ── In-projection ─────────────────────────────────────────────────────
202        let proj_btd = self.project_in(input_btd);
203        san(&proj_btd);
204        let InProjStreams {
205            z_bti,
206            x_bsi,
207            b_raw_bsMGR,
208            c_raw_btMGR,
209            dd_dt_bsh,
210            dd_a_raw_bsh,
211            lambda_raw_bsh,
212            mu_raw_bsh,
213            rot_bsa,
214            noise_bsh,
215            tropical_ab_bsh,
216        } = self.split_in_proj(proj_btd);
217        san(&z_bti);
218        san(&x_bsi);
219        san(&dd_dt_bsh);
220
221        // ── Discretisation + trapezoidal coefficients ─────────────────────────
222        // Under a Kalman gain, this call computes the decay before anything
223        // below reads it.
224        let helpers::TrapezoidCoeffs {
225            dt: dt_bsh,
226            da: da_bsh,
227            alpha: alpha_bsh,
228            nu: nu_bsh,
229            nu_interior: nu_interior_bsh,
230            gamma: gamma_bsh,
231            log_precision: log_precision_bsh,
232        } = helpers::trapezoidal_coefficients(
233            dd_dt_bsh,
234            dd_a_raw_bsh,
235            lambda_raw_bsh,
236            mu_raw_bsh,
237            self.dt_bias_h.val(),
238            self.tap_pattern_spec(),
239            self.gain_input(noise_bsh, cache.log_precision_bh.clone(), packed.as_ref()),
240        )
241        .padded(pad_bs.as_ref());
242        san(&dt_bsh);
243        san(&da_bsh);
244        san(&alpha_bsh);
245        for nu_bsh in [&nu_bsh, &nu_interior_bsh].into_iter().flatten() {
246            san(nu_bsh);
247        }
248        san(&gamma_bsh);
249
250        let x_bshp = x_bsi.reshape([batch, sequence, nheads, per_head_dim]);
251
252        // ── QK-Norm on B and C ────────────────────────────────────────────────
253        // QK-Norm over state_rank, then expand ngroups→nheads, then add a
254        // per-(head, mimo-rank) bias. The group dim is axis 3 of `_btmgr`.
255        // `C` stays at token resolution: the readout is once per token, at its
256        // last micro-step.
257        let b_bsmhr = helpers::qk_norm_expand_bias::<5, 6>(
258            b_raw_bsMGR.reshape([batch, sequence, mimo_rank, ngroups, state_rank]),
259            &self.b_norm,
260            self.b_bias_hmr.val(),
261            3,
262            nheads,
263        );
264        let c_btmhr = helpers::qk_norm_expand_bias::<5, 6>(
265            c_raw_btMGR.reshape([batch, tokens, mimo_rank, ngroups, state_rank]),
266            &self.c_norm,
267            self.c_bias_hmr.val(),
268            3,
269            nheads,
270        );
271        assert_eq!([batch, sequence, mimo_rank, nheads, state_rank], b_bsmhr.dims());
272        assert_eq!([batch, tokens, mimo_rank, nheads, state_rank], c_btmhr.dims());
273
274        // ── Data-dependent transition rotation of B and C ─────────────────────
275        // Per `RotationKind`: nothing (Real1D), the cumulative angle
276        // (Complex2D), or the cumulative quaternion(s) (Quaternion4D, Rotor4D).
277        // `C` turns at token resolution: it needs the cumulative rotation of
278        // the micro-step where it is read, a stride slice of the rotation of
279        // `B`.
280        let (b_bsmhr, c_btmhr, new_rotation) = rotate_bc_forward(
281            rot_bsa,
282            dt_bsh,
283            cache.rotation.clone(),
284            b_bsmhr,
285            c_btmhr,
286            u,
287            self.rotation_spec(),
288            packed.as_ref().map(|p| (&p.segments.start_bs, p.origin.rotation.clone())),
289        );
290        san(&b_bsmhr);
291        san(&c_btmhr);
292        new_rotation.sanity();
293
294        // ── Save the B and x of the last `lag` positions for the cache ────────
295        // These are exactly the positions whose taps the next call must pay.
296        // So the trapezoid continues unchanged across a split prefill (at
297        // `lag = u`, that window is exactly the last token). With no β tap
298        // there is nothing to continue, and the slots stay empty.
299        let (b_last_bumhr, x_last_buhp) = self.save_tap_slots(
300            &b_bsmhr,
301            &x_bshp,
302            &da_bsh,
303            self.tap_lag(),
304            end_b.clone().map(|end_b| {
305                (end_b, cache.k_state_bumhr.clone(), cache.v_state_buhp.clone())
306            }),
307            packed.as_ref(),
308        );
309
310        PlantEntry {
311            cache,
312            packed,
313            end_b,
314            z_bti,
315            x_bshp,
316            b_bsmhr,
317            c_btmhr,
318            da_bsh,
319            alpha_bsh,
320            nu_bsh,
321            nu_interior_bsh,
322            gamma_bsh,
323            log_precision_bsh,
324            tropical_ab_bsh,
325            new_rotation,
326            b_last_bumhr,
327            x_last_buhp,
328        }
329    }
330
331    /// The tail of the plant frame: from the readout of a scan to `out_proj`.
332    ///
333    /// It runs the positive systems, the `D` skip, the gate (or the gated
334    /// RMSNorm), the MIMO merge and `out_proj`. It then fills the cache fields
335    /// of the next call. The caller has already written the state.
336    ///
337    /// # Shapes
338    /// - output : `[batch, tokens, d_model]`
339    pub(crate) fn plant_tail(
340        &self,
341        tail: PlantTail,
342        mut cache: Mamba3DoubleSsdCache,
343    ) -> (Tensor<3>, Mamba3DoubleSsdCache) {
344        let PlantTail {
345            y_btmhp,
346            z_bti,
347            x_bshp,
348            log_precision_bsh,
349            tropical_ab_bsh,
350            end_b,
351            packed,
352            new_rotation,
353            b_last_bumhr,
354            x_last_buhp,
355        } = tail;
356        let [batch, tokens, mimo_rank, nheads, per_head_dim] = y_btmhp.dims();
357        let btmhp = [batch, tokens, mimo_rank, nheads, per_head_dim];
358        let d_inner = self.d_inner();
359
360        let (y_btmhp, log_precision_bh, tropical_bh) = self.positive_tail(
361            y_btmhp,
362            log_precision_bsh,
363            tropical_ab_bsh,
364            cache.tropical_bh.clone(),
365            end_b.map(|end_b| (end_b, cache.log_precision_bh.clone())),
366            packed.as_ref(),
367        );
368        cache.log_precision_bh = log_precision_bh;
369        cache.tropical_bh = tropical_bh;
370
371        // ── D skip ────────────────────────────────────────────────────────────
372        // It reads the raw `x ⊙ mimo_x` (not scaled by `γ`), at the micro-step
373        // of the readout.
374        let x_bthp = last_micro4(x_bshp, self.micro_steps);
375        let mimo_x_hmp = self.mimo_x_hmp.as_ref().map(|p| p.val());
376        let v_raw_btmhp = helpers::build_v_with_mimo::<4, 5>(x_bthp, mimo_x_hmp.as_ref(), 2);
377        let d_111h1 = self.d_h.val().unsqueeze_dims::<5>(&[0, 1, 2, 4]);
378        let y_btmhp = y_btmhp + d_111h1 * v_raw_btmhp;
379        san(&y_btmhp);
380
381        // ── Gate (or gated norm) and rank aggregation ─────────────────────────
382        // When `out_norm` is set, a per-head gated RMSNorm replaces the SiLU
383        // gate: `RmsNormGated(y, z) = norm(y) * silu(z)`, with the norm over
384        // `per_head_dim`.
385        let z_bthp = z_bti.reshape([batch, tokens, nheads, per_head_dim]);
386        let y_bti = if mimo_rank > 1 {
387            let mimo_z_hmp = self.mimo_z_hmp.as_ref().map(|p| p.val()).unwrap();
388            let mimo_o_hmp = self.mimo_o_hmp.as_ref().map(|p| p.val()).unwrap();
389
390            // zₘ = z ⊙ mimo_z[m]
391            let z_btmhp = z_bthp
392                .unsqueeze_dim::<5>(2) // z_bt1hp
393                .expand(btmhp);
394            let mimo_z_btmhp = mimo_z_hmp
395                .swap_dims(0, 1) // mimo_z_mhp
396                .unsqueeze_dims::<5>(&[0, 1]) // mimo_z_11mhp
397                .expand(btmhp);
398            let z_btmhp = z_btmhp * mimo_z_btmhp;
399            san(&z_btmhp);
400
401            let combined_btmhp = match &self.out_norm {
402                Some(norm) => norm.forward(y_btmhp, z_btmhp),
403                None => y_btmhp * Silu::new().forward(z_btmhp),
404            };
405            san(&combined_btmhp);
406
407            // out = sumₘ mimo_o[m] ⊙ combined[m]
408            let mimo_o_btmhp = mimo_o_hmp
409                .swap_dims(0, 1) // mimo_o_mhp
410                .unsqueeze_dims::<5>(&[0, 1]) // mimo_o_11mhp
411                .expand(btmhp);
412            let y_bthp: Tensor<4> = (combined_btmhp * mimo_o_btmhp)
413                .sum_dim(2) // y_bt1hp
414                .squeeze_dim(2);
415            y_bthp.reshape([batch, tokens, d_inner])
416        } else {
417            // SISO: one rank, so the gate alone.
418            let y_bthp: Tensor<4> = y_btmhp.squeeze_dim(2);
419            let combined_bthp = match &self.out_norm {
420                Some(norm) => norm.forward(y_bthp, z_bthp),
421                None => y_bthp * Silu::new().forward(z_bthp),
422            };
423            combined_bthp.reshape([batch, tokens, d_inner])
424        };
425        san(&y_bti);
426
427        // ── Out-projection ────────────────────────────────────────────────────
428        let out_btd = self.out_proj.forward(y_bti);
429        san(&out_btd);
430
431        cache.k_state_bumhr = b_last_bumhr;
432        cache.v_state_buhp = x_last_buhp;
433        cache.rotation = new_rotation;
434
435        (out_btd, cache)
436    }
437
438    /// Split the output of the in-projection into its channel groups, in the
439    /// layout of the table in [`crate::mamba3::in_proj_layout`].
440    ///
441    /// # Shapes
442    /// - `proj_btd` : `[batch, tokens, d_in_proj]`
443    pub(crate) fn split_in_proj(&self, proj_btd: Tensor<3>) -> InProjStreams {
444        use GroupKey::*;
445        let mut parts = self.in_proj_layout().split(proj_btd);
446        InProjStreams {
447            z_bti: parts.take_fixed(Z),
448            x_bsi: parts.take_fixed(X),
449            b_raw_bsMGR: parts.take_fixed(BRaw),
450            c_raw_btMGR: parts.take_fixed(CRaw),
451            dd_dt_bsh: parts.take_fixed(DdDt),
452            dd_a_raw_bsh: parts.take_fixed(DdARaw),
453            lambda_raw_bsh: parts.take(LambdaRaw),
454            mu_raw_bsh: parts.take(MuRaw),
455            rot_bsa: parts.take(Rotation),
456            noise_bsh: parts.take(KalmanNoise),
457            tropical_ab_bsh: parts.take(TropicalA).zip(parts.take(TropicalB)),
458        }
459    }
