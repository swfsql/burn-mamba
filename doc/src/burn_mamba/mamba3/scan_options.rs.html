1//! # The scan options of Mamba-3
2//!
3//! [`Mamba3ScanOptions`] holds the [SSD path](crate::ssd_path::SsdPath), the
4//! chunk length of the chunkwise SSD scan and the [`Pathway`].
5//! [`Mamba3::forward_packed`](crate::mamba3::mamba3::Mamba3::forward_packed),
6//! which `forward` calls, sends the scan options to the pathway that they
7//! select. Both pathways read the SSD path and the chunk length.
8
9use crate::mamba3::prelude::*;
10use burn::config::Config;
11
12/// The pathway of a Mamba-3 forward: how the forward maps the trapezoidal
13/// recurrence onto SSD calls. Both pathways give the same outputs, final
14/// caches and gradients (ADR-0003).
15///
16/// [`Mamba3::step`] has no pathway. It runs the double-SSD recurrence.
17///
18/// The default is [`Self::SingleSsd`].
19#[derive(Debug, Clone, Copy, PartialEq, Eq, Default, serde::Serialize, serde::Deserialize)]
20pub enum Pathway {
21    /// One standard SSD call for each term of the trapezoid: a γ-SSD for the
22    /// current sample, and one β-SSD for each `β` tap. That is two calls at
23    /// the default tap pattern, three under a two-tap pattern, and one under
24    /// [`TapPattern::None`].
25    ///
26    /// It follows the "two-SSD decomposition" of mamba3-minimal. See
27    /// [`crate::mamba3::double_ssd`].
28    DoubleSsd,
29
30    /// One SSD call, at every tap pattern. It follows the form of the official
31    /// kernels: Triton for SISO and Tilelang for MIMO. See
32    /// [`crate::mamba3::single_ssd`].
33    ///
34    /// It uses about half the training memory of [`Self::DoubleSsd`] at the
35    /// default tap pattern, and less than a third under a two-tap pattern.
36    #[default]
37    SingleSsd,
38}
39
40/// The scan options of a Mamba-3 block: the SSD path, the chunk length and
41/// the pathway.
42///
43/// All scan options give the same values and gradients.
44/// [`Mamba3::forward_packed`] runs the packed forward of the pathway of
45/// [`Self::pathway`]. That forward computes the chunk length. It builds the
46/// input bundle of the pathway at that length
47/// ([`crate::mamba3::double_ssd::ssd::Mamba3DoubleSsdInput`] or
48/// [`crate::mamba3::single_ssd::ssd::Mamba3SingleSsdInput`]). Then it gives
49/// the SSD path to the `run` of the bundle.
50///
51/// The default is [`SsdPath::SerialRecalculated`] with an unset chunk length,
52/// on [`Pathway::SingleSsd`]. For example,
53/// `Mamba3ScanOptions::new(SsdPath::Serial).with_pathway(Pathway::DoubleSsd)`
54/// selects the `Serial` SSD path on double-SSD.
55#[derive(Config, Debug)]
56pub struct Mamba3ScanOptions {
57    /// The method that computes the chunkwise SSD scan.
58    pub ssd_path: SsdPath,
59    /// The chunk length, in write positions. A larger value gives more
60    /// intra-chunk GEMM work and a shorter inter-chunk scan.
61    ///
62    /// `None` uses the optimal value, [`Self::optimal_chunk_len`]. That value
63    /// is approximately `√(state_rank · per_head_dim)`, divided by the widening
64    /// that `mimo_rank` and `micro_steps` already give the chunk.
65    pub chunk_len: Option<usize>,
66    /// The pathway of [`Mamba3::forward`] and [`Mamba3::forward_packed`].
67    #[config(default = "Pathway::SingleSsd")]
68    pub pathway: Pathway,
69}
70
71impl Default for Mamba3ScanOptions {
72    fn default() -> Self {
73        Self::new(SsdPath::default())
74    }
75}
76
77impl Mamba3ScanOptions {
78    /// Optimal chunk length, in **write positions**: `√(state_rank ·
79    /// per_head_dim)` divided by `mimo_rank`, rounded up to a multiple of 32,
80    /// clamped to `32 ..= 512`. `micro_steps` does not change this width:
81    /// [`chunk_tokens`](crate::standard_ssd::chunk_tokens) absorbs it.
82    ///
83    /// The square root is the SISO rule of thumb: it balances the intra-chunk
84    /// GEMMs against the inter-chunk scan. Two dials of the block widen a chunk
85    /// without a place in this formula, and they are **not** the same widening:
86    ///
87    /// - **`mimo_rank`** (`m`) widens *both* axes of a chunk. Every rank writes
88    ///   and every rank reads, so the intra-chunk matmuls run on the fused
89    ///   `chunk_len · m`. The rule of thumb is about that product, hence the
90    ///   divisor. An unchanged `chunk_len` would cost `m²`, and this costs `m`.
91    ///   The reference gives the same advice on its `chunk_size` argument: "64
92    ///   for SISO, 64/mimo_rank for MIMO".
93    /// - **`micro_steps`** (`u`) widens only the **write** axis. All `u`
94    ///   micro-steps of a token write to the state, but the block *reads* the
95    ///   token once, at its last micro-step. So the score is
96    ///   `[batch, nchunks, heads, T·m, T·u·m]` for `T` =
97    ///   [`chunk_tokens`](crate::standard_ssd::chunk_tokens),
98    ///   with `batch · sequence · heads · T · u · m²` elements. To keep that flat
99    ///   in `u`, keep `T · u` (the chunk on the *write* axis) flat and use fewer
100    ///   tokens.
101    ///   Memory is the binding constraint here, because no fused kernel hides it.
102    ///
103    /// So `u` subdivides the chunk and does not shorten it, and `nchunks` grows
104    /// like `u`, not `u²`. The returned length is a multiple of `micro_steps`.
105    /// So each chunk is a whole number of tokens, and their positions are a
106    /// contiguous run. It is on the 32 grid exactly when `micro_steps` divides
107    /// it.
108    ///
109    /// `info/mamba-3/architecture-deltas.md` §8.
110    pub fn optimal_chunk_len(
111        state_rank: usize,
112        per_head_dim: usize,
113        mimo_rank: usize,
114        micro_steps: usize,
115    ) -> usize {
116        let micro_steps = micro_steps.max(1);
117        let chunk_len = (state_rank * per_head_dim)
118            .isqrt()
119            .div_ceil(mimo_rank.max(1)) // the fused read axis is `chunk_tokens · m`.
120            .next_multiple_of(32) // rule-of-thumb: common plane dimension.
121            .clamp(32, 512); // rule-of-thumb: one plane minimum, ceiling at 512.
122        // A whole number of tokens per chunk, at (very nearly) that chunk length.
123        chunk_len.div_ceil(micro_steps) * micro_steps
124    }
125
126    /// The chunk length of these scan options, or [`Self::optimal_chunk_len`]
127    /// for the dimensions of `block` when unset. In both cases the length is
128    /// rounded **up to a multiple of `micro_steps`**, so a chunk is a whole
129    /// number of tokens.
130    ///
131    /// With that rounding, the read axis is a plain reshape. With
132    /// `chunk_len = T·u`, a chunk covers write positions `[cL, (c+1)L)`, that
133    /// is tokens `[cT, (c+1)T)`, and the surviving positions (`≡ u−1 mod u`) are a
134    /// contiguous run of them. At `micro_steps = 1` the rounding does nothing.
135    pub fn chunk_len_or_optimal(&self, block: &Mamba3) -> usize {
136        match self.chunk_len {
137            Some(chunk_len) => chunk_len.next_multiple_of(block.micro_steps.max(1)),
138            None => Self::optimal_chunk_len(
139                block.state_rank,
140                block.per_head_dim(),
141                block.mimo_rank,
142                block.micro_steps,
143            ),
144        }
145    }
146}
