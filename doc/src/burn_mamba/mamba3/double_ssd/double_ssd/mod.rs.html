1//! # Mamba-3 — Double-Pass SSD Forward
2//!
3//! [`Mamba3::forward_double_ssd`](crate::mamba3::mamba3::Mamba3::forward_double_ssd)
4//! and [`Mamba3::step_double_ssd`](crate::mamba3::mamba3::Mamba3::step_double_ssd):
5//! the decomposition of
6//! [`VikramLex/mamba3-minimal`](https://github.com/VikramLex/mamba3-minimal).
7//!
8//! ```text
9//!   hₚ = αₚ hₚ₋₁ + βₚ Bₚ₋ₗₐ₉ ⊗ xₚ₋ₗₐ₉ + γₚ Bₚ ⊗ xₚ   (original double-ssd trapezoidal)
10//!
11//!   forward:    h = SSD(γ-scaled V, B)   +   SSD(β-scaled V_shifted, B_shifted)
12//! ```
13//!
14//! The shift is [`TapPattern::tap_lag`](crate::mamba3::tap_pattern::TapPattern::tap_lag)
15//! write positions (`1` by default, `u` for
16//! [`TapPattern::Vertical`](crate::mamba3::tap_pattern::TapPattern::Vertical)), and
17//! `β = ν·α` carries the transport across that whole gap.
18//!
19//! A two-tap pattern adds a second `β` side at lag 1. The two shifts are
20//! different, so the passes cannot fuse, and this pathway runs **three** SSD
21//! calls. The single-SSD pathway collapses them into its key scale and stays at
22//! one call, so it suits such a pattern better.
23//!
24//! This is simple to derive and to verify (everything uses the standard SSD),
25//! but it increases the intra-chunk and chunk-state memory during training.
26//!
27//! See also: [`crate::mamba3::mamba3`] and [`crate::mamba3::single_ssd::single_ssd`].
28
29use crate::mamba3::double_ssd::prelude::*;
30use crate::mamba3::helpers;
31use crate::mamba3::plant_frame::{PlantEntry, PlantResets, PlantTail};
32use crate::mamba3::prelude::*;
33use burn_stack::modules::sanity as san;

36// ---------------------------------------------------------------------------
37// Mamba3::forward  (chunkwise double-SSD — training / prefill)
38// ---------------------------------------------------------------------------
39
40impl Mamba3 {
41    /// Process a full input sequence with the double-ssd pathway.
42    ///
43    /// For MIMO (mimo_rank>1), B/C have mimo_rank parallel rank channels. All
44    /// ranks share the state, and each rank writes to it independently.
45    ///
46    /// `pad_bt` marks a right-padded batch of tokens, as in [`Self::forward`].
47    ///
48    /// # Shapes
49    /// - `input_bsm` : `[batch, sequence, d_model]`
50    /// - `pad_bt`    : `[batch, sequence]`
51    /// - output      : `[batch, sequence, d_model]`
52    pub fn forward_double_ssd(
53        &self,
54        input_bsm: Tensor<3>,
55        cache: Option<Mamba3DoubleSsdCache>,
56        ssd_path: &Mamba3SsdPath,
57        pad_bt: Option<Tensor<2, Bool>>,
58    ) -> (Tensor<3>, Mamba3DoubleSsdCache) {
59        self.forward_double_ssd_packed(input_bsm, cache, None, ssd_path, pad_bt, None)
60    }
61
62    /// [`Self::forward_double_ssd`] over a packed batch: `reset_bt` marks the
63    /// first token of each segment, which starts from `origin`, as in
64    /// [`Self::forward_packed`].
65    #[allow(non_snake_case)]
66    pub fn forward_double_ssd_packed(
67        &self,
68        input_bsm: Tensor<3>,
69        cache: Option<Mamba3DoubleSsdCache>,
70        origin: Option<Mamba3DoubleSsdCache>,
71        ssd_path: &Mamba3SsdPath,
72        pad_bt: Option<Tensor<2, Bool>>,
73        reset_bt: Option<Tensor<2, Bool>>,
74    ) -> (Tensor<3>, Mamba3DoubleSsdCache) {
75        let [batch, tokens, _d_model] = input_bsm.dims();
76        let nheads = self.nheads();
77        let per_head_dim = self.per_head_dim();
78        let state_rank = self.state_rank;
79        let mimo_rank = self.mimo_rank;
80        let u = self.micro_steps;
81        let device = input_bsm.device();
82        let sequence = tokens * u;
83
84        // ── The entry of the plant frame ──────────────────────────────────────
85        let chunk_len = ssd_path.chunk_len_or_optimal(self);
86        let PlantEntry {
87            mut cache,
88            packed,
89            end_b,
90            z_bti,
91            x_bshp,
92            b_bsmhr,
93            c_btmhr,
94            da_bsh,
95            alpha_bsh,
96            nu_bsh,
97            nu_interior_bsh,
98            gamma_bsh,
99            log_precision_bsh,
100            tropical_ab_bsh,
101            new_rotation,
102            b_last_bumhr,
103            x_last_buhp,
104        } = self.plant_entry(
105            input_bsm,
106            cache,
107            pad_bt,
108            reset_bt.map(|reset_bt| PlantResets { reset_bt, origin, chunk_len }),
109        );
110
111        // ── The β term's shifted, β-scaled inputs ─────────────────────────────
112        //
113        // "Shift-Before-Chunking": prepend the cached xₜ₋₁ / Bₜ₋₁ at the
114        // sequence level (before SSD chunking), so the β term at t=0 sees the
115        // prior token of a continued cache. For a fresh (zero) cache this is
116        // the same as zero-padding.
117        //
118        // The shift is [`TapPattern::tap_lag`] *write* positions: 1 for the
119        // default [`TapPattern::HorizontalCarryOver`], `u` for
120        // [`TapPattern::Vertical`]. That is exactly the FIFO depth of `step`.
121        //
122        // A lag-`L` tap must be transported across *its own* gap (§9): by
123        // `Πᵈ⁼⁰..ᴸ⁻¹ αₚ₋ᵈ`, not `αₚ`. `β = ν·α` holds the `d = 0` factor, and
124        // `interior_gap_decay` the rest. For the first `L` positions, the
125        // missing factors are those that the `v` slots of the cache already
126        // carry.
127        //
128        // A two-tap pattern adds a second, lag-1 side. The two shifts are
129        // different, so they cannot share a pass, and this pathway runs three
130        // SSD calls. The single-SSD pathway fuses them into its key scale and
131        // stays at one. The prefix of the lag-1 tap of a two-tap pattern is the
132        // newest slot of the FIFO, which *is* the lag-1 slot, undecayed. It has no effect
133        // under [`TapPattern::VerticalPlusHorizontalReset`], whose `νⁱⁿᵗ` is
134        // zero at every position that would read it.
135        //
136        // Under [`TapPattern::None`] there is no left endpoint: no shift, no
137        // β-scaled copy of `x`, and (below) no second SSD call. `forward` is
138        // then one standard SSD pass, with keys scaled by `γ = Δ`.
139        let lag = self.tap_lag();
140        // The prefix of a lag-`L` tap is the newest `L` slots of a FIFO: all
141        // of it for the tap at `tap_lag`, only the newest slot for the lag-1
142        // tap of a two-tap pattern.
143        let heads = |fifo: &Mamba3DoubleSsdCache, lag: usize| -> (Tensor<4>, Tensor<5>) {
144            let slots = "a β tap keeps its (B, x) cache slots";
145            let slots_buhp = fifo.v_state_buhp.clone().expect(slots);
146            let slots_bumhr = fifo.k_state_bumhr.clone().expect(slots);
147            let newest = slots_buhp.dims()[1] - lag;
148            (slots_buhp.narrow(1, newest, lag), slots_bumhr.narrow(1, newest, lag))
149        };
150        let beta_side = |nu_bsh: Tensor<3>, lag: usize| -> (Tensor<4>, Tensor<5>) {
151            let (x_head_buhp, b_head_bumhr) = heads(&cache, lag);
152            let x_prev_bshp = helpers::shift_stream(x_bshp.clone(), x_head_buhp, lag);
153            let b_prev_bsmhr = helpers::shift_stream(b_bsmhr.clone(), b_head_bumhr, lag);
154            // A packed row: each segment starts on the prefix of the origin,
155            // as a call from it.
156            let (x_prev_bshp, b_prev_bsmhr) = match &packed {
157                Some(p) => {
158                    let (x_head_buhp, b_head_bumhr) = heads(&p.origin, lag);
159                    (
160                        p.segments.restart_heads(x_prev_bshp, 1, x_head_buhp),
161                        p.segments.restart_heads(b_prev_bsmhr, 1, b_head_bumhr),
162                    )
163                }
164                None => (x_prev_bshp, b_prev_bsmhr),
165            };
166            let beta_bsh = nu_bsh * alpha_bsh.clone();
167            let segments = packed.as_ref().map(|p| &p.segments);
168            let beta_bsh = match helpers::interior_gap_decay(da_bsh.clone(), lag, segments) {
169                Some(gap_bsh) => beta_bsh * gap_bsh,
170                None => beta_bsh,
171            };
172            // β is a per-head scalar, broadcast over mimo_rank and per_head_dim.
173            let beta_bsh1 = beta_bsh.unsqueeze_dim::<4>(3);
174            (x_prev_bshp * beta_bsh1, b_prev_bsmhr) // βₚ · xₚ₋ₗₐ₉
175        };
176        let beta_sides: Vec<(Tensor<4>, Tensor<5>)> = nu_bsh
177            .map(|nu_bsh| beta_side(nu_bsh, lag))
178            .into_iter()
179            .chain(nu_interior_bsh.map(|nu_bsh| beta_side(nu_bsh, 1)))
180            .collect();
181
182        // ── Scale the current-token input by γ ────────────────────────────────
183        let gamma_bsh1 = gamma_bsh.unsqueeze_dim::<4>(3);
184        let x_gamma_bshp = x_bshp.clone() * gamma_bsh1; // γₜ · xₜ
185
186        // ── Pad sequence to multiple of chunk_len ─────────────────────────────
187        let sequence_padded = sequence.next_multiple_of(chunk_len);
188        let pad = sequence_padded - sequence;
189
190        // Each zero block is built at most once, and the streams that need it
191        // share it (a `Tensor` clone is a handle, not a copy).
192        let pads = (pad > 0).then(|| {
193            (
194                Tensor::<4>::zeros([batch, pad, nheads, per_head_dim], &device),
195                Tensor::<3>::zeros([batch, pad, nheads], &device),
196                Tensor::<5>::zeros([batch, pad, mimo_rank, nheads, state_rank], &device),
197            )
198        });
199        let pad_hp = |t: Tensor<4>| match &pads {
200            Some((p, _, _)) => Tensor::cat(vec![t, p.clone()], 1),
201            None => t,
202        };
203        let pad_h = |t: Tensor<3>| match &pads {
204            Some((_, p, _)) => Tensor::cat(vec![t, p.clone()], 1),
205            None => t,
206        };
207        let pad_mhr = |t: Tensor<5>| match &pads {
208            Some((_, _, p)) => Tensor::cat(vec![t, p.clone()], 1),
209            None => t,
210        };
211
212        // `C` is on the read axis of the chunk, so it pads by whole tokens.
213        // `chunk_len` is a multiple of `u`, so `pad` is too.
214        let tokens_padded = sequence_padded / u;
215        let pad_tmhr = |t: Tensor<5>| match pad {
216            0 => t,
217            _ => Tensor::cat(
218                vec![
219                    t,
220                    Tensor::zeros(
221                        [batch, tokens_padded - tokens, mimo_rank, nheads, state_rank],
222                        &device,
223                    ),
224                ],
225                1,
226            ),
227        };
228
229        let x_gamma_bShp = pad_hp(x_gamma_bshp);
230        let da_bSh = pad_h(da_bsh);
231        let b_bSmhr = pad_mhr(b_bsmhr);
232        let c_bTmhr = pad_tmhr(c_btmhr);
233        let beta_sides: Vec<_> = beta_sides
234            .into_iter()
235            .map(|(x_beta, b_prev)| (pad_hp(x_beta), pad_mhr(b_prev)))
236            .collect();
237
238        // ── Reshape into chunks ───────────────────────────────────────────────
239        let nchunks = sequence_padded / chunk_len;
240        let chunk_tokens = Mamba3SsdPath::chunk_tokens(chunk_len, u);
241        let x_gamma_bnlhp = x_gamma_bShp.reshape([batch, nchunks, chunk_len, nheads, per_head_dim]);
242        let da_bnlh = da_bSh.reshape([batch, nchunks, chunk_len, nheads]);
243        let b_bnlmhr = b_bSmhr.reshape([batch, nchunks, chunk_len, mimo_rank, nheads, state_rank]);
244        let c_bntmhr =
245            c_bTmhr.reshape([batch, nchunks, chunk_tokens, mimo_rank, nheads, state_rank]);
246
247        // ── The MIMO-SSD call(s) ──────────────────────────────────────────────
248        // Build the V tensors: insert the mimo_rank axis at position 3 of
249        // `_bnlhp`. With a β tap, these are the two standard SSD passes that
250        // give the pathway its name (γ-SSM + β-SSM, summed). Under
251        // [`TapPattern::None`] the second pass does not exist, and a
252        // "double"-SSD forward is one pass.
253        let mimo_x_hmp = self.mimo_x_hmp.as_ref().map(|p| p.val());
254        let v_gamma_bnlmhp =
255            helpers::build_v_with_mimo::<5, 6>(x_gamma_bnlhp.clone(), mimo_x_hmp.as_ref(), 3);
256        // Every pass restarts its carry at the same chunks: the passes are
257        // the terms of one state. The γ pass holds the state of the origin,
258        // and a β pass restarts from zero, as it starts a call.
259        let reset_bn = packed.as_ref().map(|p| p.segments.reset_bn.clone());
260
261        let input_gamma = Mamba3DoubleSsdInput {
262            v_bnlmhp: v_gamma_bnlmhp,
263            da_bnlh: da_bnlh.clone(),
264            b_bnlmhr: b_bnlmhr.clone(),
265            c_bntmhr: c_bntmhr.clone(),
266            initial_state_bhpr: cache.ssm_bhpr,
267            init_state_hpr: self.init_state_hpr.as_ref().map(|s| s.val()),
268            read_stride: u,
269            restarts: packed.as_ref().map(|p| crate::packing::Restarts {
270                reset_bn: p.segments.reset_bn.clone(),
271                origin_bhpr: p.origin.ssm_bhpr.clone(),
272            }),
273        };
274        let (y_bntmhp, final_state_bhpr) = input_gamma.run(ssd_path);
275
276        let (y_bntmhp, final_state_bhpr) = beta_sides.into_iter().fold(
277            (y_bntmhp, final_state_bhpr),
278            |(y_bntmhp, final_state_bhpr), (x_beta_bShp, b_prev_bSmhr)| {
279                let x_beta_bnlhp =
280                    x_beta_bShp.reshape([batch, nchunks, chunk_len, nheads, per_head_dim]);
281                let b_prev_bnlmhr =
282                    b_prev_bSmhr.reshape([batch, nchunks, chunk_len, mimo_rank, nheads, state_rank]);
283                let v_beta_bnlmhp =
284                    helpers::build_v_with_mimo::<5, 6>(x_beta_bnlhp, mimo_x_hmp.as_ref(), 3);
285                let zero_bhpr = Tensor::zeros([batch, nheads, per_head_dim, state_rank], &device);
286                let input_beta = Mamba3DoubleSsdInput {
287                    v_bnlmhp: v_beta_bnlmhp,
288                    da_bnlh: da_bnlh.clone(),
289                    b_bnlmhr: b_prev_bnlmhr,
290                    c_bntmhr: c_bntmhr.clone(),
291                    initial_state_bhpr: zero_bhpr.clone(),
292                    init_state_hpr: None,
293                    read_stride: u,
294                    restarts: reset_bn.clone().map(|reset_bn| crate::packing::Restarts {
295                        reset_bn,
296                        origin_bhpr: zero_bhpr,
297                    }),
298                };
299                let (y_beta_bntmhp, final_state_beta_bhpr) = input_beta.run(ssd_path);
300                (
301                    y_bntmhp + y_beta_bntmhp,
302                    final_state_bhpr + final_state_beta_bhpr,
303                )
304            },
305        );
306
307        san(&y_bntmhp);
308        san(&final_state_bhpr);
309
310        cache.ssm_bhpr = final_state_bhpr;
311
312        // ── Unpad ─────────────────────────────────────────────────────────────
313        // The SSD returns token resolution: the readout is after all `u`
314        // writes, and the kernel computed only that position. From here, `t`
315        // (`tokens`) is the time axis again: the read axis.
316        let y_bTmhp = y_bntmhp.reshape([batch, tokens_padded, mimo_rank, nheads, per_head_dim]);
317        let y_btmhp = if pad == 0 {
318            y_bTmhp
319        } else {
320            y_bTmhp.narrow(1, 0, tokens)
321        };
322
323        // ── The tail of the plant frame ───────────────────────────────────────
324        self.plant_tail(
325            PlantTail {
326                y_btmhp,
327                z_bti,
328                x_bshp,
329                log_precision_bsh,
330                tropical_ab_bsh,
331                end_b,
332                packed,
333                new_rotation,
334                b_last_bumhr,
335                x_last_buhp,
336            },
337            cache,
338        )
339    }
340}

342// ---------------------------------------------------------------------------
343// Mamba3::step  (recurrent SSM — token-by-token decoding)
344// ---------------------------------------------------------------------------
345
346mod step {
347    use super::*;
348
349    impl Mamba3 {
350        /// State→output contraction:
351        /// `out[b, m, h, p] = Σᵣ C[b, m, h, r] · state[b, h, p, r]`
352        /// (`einsum('bhpr,bmhr->bmhp', state, C)`).
353        ///
354        /// At `mimo_rank == 1` the output axis of the GEMM is 1, so the matmul
355        /// is a matrix–vector product. [`step_readout_siso`](Mamba3::step_readout_siso)
356        /// writes it as a broadcast multiply plus a `state_rank` reduction.
357        /// [`Mamba3Config::siso_specialization_decode`](crate::mamba3::mamba3::Mamba3Config::siso_specialization_decode)
358        /// selects the branch. Both branches compute the same values and
359        /// gradients.
360        pub(crate) fn step_readout(
361            state_bhpr: Tensor<4>,
362            c_bmhr: Tensor<4>,
363            siso_specialization: bool,
364        ) -> Tensor<4> {
365            let [_batch, mimo_rank, _nheads, _state_rank] = c_bmhr.dims();
366            if mimo_rank == 1 && siso_specialization {
367                Self::step_readout_siso(state_bhpr, c_bmhr)
368            } else {
369                Self::step_readout_mimo(state_bhpr, c_bmhr)
370            }
371        }
372
373        /// SISO (`mimo_rank == 1`) state→output contraction: broadcast `C` over
374        /// `per_head_dim` and reduce `state_rank`.
375        pub(crate) fn step_readout_siso(state_bhpr: Tensor<4>, c_bmhr: Tensor<4>) -> Tensor<4> {
376            let c_bh1r: Tensor<4> = c_bmhr.squeeze_dim::<3>(1).unsqueeze_dim(2);
377            let out_bhp1: Tensor<4> = (state_bhpr * c_bh1r).sum_dim(3);
378            out_bhp1.squeeze_dim::<3>(3).unsqueeze_dim(1) // out_b1hp
379        }
380
381        /// General MIMO state→output contraction: one matmul over `state_rank`.
382        pub(crate) fn step_readout_mimo(state_bhpr: Tensor<4>, c_bmhr: Tensor<4>) -> Tensor<4> {
383            let c_bhrm = c_bmhr.permute([0, 2, 3, 1]);
384            let out_bhpm = state_bhpr.matmul(c_bhrm);
385            out_bhpm.permute([0, 3, 1, 2])
386        }
387
388        /// Process a **single token** with the pure recurrent form (the
389        /// formulas are in [`Mamba3::step`]). Both cache variants decode here:
390        /// `step_single_ssd` converts its cache and calls this function.
391        ///
392        /// At `micro_steps = u > 1` the token is `u` of those micro-steps
393        /// ([`crate::mamba3::product`]). They are evaluated **together**, not one
394        /// at a time (see the body).
395        ///
396        /// # Shapes
397        /// - `input_bd` : `[batch, d_model]`
398        /// - output     : `[batch, d_model]`
399        pub fn step_double_ssd(
400            &self,
401            input_bd: Tensor<2>,
402            cache: Option<Mamba3DoubleSsdCache>,
403        ) -> (Tensor<2>, Mamba3DoubleSsdCache) {
404            let [batch, _d_model] = input_bd.dims();
405            let nheads = self.nheads();
406            let per_head_dim = self.per_head_dim();
407            let state_rank = self.state_rank;
408            let mimo_rank = self.mimo_rank;
409            let u = self.micro_steps;
410
411            // ── The entry of the plant frame, at `tokens = 1` ─────────────────
412            // `step` adds a read axis of length 1 to its input, and removes it
413            // from the output of the tail. The write axis of the streams is
414            // then the `u` micro-steps of the token. A Kalman gain computes its
415            // decay in the entry, from the cached `ln Λ`, over those `u`
416            // micro-steps: the scan of `forward` at `len = u`.
417            let PlantEntry {
418                mut cache,
419                packed: _,
420                end_b: _,
421                z_bti: z_b1i,
422                x_bshp: x_buhp,
423                b_bsmhr: b_bumhr,
424                c_btmhr: c_b1mhr,
425                da_bsh: da_buh,
426                alpha_bsh: alpha_buh,
427                nu_bsh: nu_buh,
428                nu_interior_bsh: nu_interior_buh,
429                gamma_bsh: gamma_buh,
430                log_precision_bsh: log_precision_buh,
431                tropical_ab_bsh: tropical_ab_buh,
432                new_rotation,
433                b_last_bumhr,
434                x_last_buhp,
435            } = self.plant_entry(input_bd.unsqueeze_dim::<3>(1), cache, None, None);
436            let mimo_x_hmp = self.mimo_x_hmp.as_ref().map(|p| p.val());
437            let siso = self.use_siso_decode_kernels();
438
439            // ── The `u` micro-steps of the token at once, then one readout ──────
440            // MambaProduct evaluates a token as `u` consecutive positions of the
441            // ordinary recurrence. So a `step` call is a **write-axis block** of
442            // length `u`, and one call is `forward` on a one-token sequence
443            // (the parity tests assert this). The block is not walked one
444            // position at a time, because that is not necessary. After the RoPE
445            // trick, the transition inside the block is the *scalar* `α`,
446            // so the block has a closed form:
447            //
448            //     h = (∏ⱼ αⱼ)·h₋₁ + Σⱼ wⱼ · writeⱼ ,    wⱼ = ∏_{r>j} αᵣ
449            //
450            // Its transport `wⱼ` (the decay of a write to the end of the block)
451            // is the same quantity that [`crate::mamba3::helpers::tail_decay`]
452            // computes for the tap slots, over the whole block instead of its
453            // last `lag`.
454            //
455            // Every `writeⱼ` is an outer product into one shared state. So with
456            // the transport, and with `(u, mimo_rank)` fused into one
457            // contracted axis, each **side** of the recurrence is one
458            // [`helpers::mimo_outer_sum`] (the same collapse as the single-SSD
459            // boundary seed over its slots). Thus the op count does not depend
460            // on `u`. At `u = 1` the fused axis is `mimo_rank`, and this is the
461            // unchanged op graph of stock Mamba-3.
462            //
463            // The trapezoid is unchanged by all of this. A tap at
464            // [`TapPattern::tap_lag`] reads write position `p − lag`: the
465            // [`helpers::shift_stream`] of `forward` over this block, with the
466            // FIFO of the cache as the prefix. `β = ν·α` times
467            // [`helpers::interior_gap_decay`] is the same gap transport (§9). At
468            // lag 1 that gap has no interior, and the FIFO is one slot deep. At
469            // lag `u` the block is exactly as long as the lag, so every tap
470            // reads the FIFO, and none reads within the block.
471            let lag = self.tap_lag();
472
473            // ── The two transports of the block ───────────────────────────────
474            // `w` carries a write to the end of the block (`None` at `u = 1`,
475            // where that product is empty). `α_total` is the whole transition
476            // of the token, the one factor on the carried state.
477            let w_buh = helpers::tail_decay(da_buh.clone(), u);
478            let alpha_total_bh11 = da_buh
479                .clone()
480                .sum_dim(1)
481                .squeeze_dim::<2>(1)
482                .exp()
483                .unsqueeze_dims::<4>(&[2, 3]);
484
485            // One side of the recurrence, transported and fused:
486            //
487            //   write[b, h, p, r] = Σ_{j,m} massⱼ·wⱼ · v[j, m, h, p] · K[j, m, h, r]
488            //
489            // that is `einsum('bfhp,bfhr->bhpr', mass·w·v, K)` over the fused
490            // `f = u · mimo_rank` axis: one matmul, for any `u`.
491            let write = |mass_buh: Tensor<3>, x_buhp: Tensor<4>, k_bumhr: Tensor<5>| {
492                let mass_buh = match &w_buh {
493                    Some(w_buh) => mass_buh * w_buh.clone(),
494                    None => mass_buh,
495                };
496                let v_bumhp = helpers::build_v_with_mimo::<4, 5>(x_buhp, mimo_x_hmp.as_ref(), 2)
497                    * mass_buh.unsqueeze_dims::<5>(&[2, 4]);
498                san(&v_bumhp);
499                let write_bhpr = helpers::mimo_outer_sum(
500                    v_bumhp.reshape([batch, u * mimo_rank, nheads, per_head_dim]),
501                    k_bumhr.reshape([batch, u * mimo_rank, nheads, state_rank]),
502                    siso,
503                );
504                san(&write_bhpr);
505                write_bhpr
506            };
507
508            // A tapped side at its own lag: the `beta_side` of `forward`, over
509            // this block. The prefix is the newest `lag` slots of the FIFO: all
510            // of it for the tap at `tap_lag`, only the newest slot for the
511            // lag-1 tap of a two-tap pattern (by the FIFO convention, that slot
512            // has the empty decay product). Under [`TapPattern::None`] there is
513            // no tapped side: no shifted stream, no second outer product, one
514            // term less in the state update.
515            let tap_write = |nu_buh: Tensor<3>, lag: usize| {
516                let slots_buhp = cache
517                    .v_state_buhp
518                    .clone()
519                    .expect("a β tap keeps its (B, x) cache slots");
520                let slots_bumhr = cache
521                    .k_state_bumhr
522                    .clone()
523                    .expect("a β tap keeps its (B, x) cache slots");
524                let newest = slots_buhp.dims()[1] - lag;
525                let x_prev_buhp = helpers::shift_stream(
526                    x_buhp.clone(),
527                    slots_buhp.narrow(1, newest, lag),
528                    lag,
529                );
530                let b_prev_bumhr =
531                    helpers::shift_stream(b_bumhr.clone(), slots_bumhr.narrow(1, newest, lag), lag);
532                let beta_buh = nu_buh * alpha_buh.clone();
533                let beta_buh = match helpers::interior_gap_decay(da_buh.clone(), lag, None) {
534                    Some(gap_buh) => beta_buh * gap_buh,
535                    None => beta_buh,
536                };
537                write(beta_buh, x_prev_buhp, b_prev_bumhr)
538            };
539
540            // ── State update ──────────────────────────────────────────────────
541            let state_bhpr = alpha_total_bh11 * cache.ssm_bhpr.clone()
542                + write(gamma_buh, x_buhp.clone(), b_bumhr.clone());
543            let state_bhpr = [
544                nu_buh.map(|nu_buh| tap_write(nu_buh, lag)),
545                nu_interior_buh.map(|nu_buh| tap_write(nu_buh, 1)),
546            ]
547            .into_iter()
548            .flatten()
549            .fold(state_bhpr, |state, write| state + write);
550            san(&state_bhpr);
551
552            // ── Readout ───────────────────────────────────────────────────────
553            // yₘ[b, m, h, p] = sumᵣ C[b, m, h, r] * state[b, h, p, r], with `C`
554            // at the one read position of the block.
555            let c_bmhr = c_b1mhr.squeeze_dim::<4>(1);
556            let y_bmhp = Self::step_readout(state_bhpr.clone(), c_bmhr, siso);
557            san(&y_bmhp);
558            cache.ssm_bhpr = state_bhpr;
559
560            // ── The tail of the plant frame, on a read axis of length 1 ───────
561            // The last micro-step of the block is both the read position and
562            // the next carry of the positive systems.
563            let (out_b1d, cache) = self.plant_tail(
564                PlantTail {
565                    y_btmhp: y_bmhp.unsqueeze_dim::<5>(1),
566                    z_bti: z_b1i,
567                    x_bshp: x_buhp,
568                    log_precision_bsh: log_precision_buh,
569                    tropical_ab_bsh: tropical_ab_buh,
570                    end_b: None,
571                    packed: None,
572                    new_rotation,
573                    b_last_bumhr,
574                    x_last_buhp,
575                },
576                cache,
577            );
578            (out_b1d.squeeze_dim::<2>(1), cache)
579        }
580    }
581}
