1//! # Mamba-3 — Double-Pass SSD Forward
2//!
3//! [`Mamba3::forward_double_ssd`](crate::mamba3::mamba3::Mamba3::forward_double_ssd)
4//! and [`Mamba3::step_double_ssd`](crate::mamba3::mamba3::Mamba3::step_double_ssd):
5//! the decomposition of
6//! [`VikramLex/mamba3-minimal`](https://github.com/VikramLex/mamba3-minimal).
7//!
8//! ```text
9//!   hₚ = αₚ hₚ₋₁ + βₚ Bₚ₋ₗₐ₉ ⊗ xₚ₋ₗₐ₉ + γₚ Bₚ ⊗ xₚ   (original double-ssd trapezoidal)
10//!
11//!   forward:    h = SSD(γ-scaled V, B)   +   SSD(β-scaled V_shifted, B_shifted)
12//! ```
13//!
14//! The shift is [`TapPattern::tap_lag`](crate::mamba3::tap_pattern::TapPattern::tap_lag)
15//! write positions (`1` by default, `u` for
16//! [`TapPattern::Vertical`](crate::mamba3::tap_pattern::TapPattern::Vertical)), and
17//! `β = ν·α` carries the transport across that whole gap.
18//!
19//! A two-tap pattern adds a second `β` side at lag 1. The two shifts are
20//! different, so the passes cannot fuse, and this pathway runs **three** SSD
21//! calls. The single-SSD pathway collapses them into its key scale and stays at
22//! one call, so it suits such a pattern better.
23//!
24//! This is simple to derive and to verify (everything uses the standard SSD),
25//! but it increases the intra-chunk and chunk-state memory during training.
26//!
27//! See also: [`crate::mamba3::mamba3`] and [`crate::mamba3::single_ssd::single_ssd`].
28

37// ---------------------------------------------------------------------------
38// Mamba3::forward  (chunkwise double-SSD — training / prefill)
39// ---------------------------------------------------------------------------
40
41impl Mamba3 {
42    /// Process a full input sequence with the double-ssd pathway.
43    ///
44    /// For MIMO (mimo_rank>1), B/C have mimo_rank parallel rank channels. All
45    /// ranks share the state, and each rank writes to it independently.
46    ///
47    /// `pad_bt` marks a right-padded batch of tokens, as in [`Self::forward`].
48    ///
49    /// # Shapes
50    /// - `input_bsm` : `[batch, sequence, d_model]`
51    /// - `pad_bt`    : `[batch, sequence]`
52    /// - output      : `[batch, sequence, d_model]`
53    pub fn forward_double_ssd(
54        &self,
55        input_bsm: Tensor<3>,
56        cache: Option<Mamba3DoubleSsdCache>,
57        ssd_path: &Mamba3SsdPath,
58        pad_bt: Option<Tensor<2, Bool>>,
59    ) -> (Tensor<3>, Mamba3DoubleSsdCache) {
60        self.forward_double_ssd_packed(input_bsm, cache, None, ssd_path, pad_bt, None)
61    }
62
63    /// [`Self::forward_double_ssd`] over a packed batch: `reset_bt` marks the
64    /// first token of each segment, which starts from `origin`, as in
65    /// [`Self::forward_packed`].
66    #[allow(non_snake_case)]
67    pub fn forward_double_ssd_packed(
68        &self,
69        input_bsm: Tensor<3>,
70        cache: Option<Mamba3DoubleSsdCache>,
71        origin: Option<Mamba3DoubleSsdCache>,
72        ssd_path: &Mamba3SsdPath,
73        pad_bt: Option<Tensor<2, Bool>>,
74        reset_bt: Option<Tensor<2, Bool>>,
75    ) -> (Tensor<3>, Mamba3DoubleSsdCache) {
76        let [batch, tokens, _d_model] = input_bsm.dims();
77        let nheads = self.nheads();
78        let per_head_dim = self.per_head_dim();
79        let state_rank = self.state_rank;
80        let mimo_rank = self.mimo_rank;
81        let u = self.micro_steps;
82        let device = input_bsm.device();
83        let sequence = tokens * u;
84
85        // ── The entry of the plant frame ──────────────────────────────────────
86        let chunk_len = ssd_path.chunk_len_or_optimal(self);
87        let PlantEntry {
88            mut cache,
89            packed,
90            end_b,
91            z_bti,
92            x_bshp,
93            b_bsmhr,
94            c_btmhr,
95            da_bsh,
96            alpha_bsh,
97            nu_bsh,
98            nu_interior_bsh,
99            gamma_bsh,
100            log_precision_bsh,
101            tropical_ab_bsh,
102            new_rotation,
103            new_tap_fifo,
104        } = self.plant_entry(
105            input_bsm,
106            cache,
107            pad_bt,
108            reset_bt.map(|reset_bt| PlantResets { reset_bt, origin, chunk_len }),
109        );
110
111        // ── The β term's shifted, β-scaled inputs ─────────────────────────────
112        //
113        // "Shift-Before-Chunking": prepend the cached xₜ₋₁ / Bₜ₋₁ at the
114        // sequence level (before SSD chunking), so the β term at t=0 sees the
115        // prior token of a continued cache. For a fresh (zero) cache this is
116        // the same as zero-padding.
117        //
118        // The shift is [`TapPattern::tap_lag`] *write* positions: 1 for the
119        // default [`TapPattern::HorizontalCarryOver`], `u` for
120        // [`TapPattern::Vertical`]. That is exactly the FIFO depth of `step`.
121        //
122        // A lag-`L` tap must be transported across *its own* gap (§9): by
123        // `Πᵈ⁼⁰..ᴸ⁻¹ αₚ₋ᵈ`, not `αₚ`. `β = ν·α` holds the `d = 0` factor, and
124        // `interior_gap_decay` the rest. For the first `L` positions, the
125        // missing factors are those that the `v` slots of the cache already
126        // carry.
127        //
128        // A two-tap pattern adds a second, lag-1 side. The two shifts are
129        // different, so they cannot share a pass, and this pathway runs three
130        // SSD calls. The single-SSD pathway fuses them into its key scale and
131        // stays at one. The prefix of the lag-1 tap of a two-tap pattern is the
132        // newest slot of the FIFO, which *is* the lag-1 slot, undecayed. It has no effect
133        // under [`TapPattern::VerticalPlusHorizontalReset`], whose `νⁱⁿᵗ` is
134        // zero at every position that would read it.
135        //
136        // Under [`TapPattern::None`] there is no left endpoint: no shift, no
137        // β-scaled copy of `x`, and (below) no second SSD call. `forward` is
138        // then one standard SSD pass, with keys scaled by `γ = Δ`.
139        let lag = self.tap_lag();
140        // The prefix of a lag-`L` tap: [`TapFifo::newest`].
141        let head =|fifo: &Option<TapFifo>, lag: usize| -> TapFifo {
142            fifo.as_ref().expect("a β tap keeps its tap FIFO").newest(lag)
143        };
144        let beta_side = |nu_bsh: Tensor<3>, lag: usize| -> (Tensor<4>, Tensor<5>) {
145            let TapFifo { b_bumhr: b_head_bumhr, x_buhp: x_head_buhp } =
146                head(&cache.frame.tap_fifo, lag);
147            let x_prev_bshp = helpers::shift_stream(x_bshp.clone(), x_head_buhp, lag);
148            let b_prev_bsmhr = helpers::shift_stream(b_bsmhr.clone(), b_head_bumhr, lag);
149            // A packed row: each segment starts on the prefix of the origin,
150            // as a call from it.
151            let (x_prev_bshp, b_prev_bsmhr) = match &packed {
152                Some(p) => {
153                    let TapFifo { b_bumhr: b_head_bumhr, x_buhp: x_head_buhp } =
154                        head(&p.origin.frame.tap_fifo, lag);
155                    (
156                        p.segments.restart_heads(x_prev_bshp, 1, x_head_buhp),
157                        p.segments.restart_heads(b_prev_bsmhr, 1, b_head_bumhr),
158                    )
159                }
160                None => (x_prev_bshp, b_prev_bsmhr),
161            };
162            let beta_bsh = nu_bsh * alpha_bsh.clone();
163            let segments = packed.as_ref().map(|p| &p.segments);
164            let beta_bsh = match helpers::interior_gap_decay(da_bsh.clone(), lag, segments) {
165                Some(gap_bsh) => beta_bsh * gap_bsh,
166                None => beta_bsh,
167            };
168            // β is a per-head scalar, broadcast over mimo_rank and per_head_dim.
169            let beta_bsh1 = beta_bsh.unsqueeze_dim::<4>(3);
170            (x_prev_bshp * beta_bsh1, b_prev_bsmhr) // βₚ · xₚ₋ₗₐ₉
171        };
172        let beta_sides: Vec<(Tensor<4>, Tensor<5>)> = nu_bsh
173            .map(|nu_bsh| beta_side(nu_bsh, lag))
174            .into_iter()
175            .chain(nu_interior_bsh.map(|nu_bsh| beta_side(nu_bsh, 1)))
176            .collect();
177
178        // ── Scale the current-token input by γ ────────────────────────────────
179        let gamma_bsh1 = gamma_bsh.unsqueeze_dim::<4>(3);
180        let x_gamma_bshp = x_bshp.clone() * gamma_bsh1; // γₜ · xₜ
181
182        // ── Pad sequence to multiple of chunk_len ─────────────────────────────
183        let sequence_padded = sequence.next_multiple_of(chunk_len);
184        let pad = sequence_padded - sequence;
185
186        // Each zero block is built at most once, and the streams that need it
187        // share it (a `Tensor` clone is a handle, not a copy).
188        let pads = (pad > 0).then(|| {
189            (
190                Tensor::<4>::zeros([batch, pad, nheads, per_head_dim], &device),
191                Tensor::<3>::zeros([batch, pad, nheads], &device),
192                Tensor::<5>::zeros([batch, pad, mimo_rank, nheads, state_rank], &device),
193            )
194        });
195        let pad_hp = |t: Tensor<4>| match &pads {
196            Some((p, _, _)) => Tensor::cat(vec![t, p.clone()], 1),
197            None => t,
198        };
199        let pad_h = |t: Tensor<3>| match &pads {
200            Some((_, p, _)) => Tensor::cat(vec![t, p.clone()], 1),
201            None => t,
202        };
203        let pad_mhr = |t: Tensor<5>| match &pads {
204            Some((_, _, p)) => Tensor::cat(vec![t, p.clone()], 1),
205            None => t,
206        };
207
208        // `C` is on the read axis of the chunk, so it pads by whole tokens.
209        // `chunk_len` is a multiple of `u`, so `pad` is too.
210        let tokens_padded = sequence_padded / u;
211        let pad_tmhr = |t: Tensor<5>| match pad {
212            0 => t,
213            _ => Tensor::cat(
214                vec![
215                    t,
216                    Tensor::zeros(
217                        [batch, tokens_padded - tokens, mimo_rank, nheads, state_rank],
218                        &device,
219                    ),
220                ],
221                1,
222            ),
223        };
224
225        let x_gamma_bShp = pad_hp(x_gamma_bshp);
226        let da_bSh = pad_h(da_bsh);
227        let b_bSmhr = pad_mhr(b_bsmhr);
228        let c_bTmhr = pad_tmhr(c_btmhr);
229        let beta_sides: Vec<_> = beta_sides
230            .into_iter()
231            .map(|(x_beta, b_prev)| (pad_hp(x_beta), pad_mhr(b_prev)))
232            .collect();
233
234        // ── Reshape into chunks ───────────────────────────────────────────────
235        let nchunks = sequence_padded / chunk_len;
236        let chunk_tokens = standard_ssd::chunk_tokens(chunk_len, u);
237        let x_gamma_bnlhp = x_gamma_bShp.reshape([batch, nchunks, chunk_len, nheads, per_head_dim]);
238        let da_bnlh = da_bSh.reshape([batch, nchunks, chunk_len, nheads]);
239        let b_bnlmhr = b_bSmhr.reshape([batch, nchunks, chunk_len, mimo_rank, nheads, state_rank]);
240        let c_bntmhr =
241            c_bTmhr.reshape([batch, nchunks, chunk_tokens, mimo_rank, nheads, state_rank]);
242
243        // ── The MIMO-SSD call(s) ──────────────────────────────────────────────
244        // Build the V tensors: insert the mimo_rank axis at position 3 of
245        // `_bnlhp`. With a β tap, these are the two standard SSD passes that
246        // give the pathway its name (γ-SSM + β-SSM, summed). Under
247        // [`TapPattern::None`] the second pass does not exist, and a
248        // "double"-SSD forward is one pass.
249        let mimo_x_hmp = self.mimo_x_hmp.as_ref().map(|p| p.val());
250        let v_gamma_bnlmhp =
251            helpers::build_v_with_mimo::<5, 6>(x_gamma_bnlhp.clone(), mimo_x_hmp.as_ref(), 3);
252        // Every pass restarts its carry at the same chunks: the passes are
253        // the terms of one state. The γ pass holds the state of the origin,
254        // and a β pass restarts from zero, as it starts a call.
255        let reset_bn = packed.as_ref().map(|p| p.segments.reset_bn.clone());
256
257        let input_gamma = Mamba3DoubleSsdInput {
258            v_bnlmhp: v_gamma_bnlmhp,
259            da_bnlh: da_bnlh.clone(),
260            b_bnlmhr: b_bnlmhr.clone(),
261            c_bntmhr: c_bntmhr.clone(),
262            initial_state_bhpr: cache.ssm_bhpr,
263            init_state_hpr: self.init_state_hpr.as_ref().map(|s| s.val()),
264            read_stride: u,
265            restarts: packed.as_ref().map(|p| crate::packing::Restarts {
266                reset_bn: p.segments.reset_bn.clone(),
267                origin_bhpr: p.origin.ssm_bhpr.clone(),
268            }),
269        };
270        let (y_bntmhp, final_state_bhpr) = input_gamma.run(ssd_path);
271
272        let (y_bntmhp, final_state_bhpr) = beta_sides.into_iter().fold(
273            (y_bntmhp, final_state_bhpr),
274            |(y_bntmhp, final_state_bhpr), (x_beta_bShp, b_prev_bSmhr)| {
275                let x_beta_bnlhp =
276                    x_beta_bShp.reshape([batch, nchunks, chunk_len, nheads, per_head_dim]);
277                let b_prev_bnlmhr =
278                    b_prev_bSmhr.reshape([batch, nchunks, chunk_len, mimo_rank, nheads, state_rank]);
279                let v_beta_bnlmhp =
280                    helpers::build_v_with_mimo::<5, 6>(x_beta_bnlhp, mimo_x_hmp.as_ref(), 3);
281                let zero_bhpr = Tensor::zeros([batch, nheads, per_head_dim, state_rank], &device);
282                let input_beta = Mamba3DoubleSsdInput {
283                    v_bnlmhp: v_beta_bnlmhp,
284                    da_bnlh: da_bnlh.clone(),
285                    b_bnlmhr: b_prev_bnlmhr,
286                    c_bntmhr: c_bntmhr.clone(),
287                    initial_state_bhpr: zero_bhpr.clone(),
288                    init_state_hpr: None,
289                    read_stride: u,
290                    restarts: reset_bn.clone().map(|reset_bn| crate::packing::Restarts {
291                        reset_bn,
292                        origin_bhpr: zero_bhpr,
293                    }),
294                };
295                let (y_beta_bntmhp, final_state_beta_bhpr) = input_beta.run(ssd_path);
296                (
297                    y_bntmhp + y_beta_bntmhp,
298                    final_state_bhpr + final_state_beta_bhpr,
299                )
300            },
301        );
302
303        san(&y_bntmhp);
304        san(&final_state_bhpr);
305
306        cache.ssm_bhpr = final_state_bhpr;
307
308        // ── Unpad ─────────────────────────────────────────────────────────────
309        // The SSD returns token resolution: the readout is after all `u`
310        // writes, and the kernel computed only that position. From here, `t`
311        // (`tokens`) is the time axis again: the read axis.
312        let y_bTmhp = y_bntmhp.reshape([batch, tokens_padded, mimo_rank, nheads, per_head_dim]);
313        let y_btmhp = if pad == 0 {
314            y_bTmhp
315        } else {
316            y_bTmhp.narrow(1, 0, tokens)
317        };
318
319        // ── The tail of the plant frame ───────────────────────────────────────
320        self.plant_tail(
321            PlantTail {
322                y_btmhp,
323                z_bti,
324                x_bshp,
325                log_precision_bsh,
326                tropical_ab_bsh,
327                end_b,
328                packed,
329                new_rotation,
330                new_tap_fifo,
331            },
332            cache,
333        )
334    }
335}

338// Mamba3::step  (recurrent SSM — token-by-token decoding)
339// ---------------------------------------------------------------------------
340
341mod step {
342    use super::*;
343
344    impl Mamba3 {
345        /// State→output contraction:
346        /// `out[b, m, h, p] = Σᵣ C[b, m, h, r] · state[b, h, p, r]`
347        /// (`einsum('bhpr,bmhr->bmhp', state, C)`).
348        ///
349        /// At `mimo_rank == 1` the output axis of the GEMM is 1, so the matmul
350        /// is a matrix–vector product. [`step_readout_siso`](Mamba3::step_readout_siso)
351        /// writes it as a broadcast multiply plus a `state_rank` reduction.
352        /// [`Mamba3Config::siso_specialization_decode`](crate::mamba3::mamba3::Mamba3Config::siso_specialization_decode)
353        /// selects the branch. Both branches compute the same values and
354        /// gradients.
355        pub(crate) fn step_readout(
356            state_bhpr: Tensor<4>,
357            c_bmhr: Tensor<4>,
358            siso_specialization: bool,
359        ) -> Tensor<4> {
360            let [_batch, mimo_rank, _nheads, _state_rank] = c_bmhr.dims();
361            if mimo_rank == 1 && siso_specialization {
362                Self::step_readout_siso(state_bhpr, c_bmhr)
363            } else {
364                Self::step_readout_mimo(state_bhpr, c_bmhr)
365            }
366        }
367
368        /// SISO (`mimo_rank == 1`) state→output contraction: broadcast `C` over
369        /// `per_head_dim` and reduce `state_rank`.
370        pub(crate) fn step_readout_siso(state_bhpr: Tensor<4>, c_bmhr: Tensor<4>) -> Tensor<4> {
371            let c_bh1r: Tensor<4> = c_bmhr.squeeze_dim::<3>(1).unsqueeze_dim(2);
372            let out_bhp1: Tensor<4> = (state_bhpr * c_bh1r).sum_dim(3);
373            out_bhp1.squeeze_dim::<3>(3).unsqueeze_dim(1) // out_b1hp
374        }
375
376        /// General MIMO state→output contraction: one matmul over `state_rank`.
377        pub(crate) fn step_readout_mimo(state_bhpr: Tensor<4>, c_bmhr: Tensor<4>) -> Tensor<4> {
378            let c_bhrm = c_bmhr.permute([0, 2, 3, 1]);
379            let out_bhpm = state_bhpr.matmul(c_bhrm);
380            out_bhpm.permute([0, 3, 1, 2])
381        }
382
383        /// Process a **single token** with the pure recurrent form (the
384        /// formulas are in [`Mamba3::step`]). Both cache variants decode here:
385        /// `step_single_ssd` converts its cache and calls this function.
386        ///
387        /// At `micro_steps = u > 1` the token is `u` of those micro-steps
388        /// ([`crate::mamba3::product`]). They are evaluated **together**, not one
389        /// at a time (see the body).
390        ///
391        /// # Shapes
392        /// - `input_bd` : `[batch, d_model]`
393        /// - output     : `[batch, d_model]`
394        pub fn step_double_ssd(
395            &self,
396            input_bd: Tensor<2>,
397            cache: Option<Mamba3DoubleSsdCache>,
398        ) -> (Tensor<2>, Mamba3DoubleSsdCache) {
399            let [batch, _d_model] = input_bd.dims();
400            let nheads = self.nheads();
401            let per_head_dim = self.per_head_dim();
402            let state_rank = self.state_rank;
403            let mimo_rank = self.mimo_rank;
404            let u = self.micro_steps;
405
406            // ── The entry of the plant frame, at `tokens = 1` ─────────────────
407            // `step` adds a read axis of length 1 to its input, and removes it
408            // from the output of the tail. The write axis of the streams is
409            // then the `u` micro-steps of the token. A Kalman gain computes its
410            // decay in the entry, from the cached `ln Λ`, over those `u`
411            // micro-steps: the scan of `forward` at `len = u`.
412            let PlantEntry {
413                mut cache,
414                packed: _,
415                end_b: _,
416                z_bti: z_b1i,
417                x_bshp: x_buhp,
418                b_bsmhr: b_bumhr,
419                c_btmhr: c_b1mhr,
420                da_bsh: da_buh,
421                alpha_bsh: alpha_buh,
422                nu_bsh: nu_buh,
423                nu_interior_bsh: nu_interior_buh,
424                gamma_bsh: gamma_buh,
425                log_precision_bsh: log_precision_buh,
426                tropical_ab_bsh: tropical_ab_buh,
427                new_rotation,
428                new_tap_fifo,
429            } = self.plant_entry(input_bd.unsqueeze_dim::<3>(1), cache, None, None);
430            let mimo_x_hmp = self.mimo_x_hmp.as_ref().map(|p| p.val());
431            let siso = self.use_siso_decode_kernels();
432
433            // ── The `u` micro-steps of the token at once, then one readout ──────
434            // MambaProduct evaluates a token as `u` consecutive positions of the
435            // ordinary recurrence. So a `step` call is a **write-axis block** of
436            // length `u`, and one call is `forward` on a one-token sequence
437            // (the parity tests assert this). The block is not walked one
438            // position at a time, because that is not necessary. After the RoPE
439            // trick, the transition inside the block is the *scalar* `α`,
440            // so the block has a closed form:
441            //
442            //     h = (∏ⱼ αⱼ)·h₋₁ + Σⱼ wⱼ · writeⱼ ,    wⱼ = ∏_{r>j} αᵣ
443            //
444            // Its transport `wⱼ` (the decay of a write to the end of the block)
445            // is the same quantity that [`crate::mamba3::helpers::tail_decay`]
446            // computes for the tap FIFO, over the whole block instead of its
447            // last `lag`.
448            //
449            // Every `writeⱼ` is an outer product into one shared state. So with
450            // the transport, and with `(u, mimo_rank)` fused into one
451            // contracted axis, each **side** of the recurrence is one
452            // [`helpers::mimo_outer_sum`] (the same collapse as the single-SSD
453            // boundary seed over its slots). Thus the op count does not depend
454            // on `u`. At `u = 1` the fused axis is `mimo_rank`, and this is the
455            // unchanged op graph of stock Mamba-3.
456            //
457            // The trapezoid is unchanged by all of this. A tap at
458            // [`TapPattern::tap_lag`] reads write position `p − lag`: the
459            // [`helpers::shift_stream`] of `forward` over this block, with the
460            // FIFO of the cache as the prefix. `β = ν·α` times
461            // [`helpers::interior_gap_decay`] is the same gap transport (§9). At
462            // lag 1 that gap has no interior, and the FIFO is one slot deep. At
463            // lag `u` the block is exactly as long as the lag, so every tap
464            // reads the FIFO, and none reads within the block.
465            let lag = self.tap_lag();
466
467            // ── The two transports of the block ───────────────────────────────
468            // `w` carries a write to the end of the block (`None` at `u = 1`,
469            // where that product is empty). `α_total` is the whole transition
470            // of the token, the one factor on the carried state.
471            let w_buh = helpers::tail_decay(da_buh.clone(), u);
472            let alpha_total_bh11 = da_buh
473                .clone()
474                .sum_dim(1)
475                .squeeze_dim::<2>(1)
476                .exp()
477                .unsqueeze_dims::<4>(&[2, 3]);
478
479            // One side of the recurrence, transported and fused:
480            //
481            //   write[b, h, p, r] = Σ_{j,m} massⱼ·wⱼ · v[j, m, h, p] · K[j, m, h, r]
482            //
483            // that is `einsum('bfhp,bfhr->bhpr', mass·w·v, K)` over the fused
484            // `f = u · mimo_rank` axis: one matmul, for any `u`.
485            let write = |mass_buh: Tensor<3>, x_buhp: Tensor<4>, k_bumhr: Tensor<5>| {
486                let mass_buh = match &w_buh {
487                    Some(w_buh) => mass_buh * w_buh.clone(),
488                    None => mass_buh,
489                };
490                let v_bumhp = helpers::build_v_with_mimo::<4, 5>(x_buhp, mimo_x_hmp.as_ref(), 2)
491                    * mass_buh.unsqueeze_dims::<5>(&[2, 4]);
492                san(&v_bumhp);
493                let write_bhpr = helpers::mimo_outer_sum(
494                    v_bumhp.reshape([batch, u * mimo_rank, nheads, per_head_dim]),
495                    k_bumhr.reshape([batch, u * mimo_rank, nheads, state_rank]),
496                    siso,
497                );
498                san(&write_bhpr);
499                write_bhpr
500            };
501
502            // A tapped side at its own lag: the `beta_side` of `forward`, over
503            // this block. The prefix is the newest `lag` slots of the FIFO: all
504            // of it for the tap at `tap_lag`, only the newest slot for the
505            // lag-1 tap of a two-tap pattern (by the FIFO convention, that slot
506            // has the empty decay product). Under [`TapPattern::None`] there is
507            // no tapped side: no shifted stream, no second outer product, one
508            // term less in the state update.
509            let tap_write = |nu_buh: Tensor<3>, lag: usize| {
510                let TapFifo { b_bumhr: b_head_bumhr, x_buhp: x_head_buhp } = cache
511                    .frame
512                    .tap_fifo
513                    .as_ref()
514                    .expect("a β tap keeps its tap FIFO")
515                    .newest(lag);
516                let x_prev_buhp = helpers::shift_stream(x_buhp.clone(), x_head_buhp, lag);
517                let b_prev_bumhr = helpers::shift_stream(b_bumhr.clone(), b_head_bumhr, lag);
518                let beta_buh = nu_buh * alpha_buh.clone();
519                let beta_buh = match helpers::interior_gap_decay(da_buh.clone(), lag, None) {
520                    Some(gap_buh) => beta_buh * gap_buh,
521                    None => beta_buh,
522                };
523                write(beta_buh, x_prev_buhp, b_prev_bumhr)
524            };
525
526            // ── State update ──────────────────────────────────────────────────
527            let state_bhpr = alpha_total_bh11 * cache.ssm_bhpr.clone()
528                + write(gamma_buh, x_buhp.clone(), b_bumhr.clone());
529            let state_bhpr = [
530                nu_buh.map(|nu_buh| tap_write(nu_buh, lag)),
531                nu_interior_buh.map(|nu_buh| tap_write(nu_buh, 1)),
532            ]
533            .into_iter()
534            .flatten()
535            .fold(state_bhpr, |state, write| state + write);
536            san(&state_bhpr);
537
538            // ── Readout ───────────────────────────────────────────────────────
539            // yₘ[b, m, h, p] = sumᵣ C[b, m, h, r] * state[b, h, p, r], with `C`
540            // at the one read position of the block.
541            let c_bmhr = c_b1mhr.squeeze_dim::<4>(1);
542            let y_bmhp = Self::step_readout(state_bhpr.clone(), c_bmhr, siso);
543            san(&y_bmhp);
544            cache.ssm_bhpr = state_bhpr;
545
546            // ── The tail of the plant frame, on a read axis of length 1 ───────
547            // The last micro-step of the block is both the read position and
548            // the next carry of the positive systems.
549            let (out_b1d, cache) = self.plant_tail(
550                PlantTail {
551                    y_btmhp: y_bmhp.unsqueeze_dim::<5>(1),
552                    z_bti: z_b1i,
553                    x_bshp: x_buhp,
554                    log_precision_bsh: log_precision_buh,
555                    tropical_ab_bsh: tropical_ab_buh,
556                    end_b: None,
557                    packed: None,
558                    new_rotation,
559                    new_tap_fifo,
560                },
561                cache,
562            );
563            (out_b1d.squeeze_dim::<2>(1), cache)
564        }
565    }
566}
