1//! # Serial SSD with a custom, memory-efficient backward (Mamba-3 double-SSD)
2//!
3//! The `SerialRecalculated` path of the double-SSD pathway. The forward goes
4//! through the [`Mamba3DoubleSsdBackendExt`] trait. So an `Autodiff` backend can
5//! use a custom backward that recomputes per-chunk intermediates instead of
6//! storing them (see [`super::backward`]). Plain backends use the default body
7//! of the trait.
8//!
9//! Both the default body and the custom backward call the
10//! [standard SSD scan](crate::standard_ssd), once for each term of the
11//! trapezoid.
12
13use crate::mamba3::double_ssd::prelude::*;
14use crate::packing::OpRestarts;
15use crate::standard_ssd;
16use burn_stack::utils::fprim::F;
17use burn::backend::tensor::FloatTensor;
18use burn::backend::{Backend, Dispatch, backend_extension};
19use burn::tensor::Tensor;
20
21impl Mamba3DoubleSsdInput {
22    /// MIMO-first Serial SSD with recalculated backward.
23    ///
24    /// Sends the full K1-K5 computation to
25    /// [`Mamba3DoubleSsdBackendExt::double_ssd_serial_recalculated`]. An
26    /// autodiff backend adds the memory-efficient custom backward. A plain
27    /// backend runs the standard K1-K5 serial computation.
28    ///
29    /// # Returns
30    /// - `y_bntmhp`:         `[batch, nchunks, chunk_tokens, mimo_rank, nheads, per_head_dim]`
31    /// - `final_state_bhpr`: `[batch, nheads, per_head_dim, state_rank]`
32    pub fn double_ssd_serial_recalculated(self) -> (Tensor<6>, Tensor<4>) {
33        let input = self;
34        assert!(
35            input.init_state_hpr.is_none(),
36            "init_state_hpr not yet implemented for ssd_serial_recalculated"
37        );
38
39        let restarts = crate::packing::Restarts::op_inputs(input.restarts, &input.v_bnlmhp);
40        let (y_bntmhp, final_state_bhpr) =
41            <Dispatch as Mamba3DoubleSsdBackendExt>::double_ssd_serial_recalculated(
42                input.v_bnlmhp.into_dispatch(),
43                input.da_bnlh.into_dispatch(),
44                input.b_bnlmhr.into_dispatch(),
45                input.c_bntmhr.into_dispatch(),
46                input.initial_state_bhpr.into_dispatch(),
47                input.read_stride,
48                restarts,
49            );
50        let y_bntmhp = Tensor::from_dispatch(y_bntmhp);
51        let final_state_bhpr = Tensor::from_dispatch(final_state_bhpr);
52        (y_bntmhp, final_state_bhpr)
53    }
54}

56/// Extends the backend for the memory-efficient serial recalculated SSD.
57///
58/// The default implementation runs K1-K5 with primitive tensor operations.
59/// The Autodiff wrapper overrides it: its backward recomputes the forward
60/// intermediates instead of saving them.
61#[backend_extension(
62    // Every `#[backend_extension]` site of this crate has the same list.
63    // Change them together. `build.rs` sets `cube_backend`.
64    Cube: cfg(cube_backend),
65    Flex:  cfg(feature = "backend-flex"),
66    NdArray:  cfg(feature = "backend-ndarray"),
67    LibTorch:  cfg(any(feature = "backend-tch-cpu", feature = "backend-tch-gpu")),
68    Autodiff:  cfg(feature = "autodiff"),
69)]
70pub trait Mamba3DoubleSsdBackendExt: Backend {
71    /// Memory-efficient MIMO serial SSD.
72    ///
73    /// # Arguments
74    /// - `v_bnlmhp`:           `[batch, nchunks, chunk_len, mimo_rank, nheads, per_head_dim]`
75    /// - `da_bnlh`:            `[batch, nchunks, chunk_len, nheads]` — pre-combined Δ·A
76    /// - `b_bnlmhr`:           `[batch, nchunks, chunk_len, mimo_rank, nheads, state_rank]`
77    /// - `c_bntmhr`:           `[batch, nchunks, chunk_tokens, mimo_rank, nheads, state_rank]`
78    /// - `initial_state_bhpr`: `[batch, nheads, per_head_dim, state_rank]`
79    /// - `read_stride`:        `micro_steps` — write positions per read position, so
80    ///   `chunk_tokens = chunk_len / read_stride`
81    /// - `restarts`:           the chunks that start a new segment of a packed
82    ///   row, and the state that they start from ([`OpRestarts`])
83    ///
84    /// # Returns
85    /// - `y_bntmhp`:         `[batch, nchunks, chunk_tokens, mimo_rank, nheads, per_head_dim]`
86    /// - `final_state_bhpr`: `[batch, nheads, per_head_dim, state_rank]`
87    fn double_ssd_serial_recalculated(
88        v_bnlmhp: FloatTensor<Self>,
89        da_bnlh: FloatTensor<Self>,
90        b_bnlmhr: FloatTensor<Self>,
91        c_bntmhr: FloatTensor<Self>,
92        initial_state_bhpr: FloatTensor<Self>,
93        read_stride: usize,
94        #[extension_type] restarts: OpRestarts<Self>,
95    ) -> (FloatTensor<Self>, FloatTensor<Self>) {
96        let (y_bntmhp, final_state_bhpr) = standard_ssd::forward::<Self>(
97            F::new(v_bnlmhp),
98            F::new(da_bnlh),
99            F::new(b_bnlmhr),
100            F::new(c_bntmhr),
101            F::new(initial_state_bhpr),
102            read_stride,
103            restarts.into_option(),
104        );
105        (y_bntmhp.inner(), final_state_bhpr.inner())
106    }
107}
