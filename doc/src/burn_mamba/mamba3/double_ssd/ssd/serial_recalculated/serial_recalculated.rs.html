1//! # Serial SSD with a custom, memory-efficient backward (Mamba-3 double-SSD)
2//!
3//! The `SerialRecalculated` path for the double-SSD pathway.  The forward is the
4//! same serial scan as [`super::super::serial`], routed through the
5//! [`Mamba3DoubleSsdBackendExt`] trait so that `Autodiff` backends substitute a
6//! custom backward that recomputes per-chunk intermediates instead of storing
7//! them (see [`super::backward`] / [`super::combined_backward`]).  Plain
8//! backends use the trait's default body, which replays the serial kernels.
9//!
10//! The default body runs under a generic backend `B`, where the high-level
11//! `Tensor` (pinned to `Dispatch`) is unavailable, so the K1–K5 math goes
12//! through the rank-tagged [`F`] primitive wrapper.  K1/K2/K4 are reused by the
13//! recompute backward in [`super::combined_backward`]; K5 is forward-only.
14
15#![allow(non_snake_case)]
16

25impl Mamba3DoubleSsdInput {
26    /// MIMO-first Serial SSD with recalculated backward.
27    ///
28    /// Delegates the full K1-K5 computation to [`Mamba3DoubleSsdBackendExt::double_ssd_serial_recalculated`]
29    /// which can provide a memory-efficient custom backward for supported backends.
30    ///
31    /// Falls back to the standard K1-K5 serial computation on unsupported backends.
32    ///
33    /// # Returns
34    /// - `y_bntmhp`:         `[batch, nchunks, chunk_tokens, mimo_rank, nheads, per_head_dim]`
35    /// - `final_state_bhpr`: `[batch, nheads, per_head_dim, state_rank]`
36    pub fn double_ssd_serial_recalculated(self) -> (Tensor<6>, Tensor<4>) {
37        let input = self;
38        assert!(
39            input.init_state_hpr.is_none(),
40            "init_state_hpr not yet implemented for ssd_serial_recalculated"
41        );
42
43        let (y_bntmhp, final_state_bhpr) =
44            <Dispatch as Mamba3DoubleSsdBackendExt>::double_ssd_serial_recalculated(
45                input.v_bnlmhp.into_dispatch(),
46                input.da_bnlh.into_dispatch(),
47                input.b_bnlmhr.into_dispatch(),
48                input.c_bntmhr.into_dispatch(),
49                input.initial_state_bhpr.into_dispatch(),
50                input.read_stride,
51            );
52        let y_bntmhp = Tensor::from_dispatch(y_bntmhp);
53        let final_state_bhpr = Tensor::from_dispatch(final_state_bhpr);
54        (y_bntmhp, final_state_bhpr)
55    }
56}

58/// Extends the backend for the memory-efficient serial recalculated SSD.
59///
60/// The default implementation runs K1-K5 using primitive tensor operations.
61/// Backends that support a custom memory-efficient backward (specifically the
62/// Autodiff wrapper) override this to recompute forward intermediates during
63/// the backward pass instead of saving them.
64#[backend_extension(
65    // Every cubecl runtime — CUDA, ROCm, Metal, Vulkan, WebGPU, wgpu, CPU — is
66    // this one backend; which of them a tensor runs on is what its device says.
67    // The cfg mirrors burn's own `cube_backend`.
68    Cube: cfg(any(
69        feature = "backend-cpu",
70        feature = "backend-cuda",
71        feature = "backend-rocm",
72        feature = "backend-metal",
73        feature = "backend-vulkan",
74        feature = "backend-wgpu",
75        feature = "backend-webgpu"
76    )),
77    Flex:  cfg(feature = "backend-flex"),
78    NdArray:  cfg(feature = "backend-ndarray"),
79    LibTorch:  cfg(any(feature = "backend-tch-cpu", feature = "backend-tch-gpu")),
80    Autodiff:  cfg(feature = "autodiff"),
81)]
82pub trait Mamba3DoubleSsdBackendExt: Backend {
83    /// Memory-efficient MIMO serial SSD.
84    ///
85    /// # Arguments
86    /// - `v_bnlmhp`:           `[batch, nchunks, chunk_len, mimo_rank, nheads, per_head_dim]`
87    /// - `da_bnlh`:            `[batch, nchunks, chunk_len, nheads]` — pre-combined Δ·A
88    /// - `b_bnlmhr`:           `[batch, nchunks, chunk_len, mimo_rank, nheads, state_rank]`
89    /// - `c_bntmhr`:           `[batch, nchunks, chunk_tokens, mimo_rank, nheads, state_rank]`
90    /// - `initial_state_bhpr`: `[batch, nheads, per_head_dim, state_rank]`
91    /// - `read_stride`:        `micro_steps` — folded positions per read row, so
92    ///   `chunk_tokens = chunk_len / read_stride`
93    ///
94    /// # Returns
95    /// - `y_bntmhp`:         `[batch, nchunks, chunk_tokens, mimo_rank, nheads, per_head_dim]`
96    /// - `final_state_bhpr`: `[batch, nheads, per_head_dim, state_rank]`
97    fn double_ssd_serial_recalculated(
98        v_bnlmhp: FloatTensor<Self>,
99        da_bnlh: FloatTensor<Self>,
100        b_bnlmhr: FloatTensor<Self>,
101        c_bntmhr: FloatTensor<Self>,
102        initial_state_bhpr: FloatTensor<Self>,
103        read_stride: usize,
104    ) -> (FloatTensor<Self>, FloatTensor<Self>) {
105        // Default impl: replicate Mamba3::double_ssd_serial (K1-K5) on primitives.
106        let v_bnlmhp = F::<Self, 6>::new(v_bnlmhp);
107        let da_bnlh = F::<Self, 4>::new(da_bnlh);
108        let b_bnlmhr = F::<Self, 6>::new(b_bnlmhr);
109        let c_bntmhr = F::<Self, 6>::new(c_bntmhr);
110        let initial_state_bhpr = F::<Self, 4>::new(initial_state_bhpr);
111
112        let nchunks = v_bnlmhp.dims()[1];
113        assert!(nchunks > 0, "sequence length must be at least 1");
114
115        let (da_cumsum_bhnl, da_chunk_end_bhn) = k1_ssd_chunk_cumsum::<Self>(da_bnlh);
116        san(&da_cumsum_bhnl);
117
118        let cb_bnhTMLM = k2_ssd_bmm::<Self>(c_bntmhr.clone(), b_bnlmhr.clone());
119        san(&cb_bnhTMLM);
120
121        let intra_chunk_state_bnhpr =
122            k3_ssd_chunk_state::<Self>(v_bnlmhp.clone(), b_bnlmhr, da_cumsum_bhnl.clone());
123        san(&intra_chunk_state_bnhpr);
124
125        let (chunk_input_state_bnhpr, final_state_bhpr) = k4_ssd_state_passing::<Self>(
126            intra_chunk_state_bnhpr,
127            da_chunk_end_bhn,
128            initial_state_bhpr,
129        );
130        san(&chunk_input_state_bnhpr);
131        san(&final_state_bhpr);
132
133        let y_bntmhp = k5_ssd_chunk_scan::<Self>(
134            da_cumsum_bhnl,
135            v_bnlmhp,
136            c_bntmhr,
137            cb_bnhTMLM,
138            chunk_input_state_bnhpr,
139            read_stride,
140        );
141        san(&y_bntmhp);
142
143        (y_bntmhp.inner(), final_state_bhpr.inner())
144    }
145}

155// ─── Primitive forward kernels (K1–K5) ───────────────────────────────────────
156// Primitive ports of the high-level [`super::super::serial`] kernels, expressed
157// on `B`'s primitives via [`F`] so the trait default body can run under a
158// generic backend. K1/K2/K4 are reused by the recompute backward in
159// [`super::combined_backward`]; K5 is forward-only (the backward computes K5's
160// gradient analytically rather than recomputing it).
161
162/// Primitive port of [`super::super::serial::k1_ssd_chunk_cumsum`].
163///
164/// Returns the intra-chunk cumsum `da_cumsum_bhnl` and the per-chunk last value
165/// `da_chunk_end_bhn`.
166pub(crate) fn k1_ssd_chunk_cumsum<B: Backend>(da_bnlh: F<B, 4>) -> (F<B, 4>, F<B, 3>) {
167    let da_bhnl = da_bnlh.permute([0, 3, 1, 2]);
168    let da_cumsum_bhnl = da_bhnl.cumsum(3);
169    let da_chunk_end_bhn = da_cumsum_bhnl
170        .clone()
171        .slice(s![.., .., .., -1])
172        .squeeze_dim::<3>(3);
173    (da_cumsum_bhnl, da_chunk_end_bhn)
174}

176/// Primitive port of [`super::super::serial::k2_ssd_bmm`] (fused `L·M`).
177///
178/// Returns the intra-chunk `C·Bᵀ` block matrix `cb_bnhLMLM`.
179pub(crate) fn k2_ssd_bmm<B: Backend>(c_bntmhr: F<B, 6>, b_bnlmhr: F<B, 6>) -> F<B, 5> {
180    // `C` carries the chunk's read axis, `B` its write axis; they coincide
181    // unless `micro_steps > 1`. See the `Tensor` twin in `super::super::serial`.
182    let [batch, nchunks, chunk_tokens, mimo_rank, nheads, state_rank] = c_bntmhr.dims();
183    let [.., chunk_len, _, _, _] = b_bnlmhr.dims();
184    let c_bnTMhr = c_bntmhr.reshape([batch, nchunks, chunk_tokens * mimo_rank, nheads, state_rank]);
185    let b_bnLMhr = b_bnlmhr.reshape([batch, nchunks, chunk_len * mimo_rank, nheads, state_rank]);
186    let c_bnhTMr = c_bnTMhr.swap_dims(2, 3);
187    let b_bnhrLM = b_bnLMhr.permute([0, 1, 3, 4, 2]);
188    c_bnhTMr.matmul(b_bnhrLM)
189}

191/// Primitive port of [`super::super::serial::k3_ssd_chunk_state`] (lean:
192/// returns only the chunk-end state).
193///
194/// Returns `intra_chunk_state_bnhpr` — each chunk's contribution to its end
195/// state assuming a zero state at the chunk's start. `v_bnlmhp` is pre-scaled.
196pub(crate) fn k3_ssd_chunk_state<B: Backend>(
197    v_bnlmhp: F<B, 6>,
198    b_bnlmhr: F<B, 6>,
199    da_cumsum_bhnl: F<B, 4>,
200) -> F<B, 5> {
201    let [batch, nchunks, chunk_len, mimo_rank, nheads, per_head_dim] = v_bnlmhp.dims();
202    let [.., state_rank] = b_bnlmhr.dims();
203
204    let v_bnLMhp = v_bnlmhp.reshape([batch, nchunks, chunk_len * mimo_rank, nheads, per_head_dim]);
205    let b_bnLMhr = b_bnlmhr.reshape([batch, nchunks, chunk_len * mimo_rank, nheads, state_rank]);
206
207    let da_cumsum_last_bhn1 = da_cumsum_bhnl.clone().slice(s![.., .., .., -1]);
208    let da_cumsum_bhnLM = da_cumsum_bhnl
209        .unsqueeze_dim::<5>(4) // da_cumsum_bhnl1
210        .expand([batch, nheads, nchunks, chunk_len, mimo_rank]) // da_cumsum_bhnlm
211        .reshape([batch, nheads, nchunks, chunk_len * mimo_rank]); // da_cumsum_bhnLM
212    let decay_bhnLM = (da_cumsum_last_bhn1 - da_cumsum_bhnLM).exp();
213
214    let decay_bnLMh1 = decay_bhnLM.permute([0, 2, 3, 1]).unsqueeze_dim::<5>(4); // decay_bnLMh1
215    let decayed_v_bnLMhp = decay_bnLMh1 * v_bnLMhp;
216
217    let decayed_v_bnhpLM = decayed_v_bnLMhp.permute([0, 1, 3, 4, 2]);
218    let b_bnhLMr = b_bnLMhr.swap_dims(2, 3);
219    let intra_chunk_state_bnhpr = decayed_v_bnhpLM.matmul(b_bnhLMr);
220    assert_eq!(
221        [batch, nchunks, nheads, per_head_dim, state_rank],
222        intra_chunk_state_bnhpr.dims()
223    );
224    intra_chunk_state_bnhpr
225}

227/// Primitive port of [`super::super::serial::k4_ssd_state_passing`].
228///
229/// Returns the per-chunk input-state stream `chunk_input_state_bnhpr` and the
230/// `final_state_bhpr`.
231pub(crate) fn k4_ssd_state_passing<B: Backend>(
232    intra_chunk_state_bnhpr: F<B, 5>,
233    da_chunk_end_bhn: F<B, 3>,
234    initial_state_bhpr: F<B, 4>,
235) -> (F<B, 5>, F<B, 4>) {
236    let [batch, nchunks, nheads, per_head_dim, state_rank] = intra_chunk_state_bnhpr.dims();
237
238    let mut running_state_bhpr = initial_state_bhpr;
239    let mut chunk_input_state_vec_bhpr = Vec::with_capacity(nchunks + 1);
240    chunk_input_state_vec_bhpr.push(running_state_bhpr.clone());
241
242    for i_chunk in 0..nchunks {
243        let intra_state_bhpr = intra_chunk_state_bnhpr
244            .clone()
245            .slice(s![.., i_chunk, .., .., ..])
246            .squeeze_dim::<4>(1);
247        let decay_bhpr = da_chunk_end_bhn
248            .clone()
249            .slice(s![.., .., i_chunk])
250            .unsqueeze_dim::<4>(3)
251            .exp()
252            .expand([batch, nheads, per_head_dim, state_rank]);
253        running_state_bhpr = decay_bhpr * running_state_bhpr + intra_state_bhpr;
254        chunk_input_state_vec_bhpr.push(running_state_bhpr.clone());
255    }
256
257    let final_state_bhpr = chunk_input_state_vec_bhpr.pop().unwrap();
258    let chunk_input_state_bnhpr = F::stack(chunk_input_state_vec_bhpr, 1);
259    (chunk_input_state_bnhpr, final_state_bhpr)
260}

262/// Backward of [`k4_ssd_state_passing`] — the reverse of its scalar-decay scan,
263/// and the only part of the recompute backward that is still a walk (which is
264/// the forward's own choice, and measured there).
265///
266/// Forward, writing `sᵢ` for `chunk_input_stateᵢ`: `sᵢ₊₁ = decayᵢ·sᵢ + intraᵢ`.
267/// Reverse, seeded by `d_final`: `d_intraᵢ = d_sᵢ₊₁` and `d_sᵢ = decayᵢ·d_sᵢ₊₁ +
268/// d_chunk_input_stateᵢ`, which the walk ends on as `d_initial`. `d_decayᵢ =
269/// d_sᵢ₊₁·sᵢ` needs no walk — `decay` is one scalar per `(b, h, n)`, so it comes
270/// out of the `(p, r)` sum and the whole stream is one batched product.
271///
272/// # Returns
273/// - `d_intra_chunk_state_bnhpr` — gradient of K3's per-chunk contribution
274/// - `d_da_chunk_end_bhn` — gradient of the per-chunk log-decay
275/// - `d_initial_state_bhpr`
276pub(crate) fn k4_ssd_state_passing_backward<B: Backend>(
277    d_chunk_input_state_bnhpr: F<B, 5>,
278    chunk_input_state_bnhpr: F<B, 5>,
279    da_chunk_end_bhn: F<B, 3>,
280    d_final_state_bhpr: F<B, 4>,
281) -> (F<B, 5>, F<B, 3>, F<B, 4>) {
282    let [batch, nchunks, nheads, per_head_dim, state_rank] = chunk_input_state_bnhpr.dims();
283
284    let mut d_running_state_bhpr = d_final_state_bhpr;
285    let mut d_intra_vec_bhpr: Vec<F<B, 4>> = Vec::with_capacity(nchunks);
286    for i_chunk in (0..nchunks).rev() {
287        d_intra_vec_bhpr.push(d_running_state_bhpr.clone());
288
289        let decay_bhpr: F<B, 4> = da_chunk_end_bhn
290            .clone()
291            .slice(s![.., .., i_chunk]) // da_chunk_end_bh1
292            .exp()
293            .unsqueeze_dim::<4>(3) // decay_bh11
294            .expand([batch, nheads, per_head_dim, state_rank]);
295        let d_chunk_input_state_bhpr: F<B, 4> = d_chunk_input_state_bnhpr
296            .clone()
297            .slice(s![.., i_chunk, .., .., ..])
298            .squeeze_dim::<4>(1);
299
300        d_running_state_bhpr = decay_bhpr * d_running_state_bhpr + d_chunk_input_state_bhpr;
301        san(&d_running_state_bhpr);
302    }
303    d_intra_vec_bhpr.reverse();
304    let d_intra_chunk_state_bnhpr: F<B, 5> = F::stack(d_intra_vec_bhpr, 1);
305
306    let d_decay_bhn: F<B, 3> = (d_intra_chunk_state_bnhpr.clone() * chunk_input_state_bnhpr)
307        .reshape([batch, nchunks, nheads, per_head_dim * state_rank])
308        .sum_dim(3) // d_decay_bnh1
309        .squeeze_dim::<3>(3) // d_decay_bnh
310        .permute([0, 2, 1]); // d_decay_bhn
311    // decay = exp(da_chunk_end), so the chain rule multiplies it back in.
312    let d_da_chunk_end_bhn = d_decay_bhn * da_chunk_end_bhn.exp();
313    san(&d_da_chunk_end_bhn);
314
315    (
316        d_intra_chunk_state_bnhpr,
317        d_da_chunk_end_bhn,
318        d_running_state_bhpr,
319    )
320}

322/// Concatenate a chunk-group pass's per-group outputs back along the chunk
323/// axis — and nothing at all when the pass ran in one group, which a short
324/// enough chunk axis leaves it in (see
325/// [`Mamba3SsdPath::backward_chunk_group`](crate::mamba3::ssd_path::Mamba3SsdPath::backward_chunk_group)).
326pub(crate) fn cat_chunk_groups<B: Backend, const D: usize>(
327    mut groups: Vec<F<B, D>>,
328    dim: usize,
329) -> F<B, D> {
330    match groups.len() {
331        1 => groups.pop().expect("one group"),
332        _ => F::cat(groups, dim),
333    }
334}

336/// Primitive port of [`super::super::serial::k5_ssd_chunk_scan`].
337///
338/// Combines the intra-chunk (ORANGE, MIMO causal) and inter-chunk (BLUE,
339/// state-carried) contributions into the output `y_bnlmhp`. No `D` skip is
340/// applied — the caller handles it. Forward-only.
341fn k5_ssd_chunk_scan<B: Backend>(
342    da_cumsum_bhnl: F<B, 4>,
343    v_bnlmhp: F<B, 6>,
344    c_bntmhr: F<B, 6>,
345    cb_bnhTMLM: F<B, 5>,
346    chunk_input_state_bnhpr: F<B, 5>,
347    read_stride: usize,
348) -> F<B, 6> {
349    let [batch, nchunks, chunk_len, mimo_rank, nheads, per_head_dim] = v_bnlmhp.dims();
350    let [.., chunk_tokens, _, _, state_rank] = c_bntmhr.dims();
351    let device = v_bnlmhp.device();
352    let dtype = v_bnlmhp.dtype();
353    let fused = chunk_len * mimo_rank; // write (source) axis
354    let read = chunk_tokens * mimo_rank; // read (target) axis
355
356    let v_bnLMhp = v_bnlmhp.reshape([batch, nchunks, fused, nheads, per_head_dim]);
357    let c_bnTMhr = c_bntmhr.reshape([batch, nchunks, read, nheads, state_rank]);
358
359    // Expand base da_cumsum to fused length: [b, nheads, n, l] → [b, nheads, n, L],
360    // and to the read rows the target axis needs.
361    let da_cumsum_bhnLM = da_cumsum_bhnl
362        .clone()
363        .unsqueeze_dim::<5>(4) // da_cumsum_bhnl1
364        .expand([batch, nheads, nchunks, chunk_len, mimo_rank]) // da_cumsum_bhnlm
365        .reshape([batch, nheads, nchunks, fused]); // da_cumsum_bhnLM
366    let da_cumsum_bhnTM = read_rows::<B, 4, 5>(da_cumsum_bhnl, 3, read_stride)
367        .unsqueeze_dim::<5>(4)
368        .expand([batch, nheads, nchunks, chunk_tokens, mimo_rank])
369        .reshape([batch, nheads, nchunks, read]);
370
371    // ── BLUE (Y_off): exp(cumA[i]) · C[i] · h[n-1] ─────────────────────
372    let exp_da_bnhTMp = da_cumsum_bhnTM
373        .clone()
374        .exp()
375        .swap_dims(1, 2) // exp_da_bnhTM
376        .unsqueeze_dim::<5>(4) // exp_da_bnhTM1
377        .expand([batch, nchunks, nheads, read, per_head_dim]);
378    let c_bnhTMr = c_bnTMhr.swap_dims(2, 3);
379    let chunk_input_state_bnhrp = chunk_input_state_bnhpr.transpose();
380    let ch_bnhTMp = c_bnhTMr.matmul(chunk_input_state_bnhrp);
381    let blue_bnhTMp = ch_bnhTMp * exp_da_bnhTMp;
382
383    // ── ORANGE (Y_diag): MIMO causal decay matrix · CB @ V ────────────────────
384    let target_da_cumsum_bnhTMLM = da_cumsum_bhnTM
385        .swap_dims(1, 2) // da_cumsum_bnhTM
386        .unsqueeze_dim::<5>(4) // da_cumsum_bnhTM1
387        .expand([batch, nchunks, nheads, read, fused]);
388    let source_da_cumsum_bnhTMLM = da_cumsum_bhnLM
389        .swap_dims(1, 2) // da_cumsum_bnhLM
390        .unsqueeze_dim::<5>(3) // da_cumsum_bnh1LM
391        .expand([batch, nchunks, nheads, read, fused]);
392    let diff_da_cumsum_bnhTMLM = target_da_cumsum_bnhTMLM - source_da_cumsum_bnhTMLM;
393
394    // MIMO causal neg-inf mask: −∞ where j//m > i//m (source strictly ahead of
395    // target in time). `triu(1)` keeps the diagonal — the double-SSD sum
396    // includes the same step.
397    let neg_inf_base_bnhtl = read_causal_mask::<B>(chunk_len, read_stride, 1, &device, dtype)
398        .unsqueeze_dims::<5>(&[0, 1, 2]) // neg_inf_base_111tl
399        .expand([batch, nchunks, nheads, chunk_tokens, chunk_len]); // neg_inf_base_bnhtl
400    let neg_inf_bnhTMLM = neg_inf_base_bnhtl
401        .unsqueeze_dim::<6>(4) // neg_inf_base_bnht1l
402        .expand([batch, nchunks, nheads, chunk_tokens, mimo_rank, chunk_len]) // neg_inf_base_bnhtml
403        .reshape([batch, nchunks, nheads, read, chunk_len]) // neg_inf_base_bnhTMl
404        .unsqueeze_dim::<6>(5) // neg_inf_base_bnhTMl1
405        .expand([batch, nchunks, nheads, read, chunk_len, mimo_rank]) // neg_inf_base_bnhTMlm
406        .reshape([batch, nchunks, nheads, read, fused]); // neg_inf_bnhTMLM
407
408    let decay_bnhTMLM = (diff_da_cumsum_bnhTMLM + neg_inf_bnhTMLM).exp();
409
410    let v_bnhLMp = v_bnLMhp.swap_dims(2, 3);
411    let orange_bnhTMp = (cb_bnhTMLM * decay_bnhTMLM).matmul(v_bnhLMp);
412
413    // ── Combine and reshape ────────────────────────────────────────────────────
414    let y_bntmhp = (blue_bnhTMp + orange_bnhTMp)
415        .swap_dims(2, 3) // y_bnTMhp
416        .reshape([batch, nchunks, chunk_tokens, mimo_rank, nheads, per_head_dim]); // y_bntmhp
417    y_bntmhp
418}
