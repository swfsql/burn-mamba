1//! # Serial SSD with a custom, memory-efficient backward (Mamba-3 double-SSD)
2//!
3//! The `SerialRecalculated` path of the double-SSD pathway. The forward is the
4//! same serial scan as [`super::super::serial`], but it goes through the
5//! [`Mamba3DoubleSsdBackendExt`] trait. So an `Autodiff` backend can use a
6//! custom backward that recomputes per-chunk intermediates instead of storing
7//! them (see [`super::backward`] / [`super::combined_backward`]). Plain
8//! backends use the default body of the trait, which replays the serial
9//! kernels.
10//!
11//! The default body runs under a generic backend `B`, where the high-level
12//! `Tensor` (pinned to `Dispatch`) is not available. So the K1–K5 math uses
13//! the rank-tagged [`F`] primitive wrapper. The recompute backward in
14//! [`super::combined_backward`] uses K1/K2/K4 again. K5 is forward-only.
15
16#![allow(non_snake_case)]
17

26impl Mamba3DoubleSsdInput {
27    /// MIMO-first Serial SSD with recalculated backward.
28    ///
29    /// Sends the full K1-K5 computation to
30    /// [`Mamba3DoubleSsdBackendExt::double_ssd_serial_recalculated`]. An
31    /// autodiff backend adds the memory-efficient custom backward. A plain
32    /// backend runs the standard K1-K5 serial computation.
33    ///
34    /// # Returns
35    /// - `y_bntmhp`:         `[batch, nchunks, chunk_tokens, mimo_rank, nheads, per_head_dim]`
36    /// - `final_state_bhpr`: `[batch, nheads, per_head_dim, state_rank]`
37    pub fn double_ssd_serial_recalculated(self) -> (Tensor<6>, Tensor<4>) {
38        let input = self;
39        assert!(
40            input.init_state_hpr.is_none(),
41            "init_state_hpr not yet implemented for ssd_serial_recalculated"
42        );
43
44        let (keep_bn, origin_bhpr, resets) =
45            crate::packing::Restarts::op_inputs(input.restarts, &input.v_bnlmhp);
46        let (y_bntmhp, final_state_bhpr) =
47            <Dispatch as Mamba3DoubleSsdBackendExt>::double_ssd_serial_recalculated(
48                input.v_bnlmhp.into_dispatch(),
49                input.da_bnlh.into_dispatch(),
50                input.b_bnlmhr.into_dispatch(),
51                input.c_bntmhr.into_dispatch(),
52                input.initial_state_bhpr.into_dispatch(),
53                input.read_stride,
54                keep_bn.into_dispatch(),
55                origin_bhpr.into_dispatch(),
56                resets,
57            );
58        let y_bntmhp = Tensor::from_dispatch(y_bntmhp);
59        let final_state_bhpr = Tensor::from_dispatch(final_state_bhpr);
60        (y_bntmhp, final_state_bhpr)
61    }
62}

64/// Extends the backend for the memory-efficient serial recalculated SSD.
65///
66/// The default implementation runs K1-K5 with primitive tensor operations.
67/// The Autodiff wrapper overrides it: its backward recomputes the forward
68/// intermediates instead of saving them.
69#[backend_extension(
70    // Every cubecl runtime (CUDA, ROCm, Metal, Vulkan, WebGPU, wgpu, CPU) is
71    // this one backend. The device of a tensor tells which runtime it uses.
72    // The cfg mirrors burn's own `cube_backend`.
73    Cube: cfg(any(
74        feature = "backend-cpu",
75        feature = "backend-cuda",
76        feature = "backend-rocm",
77        feature = "backend-metal",
78        feature = "backend-vulkan",
79        feature = "backend-wgpu",
80        feature = "backend-webgpu"
81    )),
82    Flex:  cfg(feature = "backend-flex"),
83    NdArray:  cfg(feature = "backend-ndarray"),
84    LibTorch:  cfg(any(feature = "backend-tch-cpu", feature = "backend-tch-gpu")),
85    Autodiff:  cfg(feature = "autodiff"),
86)]
87pub trait Mamba3DoubleSsdBackendExt: Backend {
88    /// Memory-efficient MIMO serial SSD.
89    ///
90    /// # Arguments
91    /// - `v_bnlmhp`:           `[batch, nchunks, chunk_len, mimo_rank, nheads, per_head_dim]`
92    /// - `da_bnlh`:            `[batch, nchunks, chunk_len, nheads]` — pre-combined Δ·A
93    /// - `b_bnlmhr`:           `[batch, nchunks, chunk_len, mimo_rank, nheads, state_rank]`
94    /// - `c_bntmhr`:           `[batch, nchunks, chunk_tokens, mimo_rank, nheads, state_rank]`
95    /// - `initial_state_bhpr`: `[batch, nheads, per_head_dim, state_rank]`
96    /// - `read_stride`:        `micro_steps` — folded positions per read row, so
97    ///   `chunk_tokens = chunk_len / read_stride`
98    /// - `keep_bn`:            `[batch, nchunks]`, `0` at a chunk that starts
99    ///   a new segment of a packed row. Read only when `resets` is `true`.
100    /// - `origin_bhpr`:        `[batch, nheads, per_head_dim, state_rank]`, the
101    ///   state that such a chunk starts from. Read only when `resets` is
102    ///   `true`.
103    ///
104    /// # Returns
105    /// - `y_bntmhp`:         `[batch, nchunks, chunk_tokens, mimo_rank, nheads, per_head_dim]`
106    /// - `final_state_bhpr`: `[batch, nheads, per_head_dim, state_rank]`
107    #[allow(clippy::too_many_arguments)]
108    fn double_ssd_serial_recalculated(
109        v_bnlmhp: FloatTensor<Self>,
110        da_bnlh: FloatTensor<Self>,
111        b_bnlmhr: FloatTensor<Self>,
112        c_bntmhr: FloatTensor<Self>,
113        initial_state_bhpr: FloatTensor<Self>,
114        read_stride: usize,
115        keep_bn: FloatTensor<Self>,
116        origin_bhpr: FloatTensor<Self>,
117        resets: bool,
118    ) -> (FloatTensor<Self>, FloatTensor<Self>) {
119        // Default impl: replicate Mamba3::double_ssd_serial (K1-K5) on primitives.
120        let v_bnlmhp = F::<Self, 6>::new(v_bnlmhp);
121        let da_bnlh = F::<Self, 4>::new(da_bnlh);
122        let b_bnlmhr = F::<Self, 6>::new(b_bnlmhr);
123        let c_bntmhr = F::<Self, 6>::new(c_bntmhr);
124        let initial_state_bhpr = F::<Self, 4>::new(initial_state_bhpr);
125        let restarts = resets.then(|| (F::<Self, 2>::new(keep_bn), F::<Self, 4>::new(origin_bhpr)));
126
127        let nchunks = v_bnlmhp.dims()[1];
128        assert!(nchunks > 0, "sequence length must be at least 1");
129
130        let (da_cumsum_bhnl, da_chunk_end_bhn) = k1_ssd_chunk_cumsum::<Self>(da_bnlh);
131        san(&da_cumsum_bhnl);
132
133        let cb_bnhTMLM = k2_ssd_bmm::<Self>(c_bntmhr.clone(), b_bnlmhr.clone());
134        san(&cb_bnhTMLM);
135
136        let intra_chunk_state_bnhpr =
137            k3_ssd_chunk_state::<Self>(v_bnlmhp.clone(), b_bnlmhr, da_cumsum_bhnl.clone());
138        san(&intra_chunk_state_bnhpr);
139
140        let (chunk_input_state_bnhpr, final_state_bhpr) = k4_ssd_state_passing::<Self>(
141            intra_chunk_state_bnhpr,
142            da_chunk_end_bhn,
143            initial_state_bhpr,
144            restarts,
145            None,
146        );
147        san(&chunk_input_state_bnhpr);
148        san(&final_state_bhpr);
149
150        let y_bntmhp = k5_ssd_chunk_scan::<Self>(
151            da_cumsum_bhnl,
152            v_bnlmhp,
153            c_bntmhr,
154            cb_bnhTMLM,
155            chunk_input_state_bnhpr,
156            read_stride,
157        );
158        san(&y_bntmhp);
159
160        (y_bntmhp.inner(), final_state_bhpr.inner())
161    }
162}

172// ─── Primitive forward kernels (K1–K5) ───────────────────────────────────────
173// Primitive ports of the high-level [`super::super::serial`] kernels, on the
174// primitives of `B` through [`F`], so the default trait body can run under a
175// generic backend. The recompute backward in [`super::combined_backward`] uses
176// K1/K2/K4 again. K5 is forward-only: the backward computes the gradient of K5
177// analytically instead of recomputing it.

179/// Primitive port of [`super::super::serial::k1_ssd_chunk_cumsum`].
180///
181/// Returns the intra-chunk cumsum `da_cumsum_bhnl` and the per-chunk last value
182/// `da_chunk_end_bhn`.
183pub(crate) fn k1_ssd_chunk_cumsum<B: Backend>(da_bnlh: F<B, 4>) -> (F<B, 4>, F<B, 3>) {
184    let da_bhnl = da_bnlh.permute([0, 3, 1, 2]);
185    let da_cumsum_bhnl = da_bhnl.cumsum(3);
186    let da_chunk_end_bhn = da_cumsum_bhnl
187        .clone()
188        .slice(s![.., .., .., -1])
189        .squeeze_dim::<3>(3);
190    (da_cumsum_bhnl, da_chunk_end_bhn)
191}

193/// Primitive port of [`super::super::serial::k2_ssd_bmm`] (fused `L·M`).
194///
195/// Returns the intra-chunk `C·Bᵀ` block matrix `cb_bnhTMLM`.
196pub(crate) fn k2_ssd_bmm<B: Backend>(c_bntmhr: F<B, 6>, b_bnlmhr: F<B, 6>) -> F<B, 5> {
197    // `C` has the read axis of the chunk, and `B` its write axis. They are
198    // equal unless `micro_steps > 1`. See the `Tensor` twin in
199    // `super::super::serial`.
200    let [batch, nchunks, chunk_tokens, mimo_rank, nheads, state_rank] = c_bntmhr.dims();
201    let [.., chunk_len, _, _, _] = b_bnlmhr.dims();
202    let c_bnTMhr = c_bntmhr.reshape([batch, nchunks, chunk_tokens * mimo_rank, nheads, state_rank]);
203    let b_bnLMhr = b_bnlmhr.reshape([batch, nchunks, chunk_len * mimo_rank, nheads, state_rank]);
204    let c_bnhTMr = c_bnTMhr.swap_dims(2, 3);
205    let b_bnhrLM = b_bnLMhr.permute([0, 1, 3, 4, 2]);
206    c_bnhTMr.matmul(b_bnhrLM)
207}

209/// Primitive port of [`super::super::serial::k3_ssd_chunk_state`] (lean:
210/// returns only the chunk-end state).
211///
212/// Returns `intra_chunk_state_bnhpr`: the contribution of each chunk to its
213/// end state, from a zero state at the chunk start. `v_bnlmhp` is pre-scaled.
214pub(crate) fn k3_ssd_chunk_state<B: Backend>(
215    v_bnlmhp: F<B, 6>,
216    b_bnlmhr: F<B, 6>,
217    da_cumsum_bhnl: F<B, 4>,
218) -> F<B, 5> {
219    let [batch, nchunks, chunk_len, mimo_rank, nheads, per_head_dim] = v_bnlmhp.dims();
220    let [.., state_rank] = b_bnlmhr.dims();
221
222    let v_bnLMhp = v_bnlmhp.reshape([batch, nchunks, chunk_len * mimo_rank, nheads, per_head_dim]);
223    let b_bnLMhr = b_bnlmhr.reshape([batch, nchunks, chunk_len * mimo_rank, nheads, state_rank]);
224
225    let da_cumsum_last_bhn1 = da_cumsum_bhnl.clone().slice(s![.., .., .., -1]);
226    let da_cumsum_bhnLM = da_cumsum_bhnl
227        .unsqueeze_dim::<5>(4) // da_cumsum_bhnl1
228        .expand([batch, nheads, nchunks, chunk_len, mimo_rank]) // da_cumsum_bhnlm
229        .reshape([batch, nheads, nchunks, chunk_len * mimo_rank]); // da_cumsum_bhnLM
230    let decay_bhnLM = (da_cumsum_last_bhn1 - da_cumsum_bhnLM).exp();
231
232    let decay_bnLMh1 = decay_bhnLM.permute([0, 2, 3, 1]).unsqueeze_dim::<5>(4); // decay_bnLMh1
233    let decayed_v_bnLMhp = decay_bnLMh1 * v_bnLMhp;
234
235    let decayed_v_bnhpLM = decayed_v_bnLMhp.permute([0, 1, 3, 4, 2]);
236    let b_bnhLMr = b_bnLMhr.swap_dims(2, 3);
237    let intra_chunk_state_bnhpr = decayed_v_bnhpLM.matmul(b_bnhLMr);
238    assert_eq!(
239        [batch, nchunks, nheads, per_head_dim, state_rank],
240        intra_chunk_state_bnhpr.dims()
241    );
242    intra_chunk_state_bnhpr
243}

245/// Primitive port of [`super::super::serial::k4_ssd_state_passing`].
246///
247/// Returns the per-chunk input-state stream `chunk_input_state_bnhpr` and the
248/// `final_state_bhpr`.
249///
250/// `restarts` is `(keep_bn, origin_bhpr)`: the reset of the high-level K4 as
251/// a factor, and the origin state. The state that goes into a chunk with
252/// `keep = 0` is its restart state, `origin + seedᶜ` (`origin` without
253/// seeds). The call starts from `initial`.
254pub(crate) fn k4_ssd_state_passing<B: Backend>(
255    intra_chunk_state_bnhpr: F<B, 5>,
256    da_chunk_end_bhn: F<B, 3>,
257    initial_state_bhpr: F<B, 4>,
258    restarts: Option<(F<B, 2>, F<B, 4>)>,
259    seed_bnhpr: Option<F<B, 5>>,
260) -> (F<B, 5>, F<B, 4>) {
261    let [batch, nchunks, nheads, per_head_dim, state_rank] = intra_chunk_state_bnhpr.dims();
262    let state = [nheads, per_head_dim, state_rank];
263    let restarts = restarts.map(|(keep_bn, origin_bhpr)| {
264        let restart_bn = restart_factor(&keep_bn);
265        (keep_bn, restart_bn, origin_bhpr)
266    });
267    let restart = |origin_bhpr: &F<B, 4>, i_chunk: usize| match &seed_bnhpr {
268        Some(seed_bnhpr) => {
269            origin_bhpr.clone()
270                + seed_bnhpr
271                    .clone()
272                    .slice(s![.., i_chunk, .., .., ..])
273                    .squeeze_dim::<4>(1)
274        }
275        None => origin_bhpr.clone(),
276    };
277
278    let mut running_state_bhpr = initial_state_bhpr;
279    let mut chunk_input_state_vec_bhpr = Vec::with_capacity(nchunks + 1);
280    chunk_input_state_vec_bhpr.push(running_state_bhpr.clone());
281
282    for i_chunk in 0..nchunks {
283        // A chunk that starts a segment starts from its restart state. (A
284        // learnable `init_state_hpr` would join it, but only `Minimal` takes
285        // one.)
286        if let Some((keep_bn, restart_bn, origin_bhpr)) = &restarts {
287            running_state_bhpr = running_state_bhpr * keep_at(keep_bn, i_chunk, state)
288                + restart(origin_bhpr, i_chunk) * keep_at(restart_bn, i_chunk, state);
289            *chunk_input_state_vec_bhpr.last_mut().unwrap() = running_state_bhpr.clone();
290        }
291        let intra_state_bhpr = intra_chunk_state_bnhpr
292            .clone()
293            .slice(s![.., i_chunk, .., .., ..])
294            .squeeze_dim::<4>(1);
295        let decay_bhpr = da_chunk_end_bhn
296            .clone()
297            .slice(s![.., .., i_chunk])
298            .unsqueeze_dim::<4>(3)
299            .exp()
300            .expand([batch, nheads, per_head_dim, state_rank]);
301        running_state_bhpr = decay_bhpr * running_state_bhpr + intra_state_bhpr;
302        chunk_input_state_vec_bhpr.push(running_state_bhpr.clone());
303    }
304
305    let final_state_bhpr = chunk_input_state_vec_bhpr.pop().unwrap();
306    let chunk_input_state_bnhpr = F::stack(chunk_input_state_vec_bhpr, 1);
307    (chunk_input_state_bnhpr, final_state_bhpr)
308}

310/// `1 − keep`: `1` at a chunk that restarts.
311fn restart_factor<B: Backend>(keep_bn: &F<B, 2>) -> F<B, 2> {
312    F::<B, 2>::full(keep_bn.dims(), 1.0, &keep_bn.device(), keep_bn.dtype()) - keep_bn.clone()
313}

315/// The factor of chunk `i_chunk`, broadcast over one state
316/// (`[batch, nheads, per_head_dim, state_rank]`).
317fn keep_at<B: Backend>(keep_bn: &F<B, 2>, i_chunk: usize, [nheads, per_head_dim, state_rank]: [usize; 3]) -> F<B, 4> {
318    let batch = keep_bn.dims()[0];
319    keep_bn
320        .clone()
321        .slice(s![.., i_chunk])
322        .reshape([batch, 1, 1, 1])
323        .expand([batch, nheads, per_head_dim, state_rank])
324}

326/// Backward of [`k4_ssd_state_passing`]: the reverse of its scalar-decay scan.
327/// It is the only part of the recompute backward that is still a walk (as the
328/// forward is, where it was measured).
329///
330/// Forward, with `sᵢ` for `chunk_input_stateᵢ`: `sᵢ₊₁ = decayᵢ·sᵢ + intraᵢ`.
331/// Reverse, seeded by `d_final`: `d_intraᵢ = d_sᵢ₊₁` and
332/// `d_sᵢ = decayᵢ·d_sᵢ₊₁ + d_chunk_input_stateᵢ`. The walk ends on `d_initial`.
333/// `d_decayᵢ = d_sᵢ₊₁·sᵢ` needs no walk: `decay` is one scalar per `(b, h, n)`,
334/// so it comes out of the `(p, r)` sum, and the whole stream is one batched
335/// product.
336///
337/// Under a reset (`keep_bn`), `sᵢ = keepᵢ · (the state before chunk i) +
338/// (1 − keepᵢ) · restartᵢ`. So the gradient that goes on to chunk `i − 1` is
339/// `keepᵢ · d_sᵢ`, and `restartᵢ` gets `(1 − keepᵢ) · d_sᵢ`. The call starts
340/// from `initial`, so it gets what the walk leaves at the end. `sᵢ` in
341/// `d_decay` is already the restarted one (the caller recomputed it).
342///
343/// # Returns
344/// - `d_intra_chunk_state_bnhpr` — gradient of K3's per-chunk contribution
345/// - `d_da_chunk_end_bhn` — gradient of the per-chunk log-decay
346/// - `d_initial_state_bhpr` — what the walk leaves at the end
347/// - `d_restart_bnhpr` — the gradient of each chunk's restart state
348///   (`origin + seedᶜ`) with resets, else `None`. The origin gets its sum
349///   over the chunks, and each seed gets its own chunk.
350pub(crate) fn k4_ssd_state_passing_backward<B: Backend>(
351    d_chunk_input_state_bnhpr: F<B, 5>,
352    chunk_input_state_bnhpr: F<B, 5>,
353    da_chunk_end_bhn: F<B, 3>,
354    d_final_state_bhpr: F<B, 4>,
355    keep_bn: Option<F<B, 2>>,
356) -> (F<B, 5>, F<B, 3>, F<B, 4>, Option<F<B, 5>>) {
357    let [batch, nchunks, nheads, per_head_dim, state_rank] = chunk_input_state_bnhpr.dims();
358    let state = [nheads, per_head_dim, state_rank];
359    let restart_bn = keep_bn.as_ref().map(restart_factor);
360
361    let mut d_running_state_bhpr = d_final_state_bhpr;
362    let mut d_intra_vec_bhpr: Vec<F<B, 4>> = Vec::with_capacity(nchunks);
363    // `d_restartᵢ`, in reverse chunk order.
364    let mut d_restart_vec_bhpr: Vec<F<B, 4>> = Vec::with_capacity(nchunks);
365    for i_chunk in (0..nchunks).rev() {
366        d_intra_vec_bhpr.push(d_running_state_bhpr.clone());
367
368        let decay_bhpr: F<B, 4> = da_chunk_end_bhn
369            .clone()
370            .slice(s![.., .., i_chunk]) // da_chunk_end_bh1
371            .exp()
372            .unsqueeze_dim::<4>(3) // decay_bh11
373            .expand([batch, nheads, per_head_dim, state_rank]);
374        let d_chunk_input_state_bhpr: F<B, 4> = d_chunk_input_state_bnhpr
375            .clone()
376            .slice(s![.., i_chunk, .., .., ..])
377            .squeeze_dim::<4>(1);
378
379        d_running_state_bhpr = decay_bhpr * d_running_state_bhpr + d_chunk_input_state_bhpr;
380        if let (Some(keep_bn), Some(restart_bn)) = (&keep_bn, &restart_bn) {
381            d_restart_vec_bhpr.push(d_running_state_bhpr.clone() * keep_at(restart_bn, i_chunk, state));
382            d_running_state_bhpr = d_running_state_bhpr * keep_at(keep_bn, i_chunk, state);
383        }
384        san(&d_running_state_bhpr);
385    }
386    d_intra_vec_bhpr.reverse();
387    let d_intra_chunk_state_bnhpr: F<B, 5> = F::stack(d_intra_vec_bhpr, 1);
388
389    // The walk ends on the state before chunk 0: `initial`.
390    let d_initial_state_bhpr = d_running_state_bhpr;
391    let d_restart_bnhpr = (!d_restart_vec_bhpr.is_empty()).then(|| {
392        d_restart_vec_bhpr.reverse();
393        F::stack(d_restart_vec_bhpr, 1)
394    });
395
396    let d_decay_bhn: F<B, 3> = (d_intra_chunk_state_bnhpr.clone() * chunk_input_state_bnhpr)
397        .reshape([batch, nchunks, nheads, per_head_dim * state_rank])
398        .sum_dim(3) // d_decay_bnh1
399        .squeeze_dim::<3>(3) // d_decay_bnh
400        .permute([0, 2, 1]); // d_decay_bhn
401    // decay = exp(da_chunk_end), so the chain rule multiplies it back in.
402    let d_da_chunk_end_bhn = d_decay_bhn * da_chunk_end_bhn.exp();
403    san(&d_da_chunk_end_bhn);
404
405    (
406        d_intra_chunk_state_bnhpr,
407        d_da_chunk_end_bhn,
408        d_initial_state_bhpr,
409        d_restart_bnhpr,
410    )
411}

413/// Concatenate the per-group outputs of a chunk-group pass back along the
414/// chunk axis. Nothing to do when the pass ran in one group, which is the case
415/// for a short enough chunk axis (see
416/// [`Mamba3SsdPath::backward_chunk_group`](crate::mamba3::ssd_path::Mamba3SsdPath::backward_chunk_group)).
417pub(crate) fn cat_chunk_groups<B: Backend, const D: usize>(
418    mut groups: Vec<F<B, D>>,
419    dim: usize,
420) -> F<B, D> {
421    match groups.len() {
422        1 => groups.pop().expect("one group"),
423        _ => F::cat(groups, dim),
424    }
425}

427/// Primitive port of [`super::super::serial::k5_ssd_chunk_scan`].
428///
429/// Adds the intra-chunk (ORANGE, MIMO causal) and inter-chunk (BLUE,
430/// state-carried) contributions into the output `y_bntmhp`. It applies no `D`
431/// skip: the caller adds it. Forward-only.
432fn k5_ssd_chunk_scan<B: Backend>(
433    da_cumsum_bhnl: F<B, 4>,
434    v_bnlmhp: F<B, 6>,
435    c_bntmhr: F<B, 6>,
436    cb_bnhTMLM: F<B, 5>,
437    chunk_input_state_bnhpr: F<B, 5>,
438    read_stride: usize,
439) -> F<B, 6> {
440    let [batch, nchunks, chunk_len, mimo_rank, nheads, per_head_dim] = v_bnlmhp.dims();
441    let [.., chunk_tokens, _, _, state_rank] = c_bntmhr.dims();
442    let device = v_bnlmhp.device();
443    let dtype = v_bnlmhp.dtype();
444    let fused = chunk_len * mimo_rank; // write (source) axis
445    let read = chunk_tokens * mimo_rank; // read (target) axis
446
447    let v_bnLMhp = v_bnlmhp.reshape([batch, nchunks, fused, nheads, per_head_dim]);
448    let c_bnTMhr = c_bntmhr.reshape([batch, nchunks, read, nheads, state_rank]);
449
450    // Expand base da_cumsum to fused length: [b, nheads, n, l] → [b, nheads, n, L],
451    // and to the read rows the target axis needs.
452    let da_cumsum_bhnLM = da_cumsum_bhnl
453        .clone()
454        .unsqueeze_dim::<5>(4) // da_cumsum_bhnl1
455        .expand([batch, nheads, nchunks, chunk_len, mimo_rank]) // da_cumsum_bhnlm
456        .reshape([batch, nheads, nchunks, fused]); // da_cumsum_bhnLM
457    let da_cumsum_bhnTM = read_rows::<B, 4, 5>(da_cumsum_bhnl, 3, read_stride)
458        .unsqueeze_dim::<5>(4)
459        .expand([batch, nheads, nchunks, chunk_tokens, mimo_rank])
460        .reshape([batch, nheads, nchunks, read]);
461
462    // ── BLUE (Y_off): exp(cumA[i]) · C[i] · h[n-1] ─────────────────────
463    let exp_da_bnhTMp = da_cumsum_bhnTM
464        .clone()
465        .exp()
466        .swap_dims(1, 2) // exp_da_bnhTM
467        .unsqueeze_dim::<5>(4) // exp_da_bnhTM1
468        .expand([batch, nchunks, nheads, read, per_head_dim]);
469    let c_bnhTMr = c_bnTMhr.swap_dims(2, 3);
470    let chunk_input_state_bnhrp = chunk_input_state_bnhpr.transpose();
471    let ch_bnhTMp = c_bnhTMr.matmul(chunk_input_state_bnhrp);
472    let blue_bnhTMp = ch_bnhTMp * exp_da_bnhTMp;
473
474    // ── ORANGE (Y_diag): MIMO causal decay matrix · CB @ V ────────────────────
475    let target_da_cumsum_bnhTMLM = da_cumsum_bhnTM
476        .swap_dims(1, 2) // da_cumsum_bnhTM
477        .unsqueeze_dim::<5>(4) // da_cumsum_bnhTM1
478        .expand([batch, nchunks, nheads, read, fused]);
479    let source_da_cumsum_bnhTMLM = da_cumsum_bhnLM
480        .swap_dims(1, 2) // da_cumsum_bnhLM
481        .unsqueeze_dim::<5>(3) // da_cumsum_bnh1LM
482        .expand([batch, nchunks, nheads, read, fused]);
483    let diff_da_cumsum_bnhTMLM = target_da_cumsum_bnhTMLM - source_da_cumsum_bnhTMLM;
484
485    // MIMO causal neg-inf mask: −∞ where j//m > i//m (source strictly ahead of
486    // target in time). `triu(1)` keeps the diagonal — the double-SSD sum
487    // includes the same step.
488    let neg_inf_base_bnhtl = read_causal_mask::<B>(chunk_len, read_stride, 1, &device, dtype)
489        .unsqueeze_dims::<5>(&[0, 1, 2]) // neg_inf_base_111tl
490        .expand([batch, nchunks, nheads, chunk_tokens, chunk_len]); // neg_inf_base_bnhtl
491    let neg_inf_bnhTMLM = neg_inf_base_bnhtl
492        .unsqueeze_dim::<6>(4) // neg_inf_base_bnht1l
493        .expand([batch, nchunks, nheads, chunk_tokens, mimo_rank, chunk_len]) // neg_inf_base_bnhtml
494        .reshape([batch, nchunks, nheads, read, chunk_len]) // neg_inf_base_bnhTMl
495        .unsqueeze_dim::<6>(5) // neg_inf_base_bnhTMl1
496        .expand([batch, nchunks, nheads, read, chunk_len, mimo_rank]) // neg_inf_base_bnhTMlm
497        .reshape([batch, nchunks, nheads, read, fused]); // neg_inf_bnhTMLM
498
499    let decay_bnhTMLM = (diff_da_cumsum_bnhTMLM + neg_inf_bnhTMLM).exp();
500
501    let v_bnhLMp = v_bnLMhp.swap_dims(2, 3);
502    let orange_bnhTMp = (cb_bnhTMLM * decay_bnhTMLM).matmul(v_bnhLMp);
503
504    // ── Combine and reshape ────────────────────────────────────────────────────
505    let y_bntmhp = (blue_bnhTMp + orange_bnhTMp)
506        .swap_dims(2, 3) // y_bnTMhp
507        .reshape([batch, nchunks, chunk_tokens, mimo_rank, nheads, per_head_dim]); // y_bntmhp
508    y_bntmhp
509}
