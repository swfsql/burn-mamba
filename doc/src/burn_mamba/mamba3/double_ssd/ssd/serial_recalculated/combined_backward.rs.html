1//! # Recompute-based gradient math for the Mamba-3 double-SSD
2//!
3//! The analytic backward of the MIMO-first serial scan of each pass of the
4//! double-SSD decomposition:
5//!
6//! 1. Recompute the forward intermediates (K1–K4) from the saved leaf inputs.
7//!    They are not stored.
8//! 2. Run the chunk-**local** K5 backward batched, one chunk group at a time
9//!    ([`Mamba3SsdPath::backward_chunk_group`](crate::mamba3::ssd_path::Mamba3SsdPath::backward_chunk_group)).
10//! 3. Walk the K4 state-passing backward: the one recurrence, and the only
11//!    walk.
12//! 4. Run the K1/K2/K3 backwards batched.
13//!
14//! The fused `L·M` length carries the `mimo_rank` axis through the intra-chunk
15//! products.
16//!
17//! All the math is on backend **primitives**, through the rank-tagged `F`
18//! wrapper. The custom [`Backward`](burn::backend::autodiff::ops::Backward) node
19//! runs with a generic backend `B`, where the high-level `Tensor` is not
20//! available, so the math uses the `float_*` ops of `B`. The recomputed
21//! K1/K2/K4 kernels are primitive ports of the high-level
22//! [`serial`](crate::mamba3::double_ssd::ssd::serial) kernels.
23

36/// Per-input gradients produced by [`combined_backward`] (one field per
37/// differentiable forward input of the double-SSD scan).
38#[non_exhaustive]
39pub struct CombinedGrads<B: Backend> {
40    /// Gradient of the (pre-scaled) input `v`.
41    pub d_v_bnlmhp: F<B, 6>,
42    /// Gradient of `Δ·A` (`da`).
43    pub d_da_bnlh: F<B, 4>,
44    /// Gradient of the input projection `B`.
45    pub d_b_bnlmhr: F<B, 6>,
46    /// Gradient of the output projection `C`, on the read axis of the chunk.
47    pub d_c_bntmhr: F<B, 6>,
48    /// Gradient of the initial SSM state.
49    pub d_initial_state_bhpr: F<B, 4>,
50    /// Gradient of the origin state of the resets (`None` without resets).
51    pub d_origin_bhpr: Option<F<B, 4>>,
52}

54// ─── Recomputed forward kernels ──────────────────────────────────────────────
55// The recompute backward replays K1/K2/K4 of the forward (imported above from
56// [`super::serial_recalculated`]), plus the extended K3 below. The extended K3
57// also returns the intermediates that the gradient math needs.
58
59/// Same as `k3_ssd_chunk_state`, but it also returns the intermediates that
60/// the custom backward needs:
61/// - `intra_chunk_state_bnhpr` — the chunk-end state from a zero initial state
62/// - `decay_bhnLM` — the fused-length K3 decay factor `exp(cumA_last − cumA_fused)`
63/// - `decayed_v_bnLMhp` — V already scaled by `decay_bnLMh1`
64pub fn k3_ssd_chunk_state_extended<B: Backend>(
65    v_bnlmhp: F<B, 6>,
66    b_bnlmhr: F<B, 6>,
67    da_cumsum_bhnl: F<B, 4>,
68) -> (F<B, 5>, F<B, 4>, F<B, 5>) {
69    let [batch, nchunks, chunk_len, mimo_rank, nheads, per_head_dim] = v_bnlmhp.dims();
70    let [.., state_rank] = b_bnlmhr.dims();
71
72    let v_bnLMhp = v_bnlmhp.reshape([batch, nchunks, chunk_len * mimo_rank, nheads, per_head_dim]);
73    let b_bnLMhr = b_bnlmhr.reshape([batch, nchunks, chunk_len * mimo_rank, nheads, state_rank]);
74
75    let da_cumsum_last_bhn1 = da_cumsum_bhnl.clone().slice(s![.., .., .., -1]);
76    let da_cumsum_bhnLM = da_cumsum_bhnl
77        .unsqueeze_dim::<5>(4) // da_cumsum_bhnl1
78        .expand([batch, nheads, nchunks, chunk_len, mimo_rank]) // da_cumsum_bhnlm
79        .reshape([batch, nheads, nchunks, chunk_len * mimo_rank]); // da_cumsum_bhnLM
80    let decay_bhnLM = (da_cumsum_last_bhn1 - da_cumsum_bhnLM).exp();
81    san(&decay_bhnLM);
82
83    let decay_bnLMh1 = decay_bhnLM
84        .clone()
85        .permute([0, 2, 3, 1])
86        .unsqueeze_dim::<5>(4);
87    let decayed_v_bnLMhp = decay_bnLMh1 * v_bnLMhp;
88    san(&decayed_v_bnLMhp);
89
90    let decayed_v_bnhpLM = decayed_v_bnLMhp.clone().permute([0, 1, 3, 4, 2]);
91    let b_bnhLMr = b_bnLMhr.swap_dims(2, 3);
92    let intra_chunk_state_bnhpr = decayed_v_bnhpLM.matmul(b_bnhLMr);
93    san(&intra_chunk_state_bnhpr);
94
95    (intra_chunk_state_bnhpr, decay_bhnLM, decayed_v_bnLMhp)
96}

98/// Memory-efficient backward for the Mamba-3 MIMO-first chunkwise SSD.
99///
100/// Recomputes the forward intermediates (K1-K4) from the saved inputs. Then
101/// the chunk-local K5 (BLUE + ORANGE) backward runs batched over chunk groups,
102/// and the K4 state-passing backward walks the chunks in reverse. The K3/K2/K1
103/// backwards run as single batched ops.
104///
105/// # Arguments
106/// - `d_y_bntmhp` — upstream gradient of the SSD output (read axis)
107/// - `d_final_bhpr` — upstream gradient of the final SSM state
108/// - `v_bnlmhp`, `da_bnlh`, `b_bnlmhr`, `c_bntmhr`, `initial_state_bhpr` —
109///   the five saved forward inputs
110/// - `read_stride` — folded positions per read row (`micro_steps`)
111/// - `restarts` — `(keep_bn, origin_bhpr)`, or `None`. `keep_bn` is the
112///   constant reset factor of the forward (`0` at a chunk that starts a new
113///   segment). `origin_bhpr` is the state that those chunks start from.
114///
115/// # Returns
116/// One [`CombinedGrads`] struct with the gradients of all 5 inputs, and of
117/// the origin state.
118#[allow(clippy::too_many_arguments)]
119pub fn combined_backward<B: Backend>(
120    d_y_bntmhp: F<B, 6>,
121    d_final_bhpr: F<B, 4>,
122    //
123    v_bnlmhp: F<B, 6>,
124    da_bnlh: F<B, 4>,
125    b_bnlmhr: F<B, 6>,
126    c_bntmhr: F<B, 6>,
127    initial_state_bhpr: F<B, 4>,
128    read_stride: usize,
129    restarts: Option<(F<B, 2>, F<B, 4>)>,
130) -> CombinedGrads<B> {
131    let keep_bn = restarts.as_ref().map(|(keep_bn, _)| keep_bn.clone());
132    let [batch, nchunks, chunk_len, mimo_rank, nheads, per_head_dim] = v_bnlmhp.dims();
133    let [.., state_rank] = b_bnlmhr.dims();
134    let device = v_bnlmhp.device();
135    let dtype = v_bnlmhp.dtype();
136    // The chunk's two axes: `fused` is what it writes, `read` what it is read at.
137    let chunk_tokens = Mamba3SsdPath::chunk_tokens(chunk_len, read_stride);
138    let fused = chunk_len * mimo_rank;
139    let read = chunk_tokens * mimo_rank;
140
141    san(&d_y_bntmhp);
142    san(&d_final_bhpr);
143    san(&v_bnlmhp);
144    san(&da_bnlh);
145    san(&b_bnlmhr);
146    san(&c_bntmhr);
147    san(&initial_state_bhpr);
148
149    // ═══════════════════════════════════════════════════════════════════════
150    // RECOMPUTE FORWARD INTERMEDIATES
151    // ═══════════════════════════════════════════════════════════════════════
152
153    // K1 — pre-combined Δ·A → intra-chunk cumsum
154    let (da_cumsum_bhnl, da_chunk_end_bhn) = k1_ssd_chunk_cumsum(da_bnlh.clone());
155    san(&da_cumsum_bhnl);
156
157    // K2 — CB matrix used in K5 ORANGE
158    let cb_bnhTMLM = k2_ssd_bmm(c_bntmhr.clone(), b_bnlmhr.clone());
159    san(&cb_bnhTMLM);
160
161    // K3 — intra-chunk state + decay/decayed-V intermediates
162    let (intra_chunk_state_bnhpr, k3_decay_bhnLM, k3_decayed_v_bnLMhp) =
163        k3_ssd_chunk_state_extended(v_bnlmhp.clone(), b_bnlmhr.clone(), da_cumsum_bhnl.clone());
164
165    // K4 — chunk-input state stream consumed by K5 BLUE
166    let (chunk_input_state_bnhpr, _final_state_bhpr) = k4_ssd_state_passing(
167        intra_chunk_state_bnhpr,
168        da_chunk_end_bhn.clone(),
169        initial_state_bhpr,
170        restarts,
171        None,
172    );
173
174    // ═══════════════════════════════════════════════════════════════════════
175    // FUSED-L INTERMEDIATES USED BY THE REVERSE LOOP
176    // ═══════════════════════════════════════════════════════════════════════
177    //
178    // da_cumsum_bhnLM: cumA per fused position. The expand-then-reshape
179    // repeats each base position mimo_rank times along the fused dim, matching K5.
180    let da_cumsum_bhnLM = da_cumsum_bhnl
181        .clone()
182        .unsqueeze_dim::<5>(4) // da_cumsum_bhnl1
183        .expand([batch, nheads, nchunks, chunk_len, mimo_rank]) // da_cumsum_bhnlm
184        .reshape([batch, nheads, nchunks, fused]); // da_cumsum_bhnLM
185    let da_cumsum_bhnTM = read_rows::<B, 4, 5>(da_cumsum_bhnl.clone(), 3, read_stride)
186        .unsqueeze_dim::<5>(4)
187        .expand([batch, nheads, nchunks, chunk_tokens, mimo_rank])
188        .reshape([batch, nheads, nchunks, read]);
189
190    // d_y in (batch, nchunks, nheads, chunk_tokens * mimo_rank, per_head_dim)
191    // ordering — matches the per-chunk slicing.
192    let d_y_bnhTMp = d_y_bntmhp
193        .reshape([batch, nchunks, read, nheads, per_head_dim]) // d_y_bnTMhp
194        .swap_dims(2, 3); // d_y_bnhTMp
195    san(&d_y_bnhTMp);
196
197    // Reusable [chunk_tokens, chunk_len] -inf upper-triangular base mask for
198    // ORANGE (`triu(1)`: the double-SSD sum includes the same step).
199    let neg_inf_base_tl: F<B, 2> = read_causal_mask::<B>(chunk_len, read_stride, 1, &device, dtype);
200
201    // ═══════════════════════════════════════════════════════════════════════
202    // CHUNK-LOCAL BACKWARD — K5 (BLUE + ORANGE), batched over a chunk group
203    //
204    // Neither term is recurrent: both need chunk `i`'s own slices plus
205    // `chunk_input_state[i]`, which the recomputed K4 already produced batched.
206    // So this side mirrors the batched K5 of the forward, `group` chunks at a
207    // time. The group size sets the memory of the score `[b, group, h, TM, LM]`
208    // against the peak of that forward (see
209    // [`Mamba3SsdPath::backward_chunk_group`]). The state gradient, the one
210    // part that *is* a scan, comes below.
211    // ═══════════════════════════════════════════════════════════════════════
212    let group = Mamba3SsdPath::backward_chunk_group(nchunks);
213    let ngroups = nchunks.div_ceil(group);
214    let mut vec_orange_d_v_bnhLMp: Vec<F<B, 5>> = Vec::with_capacity(ngroups);
215    let mut vec_blue_d_c_bnhTMr: Vec<F<B, 5>> = Vec::with_capacity(ngroups);
216    let mut vec_d_cb_bnhTMLM: Vec<F<B, 5>> = Vec::with_capacity(ngroups);
217    let mut vec_blue_d_da_bnhl: Vec<F<B, 4>> = Vec::with_capacity(ngroups);
218    let mut vec_orange_d_da_bnhl: Vec<F<B, 4>> = Vec::with_capacity(ngroups);
219    let mut vec_d_chunk_input_state_bnhpr: Vec<F<B, 5>> = Vec::with_capacity(ngroups);
220
221    for start in (0..nchunks).step_by(group) {
222        // The chunk count of this group: `group`, or fewer at the tail. It is
223        // the same `n` axis, only narrowed, so the names do not change.
224        let n = group.min(nchunks - start);
225
226        // ── Group slices (fused chunk_len · mimo_rank) ─────────────────
227        let v_bnhLMp: F<B, 5> = v_bnlmhp
228            .clone()
229            .narrow(1, start, n) // v_bnlmhp
230            .reshape([batch, n, fused, nheads, per_head_dim]) // v_bnLMhp
231            .swap_dims(2, 3); // v_bnhLMp
232
233        let c_bnhTMr: F<B, 5> = c_bntmhr
234            .clone()
235            .narrow(1, start, n) // c_bntmhr
236            .reshape([batch, n, read, nheads, state_rank]) // c_bnTMhr
237            .swap_dims(2, 3); // c_bnhTMr
238
239        let cb_bnhTMLM: F<B, 5> = cb_bnhTMLM.clone().narrow(1, start, n);
240
241        let da_cumsum_bnhLM: F<B, 4> = da_cumsum_bhnLM
242            .clone()
243            .narrow(2, start, n) // da_cumsum_bhnLM
244            .swap_dims(1, 2); // da_cumsum_bnhLM
245        let da_cumsum_bnhTM: F<B, 4> = da_cumsum_bhnTM
246            .clone()
247            .narrow(2, start, n) // da_cumsum_bhnTM
248            .swap_dims(1, 2); // da_cumsum_bnhTM
249
250        let chunk_input_state_bnhpr: F<B, 5> =
251            chunk_input_state_bnhpr.clone().narrow(1, start, n);
252        san(&chunk_input_state_bnhpr);
253
254        let d_y_bnhTMp: F<B, 5> = d_y_bnhTMp.clone().narrow(1, start, n);
255
256        // ── BLUE backward ──────────────────────────────────────────────
257        //
258        //   blue[TM,p] = exp(cumA[TM]) · Σᵣ C[TM,r] · state[p,r]
259        //
260        // exp_da depends on the read position TM only — broadcast over per_head_dim.
261        let exp_da_cumsum_bnhTMp: F<B, 5> = da_cumsum_bnhTM
262            .clone()
263            .exp()
264            .unsqueeze_dim::<5>(4) // exp_da_cumsum_bnhTM1
265            .expand([batch, n, nheads, read, per_head_dim]); // exp_da_cumsum_bnhTMp
266        let d_ch_bnhTMp: F<B, 5> = d_y_bnhTMp.clone() * exp_da_cumsum_bnhTMp.clone();
267        san(&d_ch_bnhTMp);
268
269        // d_chunk_input_state[p,r] = Σ_TM C[TM,r] · d_ch[TM,p]
270        //   C^T (bnhrTM) @ d_ch (bnhTMp)  → bnhrp  → transpose → bnhpr
271        let d_chunk_input_state_bnhpr: F<B, 5> = c_bnhTMr
272            .clone()
273            .transpose() // c_bnhrTM
274            .matmul(d_ch_bnhTMp.clone()) // d_chunk_input_state_bnhrp
275            .transpose(); // d_chunk_input_state_bnhpr
276        san(&d_chunk_input_state_bnhpr);
277        vec_d_chunk_input_state_bnhpr.push(d_chunk_input_state_bnhpr);
278
279        // d_C_blue[TM,r] = Σₚ d_ch[TM,p] · state[p,r]
280        //   d_ch (bnhTMp) @ state (bnhpr)  → bnhTMr
281        let d_c_blue_bnhTMr: F<B, 5> = d_ch_bnhTMp.matmul(chunk_input_state_bnhpr.clone());
282        san(&d_c_blue_bnhTMr);
283        vec_blue_d_c_bnhTMr.push(d_c_blue_bnhTMr);
284
285        // d_da from BLUE:
286        //   ch[TM,p] = Σᵣ C[TM,r] · state[p,r]      (= C @ state_rp after transpose)
287        //   d_da[TM] = (Σₚ d_y[TM,p] · ch[TM,p]) · exp_da[TM]
288        let ch_bnhTMp: F<B, 5> = c_bnhTMr.clone().matmul(
289            chunk_input_state_bnhpr.transpose(), // chunk_input_state_bnhrp
290        ); // ch_bnhTMp
291        let d_da_blue_bnhTM: F<B, 4> = (d_y_bnhTMp.clone() * ch_bnhTMp * exp_da_cumsum_bnhTMp)
292            .sum_dim(4) // d_da_blue_bnhTM1
293            .squeeze_dim::<4>(4); // d_da_blue_bnhTM
294        san(&d_da_blue_bnhTM);
295
296        // Reduce fused TM → t (sum the mimo_rank copies K5 broadcast), then
297        // scatter back onto the folded axis: a read row's `da` is one position.
298        let d_da_blue_bnhl: F<B, 4> = scatter_read_rows::<B, 4, 5>(
299            d_da_blue_bnhTM
300                .reshape([batch, n, nheads, chunk_tokens, mimo_rank]) // d_da_blue_bnhtm
301                .sum_dim(4) // d_da_blue_bnht1
302                .squeeze_dim::<4>(4), // d_da_blue_bnht
303            3,
304            read_stride,
305        );
306        vec_blue_d_da_bnhl.push(d_da_blue_bnhl);
307
308        // ── ORANGE backward ────────────────────────────────────────────
309        //
310        //   w[LMₜ,LMₛ] = CB[LMₜ,LMₛ] · decay[LMₜ,LMₛ]   (MIMO causal mask in decay)
311        //   orange[LMₜ,p] = Σ_{LMₛ} w[LMₜ,LMₛ] · v[LMₛ,p]
312        let da_target_bnhTMLM: F<B, 5> = da_cumsum_bnhTM
313            .unsqueeze_dim::<5>(4) // da_cumsum_bnhTMₜ1
314            .expand([batch, n, nheads, read, fused]); // da_target_bnhTMₜLM
315        let da_source_bnhTMLM: F<B, 5> = da_cumsum_bnhLM
316            .unsqueeze_dim::<5>(3) // da_cumsum_bnh1LMₛ
317            .expand([batch, n, nheads, read, fused]); // da_source_bnhTMLMₛ
318        let diff_bnhTMLM = da_target_bnhTMLM - da_source_bnhTMLM;
319        san(&diff_bnhTMLM);
320
321        // MIMO causal mask: -inf where LMₛ//mimo_rank > row(TMₜ)//mimo_rank —
322        // interleaved expansion of the [t, l] upper-triangular base mask
323        // (matches K5).
324        let neg_inf_mimo_bnhTMLM: F<B, 5> = neg_inf_base_tl
325            .clone()
326            .unsqueeze_dims::<5>(&[0, 1, 2]) // neg_inf_base_111tl
327            .expand([batch, n, nheads, chunk_tokens, chunk_len]) // neg_inf_base_bnhtl
328            .unsqueeze_dim::<6>(4) // neg_inf_base_bnht1l
329            .expand([batch, n, nheads, chunk_tokens, mimo_rank, chunk_len]) // neg_inf_base_bnhtml
330            .reshape([batch, n, nheads, read, chunk_len]) // neg_inf_base_bnhTMl
331            .unsqueeze_dim::<6>(5) // neg_inf_base_bnhTMl1
332            .expand([batch, n, nheads, read, chunk_len, mimo_rank]) // neg_inf_base_bnhTMlm
333            .reshape([batch, n, nheads, read, fused]); // neg_inf_mimo_bnhTMLM
334        let decay_bnhTMLM = (diff_bnhTMLM + neg_inf_mimo_bnhTMLM).exp();
335        san(&decay_bnhTMLM);
336
337        // d_v_orange = w^T @ d_orange ; d_w = d_orange @ v^T
338        let d_orange_bnhTMp = d_y_bnhTMp;
339        let w_bnhTMLM = cb_bnhTMLM.clone() * decay_bnhTMLM.clone();
340        let d_w_bnhTMLM: F<B, 5> = d_orange_bnhTMp.clone().matmul(
341            v_bnhLMp.transpose(), // v_bnhpLM
342        ); // d_w_bnhTMₜLMₛ
343        san(&d_w_bnhTMLM);
344        let d_v_orange_bnhLMp: F<B, 5> = w_bnhTMLM
345            .transpose() // w_bnhLMₛTMₜ
346            .matmul(d_orange_bnhTMp); // d_v_orange_bnhLMₛp
347        san(&d_v_orange_bnhLMp);
348        vec_orange_d_v_bnhLMp.push(d_v_orange_bnhLMp);
349
350        // d_cb = d_w · decay ; d_decay = d_w · cb ; d_diff = d_decay · decay
351        // (masked positions where decay=0 contribute 0 to d_diff automatically)
352        let d_cb_bnhTMLM = d_w_bnhTMLM.clone() * decay_bnhTMLM.clone();
353        vec_d_cb_bnhTMLM.push(d_cb_bnhTMLM);
354
355        let d_decay_bnhTMLM = d_w_bnhTMLM * cb_bnhTMLM;
356        let d_diff_bnhTMLM = d_decay_bnhTMLM * decay_bnhTMLM;
357
358        // d_da_target[TMₜ] = Σ_{LMₛ} d_diff[TMₜ, LMₛ] (scattered onto the folded
359        // axis); d_da_source[LMₛ] = Σ_{TMₜ} d_diff[TMₜ, LMₛ] (already folded);
360        // d_da_orange = d_da_target − d_da_source  (diff = target − source).
361        let d_da_target_bnhTM: F<B, 4> = d_diff_bnhTMLM
362            .clone()
363            .sum_dim(4) // d_diff_bnhTMₜ1
364            .squeeze_dim::<4>(4); // d_da_target_bnhTMₜ
365        let d_da_source_bnhLM: F<B, 4> = d_diff_bnhTMLM
366            .sum_dim(3) // d_diff_bnh1LMₛ
367            .squeeze_dim::<4>(3); // d_da_source_bnhLMₛ
368
369        let d_da_target_bnhl: F<B, 4> = scatter_read_rows::<B, 4, 5>(
370            d_da_target_bnhTM
371                .reshape([batch, n, nheads, chunk_tokens, mimo_rank])
372                .sum_dim(4)
373                .squeeze_dim::<4>(4),
374            3,
375            read_stride,
376        );
377        let d_da_source_bnhl: F<B, 4> = d_da_source_bnhLM
378            .reshape([batch, n, nheads, chunk_len, mimo_rank]) // d_da_source_bnhlm
379            .sum_dim(4) // d_da_source_bnhl1
380            .squeeze_dim::<4>(4); // d_da_source_bnhl
381        let d_da_orange_bnhl = d_da_target_bnhl - d_da_source_bnhl;
382        san(&d_da_orange_bnhl);
383        vec_orange_d_da_bnhl.push(d_da_orange_bnhl);
384    }
385
386    // ── Rejoin the groups along the chunk axis ────────────────────────────
387    let d_v_orange_bnhLMp: F<B, 5> = cat_chunk_groups(vec_orange_d_v_bnhLMp, 1);
388    let d_c_blue_bnhTMr: F<B, 5> = cat_chunk_groups(vec_blue_d_c_bnhTMr, 1);
389    let d_cb_bnhTMLM: F<B, 5> = cat_chunk_groups(vec_d_cb_bnhTMLM, 1);
390    let d_da_blue_bhnl: F<B, 4> = cat_chunk_groups(vec_blue_d_da_bnhl, 1).swap_dims(1, 2);
391    let d_da_orange_bhnl: F<B, 4> = cat_chunk_groups(vec_orange_d_da_bnhl, 1).swap_dims(1, 2);
392    let d_chunk_input_state_bnhpr: F<B, 5> = cat_chunk_groups(vec_d_chunk_input_state_bnhpr, 1);
393
394    // ── K4 backward — the reverse of the state-passing scan ───────────────
395    let (d_intra_chunk_state_bnhpr, d_da_end_bhn, d_initial_state_bhpr, d_restart_bnhpr) =
396        k4_ssd_state_passing_backward(
397            d_chunk_input_state_bnhpr,
398            chunk_input_state_bnhpr,
399            da_chunk_end_bhn,
400            d_final_bhpr,
401            keep_bn,
402        );
403    // Every chunk restarts from the same origin state.
404    let d_origin_bhpr = d_restart_bnhpr.map(|d_restart_bnhpr| d_restart_bnhpr.sum_dim(1).squeeze_dim::<4>(1));
405    // d_da_end: [batch,nheads,nchunks] scattered into the last `l` of d_da_cumsum_k4.
406    let d_da_cumsum_k4_bhnl: F<B, 4> = {
407        let zeros = F::<B, 4>::zeros([batch, nheads, nchunks, chunk_len - 1], &device, dtype);
408        let d_da_end_bhn1 = d_da_end_bhn.unsqueeze_dim::<4>(3);
409        F::cat(vec![zeros, d_da_end_bhn1], 3)
410    };
411
412    // ═══════════════════════════════════════════════════════════════════════
413    // K3 BACKWARD (batched)
414    //
415    // Forward (recap):
416    //   v_bnLMhp        = v.reshape
417    //   b_bnLMhr        = b.reshape
418    //   decay_bhnLM     = exp(cumA_last − cumA)
419    //   decay_bnLMh1    = decay_bhnLM.permute([0,2,3,1]).unsqueeze(4)
420    //   decayed_v_bnLMhp = decay_bnLMh1 · v_bnLMhp                      (elementwise)
421    //   decayed_v_bnhpLM = decayed_v_bnLMhp.permute([0,1,3,4,2])
422    //   b_bnhLMr        = b_bnLMhr.swap_dims(2, 3)
423    //   intra_state    = decayed_v_bnhpLM @ b_bnhLMr
424    // ═══════════════════════════════════════════════════════════════════════
425    let v_bnLMhp =
426        v_bnlmhp
427            .clone()
428            .reshape([batch, nchunks, chunk_len * mimo_rank, nheads, per_head_dim]);
429    let b_bnLMhr =
430        b_bnlmhr
431            .clone()
432            .reshape([batch, nchunks, chunk_len * mimo_rank, nheads, state_rank]);
433    let b_bnhLMr = b_bnLMhr.clone().swap_dims(2, 3);
434    let decayed_v_bnhpLM = k3_decayed_v_bnLMhp.permute([0, 1, 3, 4, 2]);
435
436    let d_decayed_v_bnhpLM: F<B, 5> = d_intra_chunk_state_bnhpr.clone().matmul(
437        b_bnhLMr.transpose(), // b_bnhrLM
438    ); // d_decayed_v_bnhpLM
439    let d_b_k3_bnhLMr: F<B, 5> = decayed_v_bnhpLM
440        .transpose() // decayed_v_bnhLMp
441        .matmul(d_intra_chunk_state_bnhpr); // d_b_k3_bnhLMr
442
443    let d_decayed_v_bnLMhp = d_decayed_v_bnhpLM.permute([0, 1, 4, 2, 3]);
444    let d_decay_bhnLM: F<B, 4> = (d_decayed_v_bnLMhp.clone() * v_bnLMhp)
445        .sum_dim(4) // d_decay_bnLMh1
446        .squeeze_dim::<4>(4) // d_decay_bnLMh
447        .permute([0, 3, 1, 2]); // d_decay_bhnLM
448
449    // d_v_k3_bnLMhp = d_decayed_v · decay (broadcast)
450    let k3_decay_bnLMh1 = k3_decay_bhnLM
451        .clone()
452        .permute([0, 2, 3, 1]) // k3_decay_bnLMh
453        .unsqueeze_dim::<5>(4); // k3_decay_bnLMh1
454    let d_v_k3_bnLMhp: F<B, 5> = d_decayed_v_bnLMhp * k3_decay_bnLMh1;
455    let d_v_k3_bnlrhp: F<B, 6> =
456        d_v_k3_bnLMhp.reshape([batch, nchunks, chunk_len, mimo_rank, nheads, per_head_dim]);
457
458    // d(cumA_last − cumA) = d_decay · decay
459    let d_decay_times_decay_bhnLM = d_decay_bhnLM * k3_decay_bhnLM;
460    // d_a_cumsum_last: Σ over LM (broadcast dim).
461    let d_a_cumsum_last_bhn: F<B, 3> = d_decay_times_decay_bhnLM
462        .clone()
463        .sum_dim(3) // d_decay_times_decay_bhn1
464        .squeeze_dim::<3>(3); // d_a_cumsum_last_bhn
465    // d_a_cumsum: negated (subtraction).
466    let d_da_cumsum_bhnLM = -d_decay_times_decay_bhnLM;
467
468    // Contribution to d_da_cumsum from the fused-cumA expand (sum mimo_rank copies).
469    let d_da_cumsum_k3_from_fused_bhnl: F<B, 4> = d_da_cumsum_bhnLM
470        .reshape([batch, nheads, nchunks, chunk_len, mimo_rank]) // d_da_cumsum_bhnlm
471        .sum_dim(4) // d_da_cumsum_bhnl1
472        .squeeze_dim::<4>(4); // d_da_cumsum_k3_from_fused_bhnl
473    // Contribution from cumA_last: only the last-l position.
474    let d_da_cumsum_k3_from_last_bhnl: F<B, 4> = {
475        let zeros = F::<B, 4>::zeros([batch, nheads, nchunks, chunk_len - 1], &device, dtype);
476        let d_last = d_a_cumsum_last_bhn.unsqueeze_dim::<4>(3);
477        F::cat(vec![zeros, d_last], 3)
478    };
479    let d_da_cumsum_k3_bhnl = d_da_cumsum_k3_from_fused_bhnl + d_da_cumsum_k3_from_last_bhnl;
480
481    // d_b_k3
482    let d_b_k3_bnLMhr = d_b_k3_bnhLMr.swap_dims(2, 3);
483    let d_b_k3_bnlmhr: F<B, 6> =
484        d_b_k3_bnLMhr.reshape([batch, nchunks, chunk_len, mimo_rank, nheads, state_rank]);
485
486    // ═══════════════════════════════════════════════════════════════════════
487    // K2 BACKWARD (batched)
488    //
489    //   cb_bnhLMLM = c_bnhLMr @ b_bnhrLM   (contracts state_rank)
490    //   d_c_bnhLMr = d_cb @ b_bnhLMr      (= d_cb @ b_bnhrLM^T)
491    //   d_b_bnhrLM = c_bnhrLM @ d_cb      (= c_bnhLMr^T @ d_cb)
492    // ═══════════════════════════════════════════════════════════════════════
493    let c_bnhTMr = c_bntmhr
494        .clone()
495        .reshape([batch, nchunks, read, nheads, state_rank]) // c_bnTMhr
496        .swap_dims(2, 3); // c_bnhTMr
497    let b_for_k2_bnhLMr = b_bnLMhr.swap_dims(2, 3);
498
499    let d_c_k2_bnhTMr: F<B, 5> = d_cb_bnhTMLM.clone().matmul(b_for_k2_bnhLMr);
500    let d_b_k2_bnhrLM: F<B, 5> = c_bnhTMr
501        .transpose() // c_bnhrTM
502        .matmul(d_cb_bnhTMLM); // d_b_k2_bnhrLM
503
504    // Undo permutes and reshape back
505    let d_c_k2_bntmhr: F<B, 6> = d_c_k2_bnhTMr
506        .swap_dims(2, 3) // d_c_k2_bnTMhr
507        .reshape([batch, nchunks, chunk_tokens, mimo_rank, nheads, state_rank]); // d_c_k2_bntmhr
508    let d_b_k2_bnlmhr: F<B, 6> = d_b_k2_bnhrLM
509        .permute([0, 1, 4, 2, 3]) // d_b_k2_bnLMhr
510        .reshape([batch, nchunks, chunk_len, mimo_rank, nheads, state_rank]); // d_b_k2_bnlmhr
511
512    // ── Unstack d_c_blue / d_v_orange and reshape back ────────────────────
513    let d_c_blue_bntmhr: F<B, 6> = d_c_blue_bnhTMr
514        .swap_dims(2, 3) // d_c_blue_bnTMhr
515        .reshape([batch, nchunks, chunk_tokens, mimo_rank, nheads, state_rank]); // d_c_blue_bntmhr
516    let d_v_orange_bnlrhp: F<B, 6> = d_v_orange_bnhLMp
517        .swap_dims(2, 3) // d_v_orange_bnLMhp
518        .reshape([batch, nchunks, chunk_len, mimo_rank, nheads, per_head_dim]); // d_v_orange_bnlrhp
519
520    // ═══════════════════════════════════════════════════════════════════════
521    // K1 BACKWARD + SUM CONTRIBUTIONS
522    // ═══════════════════════════════════════════════════════════════════════
523    let d_da_cumsum_bhnl =
524        d_da_blue_bhnl + d_da_orange_bhnl + d_da_cumsum_k3_bhnl + d_da_cumsum_k4_bhnl;
525    san(&d_da_cumsum_bhnl);
526
527    // K1 inverse: da_cumsum[l] = cumsum(da)[l]  →  d_da[l] = Σ_{k ≥ l} d_da_cumsum[k]
528    //
529    // Suffix sum:  d_da[l] = total_sum − cumsum(d_da_cumsum)[l-1] (cumsum[-1] = 0).
530    let d_da_bhnl = {
531        let d_total_bhnl = d_da_cumsum_bhnl
532            .clone()
533            .sum_dim(3) // d_da_cumsum_bhn1
534            .expand([batch, nheads, nchunks, chunk_len]); // d_total_bhnl
535        let prefix_bhnl = d_da_cumsum_bhnl.cumsum(3);
536        let zeros_bhn1 = F::<B, 4>::zeros([batch, nheads, nchunks, 1], &device, dtype);
537        let prefix_shifted_bhnl =
538            F::cat(vec![zeros_bhn1, prefix_bhnl.narrow(3, 0, chunk_len - 1)], 3);
539        d_total_bhnl - prefix_shifted_bhnl
540    };
541    san(&d_da_bhnl);
542    // Undo permute
543    let d_da_bnlh = d_da_bhnl.permute([0, 2, 3, 1]);
544
545    // ── Combine per-input gradient contributions ──────────────────────────
546    let d_v_bnlmhp = d_v_k3_bnlrhp + d_v_orange_bnlrhp;
547    let d_b_bnlmhr = d_b_k2_bnlmhr + d_b_k3_bnlmhr;
548    let d_c_bntmhr = d_c_k2_bntmhr + d_c_blue_bntmhr;
549
550    san(&d_v_bnlmhp);
551    san(&d_da_bnlh);
552    san(&d_b_bnlmhr);
553    san(&d_c_bntmhr);
554    san(&d_initial_state_bhpr);
555
556    CombinedGrads {
557        d_v_bnlmhp,
558        d_da_bnlh,
559        d_b_bnlmhr,
560        d_c_bntmhr,
561        d_initial_state_bhpr,
562        d_origin_bhpr,
563    }
564}
