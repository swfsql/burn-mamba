1//! # Recompute-based gradient math for the Mamba-3 double-SSD
2//!
3//! The analytic backward of the MIMO-first serial scan used by each pass of the
4//! double-SSD decomposition.  The forward intermediates (K1–K4) are recomputed
5//! from the saved leaf inputs rather than stashed, then the chunk-**local** K5
6//! backward runs batched, a chunk group at a time
7//! ([`Mamba3SsdPath::backward_chunk_group`](crate::mamba3::ssd_path::Mamba3SsdPath::backward_chunk_group)); only the K4 state-passing
8//! backward, the one recurrence, is a walk.  K1/K2/K3 backwards run batched
9//! after it.  The fused `L·M` length carries the `mimo_rank` axis through the
10//! intra-chunk products.
11//!
12//! Everything operates on backend **primitives** through the rank-tagged `F`
13//! wrapper: the custom [`Backward`](burn::backend::autodiff::ops::Backward) node
14//! runs with a generic backend `B`, so the high-level `Tensor` is unavailable
15//! and the math uses `B`'s `float_*` ops.  The recomputed K1/K2/K4 kernels are
16//! local primitive ports of the high-level [`super::super::serial`](crate::mamba3::double_ssd::ssd::serial) kernels.
17
18#![allow(non_snake_case)]

30/// Per-input gradients produced by [`combined_backward`] (one field per
31/// differentiable forward input of the double-SSD scan).
32#[non_exhaustive]
33pub struct CombinedGrads<B: Backend> {
34    /// Gradient of the (pre-scaled) input `v`.
35    pub d_v_bnlmhp: F<B, 6>,
36    /// Gradient of `Δ·A` (`da`).
37    pub d_da_bnlh: F<B, 4>,
38    /// Gradient of the input projection `B`.
39    pub d_b_bnlmhr: F<B, 6>,
40    /// Gradient of the output projection `C`, on the chunk's read axis.
41    pub d_c_bntmhr: F<B, 6>,
42    /// Gradient of the initial SSM state.
43    pub d_initial_state_bhpr: F<B, 4>,
44}

46// ─── Recomputed forward kernels ──────────────────────────────────────────────
47// The recompute backward replays the forward's K1/K2/K4 (imported above from
48// [`super::serial_recalculated`]) plus the extended K3 below, which returns the
49// extra intermediates the gradient math needs.
50
51/// Same as `k3_ssd_chunk_state` but
52/// also returns intermediates needed by the custom backward:
53/// - `intra_chunk_state_bnhpr` — the chunk-end state assuming zero initial state
54/// - `decay_bhnLM` — the fused-length K3 decay factor `exp(cumA_last − cumA_fused)`
55/// - `decayed_v_bnLMhp` — V already scaled by `decay_bnLMh1`
56pub fn k3_ssd_chunk_state_extended<B: Backend>(
57    v_bnlmhp: F<B, 6>,
58    b_bnlmhr: F<B, 6>,
59    da_cumsum_bhnl: F<B, 4>,
60) -> (F<B, 5>, F<B, 4>, F<B, 5>) {
61    let [batch, nchunks, chunk_len, mimo_rank, nheads, per_head_dim] = v_bnlmhp.dims();
62    let [.., state_rank] = b_bnlmhr.dims();
63
64    let v_bnLMhp = v_bnlmhp.reshape([batch, nchunks, chunk_len * mimo_rank, nheads, per_head_dim]);
65    let b_bnLMhr = b_bnlmhr.reshape([batch, nchunks, chunk_len * mimo_rank, nheads, state_rank]);
66
67    let da_cumsum_last_bhn1 = da_cumsum_bhnl.clone().slice(s![.., .., .., -1]);
68    let da_cumsum_bhnLM = da_cumsum_bhnl
69        .unsqueeze_dim::<5>(4) // da_cumsum_bhnl1
70        .expand([batch, nheads, nchunks, chunk_len, mimo_rank]) // da_cumsum_bhnlm
71        .reshape([batch, nheads, nchunks, chunk_len * mimo_rank]); // da_cumsum_bhnLM
72    let decay_bhnLM = (da_cumsum_last_bhn1 - da_cumsum_bhnLM).exp();
73    san(&decay_bhnLM);
74
75    let decay_bnLMh1 = decay_bhnLM
76        .clone()
77        .permute([0, 2, 3, 1])
78        .unsqueeze_dim::<5>(4);
79    let decayed_v_bnLMhp = decay_bnLMh1 * v_bnLMhp;
80    san(&decayed_v_bnLMhp);
81
82    let decayed_v_bnhpLM = decayed_v_bnLMhp.clone().permute([0, 1, 3, 4, 2]);
83    let b_bnhLMr = b_bnLMhr.swap_dims(2, 3);
84    let intra_chunk_state_bnhpr = decayed_v_bnhpLM.matmul(b_bnhLMr);
85    san(&intra_chunk_state_bnhpr);
86
87    (intra_chunk_state_bnhpr, decay_bhnLM, decayed_v_bnLMhp)
88}

90/// Memory-efficient backward for the Mamba-3 MIMO-first chunkwise SSD.
91///
92/// Recomputes the forward intermediates (K1-K4) from the saved inputs, then
93/// runs a reverse per-chunk loop that fuses the K5 (BLUE + ORANGE) backward
94/// with the K4 state-passing backward.  K3/K2/K1 backwards run as single
95/// batched ops once the loop has collected all per-chunk slices.
96///
97/// # Arguments
98/// - `d_y_bnlmhp` — upstream gradient of the SSD output
99/// - `d_final_bhpr` — upstream gradient of the final SSM state
100/// - `v_bnlmhp`, `da_bnlh`, `b_bnlmhr`, `c_bnlmhr`, `initial_state_bhpr` —
101///   the five saved forward inputs
102///
103/// # Returns
104/// One [`CombinedGrads`] struct containing gradients for all 5 inputs.
105pub fn combined_backward<B: Backend>(
106    d_y_bntmhp: F<B, 6>,
107    d_final_bhpr: F<B, 4>,
108    //
109    v_bnlmhp: F<B, 6>,
110    da_bnlh: F<B, 4>,
111    b_bnlmhr: F<B, 6>,
112    c_bntmhr: F<B, 6>,
113    initial_state_bhpr: F<B, 4>,
114    read_stride: usize,
115) -> CombinedGrads<B> {
116    let [batch, nchunks, chunk_len, mimo_rank, nheads, per_head_dim] = v_bnlmhp.dims();
117    let [.., state_rank] = b_bnlmhr.dims();
118    let device = v_bnlmhp.device();
119    let dtype = v_bnlmhp.dtype();
120    // The chunk's two axes: `fused` is what it writes, `read` what it is read at.
121    let chunk_tokens = Mamba3SsdPath::chunk_tokens(chunk_len, read_stride);
122    let fused = chunk_len * mimo_rank;
123    let read = chunk_tokens * mimo_rank;
124
125    san(&d_y_bntmhp);
126    san(&d_final_bhpr);
127    san(&v_bnlmhp);
128    san(&da_bnlh);
129    san(&b_bnlmhr);
130    san(&c_bntmhr);
131    san(&initial_state_bhpr);
132
133    // ═══════════════════════════════════════════════════════════════════════
134    // RECOMPUTE FORWARD INTERMEDIATES
135    // ═══════════════════════════════════════════════════════════════════════
136
137    // K1 — pre-combined Δ·A → intra-chunk cumsum
138    let (da_cumsum_bhnl, da_chunk_end_bhn) = k1_ssd_chunk_cumsum(da_bnlh.clone());
139    san(&da_cumsum_bhnl);
140
141    // K2 — CB matrix used in K5 ORANGE
142    let cb_bnhTMLM = k2_ssd_bmm(c_bntmhr.clone(), b_bnlmhr.clone());
143    san(&cb_bnhTMLM);
144
145    // K3 — intra-chunk state + decay/decayed-V intermediates
146    let (intra_chunk_state_bnhpr, k3_decay_bhnLM, k3_decayed_v_bnLMhp) =
147        k3_ssd_chunk_state_extended(v_bnlmhp.clone(), b_bnlmhr.clone(), da_cumsum_bhnl.clone());
148
149    // K4 — chunk-input state stream consumed by K5 BLUE
150    let (chunk_input_state_bnhpr, _final_state_bhpr) = k4_ssd_state_passing(
151        intra_chunk_state_bnhpr,
152        da_chunk_end_bhn.clone(),
153        initial_state_bhpr,
154    );
155
156    // ═══════════════════════════════════════════════════════════════════════
157    // FUSED-L INTERMEDIATES USED BY THE REVERSE LOOP
158    // ═══════════════════════════════════════════════════════════════════════
159    //
160    // da_cumsum_bhnLM: cumA per fused position. The expand-then-reshape
161    // repeats each base position mimo_rank times along the fused dim, matching K5.
162    let da_cumsum_bhnLM = da_cumsum_bhnl
163        .clone()
164        .unsqueeze_dim::<5>(4) // da_cumsum_bhnl1
165        .expand([batch, nheads, nchunks, chunk_len, mimo_rank]) // da_cumsum_bhnlm
166        .reshape([batch, nheads, nchunks, fused]); // da_cumsum_bhnLM
167    let da_cumsum_bhnTM = read_rows::<B, 4, 5>(da_cumsum_bhnl.clone(), 3, read_stride)
168        .unsqueeze_dim::<5>(4)
169        .expand([batch, nheads, nchunks, chunk_tokens, mimo_rank])
170        .reshape([batch, nheads, nchunks, read]);
171
172    // d_y in (batch, nchunks, nheads, chunk_tokens * mimo_rank, per_head_dim)
173    // ordering — matches the per-chunk slicing.
174    let d_y_bnhTMp = d_y_bntmhp
175        .reshape([batch, nchunks, read, nheads, per_head_dim]) // d_y_bnTMhp
176        .swap_dims(2, 3); // d_y_bnhTMp
177    san(&d_y_bnhTMp);
178
179    // Reusable [chunk_tokens, chunk_len] -inf upper-triangular base mask for
180    // ORANGE (`triu(1)`: the double-SSD sum includes the same step).
181    let neg_inf_base_tl: F<B, 2> = read_causal_mask::<B>(chunk_len, read_stride, 1, &device, dtype);
182
183    // ═══════════════════════════════════════════════════════════════════════
184    // CHUNK-LOCAL BACKWARD — K5 (BLUE + ORANGE), batched over a chunk group
185    //
186    // Neither term is recurrent: both need chunk `i`'s own slices plus
187    // `chunk_input_state[i]`, which the recomputed K4 already produced batched.
188    // So this side mirrors the forward's own batched K5, `group` chunks at a
189    // time — the group being what prices the score `[b, group, h, TM, LM]`
190    // against that forward's peak (see [`Mamba3SsdPath::backward_chunk_group`]).
191    // The state gradient, the one part that *is* a scan, follows below.
192    // ═══════════════════════════════════════════════════════════════════════
193    let group = Mamba3SsdPath::backward_chunk_group(nchunks);
194    let ngroups = nchunks.div_ceil(group);
195    let mut vec_orange_d_v_bnhLMp: Vec<F<B, 5>> = Vec::with_capacity(ngroups);
196    let mut vec_blue_d_c_bnhTMr: Vec<F<B, 5>> = Vec::with_capacity(ngroups);
197    let mut vec_d_cb_bnhTMLM: Vec<F<B, 5>> = Vec::with_capacity(ngroups);
198    let mut vec_blue_d_da_bnhl: Vec<F<B, 4>> = Vec::with_capacity(ngroups);
199    let mut vec_orange_d_da_bnhl: Vec<F<B, 4>> = Vec::with_capacity(ngroups);
200    let mut vec_d_chunk_input_state_bnhpr: Vec<F<B, 5>> = Vec::with_capacity(ngroups);
201
202    for start in (0..nchunks).step_by(group) {
203        // The group's own chunk count — `group`, short at the tail. It is the
204        // same `n` axis throughout, just narrowed, so the names do not change.
205        let n = group.min(nchunks - start);
206
207        // ── Group slices (fused chunk_len · mimo_rank) ─────────────────
208        let v_bnhLMp: F<B, 5> = v_bnlmhp
209            .clone()
210            .narrow(1, start, n) // v_bnlmhp
211            .reshape([batch, n, fused, nheads, per_head_dim]) // v_bnLMhp
212            .swap_dims(2, 3); // v_bnhLMp
213
214        let c_bnhTMr: F<B, 5> = c_bntmhr
215            .clone()
216            .narrow(1, start, n) // c_bntmhr
217            .reshape([batch, n, read, nheads, state_rank]) // c_bnTMhr
218            .swap_dims(2, 3); // c_bnhTMr
219
220        let cb_bnhTMLM: F<B, 5> = cb_bnhTMLM.clone().narrow(1, start, n);
221
222        let da_cumsum_bnhLM: F<B, 4> = da_cumsum_bhnLM
223            .clone()
224            .narrow(2, start, n) // da_cumsum_bhnLM
225            .swap_dims(1, 2); // da_cumsum_bnhLM
226        let da_cumsum_bnhTM: F<B, 4> = da_cumsum_bhnTM
227            .clone()
228            .narrow(2, start, n) // da_cumsum_bhnTM
229            .swap_dims(1, 2); // da_cumsum_bnhTM
230
231        let chunk_input_state_bnhpr: F<B, 5> =
232            chunk_input_state_bnhpr.clone().narrow(1, start, n);
233        san(&chunk_input_state_bnhpr);
234
235        let d_y_bnhTMp: F<B, 5> = d_y_bnhTMp.clone().narrow(1, start, n);
236
237        // ── BLUE backward ──────────────────────────────────────────────
238        //
239        //   blue[TM,p] = exp(cumA[TM]) · Σᵣ C[TM,r] · state[p,r]
240        //
241        // exp_da depends on the read position TM only — broadcast over per_head_dim.
242        let exp_da_cumsum_bnhTMp: F<B, 5> = da_cumsum_bnhTM
243            .clone()
244            .exp()
245            .unsqueeze_dim::<5>(4) // exp_da_cumsum_bnhTM1
246            .expand([batch, n, nheads, read, per_head_dim]); // exp_da_cumsum_bnhTMp
247        let d_ch_bnhTMp: F<B, 5> = d_y_bnhTMp.clone() * exp_da_cumsum_bnhTMp.clone();
248        san(&d_ch_bnhTMp);
249
250        // d_chunk_input_state[p,r] = Σ_TM C[TM,r] · d_ch[TM,p]
251        //   C^T (bnhrTM) @ d_ch (bnhTMp)  → bnhrp  → transpose → bnhpr
252        let d_chunk_input_state_bnhpr: F<B, 5> = c_bnhTMr
253            .clone()
254            .transpose() // c_bnhrTM
255            .matmul(d_ch_bnhTMp.clone()) // d_chunk_input_state_bnhrp
256            .transpose(); // d_chunk_input_state_bnhpr
257        san(&d_chunk_input_state_bnhpr);
258        vec_d_chunk_input_state_bnhpr.push(d_chunk_input_state_bnhpr);
259
260        // d_C_blue[TM,r] = Σₚ d_ch[TM,p] · state[p,r]
261        //   d_ch (bnhTMp) @ state (bnhpr)  → bnhTMr
262        let d_c_blue_bnhTMr: F<B, 5> = d_ch_bnhTMp.matmul(chunk_input_state_bnhpr.clone());
263        san(&d_c_blue_bnhTMr);
264        vec_blue_d_c_bnhTMr.push(d_c_blue_bnhTMr);
265
266        // d_da from BLUE:
267        //   ch[TM,p] = Σᵣ C[TM,r] · state[p,r]      (= C @ state_rp after transpose)
268        //   d_da[TM] = (Σₚ d_y[TM,p] · ch[TM,p]) · exp_da[TM]
269        let ch_bnhTMp: F<B, 5> = c_bnhTMr.clone().matmul(
270            chunk_input_state_bnhpr.transpose(), // chunk_input_state_bnhrp
271        ); // ch_bnhTMp
272        let d_da_blue_bnhTM: F<B, 4> = (d_y_bnhTMp.clone() * ch_bnhTMp * exp_da_cumsum_bnhTMp)
273            .sum_dim(4) // d_da_blue_bnhTM1
274            .squeeze_dim::<4>(4); // d_da_blue_bnhTM
275        san(&d_da_blue_bnhTM);
276
277        // Reduce fused TM → t (sum the mimo_rank copies K5 broadcast), then
278        // scatter back onto the folded axis: a read row's `da` is one position.
279        let d_da_blue_bnhl: F<B, 4> = scatter_read_rows::<B, 4, 5>(
280            d_da_blue_bnhTM
281                .reshape([batch, n, nheads, chunk_tokens, mimo_rank]) // d_da_blue_bnhtm
282                .sum_dim(4) // d_da_blue_bnht1
283                .squeeze_dim::<4>(4), // d_da_blue_bnht
284            3,
285            read_stride,
286        );
287        vec_blue_d_da_bnhl.push(d_da_blue_bnhl);
288
289        // ── ORANGE backward ────────────────────────────────────────────
290        //
291        //   w[LMₜ,LMₛ] = CB[LMₜ,LMₛ] · decay[LMₜ,LMₛ]   (MIMO causal mask in decay)
292        //   orange[LMₜ,p] = Σ_{LMₛ} w[LMₜ,LMₛ] · v[LMₛ,p]
293        let da_target_bnhTMLM: F<B, 5> = da_cumsum_bnhTM
294            .unsqueeze_dim::<5>(4) // da_cumsum_bnhTMₜ1
295            .expand([batch, n, nheads, read, fused]); // da_target_bnhTMₜLM
296        let da_source_bnhTMLM: F<B, 5> = da_cumsum_bnhLM
297            .unsqueeze_dim::<5>(3) // da_cumsum_bnh1LMₛ
298            .expand([batch, n, nheads, read, fused]); // da_source_bnhTMLMₛ
299        let diff_bnhTMLM = da_target_bnhTMLM - da_source_bnhTMLM;
300        san(&diff_bnhTMLM);
301
302        // MIMO causal mask: -inf where LMₛ//mimo_rank > row(TMₜ)//mimo_rank —
303        // interleaved expansion of the [t, l] upper-triangular base mask
304        // (matches K5).
305        let neg_inf_mimo_bnhTMLM: F<B, 5> = neg_inf_base_tl
306            .clone()
307            .unsqueeze_dims::<5>(&[0, 1, 2]) // neg_inf_base_111tl
308            .expand([batch, n, nheads, chunk_tokens, chunk_len]) // neg_inf_base_bnhtl
309            .unsqueeze_dim::<6>(4) // neg_inf_base_bnht1l
310            .expand([batch, n, nheads, chunk_tokens, mimo_rank, chunk_len]) // neg_inf_base_bnhtml
311            .reshape([batch, n, nheads, read, chunk_len]) // neg_inf_base_bnhTMl
312            .unsqueeze_dim::<6>(5) // neg_inf_base_bnhTMl1
313            .expand([batch, n, nheads, read, chunk_len, mimo_rank]) // neg_inf_base_bnhTMlm
314            .reshape([batch, n, nheads, read, fused]); // neg_inf_mimo_bnhTMLM
315        let decay_bnhTMLM = (diff_bnhTMLM + neg_inf_mimo_bnhTMLM).exp();
316        san(&decay_bnhTMLM);
317
318        // d_v_orange = w^T @ d_orange ; d_w = d_orange @ v^T
319        let d_orange_bnhTMp = d_y_bnhTMp;
320        let w_bnhTMLM = cb_bnhTMLM.clone() * decay_bnhTMLM.clone();
321        let d_w_bnhTMLM: F<B, 5> = d_orange_bnhTMp.clone().matmul(
322            v_bnhLMp.transpose(), // v_bnhpLM
323        ); // d_w_bnhTMₜLMₛ
324        san(&d_w_bnhTMLM);
325        let d_v_orange_bnhLMp: F<B, 5> = w_bnhTMLM
326            .transpose() // w_bnhLMₛTMₜ
327            .matmul(d_orange_bnhTMp); // d_v_orange_bnhLMₛp
328        san(&d_v_orange_bnhLMp);
329        vec_orange_d_v_bnhLMp.push(d_v_orange_bnhLMp);
330
331        // d_cb = d_w · decay ; d_decay = d_w · cb ; d_diff = d_decay · decay
332        // (masked positions where decay=0 contribute 0 to d_diff automatically)
333        let d_cb_bnhTMLM = d_w_bnhTMLM.clone() * decay_bnhTMLM.clone();
334        vec_d_cb_bnhTMLM.push(d_cb_bnhTMLM);
335
336        let d_decay_bnhTMLM = d_w_bnhTMLM * cb_bnhTMLM;
337        let d_diff_bnhTMLM = d_decay_bnhTMLM * decay_bnhTMLM;
338
339        // d_da_target[TMₜ] = Σ_{LMₛ} d_diff[TMₜ, LMₛ] (scattered onto the folded
340        // axis); d_da_source[LMₛ] = Σ_{TMₜ} d_diff[TMₜ, LMₛ] (already folded);
341        // d_da_orange = d_da_target − d_da_source  (diff = target − source).
342        let d_da_target_bnhTM: F<B, 4> = d_diff_bnhTMLM
343            .clone()
344            .sum_dim(4) // d_diff_bnhTMₜ1
345            .squeeze_dim::<4>(4); // d_da_target_bnhTMₜ
346        let d_da_source_bnhLM: F<B, 4> = d_diff_bnhTMLM
347            .sum_dim(3) // d_diff_bnh1LMₛ
348            .squeeze_dim::<4>(3); // d_da_source_bnhLMₛ
349
350        let d_da_target_bnhl: F<B, 4> = scatter_read_rows::<B, 4, 5>(
351            d_da_target_bnhTM
352                .reshape([batch, n, nheads, chunk_tokens, mimo_rank])
353                .sum_dim(4)
354                .squeeze_dim::<4>(4),
355            3,
356            read_stride,
357        );
358        let d_da_source_bnhl: F<B, 4> = d_da_source_bnhLM
359            .reshape([batch, n, nheads, chunk_len, mimo_rank]) // d_da_source_bnhlm
360            .sum_dim(4) // d_da_source_bnhl1
361            .squeeze_dim::<4>(4); // d_da_source_bnhl
362        let d_da_orange_bnhl = d_da_target_bnhl - d_da_source_bnhl;
363        san(&d_da_orange_bnhl);
364        vec_orange_d_da_bnhl.push(d_da_orange_bnhl);
365    }
366
367    // ── Rejoin the groups along the chunk axis ────────────────────────────
368    let d_v_orange_bnhLMp: F<B, 5> = cat_chunk_groups(vec_orange_d_v_bnhLMp, 1);
369    let d_c_blue_bnhTMr: F<B, 5> = cat_chunk_groups(vec_blue_d_c_bnhTMr, 1);
370    let d_cb_bnhTMLM: F<B, 5> = cat_chunk_groups(vec_d_cb_bnhTMLM, 1);
371    let d_da_blue_bhnl: F<B, 4> = cat_chunk_groups(vec_blue_d_da_bnhl, 1).swap_dims(1, 2);
372    let d_da_orange_bhnl: F<B, 4> = cat_chunk_groups(vec_orange_d_da_bnhl, 1).swap_dims(1, 2);
373    let d_chunk_input_state_bnhpr: F<B, 5> = cat_chunk_groups(vec_d_chunk_input_state_bnhpr, 1);
374
375    // ── K4 backward — the reverse of the state-passing scan ───────────────
376    let (d_intra_chunk_state_bnhpr, d_da_end_bhn, d_initial_state_bhpr) =
377        k4_ssd_state_passing_backward(
378            d_chunk_input_state_bnhpr,
379            chunk_input_state_bnhpr,
380            da_chunk_end_bhn,
381            d_final_bhpr,
382        );
383    // d_da_end: [batch,nheads,nchunks] scattered into the last `l` of d_da_cumsum_k4.
384    let d_da_cumsum_k4_bhnl: F<B, 4> = {
385        let zeros = F::<B, 4>::zeros([batch, nheads, nchunks, chunk_len - 1], &device, dtype);
386        let d_da_end_bhn1 = d_da_end_bhn.unsqueeze_dim::<4>(3);
387        F::cat(vec![zeros, d_da_end_bhn1], 3)
388    };
389
390    // ═══════════════════════════════════════════════════════════════════════
391    // K3 BACKWARD (batched)
392    //
393    // Forward (recap):
394    //   v_bnLMhp        = v.reshape
395    //   b_bnLMhr        = b.reshape
396    //   decay_bhnLM     = exp(cumA_last − cumA)
397    //   decay_bnLMh1    = decay_bhnLM.permute([0,2,3,1]).unsqueeze(4)
398    //   decayed_v_bnLMhp = decay_bnLMh1 · v_bnLMhp                      (elementwise)
399    //   decayed_v_bnhpLM = decayed_v_bnLMhp.permute([0,1,3,4,2])
400    //   b_bnhLMr        = b_bnLMhr.swap_dims(2, 3)
401    //   intra_state    = decayed_v_bnhpLM @ b_bnhLMr
402    // ═══════════════════════════════════════════════════════════════════════
403    let v_bnLMhp =
404        v_bnlmhp
405            .clone()
406            .reshape([batch, nchunks, chunk_len * mimo_rank, nheads, per_head_dim]);
407    let b_bnLMhr =
408        b_bnlmhr
409            .clone()
410            .reshape([batch, nchunks, chunk_len * mimo_rank, nheads, state_rank]);
411    let b_bnhLMr = b_bnLMhr.clone().swap_dims(2, 3);
412    let decayed_v_bnhpLM = k3_decayed_v_bnLMhp.permute([0, 1, 3, 4, 2]);
413
414    let d_decayed_v_bnhpLM: F<B, 5> = d_intra_chunk_state_bnhpr.clone().matmul(
415        b_bnhLMr.transpose(), // b_bnhrLM
416    ); // d_decayed_v_bnhpLM
417    let d_b_k3_bnhLMr: F<B, 5> = decayed_v_bnhpLM
418        .transpose() // decayed_v_bnhLMp
419        .matmul(d_intra_chunk_state_bnhpr); // d_b_k3_bnhLMr
420
421    let d_decayed_v_bnLMhp = d_decayed_v_bnhpLM.permute([0, 1, 4, 2, 3]);
422    let d_decay_bhnLM: F<B, 4> = (d_decayed_v_bnLMhp.clone() * v_bnLMhp)
423        .sum_dim(4) // d_decay_bnLMh1
424        .squeeze_dim::<4>(4) // d_decay_bnLMh
425        .permute([0, 3, 1, 2]); // d_decay_bhnLM
426
427    // d_v_k3_bnLMhp = d_decayed_v · decay (broadcast)
428    let k3_decay_bnLMh1 = k3_decay_bhnLM
429        .clone()
430        .permute([0, 2, 3, 1]) // k3_decay_bnLMh
431        .unsqueeze_dim::<5>(4); // k3_decay_bnLMh1
432    let d_v_k3_bnLMhp: F<B, 5> = d_decayed_v_bnLMhp * k3_decay_bnLMh1;
433    let d_v_k3_bnlrhp: F<B, 6> =
434        d_v_k3_bnLMhp.reshape([batch, nchunks, chunk_len, mimo_rank, nheads, per_head_dim]);
435
436    // d(cumA_last − cumA) = d_decay · decay
437    let d_decay_times_decay_bhnLM = d_decay_bhnLM * k3_decay_bhnLM;
438    // d_a_cumsum_last: Σ over LM (broadcast dim).
439    let d_a_cumsum_last_bhn: F<B, 3> = d_decay_times_decay_bhnLM
440        .clone()
441        .sum_dim(3) // d_decay_times_decay_bhn1
442        .squeeze_dim::<3>(3); // d_a_cumsum_last_bhn
443    // d_a_cumsum: negated (subtraction).
444    let d_da_cumsum_bhnLM = -d_decay_times_decay_bhnLM;
445
446    // Contribution to d_da_cumsum from the fused-cumA expand (sum mimo_rank copies).
447    let d_da_cumsum_k3_from_fused_bhnl: F<B, 4> = d_da_cumsum_bhnLM
448        .reshape([batch, nheads, nchunks, chunk_len, mimo_rank]) // d_da_cumsum_bhnlm
449        .sum_dim(4) // d_da_cumsum_bhnl1
450        .squeeze_dim::<4>(4); // d_da_cumsum_k3_from_fused_bhnl
451    // Contribution from cumA_last: only the last-l position.
452    let d_da_cumsum_k3_from_last_bhnl: F<B, 4> = {
453        let zeros = F::<B, 4>::zeros([batch, nheads, nchunks, chunk_len - 1], &device, dtype);
454        let d_last = d_a_cumsum_last_bhn.unsqueeze_dim::<4>(3);
455        F::cat(vec![zeros, d_last], 3)
456    };
457    let d_da_cumsum_k3_bhnl = d_da_cumsum_k3_from_fused_bhnl + d_da_cumsum_k3_from_last_bhnl;
458
459    // d_b_k3
460    let d_b_k3_bnLMhr = d_b_k3_bnhLMr.swap_dims(2, 3);
461    let d_b_k3_bnlmhr: F<B, 6> =
462        d_b_k3_bnLMhr.reshape([batch, nchunks, chunk_len, mimo_rank, nheads, state_rank]);
463
464    // ═══════════════════════════════════════════════════════════════════════
465    // K2 BACKWARD (batched)
466    //
467    //   cb_bnhLMLM = c_bnhLMr @ b_bnhrLM   (contracts state_rank)
468    //   d_c_bnhLMr = d_cb @ b_bnhLMr      (= d_cb @ b_bnhrLM^T)
469    //   d_b_bnhrLM = c_bnhrLM @ d_cb      (= c_bnhLMr^T @ d_cb)
470    // ═══════════════════════════════════════════════════════════════════════
471    let c_bnhTMr = c_bntmhr
472        .clone()
473        .reshape([batch, nchunks, read, nheads, state_rank]) // c_bnTMhr
474        .swap_dims(2, 3); // c_bnhTMr
475    let b_for_k2_bnhLMr = b_bnLMhr.swap_dims(2, 3);
476
477    let d_c_k2_bnhTMr: F<B, 5> = d_cb_bnhTMLM.clone().matmul(b_for_k2_bnhLMr);
478    let d_b_k2_bnhrLM: F<B, 5> = c_bnhTMr
479        .transpose() // c_bnhrTM
480        .matmul(d_cb_bnhTMLM); // d_b_k2_bnhrLM
481
482    // Undo permutes and reshape back
483    let d_c_k2_bntmhr: F<B, 6> = d_c_k2_bnhTMr
484        .swap_dims(2, 3) // d_c_k2_bnTMhr
485        .reshape([batch, nchunks, chunk_tokens, mimo_rank, nheads, state_rank]); // d_c_k2_bntmhr
486    let d_b_k2_bnlmhr: F<B, 6> = d_b_k2_bnhrLM
487        .permute([0, 1, 4, 2, 3]) // d_b_k2_bnLMhr
488        .reshape([batch, nchunks, chunk_len, mimo_rank, nheads, state_rank]); // d_b_k2_bnlmhr
489
490    // ── Unstack d_c_blue / d_v_orange and reshape back ────────────────────
491    let d_c_blue_bntmhr: F<B, 6> = d_c_blue_bnhTMr
492        .swap_dims(2, 3) // d_c_blue_bnTMhr
493        .reshape([batch, nchunks, chunk_tokens, mimo_rank, nheads, state_rank]); // d_c_blue_bntmhr
494    let d_v_orange_bnlrhp: F<B, 6> = d_v_orange_bnhLMp
495        .swap_dims(2, 3) // d_v_orange_bnLMhp
496        .reshape([batch, nchunks, chunk_len, mimo_rank, nheads, per_head_dim]); // d_v_orange_bnlrhp
497
498    // ═══════════════════════════════════════════════════════════════════════
499    // K1 BACKWARD + SUM CONTRIBUTIONS
500    // ═══════════════════════════════════════════════════════════════════════
501    let d_da_cumsum_bhnl =
502        d_da_blue_bhnl + d_da_orange_bhnl + d_da_cumsum_k3_bhnl + d_da_cumsum_k4_bhnl;
503    san(&d_da_cumsum_bhnl);
504
505    // K1 inverse: da_cumsum[l] = cumsum(da)[l]  →  d_da[l] = Σ_{k ≥ l} d_da_cumsum[k]
506    //
507    // Suffix sum:  d_da[l] = total_sum − cumsum(d_da_cumsum)[l-1] (cumsum[-1] = 0).
508    let d_da_bhnl = {
509        let d_total_bhnl = d_da_cumsum_bhnl
510            .clone()
511            .sum_dim(3) // d_da_cumsum_bhn1
512            .expand([batch, nheads, nchunks, chunk_len]); // d_total_bhnl
513        let prefix_bhnl = d_da_cumsum_bhnl.cumsum(3);
514        let zeros_bhn1 = F::<B, 4>::zeros([batch, nheads, nchunks, 1], &device, dtype);
515        let prefix_shifted_bhnl =
516            F::cat(vec![zeros_bhn1, prefix_bhnl.narrow(3, 0, chunk_len - 1)], 3);
517        d_total_bhnl - prefix_shifted_bhnl
518    };
519    san(&d_da_bhnl);
520    // Undo permute
521    let d_da_bnlh = d_da_bhnl.permute([0, 2, 3, 1]);
522
523    // ── Combine per-input gradient contributions ──────────────────────────
524    let d_v_bnlmhp = d_v_k3_bnlrhp + d_v_orange_bnlrhp;
525    let d_b_bnlmhr = d_b_k2_bnlmhr + d_b_k3_bnlmhr;
526    let d_c_bntmhr = d_c_k2_bntmhr + d_c_blue_bntmhr;
527
528    san(&d_v_bnlmhp);
529    san(&d_da_bnlh);
530    san(&d_b_bnlmhr);
531    san(&d_c_bntmhr);
532    san(&d_initial_state_bhpr);
533
534    CombinedGrads {
535        d_v_bnlmhp,
536        d_da_bnlh,
537        d_b_bnlmhr,
538        d_c_bntmhr,
539        d_initial_state_bhpr,
540    }
