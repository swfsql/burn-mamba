1//! # `RecomputeOp` of the Mamba-3 double-SSD
2//!
3//! Implements [`Mamba3DoubleSsdBackendExt`] for `Autodiff<B>` with one
4//! [recompute node](burn_stack::utils::recompute_node). The node saves the
5//! inputs of the forward and not its large intermediates. The backward replays
6//! the serial kernels and the gradient math of the
7//! [standard SSD scan](crate::standard_ssd).
8//!
9//! Without resets, `origin_bhpr` is a placeholder with one element
10//! (`packing::placeholder`), which the node saves at no real cost.
11
12use crate::mamba3::double_ssd::ssd::serial_recalculated::Mamba3DoubleSsdBackendExt;
13use crate::packing::OpResets;
14use crate::standard_ssd::combined_backward;
15use burn::backend::Backend;
16use burn::backend::autodiff::{Autodiff, checkpoint::strategy::CheckpointStrategy};
17use burn::backend::tensor::FloatTensor;
18use burn_stack::utils::fprim::F;
19use burn_stack::utils::recompute_node::{Prim, RecomputeOp, recompute_pair};
20
21burn_stack::recompute_inputs!(Inputs {
22    v_bnlmhp,
23    da_bnlh,
24    b_bnlmhr,
25    c_bntmhr,
26    initial_state_bhpr,
27    origin_bhpr,
28});
29
30/// The values that the op takes with no gradient.
31#[derive(Clone, Debug)]
32struct Extra<B: Backend> {
33    read_stride: usize,
34    /// `None` without resets.
35    keep_bn: Option<Prim<B>>,
36}
37
38/// The [`RecomputeOp`] of the Mamba-3 double-SSD.
39struct SsdRecompute;
40
41impl<B: Backend + Mamba3DoubleSsdBackendExt> RecomputeOp<B, 6> for SsdRecompute {
42    type Inputs<T> = Inputs<T>;
43    type Extra = Extra<B>;
44
45    fn forward(inputs: Inputs<Prim<B>>, extra: &Extra<B>) -> (Prim<B>, Prim<B>) {
46        B::double_ssd_serial_recalculated(
47            inputs.v_bnlmhp,
48            inputs.da_bnlh,
49            inputs.b_bnlmhr,
50            inputs.c_bntmhr,
51            inputs.initial_state_bhpr,
52            extra.read_stride,
53            OpResets::from_recompute(extra.keep_bn.clone(), inputs.origin_bhpr),
54        )
55    }
56
57    fn backward(
58        d_y_bntmhp: Prim<B>,
59        d_final_state_bhpr: Prim<B>,
60        inputs: Inputs<Prim<B>>,
61        extra: Extra<B>,
62    ) -> Inputs<Option<Prim<B>>> {
63        let grads = combined_backward::<B>(
64            F::new(d_y_bntmhp),
65            F::new(d_final_state_bhpr),
66            F::new(inputs.v_bnlmhp),
67            F::new(inputs.da_bnlh),
68            F::new(inputs.b_bnlmhr),
69            F::new(inputs.c_bntmhr),
70            F::new(inputs.initial_state_bhpr),
71            extra.read_stride,
72            OpResets::from_recompute(extra.keep_bn, inputs.origin_bhpr).into_option(),
73        );
74        Inputs {
75            v_bnlmhp: Some(grads.d_v_bnlmhp.inner()),
76            da_bnlh: Some(grads.d_da_bnlh.inner()),
77            b_bnlmhr: Some(grads.d_b_bnlmhr.inner()),
78            c_bntmhr: Some(grads.d_c_bntmhr.inner()),
79            initial_state_bhpr: Some(grads.d_initial_state_bhpr.inner()),
80            origin_bhpr: grads.d_origin_bhpr.map(F::inner),
81        }
82    }
83}
84
85impl<B: Backend + Mamba3DoubleSsdBackendExt, C: CheckpointStrategy> Mamba3DoubleSsdBackendExt
86    for Autodiff<B, C>
87{
88    fn double_ssd_serial_recalculated(
89        v_bnlmhp: FloatTensor<Self>,
90        da_bnlh: FloatTensor<Self>,
91        b_bnlmhr: FloatTensor<Self>,
92        c_bntmhr: FloatTensor<Self>,
93        initial_state_bhpr: FloatTensor<Self>,
94        read_stride: usize,
95        resets: OpResets<Self>,
96    ) -> (FloatTensor<Self>, FloatTensor<Self>) {
97        let (keep_bn, origin_bhpr) = resets.into_recompute(&v_bnlmhp);
98        recompute_pair::<B, C, SsdRecompute, _>(
99            Inputs { v_bnlmhp, da_bnlh, b_bnlmhr, c_bntmhr, initial_state_bhpr, origin_bhpr },
100            Extra { read_stride, keep_bn },
101        )
102    }
103}
