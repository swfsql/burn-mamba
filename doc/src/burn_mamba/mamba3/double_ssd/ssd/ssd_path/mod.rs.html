1//! # SSD input bundle for the Mamba-3 double-SSD pathway
2//!
3//! [`Mamba3DoubleSsdInput`] bundles the pre-processed tensors that one standard
4//! SSD pass uses. [`Mamba3DoubleSsdInput::run`] runs the SSD path that the
5//! shared [`SsdPath`](crate::ssd_path::SsdPath) selects.
6
7use crate::mamba3::prelude::*;
8use burn::prelude::*;
9
10/// MIMO-first SSD input.
11///
12/// All tensors are pre-processed:
13///
14/// - B/C are QK-normed, bias-added, rotated, and expanded to per-head (not
15///   per-group),
16/// - V is scaled by the trapezoidal coefficient (γ or β) of its pass,
17/// - the log-decay `da` (`Δ·A`, or the Kalman decay) is pre-computed.
18///
19/// The caller adds the D skip.
20pub struct Mamba3DoubleSsdInput {
21    /// Value tensor, already scaled by the trapezoidal coefficient (γ or β).
22    ///
23    /// # Shape
24    /// - `[batch, nchunks, chunk_len, mimo_rank, nheads, per_head_dim]`
25    pub v_bnlmhp: Tensor<6>,
26
27    /// Pre-computed log-decay (negative): `Δ·A`, or the Kalman decay.
28    ///
29    /// # Shape
30    /// - `[batch, nchunks, chunk_len, nheads]`
31    pub da_bnlh: Tensor<4>,
32
33    /// Key/B tensor: QK-normed, bias-added, rotated, expanded to per-head,
34    /// per-rank.
35    ///
36    /// # Shape
37    /// - `[batch, nchunks, chunk_len, mimo_rank, nheads, state_rank]`
38    pub b_bnlmhr: Tensor<6>,
39
40    /// Query/C tensor: the same processing as B, but on the **read** axis of
41    /// the chunk. That is one position per token, not per write position, because
42    /// the readout reads the state once for each token, at its last micro-step.
43    /// See
44    /// [`Mamba3SingleSsdInput::c_bntmhr`](crate::mamba3::single_ssd::ssd::Mamba3SingleSsdInput)
45    /// and [the read axis](crate::mamba3::product).
46    ///
47    /// # Shape
48    /// - `[batch, nchunks, chunk_tokens, mimo_rank, nheads, state_rank]`,
49    ///   `chunk_tokens = chunk_len / read_stride`
50    pub c_bntmhr: Tensor<6>,
51
52    /// The read stride of the chunk: `micro_steps`, the write positions per
53    /// read position. `1` for stock Mamba-3, where the two axes are equal.
54    pub read_stride: usize,
55
56    /// Initial state.
57    ///
58    /// # Shape
59    /// - `[batch, nheads, per_head_dim, state_rank]`
60    pub initial_state_bhpr: Tensor<4>,
61
62    /// Optional learnable initial state (broadcast over batch). Only the
63    /// `Minimal` path supports it: the serial paths panic when it is `Some`.
64    ///
65    /// # Shape
66    /// - `[nheads, per_head_dim, state_rank]`
67    pub init_state_hpr: Option<Tensor<3>>,
68
69    /// The chunks that start a new segment of a packed row, and the origin
70    /// state. The start state of each of these chunks is the origin state
71    /// (see [`crate::packing`]). The call itself starts from
72    /// `initial_state_bhpr`.
73    /// `None` ⇒ no reset. Only the two serial paths take it. `Minimal` panics
74    /// when it is `Some`.
75    pub resets: Option<crate::packing::SsdResets>,
76}
77
78impl Mamba3DoubleSsdInput {
79    /// Run the [`NaN`/`Inf` guards](burn_stack::modules::misc::sanity) on every input tensor.
80    pub fn sanity(&self) {
81        use burn_stack::modules::sanity as san;
82        san(&self.v_bnlmhp);
83        san(&self.da_bnlh);
84        san(&self.b_bnlmhr);
85        san(&self.c_bntmhr);
86        san(&self.initial_state_bhpr);
87        if let Some(ref init_state_hpr) = self.init_state_hpr {
88            san(init_state_hpr);
89        }
90    }
91}
92
93impl Mamba3DoubleSsdInput {
94    /// Run the selected double-ssd SSD path on this MIMO-first input:
95    /// `double_ssd_minimal`, `double_ssd_serial`, or
96    /// `double_ssd_serial_recalculated`, by [`SsdPath`] variant. The chunk
97    /// length is already in the tensor shapes.
98    ///
99    /// # Returns
100    /// - `y_bntmhp`: `[batch, nchunks, chunk_tokens, mimo_rank, nheads, per_head_dim]`,
101    ///   at token resolution, the resolution of the readout (see [`Self::c_bntmhr`])
102    /// - `final_state_bhpr`: `[batch, nheads, per_head_dim, state_rank]`
103    pub fn run(self, ssd_path: SsdPath) -> (Tensor<6>, Tensor<4>) {
104        match ssd_path {
105            SsdPath::Minimal => {
106                assert!(
107                    self.resets.is_none(),
108                    "Minimal does not take resets: use Serial or SerialRecalculated"
109                );
110                self.double_ssd_minimal()
111            }
112            SsdPath::Serial => self.double_ssd_serial(),
113            SsdPath::SerialRecalculated => self.double_ssd_serial_recalculated(),
114        }
115    }
116}
