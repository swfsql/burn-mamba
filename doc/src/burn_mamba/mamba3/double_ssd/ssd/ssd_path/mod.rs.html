1//! # SSD input bundle for the Mamba-3 double-SSD pathway
2//!
3//! [`Mamba3DoubleSsdInput`] bundles the pre-processed tensors that one standard
4//! SSD pass uses. [`Mamba3DoubleSsdInput::run`] calls the algorithm that the
5//! shared [`Mamba3SsdPath`] selects.
6//!
7//! [`Mamba3SsdPath`]: crate::mamba3::ssd_path::Mamba3SsdPath
8
9use crate::mamba3::prelude::*;
10use burn::prelude::*;
11
12/// MIMO-first SSD input.
13///
14/// All tensors are pre-processed:
15///
16/// - B/C are QK-normed, bias-added, rotated, and expanded to per-head (not
17///   per-group),
18/// - V is scaled by the trapezoidal coefficient (γ or β) of its pass,
19/// - the log-decay `da` (`Δ·A`, or the Kalman decay) is pre-computed.
20///
21/// The caller adds the D skip.
22pub struct Mamba3DoubleSsdInput {
23    /// Value tensor, already scaled by the trapezoidal coefficient (γ or β).
24    ///
25    /// # Shape
26    /// - `[batch, nchunks, chunk_len, mimo_rank, nheads, per_head_dim]`
27    pub v_bnlmhp: Tensor<6>,
28
29    /// Pre-computed log-decay (negative): `Δ·A`, or the Kalman decay.
30    ///
31    /// # Shape
32    /// - `[batch, nchunks, chunk_len, nheads]`
33    pub da_bnlh: Tensor<4>,
34
35    /// Key/B tensor: QK-normed, bias-added, rotated, expanded to per-head,
36    /// per-rank.
37    ///
38    /// # Shape
39    /// - `[batch, nchunks, chunk_len, mimo_rank, nheads, state_rank]`
40    pub b_bnlmhr: Tensor<6>,
41
42    /// Query/C tensor: the same processing as B, but on the **read** axis of
43    /// the chunk. That is one row per token, not per folded position, because
44    /// a token is read once, at its last micro-step. See
45    /// [`Mamba3SingleSsdInput::c_bntmhr`](crate::mamba3::single_ssd::ssd::Mamba3SingleSsdInput)
46    /// and [the read axis](crate::mamba3::product).
47    ///
48    /// # Shape
49    /// - `[batch, nchunks, chunk_tokens, mimo_rank, nheads, state_rank]`,
50    ///   `chunk_tokens = chunk_len / read_stride`
51    pub c_bntmhr: Tensor<6>,
52
53    /// The read stride of the chunk: `micro_steps`, the folded positions per
54    /// read row. `1` for stock Mamba-3, where the two axes are equal.
55    pub read_stride: usize,
56
57    /// Initial SSM hidden state.
58    ///
59    /// # Shape
60    /// - `[batch, nheads, per_head_dim, state_rank]`
61    pub initial_state_bhpr: Tensor<4>,
62
63    /// Optional learnable initial state (broadcast over batch). Only the
64    /// `Minimal` path supports it: the serial paths panic when it is `Some`.
65    ///
66    /// # Shape
67    /// - `[nheads, per_head_dim, state_rank]`
68    pub init_state_hpr: Option<Tensor<3>>,
69}
70
71impl Mamba3DoubleSsdInput {
72    /// Run the [`NaN`/`Inf` guards](burn_stack::modules::misc::sanity) on every input tensor.
73    pub fn sanity(&self) {
74        use burn_stack::modules::sanity as san;
75        san(&self.v_bnlmhp);
76        san(&self.da_bnlh);
77        san(&self.b_bnlmhr);
78        san(&self.c_bntmhr);
79        san(&self.initial_state_bhpr);
80        if let Some(ref init_state_hpr) = self.init_state_hpr {
81            san(init_state_hpr);
82        }
83    }
84}
85
86impl Mamba3DoubleSsdInput {
87    /// Run the selected double-ssd algorithm on this MIMO-first input:
88    /// `double_ssd_minimal`, `double_ssd_serial`, or
89    /// `double_ssd_serial_recalculated`, by [`Mamba3SsdPath`] variant.
90    ///
91    /// # Returns
92    /// - `y_bntmhp`: `[batch, nchunks, chunk_tokens, mimo_rank, nheads, per_head_dim]`,
93    ///   at token resolution, the resolution of the readout (see [`Self::c_bntmhr`])
94    /// - `final_state_bhpr`: `[batch, nheads, per_head_dim, state_rank]`
95    pub fn run(self, path: &Mamba3SsdPath) -> (Tensor<6>, Tensor<4>) {
96        match path {
97            Mamba3SsdPath::Minimal(_) => self.double_ssd_minimal(),
98            Mamba3SsdPath::Serial(_) => self.double_ssd_serial(),
99            Mamba3SsdPath::SerialRecalculated(_) => self.double_ssd_serial_recalculated(),
100        }
101    }
102}
