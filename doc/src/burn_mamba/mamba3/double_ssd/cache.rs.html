1//! # Mamba-3 Inference Caches
2//!
3//! During autoregressive (token-by-token) generation, three pieces of state
4//! must be preserved between calls:
5//!
6//! 1. **SSM hidden state** — `hₜ ∈ ℝ^{per_head_dim×state_rank}` per head, compressed context.
7//! 2. **Previous K state** — the last `lag` positions' `B`, oldest first
8//!    `[batch, lag, mimo_rank, nheads, state_rank]`, needed for the β term of
9//!    the (double-ssd) trapezoidal recurrence.
10//! 3. **Previous V state** — the matching `x`
11//!    `[batch, lag, nheads, per_head_dim]`, paired with k_state to reconstruct
12//!    `β Bₚ₋ₗₐ₉ ⊗ xₚ₋ₗₐ₉`.
13//!    Both are the trapezoid's **tap slots** and exist only when the block's
14//!    [`Trapezoid`](crate::mamba3::trapezoid::Trapezoid) has a β tap; the FIFO is
15//!    as deep as its [`tap_lag`](crate::mamba3::trapezoid::Trapezoid::tap_lag)
16//!    (`1` for the default, `u` for
17//!    [`Trapezoid::Vertical`](crate::mamba3::trapezoid::Trapezoid::Vertical)).
18//! 4. **Cumulative RoPE angle** — the accumulated rotation angle up to position
19//!    `t`, needed to correctly continue data-dependent rotary embeddings.
20//!
21//! Note: Mamba-3 has **no conv cache** (the short 1-dimensional convolution present in
22//! Mamba-3 is removed; its role is absorbed by the trapezoidal discretization
23//! and the learnable B/C biases).
24

31// Mamba3DoubleSsdCaches  (one cache entry per layer)

34/// A collection of per-layer caches for a complete Mamba-3 network.
35#[derive(Module, Debug)]
36pub struct Mamba3DoubleSsdCaches {
37    /// Per-layer caches.  Length equals the number of virtual layers.
38    pub caches: Vec<Mamba3DoubleSsdCache>,
39}
40
41/// Configuration / factory for [`Mamba3DoubleSsdCaches`].
42#[derive(Config, Debug)]
43pub struct Mamba3DoubleSsdCachesConfig {
44    /// Number of cache slots (= number of virtual layers).
45    pub n_real_caches: usize,
46
47    /// Shared configuration that determines the shape of each cache.
48    pub cache: Mamba3DoubleSsdCacheConfig,
49}
50
51impl Mamba3DoubleSsdCachesConfig {
52    /// Convenience constructor from a block config.
53    pub fn new_from_block_config(
54        n_real_caches: usize,
55        batch: usize,
56        block_config: Mamba3Config,
57    ) -> Self {
58        Self {
59            n_real_caches,
60            cache: Mamba3DoubleSsdCacheConfig::new_from_block_config(batch, block_config),
61        }
62    }
63
64    /// Allocate all cache tensors (zero-initialised) on `device`.
65    pub fn init(&self, device: &Device) -> Mamba3DoubleSsdCaches {
66        let caches = (0..self.n_real_caches)
67            .map(|_| self.cache.clone().init(device))
68            .collect();
69        Mamba3DoubleSsdCaches { caches }
70    }
71}

73// ---------------------------------------------------------------------------
74// Mamba3DoubleSsdCache  (state for a single layer)
75// ---------------------------------------------------------------------------
76
77/// The mutable state carried between decoding steps for a **single** Mamba-3 layer.
78///
79/// All tensors are updated at every call to [`crate::mamba3::mamba3::Mamba3::step`].
80#[derive(Module, Debug)]
81pub struct Mamba3DoubleSsdCache {
82    /// **SSM hidden state** `hₜ`.
83    ///
84    /// Updated via the (double-ssd) trapezoidal recurrence:
85    /// `hₜ = αₜ hₜ₋₁ + βₜ (sumₘ Kₜ₋₁[m] ⊗ (Vₜ₋₁ * mimo_x[m])) + γₜ (sumₘ Bₜ[m] ⊗ (xₜ * mimo_x[m]))`
86    ///
87    /// Shape: `[batch, nheads, per_head_dim, state_rank]`
88    pub ssm_bhpr: Tensor<4>,
89
90    /// **The tap FIFO's B**, one slot per lagged position, **oldest first**.
91    ///
92    /// Used to reconstruct the β term: `β * sumₘ Bₚ₋ₗₐ₉[m] ⊗ (xₚ₋ₗₐ₉ * mimo_x[m])`.
93    /// Stored **as rotated**, at its own position: the relative-rotation
94    /// factoring `C̄ₜᵀB̄ₛ` then reconstructs the transport, so no rotation is
95    /// re-applied on the next call.
96    ///
97    /// The trapezoid's tap slots: `None` — allocated nowhere, not zeroed — under
98    /// [`Trapezoid::None`], which has no β term. Present exactly when
99    /// [`Self::v_state_buhp`] is (see [`Trapezoid::tap_slots`]).
100    ///
101    /// Shape: `[batch, tap_slots, mimo_rank, nheads, state_rank]`
102    pub k_state_bumhr: Option<Tensor<5>>,
103
104    /// **The tap FIFO's x**, matching [`Self::k_state_bumhr`] slot for slot, and
105    /// `None` with it.
106    ///
107    /// Each slot is **pre-scaled by the decay it has accumulated since its own
108    /// position** (`helpers::tail_decay`), which is what carries a lag-`u` tap's
109    /// gap transport across the call boundary. At `lag = 1` that product is
110    /// empty, so the slot is the plain `xₜ₋₁`.
111    ///
112    /// Shape: `[batch, tap_slots, nheads, per_head_dim]`
113    pub v_state_buhp: Option<Tensor<4>>,
114
115    /// **Cumulative data-dependent rotation** up to the current position
116    /// ([`RotationState`]): the abelian RoPE angle for
117    /// [`Complex2D`](crate::mamba3::rotation::RotationKind::Complex2D) (each step
118    /// `cum_angleₜ = cum_angleₜ₋₁ + Δₜ · tanh(θₜ) · π`), or the cumulative unit
119    /// quaternion for [`Quaternion4D`](crate::mamba3::rotation::RotationKind::Quaternion4D).
120    ///
121    /// Starts at the identity for fresh sequences; continued across calls for
122    /// streaming.
123    pub rotation: RotationState,
124}
125
126impl Mamba3DoubleSsdCache {
127    /// Run the [`NaN`/`Inf` guards](burn_stack::modules::misc::sanity) on every cached tensor.
128    pub fn sanity(&self) {
129        san(&self.ssm_bhpr);
130        assert_eq!(
131            self.k_state_bumhr.is_some(),
132            self.v_state_buhp.is_some(),
133            "the trapezoid's tap slots are present or absent together"
134        );
135        if let Some(k_state_bumhr) = &self.k_state_bumhr {
136            san(k_state_bumhr);
137        }
138        if let Some(v_state_buhp) = &self.v_state_buhp {
139            san(v_state_buhp);
140        }
141        self.rotation.sanity();
142    }
143}

145/// Configuration / factory for a single [`Mamba3DoubleSsdCache`].
146#[derive(Config, Debug)]
147pub struct Mamba3DoubleSsdCacheConfig {
148    /// Batch size.
149    pub batch: usize,
150
151    /// State rank.
152    #[config(default = 128)]
153    pub state_rank: usize,
154
155    /// Head dimension per_head_dim.
156    #[config(default = 64)]
157    pub per_head_dim: usize,
158
159    /// Number of SSM heads.
160    pub nheads: usize,
161
162    /// MIMO rank.  1 = SISO.
163    #[config(default = 1)]
164    pub mimo_rank: usize,
165
166    /// Number of RoPE angle pairs = `rope_dim / 2` = `(state_rank * rope_fraction) / 2`
167    /// (rounded down to even via `Mamba3Config::rope_dim`).
168    pub num_rope_angles: usize,
169
170    /// Which transition rotation the block uses ([`RotationKind`]); selects the
171    /// accumulator variant — [`RotationState::Quaternion`] for
172    /// [`RotationKind::Quaternion4D`], [`RotationState::Rotor`] for
173    /// [`RotationKind::Rotor4D`], else [`RotationState::Angle`].
174    #[config(default = "crate::mamba3::rotation::RotationKind::Complex2D")]
175    pub rotation: RotationKind,
176
177    /// Number of quaternion blocks (`rope_dim / 4`); only used for
178    /// [`RotationKind::Quaternion4D`] / [`RotationKind::Rotor4D`].
179    #[config(default = 1)]
180    pub num_quat_blocks: usize,
181
182    /// The block's tap pattern ([`Trapezoid`]) — it decides whether the tap
183    /// slots exist at all, and with [`Self::micro_steps`] how many there are
184    /// (see [`Trapezoid::tap_slots`]).
185    #[config(default = "crate::mamba3::trapezoid::Trapezoid::HorizontalCarryOver")]
186    pub trapezoid: Trapezoid,
187
188    /// Recurrence micro-steps per token (`u`); only the lag-`u` tap patterns
189    /// read it (see [`Trapezoid::tap_lag`]).
190    #[config(default = 1)]
191    pub micro_steps: usize,
192}
193
194impl Mamba3DoubleSsdCacheConfig {
195    /// Derive cache shapes from a Mamba-3 block configuration plus a batch size.
196    pub fn new_from_block_config(batch: usize, block_config: Mamba3Config) -> Self {
197        Self {
198            batch,
199            state_rank: block_config.state_rank,
200            per_head_dim: block_config.per_head_dim,
201            nheads: block_config.nheads(),
202            mimo_rank: block_config.mimo_rank,
203            num_rope_angles: block_config.num_rope_angles(),
204            rotation: block_config.rotation,
205            num_quat_blocks: block_config.num_quat_blocks(),
206            trapezoid: block_config.trapezoid,
207            micro_steps: block_config.micro_steps,
208        }
209    }
210
211    /// Allocate zero/identity-initialised cache tensors on `device`.
212    pub fn init(&self, device: &Device) -> Mamba3DoubleSsdCache {
213        let ssm_bhpr = Tensor::zeros(
214            [self.batch, self.nheads, self.per_head_dim, self.state_rank],
215            device,
216        );
217        let slots = self.trapezoid.tap_slots(self.micro_steps);
218        let tap = slots > 0;
219        let k_state_bumhr = tap.then(|| {
220            Tensor::zeros(
221                [
222                    self.batch,
223                    slots,
224                    self.mimo_rank,
225                    self.nheads,
226                    self.state_rank,
227                ],
228                device,
229            )
230        });
231        let v_state_buhp = tap.then(|| {
232            Tensor::zeros(
233                [self.batch, slots, self.nheads, self.per_head_dim],
234                device,
235            )
236        });
237        let rotation = RotationState::identity(
238            self.rotation,
239            self.batch,
240            self.nheads,
241            self.num_rope_angles,
242            self.num_quat_blocks,
243            device,
244        );
245        Mamba3DoubleSsdCache {
246            ssm_bhpr,
247            k_state_bumhr,
248            v_state_buhp,
249            rotation,
250        }
251    }
