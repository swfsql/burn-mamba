1//! # Mamba-3 — In-Projection Layout
2//!
3//! One table holds the channel groups of the in-projection. Each row gives the
4//! key, the name, the width, the axis, the optimizer and the place of one
5//! group. The widths of the in-projection, the Muon plan and the split read
6//! this table. A new optional group needs one [`GroupKey`] variant, one row,
7//! and one field of [`InProjStreams`](crate::mamba3::plant_frame::InProjStreams)
8//! with its `take` line.
9//!
10//! The doc of [`Mamba3::in_proj`](crate::mamba3::mamba3::Mamba3::in_proj) gives
11//! the layout in full.
12//!
13//! Burn has no zero-width tensors, and `split_with_sizes` drops a part of zero
14//! length. So the table lists only the groups that the block projects, and each
15//! row has a width above zero. Each row then has exactly one part of the split.
16
17use crate::mamba3::positive::{Gain, Tropical};
18use crate::mamba3::product::unfold_micro_bs;
19use crate::mamba3::tap_pattern::TapPattern;
20use burn::prelude::*;

22/// The key of a channel group. There is one variant for each group, in layout
23/// order.
24#[derive(Clone, Copy, Debug, PartialEq, Eq)]
25pub(crate) enum GroupKey {
26    Z,
27    X,
28    BRaw,
29    CRaw,
30    DdDt,
31    DdARaw,
32    LambdaRaw,
33    MuRaw,
34    Rotation,
35    KalmanNoise,
36    TropicalA,
37    TropicalB,
38}
39
40/// The axis of the stream of a channel group.
41#[derive(Clone, Copy, Debug, PartialEq, Eq)]
42pub(crate) enum GroupAxis {
43    /// One sub-projection. The stream is on the read axis.
44    Token,
45    /// One sub-projection for each micro-step. The stream is on the write axis.
46    MicroStep,
47}
48
49impl GroupAxis {
50    /// The number of sub-projections of a group on this axis.
51    fn sub_projections(self, micro_steps: usize) -> usize {
52        match self {
53            Self::Token => 1,
54            Self::MicroStep => micro_steps,
55        }
56    }
57}
58
59/// The optimizer of a channel group.
60#[derive(Clone, Copy, Debug, PartialEq, Eq)]
61#[cfg_attr(not(feature = "optim"), allow(dead_code))]
62pub(crate) enum GroupOptimizer {
63    /// Muon orthogonalises each sub-projection of the group.
64    Muon,
65    /// The group stays on the fallback optimizer.
66    Fallback,
67}
68
69/// The weight that holds a channel group under
70/// [`Mamba3Untied::InProjTail`](crate::mamba3::mamba3::Mamba3Untied::InProjTail).
71#[derive(Clone, Copy, Debug, PartialEq, Eq)]
72pub(crate) enum GroupPlace {
73    /// `in_proj`.
74    InProj,
75    /// `in_proj_tail`.
76    InProjTail,
77}

79/// One row of the table.
80#[derive(Clone, Copy, Debug)]
81#[cfg_attr(not(feature = "optim"), allow(dead_code))]
82pub(crate) struct ChannelGroup {
83    pub key: GroupKey,
84    pub name: &'static str,
85    /// The channels of one micro-step, or of the token for a per-token group.
86    pub width: usize,
87    pub axis: GroupAxis,
88    pub optimizer: GroupOptimizer,
89    pub place: GroupPlace,
90}

92/// The settings of a block that set the table. `Mamba3Config` and `Mamba3` both
93/// hold them.
94pub(crate) struct InProjSettings {
95    pub d_inner: usize,
96    pub nheads: usize,
97    /// `ngroups · state_rank · mimo_rank`.
98    pub bc_width: usize,
99    pub num_rotation_channels: usize,
100    pub micro_steps: usize,
101    pub tap_pattern: TapPattern,
102    pub gain: Gain,
103    pub tropical: Tropical,
104}

106/// The channel groups that a block projects, in layout order.
107pub(crate) struct InProjLayout {
108    groups: Vec<ChannelGroup>,
109    micro_steps: usize,
110}
111
112impl InProjLayout {
113    /// The table of a block with these settings.
114    ///
115    /// # Panics
116    /// If a group that the block projects has a width of zero.
117    pub(crate) fn new(settings: InProjSettings) -> Self {
118        use GroupAxis::{MicroStep, Token};
119        use GroupKey::*;
120        use GroupOptimizer::{Fallback, Muon};
121        use GroupPlace::{InProj, InProjTail};
122        let InProjSettings {
123            d_inner,
124            nheads,
125            bc_width,
126            num_rotation_channels,
127            micro_steps,
128            tap_pattern,
129            gain,
130            tropical,
131        } = settings;
132
133        // The first column is the presence condition of the group.
134        #[rustfmt::skip]
135        let rows = [
136            (true,                                  Z,           "z",            d_inner,               Token,     Muon,     InProj),
137            (true,                                  X,           "x",            d_inner,               MicroStep, Muon,     InProj),
138            (true,                                  BRaw,        "b_raw",        bc_width,              MicroStep, Muon,     InProj),
139            (true,                                  CRaw,        "c_raw",        bc_width,              Token,     Muon,     InProj),
140            (true,                                  DdDt,        "dd_dt",        nheads,                MicroStep, Fallback, InProjTail),
141            (true,                                  DdARaw,      "dd_a_raw",     nheads,                MicroStep, Fallback, InProjTail),
142            (tap_pattern.has_beta_tap(),            LambdaRaw,   "lambda_raw",   nheads,                MicroStep, Fallback, InProjTail),
143            (tap_pattern.has_two_taps(micro_steps), MuRaw,       "mu_raw",       nheads,                MicroStep, Fallback, InProjTail),
144            (num_rotation_channels > 0,             Rotation,    "rotation",     num_rotation_channels, MicroStep, Muon,     InProjTail),
145            (gain.projects_noise(),                 KalmanNoise, "kalman_noise", nheads,                MicroStep, Fallback, InProjTail),
146            (tropical.is_on(),                      TropicalA,   "tropical_a",   nheads,                MicroStep, Fallback, InProjTail),
147            (tropical.is_on(),                      TropicalB,   "tropical_b",   nheads,                MicroStep, Fallback, InProjTail),
148        ];
149        let groups: Vec<ChannelGroup> = rows
150            .into_iter()
151            .filter(|row| row.0)
152            .map(|(_, key, name, width, axis, optimizer, place)| ChannelGroup {
153                key,
154                name,
155                width,
156                axis,
157                optimizer,
158                place,
159            })
160            .collect();
161        for group in &groups {
162            assert!(group.width > 0, "the channel group {} has a width of zero", group.name);
163        }
164        Self { groups, micro_steps }
165    }
166
167    /// The channels of `group` over all of its sub-projections.
168    fn channels(&self, group: &ChannelGroup) -> usize {
169        group.width * group.axis.sub_projections(self.micro_steps)
170    }
171
172    /// The output width of the in-projection.
173    pub(crate) fn d_in_proj(&self) -> usize {
174        self.groups.iter().map(|group| self.channels(group)).sum()
175    }
176
177    /// The output width of the groups at [`GroupPlace::InProjTail`].
178    pub(crate) fn d_in_proj_tail(&self) -> usize {
179        self.groups
180            .iter()
181            .filter(|group| group.place == GroupPlace::InProjTail)
182            .map(|group| self.channels(group))
183            .sum()
184    }
185
186    /// The channels of the group `key`, or `0` when the block does not project
187    /// it.
188    #[cfg(all(test, feature = "_dev-test"))]
189    pub(crate) fn channels_of(&self, key: GroupKey) -> usize {
190        self.groups
191            .iter()
192            .find(|group| group.key == key)
193            .map_or(0, |group| self.channels(group))
194    }
195
196    /// The sub-projections of the groups at `place`, one for each micro-step
197    /// of a per-micro-step group.
198    #[cfg(feature = "optim")]
199    pub(crate) fn sub_projections(
200        &self,
201        place: GroupPlace,
202    ) -> Vec<burn_stack::optim::SubProjection> {
203        use burn_stack::optim::SubProjection;
204        self.groups
205            .iter()
206            .filter(|group| group.place == place)
207            .flat_map(|group| {
208                let sub = match group.optimizer {
209                    GroupOptimizer::Muon => SubProjection::muon(group.name, group.width),
210                    GroupOptimizer::Fallback => SubProjection::adamw(group.name, group.width),
211                };
212                std::iter::repeat_n(sub, group.axis.sub_projections(self.micro_steps))
213            })
214            .collect()
215    }
216
217    /// Split the output of the in-projection into the stream of each group.
218    ///
219    /// The stream of a per-micro-step group is on the write axis. See
220    /// [`crate::mamba3::product`].
221    ///
222    /// # Shapes
223    /// - `proj_btd` : `[batch, tokens, d_in_proj]`
224    pub(crate) fn split(&self, proj_btd: Tensor<3>) -> InProjParts {
225        let sizes = self.groups.iter().map(|group| self.channels(group)).collect();
226        let parts = proj_btd
227            .split_with_sizes(sizes, 2)
228            .into_iter()
229            .zip(&self.groups)
230            .map(|(part, group)| {
231                let stream = match group.axis {
232                    GroupAxis::Token => part,
233                    GroupAxis::MicroStep => unfold_micro_bs(part, self.micro_steps),
234                };
235                (group.key, stream)
236            })
237            .collect();
238        InProjParts { parts }
239    }
240}
241
242/// The streams of the channel groups of one in-projection output.
243pub(crate) struct InProjParts {
244    parts: Vec<(GroupKey, Tensor<3>)>,
245}
246
247impl InProjParts {
248    /// Remove the stream of `key`. `None` when the block does not project the
249    /// group.
250    pub(crate) fn take(&mut self, key: GroupKey) -> Option<Tensor<3>> {
251        let at = self.parts.iter().position(|(part_key, _)| *part_key == key)?;
252        Some(self.parts.swap_remove(at).1)
253    }
254
255    /// Remove the stream of `key`, a group that each block projects.
256    ///
257    /// # Panics
258    /// If the table has no group `key`.
259    pub(crate) fn take_fixed(&mut self, key: GroupKey) -> Tensor<3> {
260        self.take(key).unwrap_or_else(|| panic!("the in-projection has no channel group {key:?}"))
261    }
