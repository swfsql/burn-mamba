1//! # Pathway-agnostic SSD path selection (Mamba-3)
2//!
3//! [`Mamba3SsdPath`] selects the chunkwise *SSD path* (Minimal / Serial /
4//! SerialRecalculated) and the chunk length. It is independent of the
5//! double-vs-single *pathway*, which the cache variant selects. It converts
6//! into the per-pathway path types with `From`, and
7//! [`Mamba3::forward`](crate::mamba3::mamba3::Mamba3::forward) sends it to the
8//! pathway of the cache.
9
10use crate::mamba3::prelude::*;
11
12/// The SSD path of the Mamba-3 chunkwise SSD.
13///
14/// This selects the chunkwise *SSD path*. The cache variant selects the
15/// *pathway* (double- vs single-ssd, see
16/// [`crate::mamba3::cache::Mamba3Caches`]). [`Mamba3::forward`] converts this
17/// selection into the input bundle of that pathway
18/// ([`crate::mamba3::double_ssd::ssd::Mamba3DoubleSsdInput`] or
19/// [`crate::mamba3::single_ssd::ssd::Mamba3SingleSsdInput`]) and calls its
20/// `run`.
21///
22/// Each variant carries an optional chunk length. A larger value gives more
23/// intra-chunk GEMM work and a shorter inter-chunk scan. The optimal value is
24/// approximately `√(state_rank · per_head_dim)`, divided by the widening that
25/// `mimo_rank` and `micro_steps` already give the chunk (see
26/// [`Self::optimal_chunk_len`]). `None` uses that optimal value.
27///
28/// The default is [`Self::SerialRecalculated`] with an unset chunk length.
29#[derive(Debug, Clone)]
30pub enum Mamba3SsdPath {
31    /// Minimal/segsum SSD: mostly batched matmuls, with the autodiff backward.
32    ///
33    /// See [`crate::mamba3::double_ssd::ssd::Mamba3DoubleSsdInput::double_ssd_minimal`]
34    /// / [`crate::mamba3::single_ssd::ssd::Mamba3SingleSsdInput::single_ssd_minimal`].
35    /// For training, prefer [`Self::SerialRecalculated`]. This is the only path
36    /// that supports a learnable initial state.
37    Minimal(Option<usize>),
38
39    /// (Hybrid) serial SSD: a serial loop over the chunks plus batched
40    /// matmuls, with the autodiff backward.
41    ///
42    /// See [`crate::mamba3::double_ssd::ssd::Mamba3DoubleSsdInput::double_ssd_serial`]
43    /// / [`crate::mamba3::single_ssd::ssd::Mamba3SingleSsdInput::single_ssd_serial`].
44    /// For a memory-saving custom backward, see [`Self::SerialRecalculated`].
45    Serial(Option<usize>),
46
47    /// (Hybrid) serial SSD with a custom, memory-efficient backward. The
48    /// backward recomputes the forward intermediates instead of storing them.
49    ///
50    /// See [`crate::mamba3::double_ssd::ssd::Mamba3DoubleSsdInput::double_ssd_serial_recalculated`]
51    /// / [`crate::mamba3::single_ssd::ssd::Mamba3SingleSsdInput::single_ssd_serial_recalculated`].
52    /// For a plain autodiff backward, see [`Self::Serial`].
53    SerialRecalculated(Option<usize>),
54}
55
56impl Mamba3SsdPath {
57    /// Optimal chunk length, in **write positions**: `√(state_rank ·
58    /// per_head_dim)` divided by `mimo_rank`, rounded up to a multiple of 32,
59    /// clamped to `32 ..= 512`. `micro_steps` does not change this width:
60    /// [`chunk_tokens`](crate::standard_ssd::chunk_tokens) absorbs it.
61    ///
62    /// The square root is the SISO rule of thumb: it balances the intra-chunk
63    /// GEMMs against the inter-chunk scan. Two dials of the block widen a chunk
64    /// without a place in this formula, and they are **not** the same widening:
65    ///
66    /// - **`mimo_rank`** (`m`) widens *both* axes of a chunk. Every rank writes
67    ///   and every rank reads, so the intra-chunk matmuls run on the fused
68    ///   `chunk_len · m`. The rule of thumb is about that product, hence the
69    ///   divisor. An unchanged `chunk_len` would cost `m²`, and this costs `m`.
70    ///   The reference gives the same advice on its `chunk_size` argument: "64
71    ///   for SISO, 64/mimo_rank for MIMO".
72    /// - **`micro_steps`** (`u`) widens only the **write** axis. All `u`
73    ///   micro-steps of a token write to the state, but the block *reads* the
74    ///   token once, at its last micro-step. So the score is
75    ///   `[batch, nchunks, heads, T·m, T·u·m]` for `T` =
76    ///   [`chunk_tokens`](crate::standard_ssd::chunk_tokens),
77    ///   with `batch · sequence · heads · T · u · m²` elements. To keep that flat
78    ///   in `u`, keep `T · u` (the chunk on the *write* axis) flat and use fewer
79    ///   tokens.
80    ///   Memory is the binding constraint here, because no fused kernel hides it.
81    ///
82    /// So `u` subdivides the chunk and does not shorten it, and `nchunks` grows
83    /// like `u`, not `u²`. The returned length is a multiple of `micro_steps`.
84    /// So each chunk is a whole number of tokens, and their positions are a
85    /// contiguous run. It is on the 32 grid exactly when `micro_steps` divides
86    /// it.
87    ///
88    /// `info/mamba-3/architecture-deltas.md` §8.
89    pub fn optimal_chunk_len(
90        state_rank: usize,
91        per_head_dim: usize,
92        mimo_rank: usize,
93        micro_steps: usize,
94    ) -> usize {
95        let micro_steps = micro_steps.max(1);
96        let chunk_len = (state_rank * per_head_dim)
97            .isqrt()
98            .div_ceil(mimo_rank.max(1)) // the fused read axis is `chunk_tokens · m`.
99            .next_multiple_of(32) // rule-of-thumb: common plane dimension.
100            .clamp(32, 512); // rule-of-thumb: one plane minimum, ceiling at 512.
101        // A whole number of tokens per chunk, at (very nearly) that chunk length.
102        chunk_len.div_ceil(micro_steps) * micro_steps
103    }
104
105    /// The chunk length carried by this variant, if any.
106    pub fn chunk_len(&self) -> Option<usize> {
107        match self {
108            Self::Minimal(chunk_len)
109            | Self::Serial(chunk_len)
110            | Self::SerialRecalculated(chunk_len) => *chunk_len,
111        }
112    }
113
114    /// The chunk length carried by this variant, or [`Self::optimal_chunk_len`]
115    /// for the dimensions of `block` when unset. In both cases the length is
116    /// rounded **up to a multiple of `micro_steps`**, so a chunk is a whole
117    /// number of tokens.
118    ///
119    /// With that rounding, the read axis is a plain reshape. With
120    /// `chunk_len = T·u`, a chunk covers write positions `[cL, (c+1)L)`, that
121    /// is tokens `[cT, (c+1)T)`, and the surviving positions (`≡ u−1 mod u`) are a
122    /// contiguous run of them. At `micro_steps = 1` the rounding does nothing.
123    pub fn chunk_len_or_optimal(&self, block: &Mamba3) -> usize {
124        match self.chunk_len() {
125            Some(chunk_len) => chunk_len.next_multiple_of(block.micro_steps.max(1)),
126            None => Self::optimal_chunk_len(
127                block.state_rank,
128                block.per_head_dim(),
129                block.mimo_rank,
130                block.micro_steps,
131            ),
132        }
133    }
134
135    /// The recommended default path for a given block: [`Self::SerialRecalculated`]
136    /// with [`Self::optimal_chunk_len`] for the block's dimensions.
137    pub fn default_optimal_from_block(block: &Mamba3) -> Self {
138        let chunk_len = Self::optimal_chunk_len(
139            block.state_rank,
140            block.per_head_dim(),
141            block.mimo_rank,
142            block.micro_steps,
143        );
144        Self::SerialRecalculated(Some(chunk_len))
145    }
146}
147
148impl Default for Mamba3SsdPath {
149    fn default() -> Self {
150        // Defaults to the SerialRecalculated SSD path with the optimal chunk length.
151        Self::SerialRecalculated(None)
152    }
153}
