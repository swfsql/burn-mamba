1//! # Mamba-1 Inference Caches
2//!
3//! [`Mamba1Cache`] is the state that one block carries between calls. The
4//! caches of a layer stack are a `Vec<Mamba1Cache>`, with one cache for each
5//! virtual layer.
6//!
7//! In *training* or *prefill*, [`Mamba1::forward`] processes the whole
8//! sequence at once. In *decoding*, [`Mamba1::step`] processes one token per
9//! call. Both modes carry the same two pieces of state:
10//!
11//! 1. **Convolution window** — the last `conv_kernel` pre-activation inputs to
12//!    the depthwise Conv1d. With it, each step applies the causal filter
13//!    without processing earlier tokens again.
14//!
15//! 2. **State** — the per-channel state matrix. It compresses the
16//!    full past into a fixed size, for any number of tokens.
17//!
18//! [`Mamba1::zero_cache`] builds the cache of a fresh sequence. Mirrors
19//! [`crate::mamba2::cache`]. The Mamba-1 state has no head axis.
20

22use burn_stack::modules::sanity as san;

25// ---------------------------------------------------------------------------
26// Mamba1Cache  (cache for a single block)
27// ---------------------------------------------------------------------------
28
29/// The cache that a **single** Mamba-1 block carries between calls.
30#[derive(Module, Debug)]
31pub struct Mamba1Cache {
32    /// **Convolution rolling window.**
33    ///
34    /// The last `conv_kernel` feature vectors that went into the depthwise
35    /// Conv1d. Each step removes the oldest column and appends the projection
36    /// of the new token on the right.
37    ///
38    /// Shape: `[batch, d_inner, conv_kernel]`
39    pub conv_bik: Tensor<3>,
40
41    /// **State.**
42    ///
43    /// The O(d_inner·state_rank) compressed summary of all earlier tokens. The
44    /// selective-scan recurrence updates it at each token.
45    ///
46    /// Shape: `[batch, d_inner, state_rank]`
47    pub ssm_bir: Tensor<3>,
48}
49
50impl Mamba1Cache {
51    /// Run the [`NaN`/`Inf` guards](burn_stack::modules::misc::sanity) on every cached tensor.
52    pub fn sanity(&self) {
53        san(&self.conv_bik);
54        san(&self.ssm_bir);
55    }
56}
57
58impl Mamba1 {
59    /// The cache of a fresh sequence, for `batch` rows. A `forward` or a
60    /// `step` without a cache starts from it. The shapes come from the
61    /// weights of the block.
62    ///
63    /// Zeros are correct: a zero convolution window means "no previous tokens"
64    /// (the causal zero padding), and a zero state is the standard initial
65    /// condition `h₀ = 0`.
66    pub fn zero_cache(&self, batch: usize, device: &Device) -> Mamba1Cache {
67        let [d_inner, state_rank] = self.a_log.dims();
68        let [_, _, conv_kernel] = self.conv1d.weight.dims();
69        let conv_bik = Tensor::zeros([batch, d_inner, conv_kernel], device);
70        let ssm_bir = Tensor::zeros([batch, d_inner, state_rank], device);
71        Mamba1Cache { conv_bik, ssm_bir }
72    }
73}
