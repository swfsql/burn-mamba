1//! # The analytic backward of the standard SSD scan
2//!
3//! The steps of the backward of one SSD call:
4//!
5//! 1. Recompute the forward intermediates (K1–K4) from the saved leaf inputs.
6//!    They are not stored.
7//! 2. Run the chunk-**local** K5 backward batched, one chunk group at a time.
8//! 3. Walk the K4 state-passing backward: the one recurrence, and the only
9//!    walk.
10//! 4. Run the K3, K2 and K1 backwards batched.
11//!
12//! The backward of each kernel is in [`super::kernels`], beside its forward.
13//! This file assembles them. The single-SSD pathway assembles the same
14//! functions, with its own terms.
15//!
16//! The fused `L·M` length carries the `mimo_rank` axis through the intra-chunk
17//! products.
18//!
19//! All the math is on backend **primitives**, through the rank-tagged `F`
20//! wrapper. The [recompute node](burn_stack::utils::recompute_node)
21//! runs with a generic backend `B`, where the high-level `Tensor` is not
22//! available, so the math uses the `float_*` ops of `B`.
23
24#![allow(non_snake_case)]
25
26use super::chunk_tokens;
27use super::kernels::{
28    K5Grads, k1_ssd_chunk_cumsum, k1_ssd_chunk_cumsum_backward, k2_ssd_bmm, k2_ssd_bmm_backward,
29    k3_ssd_chunk_state_backward, k3_ssd_chunk_state_extended, k4_ssd_state_passing,
30    k4_ssd_state_passing_backward, k5_da_cumsum_on_chunk_axes, k5_ssd_chunk_scan_backward,
31    scatter_chunk_end, split_fused_axis,
32};
33use burn_stack::utils::fprim::{F, san};
34use burn::backend::Backend;

36/// Per-input gradients produced by [`combined_backward`] (one field per
37/// differentiable forward input of one SSD call).
38#[non_exhaustive]
39pub struct CombinedGrads<B: Backend> {
40    /// Gradient of the (pre-scaled) input `v`.
41    pub d_v_bnlmhp: F<B, 6>,
42    /// Gradient of `Δ·A` (`da`).
43    pub d_da_bnlh: F<B, 4>,
44    /// Gradient of the input projection `B`.
45    pub d_b_bnlmhr: F<B, 6>,
46    /// Gradient of the output projection `C`, on the read axis of the chunk.
47    pub d_c_bntmhr: F<B, 6>,
48    /// Gradient of the initial state.
49    pub d_initial_state_bhpr: F<B, 4>,
50    /// Gradient of the origin state of the resets (`None` without resets).
51    pub d_origin_bhpr: Option<F<B, 4>>,
52    /// Local same-chunk part of `d_da_cumsum`, from BLUE and ORANGE only (no K3
53    /// and no K4 contribution). Test builds only.
54    #[cfg(test)]
55    pub d_da_local_bhnl: F<B, 4>,
56    /// The ORANGE part of `d_v`. Test builds only.
57    #[cfg(test)]
58    pub d_v_orange_bnlmhp: F<B, 6>,
59}
60
61/// Memory-efficient backward of one SSD call.
62///
63/// Recomputes the forward intermediates (K1-K4) from the saved inputs. Then
64/// the chunk-local K5 (BLUE + ORANGE) backward runs batched over chunk groups,
65/// and the K4 state-passing backward walks the chunks in reverse. The K3/K2/K1
66/// backwards run as single batched ops.
67///
68/// # Arguments
69/// - `d_y_bntmhp` — upstream gradient of the SSD output (read axis)
70/// - `d_final_bhpr` — upstream gradient of the final state
71/// - `v_bnlmhp`, `da_bnlh`, `b_bnlmhr`, `c_bntmhr`, `initial_state_bhpr` —
72///   the five saved forward inputs
73/// - `read_stride` — write positions per read position (`micro_steps`)
74/// - `restarts` — `(keep_bn, origin_bhpr)`, or `None`. `keep_bn` is the
75///   constant reset factor of the forward (`0` at a chunk that starts a new
76///   segment). `origin_bhpr` is the state that those chunks start from.
77///
78/// # Returns
79/// One [`CombinedGrads`] struct with the gradients of all 5 inputs, and of
80/// the origin state.
81#[allow(clippy::too_many_arguments)]
82pub fn combined_backward<B: Backend>(
83    d_y_bntmhp: F<B, 6>,
84    d_final_bhpr: F<B, 4>,
85    //
86    v_bnlmhp: F<B, 6>,
87    da_bnlh: F<B, 4>,
88    b_bnlmhr: F<B, 6>,
89    c_bntmhr: F<B, 6>,
90    initial_state_bhpr: F<B, 4>,
91    read_stride: usize,
92    restarts: Option<(F<B, 2>, F<B, 4>)>,
93) -> CombinedGrads<B> {
94    let keep_bn = restarts.as_ref().map(|(keep_bn, _)| keep_bn.clone());
95    let [batch, nchunks, chunk_len, mimo_rank, nheads, per_head_dim] = v_bnlmhp.dims();
96    let [.., state_rank] = b_bnlmhr.dims();
97    // The chunk's two axes: `fused` is what it writes, `read` what it is read at.
98    let chunk_tokens = chunk_tokens(chunk_len, read_stride);
99    let fused = chunk_len * mimo_rank;
100    let read = chunk_tokens * mimo_rank;
101
102    san(&d_y_bntmhp);
103    san(&d_final_bhpr);
104    san(&v_bnlmhp);
105    san(&da_bnlh);
106    san(&b_bnlmhr);
107    san(&c_bntmhr);
108    san(&initial_state_bhpr);
109
110    // ── Recompute the forward intermediates ───────────────────────────────
111
112    // K1: pre-combined Δ·A to the intra-chunk cumsum.
113    let (da_cumsum_bhnl, da_chunk_end_bhn) = k1_ssd_chunk_cumsum(da_bnlh);
114    san(&da_cumsum_bhnl);
115
116    // K2: the CB matrix of ORANGE.
117    let cb_bnhTMLM = k2_ssd_bmm(c_bntmhr.clone(), b_bnlmhr.clone());
118    san(&cb_bnhTMLM);
119
120    // K3: the intra-chunk state, and the intermediates of its backward.
121    let (intra_chunk_state_bnhpr, k3_decay_bhnLM, k3_decayed_v_bnLMhp) =
122        k3_ssd_chunk_state_extended(v_bnlmhp.clone(), b_bnlmhr.clone(), da_cumsum_bhnl.clone());
123
124    // K4: the chunk-input state stream of BLUE.
125    let (chunk_input_state_bnhpr, _final_state_bhpr) = k4_ssd_state_passing(
126        intra_chunk_state_bnhpr,
127        da_chunk_end_bhn.clone(),
128        initial_state_bhpr,
129        restarts,
130        None,
131    );
132
133    let (da_cumsum_bhnLM, da_cumsum_bhnTM) =
134        k5_da_cumsum_on_chunk_axes(da_cumsum_bhnl, mimo_rank, read_stride);
135    let d_y_bnhTMp = d_y_bntmhp
136        .reshape([batch, nchunks, read, nheads, per_head_dim]) // d_y_bnTMhp
137        .swap_dims(2, 3); // d_y_bnhTMp
138    san(&d_y_bnhTMp);
139
140    // ── K5 backward (BLUE + ORANGE) ───────────────────────────────────────
141    let (
142        K5Grads {
143            d_chunk_input_state_bnhpr,
144            d_c_blue_bnhTMr,
145            d_da_blue_bhnl,
146            d_v_intra_bnhLMp: d_v_orange_bnhLMp,
147            d_cb_bnhTMLM,
148            d_da_intra_bhnl: d_da_orange_bhnl,
149        },
150        _,
151    ) = k5_ssd_chunk_scan_backward(
152        d_y_bnhTMp,
153        v_bnlmhp.clone(),
154        c_bntmhr.clone(),
155        cb_bnhTMLM,
156        da_cumsum_bhnLM,
157        da_cumsum_bhnTM,
158        chunk_input_state_bnhpr.clone(),
159        None,
160        1, // `triu(1)`: the sum of this scan includes the same position.
161        read_stride,
162    );
163
164    // ── K4 backward: the reverse of the state-passing scan ────────────────
165    let (d_intra_chunk_state_bnhpr, d_da_end_bhn, d_initial_state_bhpr, d_restart_bnhpr) =
166        k4_ssd_state_passing_backward(
167            d_chunk_input_state_bnhpr,
168            chunk_input_state_bnhpr,
169            da_chunk_end_bhn,
170            d_final_bhpr,
171            keep_bn,
172        );
173    // Every chunk restarts from the same origin state.
174    let d_origin_bhpr = d_restart_bnhpr.map(|d_restart_bnhpr| d_restart_bnhpr.sum_dim(1).squeeze_dim::<4>(1));
175    let d_da_cumsum_k4_bhnl = scatter_chunk_end(d_da_end_bhn, chunk_len);
176
177    // ── K3 and K2 backwards ───────────────────────────────────────────────
178    let b_bnLMhr = b_bnlmhr.reshape([batch, nchunks, fused, nheads, state_rank]);
179    let (d_v_k3_bnlmhp, d_b_k3_bnlmhr, d_da_cumsum_k3_bhnl) = k3_ssd_chunk_state_backward(
180        d_intra_chunk_state_bnhpr,
181        k3_decay_bhnLM,
182        k3_decayed_v_bnLMhp,
183        v_bnlmhp,
184        b_bnLMhr.clone(),
185    );
186    let (d_c_k2_bntmhr, d_b_k2_bnlmhr) = k2_ssd_bmm_backward(d_cb_bnhTMLM, c_bntmhr, b_bnLMhr);
187
188    let d_c_blue_bntmhr = split_fused_axis(d_c_blue_bnhTMr, chunk_tokens);
189    let d_v_orange_bnlmhp = split_fused_axis(d_v_orange_bnhLMp, chunk_len);
190
191    // Test-only copies for the oracle of the local same-chunk `d_da`.
192    // Production builds compute neither the add nor the copy.
193    #[cfg(test)]
194    let d_da_local_bhnl = d_da_blue_bhnl.clone() + d_da_orange_bhnl.clone();
195    #[cfg(test)]
196    let d_v_orange_save_bnlmhp = d_v_orange_bnlmhp.clone();
197
198    // ── K1 backward and the sum of the contributions ──────────────────────
199    let d_da_cumsum_bhnl =
200        d_da_blue_bhnl + d_da_orange_bhnl + d_da_cumsum_k3_bhnl + d_da_cumsum_k4_bhnl;
201    san(&d_da_cumsum_bhnl);
202    let d_da_bnlh = k1_ssd_chunk_cumsum_backward(d_da_cumsum_bhnl);
203
204    let d_v_bnlmhp = d_v_k3_bnlmhp + d_v_orange_bnlmhp;
205    let d_b_bnlmhr = d_b_k2_bnlmhr + d_b_k3_bnlmhr;
206    let d_c_bntmhr = d_c_k2_bntmhr + d_c_blue_bntmhr;
207
208    san(&d_v_bnlmhp);
209    san(&d_da_bnlh);
210    san(&d_b_bnlmhr);
211    san(&d_c_bntmhr);
212    san(&d_initial_state_bhpr);
213
214    CombinedGrads {
215        d_v_bnlmhp,
216        d_da_bnlh,
217        d_b_bnlmhr,
218        d_c_bntmhr,
219        d_initial_state_bhpr,
220        d_origin_bhpr,
221        #[cfg(test)]
222        d_da_local_bhnl,
223        #[cfg(test)]
224        d_v_orange_bnlmhp: d_v_orange_save_bnlmhp,
225    }
226}
