9// ===========================================================================
10// Class tokens / latents (learnable sequence-inserted tokens)
11// ===========================================================================
12//
13// A *class token* / *class latent* is a learnable embedding spliced into the
14// sequence — a transformer-`[CLS]`-style register the model can read/write
15// through. They are inserted at the input boundary of a container (a network's
16// input for [`ClassToken`], width = the input feature width; a layer's working
17// sequence for [`ClassLatent`], width = `d_model`), permanently lengthening the
18// sequence for everything downstream. A container can carry any number; the
19// markers below say *where* each one lands, while a single `Param<Tensor<2>>`
20// of shape `[num_markers, width]` holds the embeddings (row `i` ↔ marker `i`).
21//
22// Insertion order (all relative to the *original* length `L`): every `Start`
23// first (index 0), then `Middle` (index `L/2`, splitting the original
24// sequence), then `End` (index `L`), then `Custom(index)` (explicit index,
25// inserted last). Markers sharing an index keep their `Vec` order.
26//
27// One rule places three of the four kinds: the marker names an index into the
28// *original* sequence and is emitted immediately **before** the user token
29// sitting there — `Start` (0) opens the sequence, `Middle` (`L/2`) splits it,
30// `Custom(k)` precedes token `k`. `End` is the sole exception, having no token
31// to precede: it **closes** the sequence, trailing the last user token.
32//
33// `Custom` is uniform in `k`, so a `Custom(k ≥ L)` simply never lands — there is
34// no token `k` to precede. Should the caller keep feeding tokens past the
35// announced `L`, it lands then, still *before* the next token; it never trails.
36// An open-ended stream (no hint) is likewise never closed — the same reason
37// `End` needs the length hint below.
38//
39// `step` therefore returns the output of the **last** token it emitted (and
40// leaves the state after it): the user token, unless `End` follows it — `End`
41// being then the sequence's latest token. Markers emitted before the user token
42// only leave their mark on the state.
43//
44// Placement is **streamed**, identically for `forward` (a chunk of the
45// sequence) and `step` (a single token): [`ClassCursors`] carries one
46// `full_len` hint — the length of the whole sequence the call is part of — plus
47// one cursor per level, each recording how much of *that* level's output
48// sequence earlier calls already emitted. From those:
49//
50//   * a marker whose output position is behind the cursor was emitted by an
51//     earlier call and is skipped — so `Start` fires only while the cursor is
52//     still at 0, and a resumed stream does not re-insert it;
53//   * `Middle`/`End` resolve only against the whole sequence, so they **panic**
54//     without a `full_len` hint (`Start`/`Custom` are length-independent and
55//     work on an open-ended stream);
56//   * a marker landing exactly at a chunk's end is emitted by that chunk only
57//     if the chunk *closes* the sequence — otherwise it opens the next one, so
58//     splitting a sequence anywhere leaves the placement unchanged.
59//
60// No cursors (`None`) keeps the two calls' historical defaults: `forward` treats
61// its argument as the whole sequence (`full_len` = its length, cursors at 0),
62// `step` injects nothing at all.
63
64/// Position marker for a learnable class **token** inserted into a *network's*
65/// input sequence (embedding width = the network input width / "d_input").
66#[derive(Config, Debug)]
67pub enum ClassToken {
68    /// Prepend before the whole sequence (index 0).
69    Start,
70    /// Insert before the middle token of the original sequence (index `L/2`).
71    /// Needs a [`ClassCursors::full_len`] hint.
72    Middle,
73    /// **Close** the sequence: appended after its last token (index `L`) — the
74    /// only marker that trails one instead of preceding it, and so the token a
75    /// closing `step` returns. Needs a [`ClassCursors::full_len`] hint.
76    End,
77    /// Insert before the original sequence's token `index` — for any `index`,
78    /// so one at or past the end never lands (no such token), unless the caller
79    /// feeds tokens past the announced length and it precedes the next one.
80    Custom(usize),
81}

85#[derive(Config, Debug)]

93    /// only marker that trails one instead of preceding it, and so the token a
94    /// closing `step` returns. Needs a [`ClassCursors::full_len`] hint.
95    End,

102/// Shared behaviour of the [`ClassToken`] / [`ClassLatent`] position markers,
103/// letting one generic helper place either kind.
104pub trait ClassMarker: Clone {
105    /// Insertion index measured against the *original* sequence length `orig_len`.
106    fn insert_pos(&self, orig_len: usize) -> usize;
107    /// Tie-break rank among markers sharing an index (`Start`<`Middle`<`End`<`Custom`).
108    fn group_rank(&self) -> usize;
109    /// Whether this marker's position is only defined against the whole sequence
110    /// (`Middle`/`End`), so placing it requires a [`ClassCursor::full_len`] hint.
111    fn needs_full_len(&self) -> bool;
112    /// Whether this marker *closes* the sequence — trailing its last token
113    /// rather than preceding one. `End` alone does.
114    fn closes_sequence(&self) -> bool;
115}
116
117macro_rules! impl_class_marker {
118    ($ty:ty) => {
119        impl ClassMarker for $ty {
120            fn insert_pos(&self, orig_len: usize) -> usize {
121                match self {
122                    Self::Start => 0,
123                    Self::Middle => orig_len / 2,
124                    Self::End => orig_len,
125                    Self::Custom(index) => *index,
126                }
127            }
128            fn group_rank(&self) -> usize {
129                match self {
130                    Self::Start => 0,
131                    Self::Middle => 1,
132                    Self::End => 2,
133                    Self::Custom(_) => 3,
134                }
135            }
136            fn needs_full_len(&self) -> bool {
137                matches!(self, Self::Middle | Self::End)
138            }
139            fn closes_sequence(&self) -> bool {
140                matches!(self, Self::End)
141            }
142        }
143    };
144}
145impl_class_marker!(ClassToken);
146impl_class_marker!(ClassLatent);

148/// Placement state of **one** class-marker level (one container's own markers):
149/// how far into that level's *output* sequence the previous calls got, and the
150/// length that positions are measured against.
151#[derive(Debug, Clone, Copy, Default, PartialEq, Eq)]
152pub struct ClassCursor {
153    /// Output-sequence position reached so far — user tokens *and* already
154    /// emitted class markers (0 ⇒ the sequence has not started).
155    pub offset: usize,
156    /// Length of the whole sequence this level receives, its own markers
157    /// excluded. `None` ⇒ an open-ended stream: `Start`/`Custom` still place
158    /// exactly, `Middle`/`End` panic.
159    pub full_len: Option<usize>,
160}
161
162impl ClassCursor {
163    /// A cursor at `offset` measured against `full_len`.
164    pub fn at(offset: usize, full_len: Option<usize>) -> Self {
165        Self { offset, full_len }
166    }
167
168    /// A fresh cursor for a call that covers the entire sequence of length
169    /// `len` — what `forward` assumes when given no cursors.
170    pub fn whole(len: usize) -> Self {
171        Self {
172            offset: 0,
173            full_len: Some(len),
174        }
175    }
176}

178/// Everything a `forward` (chunk) or `step` (single token) call needs in order
179/// to place the class tokens / class latents of a whole network: one
180/// full-length hint plus one cursor per class-marker level. Pass the **same**
181/// value to every call of a sequence — each call advances the cursors it uses,
182/// so the next one resumes exactly where this one stopped.
183///
184/// The levels nest, and each cursor counts the sequence *its* level sees (which
185/// already includes whatever the levels below it spliced in):
186///
187/// ```text
188/// network     LatentNetwork's own ClassTokens  (before `in_proj`)
189/// stack       Layers'/BidiLayers' own ClassLatents
190/// per_layer   one cursor per virtual layer, for that Layer's ClassLatents
191/// ```
192///
193/// [`Self::full_len`] is the length of the user sequence handed to the
194/// outermost call; the inner levels' lengths are derived from it.
195///
196/// To read a marker back out of a *chunked* `forward`: its position in the whole
197/// output is the container's `class_*_output_indices(full_len)`; subtracting the
198/// level's cursor as it was *before* that call gives its index inside the
199/// chunk's own output. A `step` returns one token — the last it emitted, which
200/// is the user token unless an `End` marker trails it, `End` being then the
201/// sequence's true last token.
202#[derive(Debug, Clone, Default, PartialEq, Eq)]
203pub struct ClassCursors {
204    /// Total length of the user sequence all the calls form together, or `None`
205    /// for an open-ended stream (then `Middle`/`End` markers panic).
206    pub full_len: Option<usize>,
207    /// Cursor of a network's own [`ClassToken`]s (unused by a bare layer stack).
208    pub network: usize,
209    /// Cursor of a layer container's own [`ClassLatent`]s.
210    pub stack: usize,
211    /// One cursor per **virtual** layer, for the per-layer [`ClassLatent`]s.
212    /// Left empty it is sized (with zeros) on the first call.
213    pub per_layer: Vec<usize>,
214}
215
216impl ClassCursors {
217    /// Cursors at the start of a sequence of known total length — the form that
218    /// enables `Middle`/`End` markers, and what a whole-sequence `forward` uses
219    /// when given none.
220    pub fn new(full_len: usize) -> Self {
221        Self {
222            full_len: Some(full_len),
223            ..Default::default()
224        }
225    }
226
227    /// Cursors at the start of an open-ended stream (unknown total length):
228    /// `Start`/`Custom` markers place exactly, `Middle`/`End` panic.
229    pub fn stream() -> Self {
230        Self::default()
231    }
232
233    /// Size [`Self::per_layer`] for a stack of `n` virtual layers (idempotent).
234    pub(crate) fn fit(&mut self, n: usize) {
235        if self.per_layer.is_empty() {
236            self.per_layer = vec![0; n];
237        }
238        assert_eq!(
239            self.per_layer.len(),
240            n,
241            "one class-latent cursor per virtual layer"
242        );
243    }
244
245    /// Enter the inner level, whose sequence is longer by the `extra` markers
246    /// this level splices in. Returns the previous hint, for [`Self::leave`].
247    pub(crate) fn enter(&mut self, extra: usize) -> Option<usize> {
248        let saved = self.full_len;
249        self.full_len = saved.map(|l| l + extra);
250        saved
251    }
252
253    /// Leave the inner level, restoring the hint [`Self::enter`] returned.
254    pub(crate) fn leave(&mut self, saved: Option<usize>) {
255        self.full_len = saved;
256    }
257}

259/// Panic if any marker's position needs the whole sequence length while none is
260/// known — `Middle`/`End` cannot be placed from a chunk (or a single token).
261pub(crate) fn assert_full_len_known<M: ClassMarker>(
262    markers: &[M],
263    full_len: Option<usize>,
264    who: &str,
265) {
266    assert!(
267        full_len.is_some() || !markers.iter().any(|m| m.needs_full_len()),
268        "{who}: Middle/End class markers need a full-length hint (ClassCursors::new)"
269    );
270}

272/// Which of `markers` fall inside the next `chunk_len` user tokens, and where.
273///
274/// Returns `(at, marker)` pairs in output order: insert `markers[marker]`
275/// *before* the chunk's `at`-th token (`at == chunk_len` ⇒ after the last one).
276/// `cursor` is advanced past the whole chunk, its insertions included.
277pub(crate) fn class_chunk_plan<M: ClassMarker>(
278    markers: &[M],
279    chunk_len: usize,
280    cursor: &mut ClassCursor,
281    who: &str,
282) -> Vec<(usize, usize)> {
283    if markers.is_empty() {
284        cursor.offset += chunk_len;
285        return Vec::new();
286    }
287    assert_full_len_known(markers, cursor.full_len, who);
288    let positions = class_marker_output_indices(markers, cursor.full_len.unwrap_or(usize::MAX));
289
290    // User tokens consumed once this chunk is done: the output positions behind
291    // the cursor, minus the markers among them, plus this chunk. Feeding more
292    // than an announced `full_len` is allowed — everything is already placed by
293    // then, so the extra tokens simply stream on.
294    let start = cursor.offset;
295    let consumed = start - positions.iter().filter(|&&p| p < start).count() + chunk_len;
296    // Whether this chunk reaches the announced end, i.e. carries the last user
297    // token — the one and only place an `End` can go.
298    let closes = cursor.full_len == Some(consumed);
299
300    let mut order: Vec<usize> = (0..markers.len()).collect();
301    order.sort_by_key(|&i| positions[i]);
302
303    let mut out = start; // running output position
304    let mut at = 0usize; // chunk tokens placed before it
305    let mut plan = Vec::new();
306    for i in order {
307        let p = positions[i];
308        if p < start {
309            continue; // emitted by an earlier call
310        }
311        let need = p - out; // user tokens preceding this marker
312        if at + need > chunk_len {
313            break; // the token it precedes is in a later chunk, as is it
314        }
315        if at + need == chunk_len && !(closes && markers[i].closes_sequence()) {
316            // Nothing left in this chunk to precede: only a closing `End` on the
317            // chunk that ends the sequence belongs here. A `Custom` waits for
318            // its token — for one at/past the end, forever.
319            break;
320        }
321        at += need;
322        out = p + 1;
323        plan.push((at, i));
324    }
325    cursor.offset = out + (chunk_len - at);
326    plan
327}

329/// Splice the learnable class markers `emb` (`[k, width]`, row `i` ↔
330/// `markers[i]`) that fall inside the chunk `x` (`[batch, chunk_len, width]`),
331/// returning the lengthened chunk and advancing `cursor` past it.
332///
333/// `markers` empty (or none of them landing in this chunk) ⇒ `x` unchanged.
334pub(crate) fn insert_class_markers<M: ClassMarker>(
335    x: Tensor<3>,
336    markers: &[M],
337    emb: Option<&Param<Tensor<2>>>,
338    cursor: &mut ClassCursor,
339    who: &str,
340) -> Tensor<3> {
341    let [batch, chunk_len, width] = x.dims();
342    let plan = class_chunk_plan(markers, chunk_len, cursor, who);
343    if plan.is_empty() {
344        return x;
345    }
346    let emb = emb
347        .expect("class-token markers present but no embedding param")
348        .val();
349    assert_eq!(
350        emb.dims(),
351        [markers.len(), width],
352        "one embedding row per class marker"
353    );
354
355    let mut segments: Vec<Tensor<3>> = Vec::with_capacity(2 * plan.len() + 1);
356    let mut taken = 0usize; // chunk tokens emitted so far
357    for (at, i) in plan {
358        if at > taken {
359            segments.push(x.clone().narrow(1, taken, at - taken));
360            taken = at;
361        }
362        segments.push(
363            emb.clone()
364                .narrow(0, i, 1) // [1, width]
365                .unsqueeze_dim::<3>(0) // [1, 1, width]
366                .expand([batch, 1, width]),
367        );
368    }
369    if taken < chunk_len {
370        segments.push(x.narrow(1, taken, chunk_len - taken));
371    }
372    Tensor::cat(segments, 1)
373}

375/// The embedding row of marker `i` as one broadcast token (`[batch, width]`) —
376/// the `step` counterpart of a slice of [`insert_class_markers`].
377pub(crate) fn class_row(
378    emb: Option<&Param<Tensor<2>>>,
379    i: usize,
380    batch: usize,
381    width: usize,
382) -> Tensor<2> {
383    emb.expect("class-token markers present but no embedding param")
384        .val()
385        .narrow(0, i, 1)
386        .expand([batch, width])
387}

389/// The output-sequence position of each marker (in `Vec` order) for an input of
390/// length `orig_len`, without materialising any tensor. Mirrors the placement in
391/// [`insert_class_markers`] — useful for reading a class token back out.
392///
393/// A marker that never lands (a `Custom` at or past the end — it has no token to
394/// precede) reports the position it *would* take, which is then `>= orig_len +
395/// (number of markers that do land)`, i.e. past the emitted sequence.
396pub(crate) fn class_marker_output_indices<M: ClassMarker>(
397    markers: &[M],
398    orig_len: usize,
399) -> Vec<usize> {
400    let k = markers.len();
401    let mut order: Vec<usize> = (0..k).collect();
402    order.sort_by_key(|&i| (markers[i].insert_pos(orig_len), markers[i].group_rank(), i));
403    let mut cursor = 0usize;
404    let mut out_len = 0usize;
405    let mut out_index = vec![0usize; k];
406    for &i in &order {
407        let p = markers[i].insert_pos(orig_len).min(orig_len);
408        if p > cursor {
409            out_len += p - cursor;
410            cursor = p;
411        }
412        out_index[i] = out_len;
413        out_len += 1;
414    }
415    out_index
416}

418/// Build the embedding param for `n` class markers of the given `width`
419/// (`None` when there are none — Burn has no zero-width tensors).
420pub(crate) fn init_class_emb(n: usize, width: usize, device: &Device) -> Option<Param<Tensor<2>>> {
421    (n > 0).then(|| {
422        Initializer::Normal {
423            mean: 0.0,
424            std: 0.02,
425        }
426        .init([n, width], device)
