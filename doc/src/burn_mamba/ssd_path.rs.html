1//! # The SSD path (Mamba-2 and Mamba-3)
2//!
3//! [`SsdPath`](crate::ssd_path::SsdPath) selects the method that computes the
4//! chunkwise SSD scan. All SSD paths give the same values and gradients. The
5//! scan options of a block hold an SSD path and a chunk length:
6//!
7//! - [`Mamba2ScanOptions`](crate::mamba2::scan_options::Mamba2ScanOptions) for
8//!   Mamba-2.
9//! - [`Mamba3ScanOptions`](crate::mamba3::scan_options::Mamba3ScanOptions) for
10//!   Mamba-3, on each pathway.
11//!
12//! The `run` of each input bundle reads only the SSD path. The chunk length is
13//! already in the shapes of its tensors.
14
15/// The method that computes the chunkwise SSD scan.
16///
17/// The default is [`Self::SerialRecalculated`].
18#[derive(Debug, Clone, Copy, PartialEq, Eq, Default, serde::Serialize, serde::Deserialize)]
19pub enum SsdPath {
20    /// Minimal SSD: mostly batched matmuls and a `segsum` mask, with the
21    /// autodiff backward.
22    ///
23    /// For training, prefer [`Self::SerialRecalculated`]. This is the only path
24    /// that supports a learnable initial state.
25    ///
26    /// See these functions:
27    ///
28    /// - Mamba-2:
29    ///   [`Mamba2SsdInput::ssd_minimal`](crate::mamba2::ssd::Mamba2SsdInput::ssd_minimal).
30    ///   It is based on `/mamba_ssm/modules/ssd_minimal.py` in the
31    ///   `state-spaces/mamba` GitHub reference.
32    /// - Mamba-3:
33    ///   [`Mamba3DoubleSsdInput::double_ssd_minimal`](crate::mamba3::double_ssd::ssd::Mamba3DoubleSsdInput::double_ssd_minimal)
34    ///   and
35    ///   [`Mamba3SingleSsdInput::single_ssd_minimal`](crate::mamba3::single_ssd::ssd::Mamba3SingleSsdInput::single_ssd_minimal).
36    Minimal,
37
38    /// (Hybrid) serial SSD: a serial loop over the chunks plus batched
39    /// matmuls, with the autodiff backward.
40    ///
41    /// For a memory-saving custom backward, see [`Self::SerialRecalculated`].
42    ///
43    /// See these functions:
44    ///
45    /// - Mamba-2:
46    ///   [`Mamba2SsdInput::ssd_serial`](crate::mamba2::ssd::Mamba2SsdInput::ssd_serial).
47    ///   It is based on 5 kernels under `/mamba_ssm/ops/triton/` in the
48    ///   `state-spaces/mamba` GitHub reference:
49    ///   - `ssd_chunk_state.py` (K1, K3).
50    ///   - `ssd_bmm.py` (K2).
51    ///   - `ssd_state_passing.py` (K4).
52    ///   - `ssd_chunk_scan.py` (K5).
53    /// - Mamba-3:
54    ///   [`Mamba3DoubleSsdInput::double_ssd_serial`](crate::mamba3::double_ssd::ssd::Mamba3DoubleSsdInput::double_ssd_serial)
55    ///   and
56    ///   [`Mamba3SingleSsdInput::single_ssd_serial`](crate::mamba3::single_ssd::ssd::Mamba3SingleSsdInput::single_ssd_serial).
57    Serial,
58
59    /// (Hybrid) serial SSD with a custom, memory-efficient backward. The
60    /// backward recomputes the forward intermediates instead of storing them.
61    ///
62    /// For a plain autodiff backward, see [`Self::Serial`].
63    ///
64    /// See these functions:
65    ///
66    /// - Mamba-2:
67    ///   [`Mamba2SsdInput::ssd_serial_recalculated`](crate::mamba2::ssd::Mamba2SsdInput::ssd_serial_recalculated).
68    ///   It calls the [standard SSD scan](crate::standard_ssd). The gradients
69    ///   are those of the combined kernel `/mamba_ssm/ops/triton/ssd_combined.py`
70    ///   in the `state-spaces/mamba` GitHub reference. The order of the math is
71    ///   that of the scan, and not that of the kernel (ADR-0010).
72    /// - Mamba-3:
73    ///   [`Mamba3DoubleSsdInput::double_ssd_serial_recalculated`](crate::mamba3::double_ssd::ssd::Mamba3DoubleSsdInput::double_ssd_serial_recalculated)
74    ///   and
75    ///   [`Mamba3SingleSsdInput::single_ssd_serial_recalculated`](crate::mamba3::single_ssd::ssd::Mamba3SingleSsdInput::single_ssd_serial_recalculated).
76    #[default]
77    SerialRecalculated,
78}
