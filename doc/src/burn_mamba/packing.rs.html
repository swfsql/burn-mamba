1//! Packed segments inside a block: several sequences one after another in one
2//! row, each from one origin cache.
3//!
4//! `reset_bt` (`[batch, tokens]`, `true` at the first token of a segment)
5//! marks where a row starts a new sequence. At that token, the block restarts
6//! from the **origin** cache (`origin`, a fresh one for `origin = None`). The
7//! tokens before the first reset of a row continue from the incoming cache
8//! (`cache`), as in a plain call. So a stream of packed calls can continue a
9//! sequence across a call boundary. A reset at the first token of a call
10//! starts a new sequence.
11//!
12//! Each sequence gives what its own row gives in a batch of plain calls from
13//! the origin: the outputs, the last cache and the gradients. This is true for
14//! any split of the stream into calls. The previous segment gets no gradient
15//! through a restart. The origin gets the gradients of every segment that
16//! restarts from it. `origin = cache` restarts every segment from the incoming
17//! cache (for example, a shared prefix).
18//!
19//! # Only at a chunk start
20//!
21//! A reset must be at the first token of a chunk (`chunk_len` tokens, or
22//! `chunk_len / micro_steps` for Mamba-3). Then the chunkwise scan changes at
23//! one place only, the carry between chunks (K4). Every other site that reads
24//! an earlier sample (a conv tap, a trapezoid tap) crosses a reset exactly when
25//! two conditions are true:
26//!
27//! - its chunk starts with a reset,
28//! - it reads further back than its own index in the chunk.
29//!
30//! Such a read takes the value of the origin instead (`restart_heads`). So a
31//! per-chunk flag is all that those sites need, and no scan runs over the row
32//! for them. This needs a read that reaches back one chunk at most:
33//!
34//! - A Mamba-2 conv tap reads `conv_kernel − 1` rows back. So
35//!   `Mamba2::forward_packed` asserts `chunk_len ≥ conv_kernel − 1`.
36//! - A Mamba-3 tap reads `u` positions (one token) back at most.
37//!
38//! A segment can end with pad rows (`pad`), up to the next chunk start. The official Mamba-3 varlen kernels also start each
39//! sequence at a chunk start, with a padded last chunk.
40//!
41//! Some scans run over the whole row: the cumulative rotation of Mamba-3 and
42//! its positive systems. They restart at the first position of each segment,
43//! from the carry of the origin.
44//!
45//! The cache that a call returns is that of the last segment of each row. Its
46//! "last samples" fields (conv window, tap FIFO, carries) read the origin for
47//! the entries before the start of that segment (`last_origin`,
48//! `restart_window`).
49//!
50//! Only the serial SSD paths take resets. The inter-chunk decay of `Minimal`
51//! is a `nchunks × nchunks` matrix, which a long packed row makes large.
52//!
53//! The entry points are `Mamba2::forward_packed` and
54//! `Mamba3::forward_packed`.
55

58/// The resets of a packed row as an SSD pass takes them: the chunks that
59/// restart, and the state that they restart from.
60#[derive(Clone, Debug)]
61pub struct Restarts {
62    /// `[batch, nchunks]`: `true` at a chunk that starts a new segment.
63    pub reset_bn: Tensor<2, Bool>,
64    /// `[batch, nheads, per_head_dim, state_rank]`: the state of the origin
65    /// cache. Each chunk of `reset_bn` starts from it.
66    pub origin_bhpr: Tensor<4>,
67}
68
69impl Restarts {
70    /// The restarts as a recompute op takes them. The extension macro takes
71    /// no `Option` tensor, so the op gets `(keep_bn, origin_bhpr, resets)`:
72    ///
73    /// - `keep_bn`: `0` at a chunk that starts a new segment, `1` elsewhere.
74    ///   It is a constant (no gradient), with the dtype of `like`.
75    /// - `origin_bhpr`: the state that those chunks start from.
76    /// - `resets`: `false` for `None`. Then the two tensors are placeholders
77    ///   (each dimension 1) that the op does not read.
78    ///
79    /// `like` is `[batch, nchunks, …]`.
80    pub(crate) fn op_inputs<const D: usize>(
81        restarts: Option<Self>,
82        like: &Tensor<D>,
83    ) -> (Tensor<2>, Tensor<4>, bool) {
84        let dims = like.dims();
85        let options = (&like.device(), like.dtype());
86        match restarts {
87            Some(Restarts { reset_bn, origin_bhpr }) => (
88                Tensor::ones([dims[0], dims[1]], options).mask_fill(reset_bn, 0.0),
89                origin_bhpr,
90                true,
91            ),
92            None => (
93                Tensor::ones([1, 1], options),
94                Tensor::zeros([1, 1, 1, 1], options),
95                false,
96            ),
97        }
98    }
99}

101/// `cache`, where each row with a reset takes the row of `origin` instead:
102/// the field that the last segment of each row starts from. Axis 0 is the
103/// batch.
104pub(crate) fn last_origin<const D: usize>(
105    cache: Tensor<D>,
106    origin: Tensor<D>,
107    reset_bn: &Tensor<2, Bool>,
108) -> Tensor<D> {
109    let dims = cache.dims();
110    let mut shape = [1usize; D];
111    shape[0] = dims[0];
112    let restarted = reset_bn.clone().any_dim(1).reshape(shape).expand(dims);
113    cache.mask_where(restarted, origin)
114}

116/// The reset flag of each chunk, `[batch, nchunks]`: the flag of its first
117/// token. `nchunks = ⌈tokens / chunk_tokens⌉`, so a partial last chunk also
118/// has a flag.
119///
120/// With `debug_assertions`, it panics if a reset is not at a chunk start (this
121/// check reads the device).
122pub(crate) fn chunk_resets(reset_bt: Tensor<2, Bool>, chunk_tokens: usize) -> Tensor<2, Bool> {
123    let [batch, tokens] = reset_bt.dims();
124    let nchunks = tokens.div_ceil(chunk_tokens);
125    let pad = nchunks * chunk_tokens - tokens;
126    // The partial last chunk is padded with "no reset".
127    let whole_bt = if pad == 0 {
128        reset_bt
129    } else {
130        let device = reset_bt.device();
131        let none_bt = Tensor::<2, Int>::zeros([batch, pad], &device).bool();
132        Tensor::cat(vec![reset_bt, none_bt], 1)
133    };
134    let reset_bnt = whole_bt.reshape([batch, nchunks, chunk_tokens]);
135    if cfg!(debug_assertions) && chunk_tokens > 1 {
136        let inside = reset_bnt.clone().narrow(2, 1, chunk_tokens - 1).any();
137        assert!(
138            !inside.into_scalar::<bool>().to_bool(),
139            "a reset must be at the first token of a chunk (every {chunk_tokens} tokens)"
140        );
141    }
142    reset_bnt.narrow(2, 0, 1).reshape([batch, nchunks])
143}

145/// `[batch, len]`: `true` at the first `width` positions of every chunk that
146/// starts with a reset. `len ≤ nchunks · chunk_len`.
147///
148/// These are the positions whose reads `width` positions back cross the reset.
149pub(crate) fn chunk_heads(
150    reset_bn: &Tensor<2, Bool>,
151    chunk_len: usize,
152    width: usize,
153    len: usize,
154) -> Tensor<2, Bool> {
155    let [batch, nchunks] = reset_bn.dims();
156    let device = reset_bn.device();
157    let head_bnl = Tensor::<1, Int>::arange(0..chunk_len as i64, &device)
158        .lower_elem(width as i64)
159        .reshape([1, 1, chunk_len])
160        .expand([batch, nchunks, chunk_len]);
161    reset_bn
162        .clone()
163        .unsqueeze_dim::<3>(2)
164        .expand([batch, nchunks, chunk_len])
165        .bool_and(head_bnl)
166        .reshape([batch, nchunks * chunk_len])
167        .narrow(1, 0, len)
168}

170/// `x`, where `head` replaces the first `width` positions (along `axis`) of
171/// every chunk that starts with a reset. `head` holds the values of the origin
172/// cache that those positions read at a restart: `x` with `width` entries
173/// along `axis`, one set per row. Axis 0 is the batch.
174pub(crate) fn restart_heads<const D: usize>(
175    x: Tensor<D>,
176    axis: usize,
177    head: Tensor<D>,
178    reset_bn: &Tensor<2, Bool>,
179    chunk_len: usize,
180) -> Tensor<D> {
181    let dims = x.dims();
182    let len = dims[axis];
183    let width = head.dims()[axis];
184    // `head` at every position, by its index in the chunk (clamped: the
185    // positions past `width` are not selected). A gather, because its
186    // backward is a scatter-add, and the backward of `repeat_dim` is wrong in
187    // this Burn.
188    let mut line = [1usize; D];
189    line[axis] = len;
190    let idx = Tensor::<1, Int>::arange(0..len as i64, &x.device())
191        .remainder_scalar(chunk_len as i64)
192        .clamp_max(width as i64 - 1)
193        .reshape(line)
194        .expand(dims);
195    let tiled = head.gather(axis, idx);
196    let mut shape = [1usize; D];
197    shape[0] = dims[0];
198    shape[axis] = len;
199    let heads = chunk_heads(reset_bn, chunk_len, width, len)
200        .reshape(shape)
201        .expand(dims);
202    x.mask_where(heads, tiled)
203}

205/// `[batch, len]`: the first position of the segment that holds each
206/// position. It is `-1` where no reset comes before the position in its row:
207/// the incoming cache continues there.
208pub(crate) fn segment_start(reset_bn: &Tensor<2, Bool>, chunk_len: usize, len: usize) -> Tensor<2, Int> {
209    let [batch, nchunks] = reset_bn.dims();
210    let device = reset_bn.device();
211    let first_bn = (Tensor::<1, Int>::arange(0..nchunks as i64, &device) * chunk_len as i64)
212        .reshape([1, nchunks])
213        .expand([batch, nchunks]);
214    Tensor::<2, Int>::full([batch, nchunks], -1, &device)
215        .mask_where(reset_bn.clone(), first_bn)
216        .cummax(1)
217        .unsqueeze_dim::<3>(2)
218        .expand([batch, nchunks, chunk_len])
219        .reshape([batch, nchunks * chunk_len])
220        .narrow(1, 0, len)
221}

223/// `[batch]`: the first position of the last segment of each row, or `0` for
224/// a row with no reset.
225pub(crate) fn last_start_b(reset_bn: &Tensor<2, Bool>, chunk_len: usize) -> Tensor<1, Int> {
226    let [batch, nchunks] = reset_bn.dims();
227    let device = reset_bn.device();
228    (Tensor::<1, Int>::arange(0..nchunks as i64, &device) * chunk_len as i64)
229        .reshape([1, nchunks])
230        .expand([batch, nchunks])
231        .mask_fill(reset_bn.clone().bool_not(), 0)
232        .max_dim(1)
233        .reshape([batch])
234}

236/// The resets of a packed call on its folded axis (Mamba-3: `micro_steps`
237/// positions per token), with what the sites that read the past need.
238pub(crate) struct Segments {
239    /// `[batch, nchunks]`: the chunks that start a segment.
240    pub reset_bn: Tensor<2, Bool>,
241    /// Positions per chunk.
242    pub chunk_len: usize,
243    /// `[batch, len]`: see [`segment_start`].
244    pub start_bs: Tensor<2, Int>,
245}
246
247impl Segments {
248    /// `len` is the number of positions of the call (not padded to a chunk).
249    pub fn new(reset_bn: Tensor<2, Bool>, chunk_len: usize, len: usize) -> Self {
250        let start_bs = segment_start(&reset_bn, chunk_len, len);
251        Self {
252            reset_bn,
253            chunk_len,
254            start_bs,
255        }
256    }
257
258    /// `[batch, len]`: `true` at the first `width` positions of each segment
259    /// (see [`chunk_heads`]).
260    pub fn heads(&self, width: usize) -> Tensor<2, Bool> {
261        let len = self.start_bs.dims()[1];
262        chunk_heads(&self.reset_bn, self.chunk_len, width, len)
263    }
264
265    /// See [`restart_heads`].
266    pub fn restart_heads<const D: usize>(&self, x: Tensor<D>, axis: usize, head: Tensor<D>) -> Tensor<D> {
267        restart_heads(x, axis, head, &self.reset_bn, self.chunk_len)
268    }
269
270    /// See [`last_start_b`].
271    pub fn last_start_b(&self) -> Tensor<1, Int> {
272        last_start_b(&self.reset_bn, self.chunk_len)
273    }
274
275    /// See [`last_origin`].
276    pub fn last_origin<const D: usize>(&self, cache: Tensor<D>, origin: Tensor<D>) -> Tensor<D> {
277        last_origin(cache, origin, &self.reset_bn)
278    }
279}

281/// [`crate::padding::window`] of `x_all` (`[start | row]` along `axis`, where
282/// the `start` field is `width` entries) at `end_b`, as the last segment of
283/// each row sees it. That segment starts from the `start` field, so an entry
284/// from before its start is read from that field instead. `start` is the
285/// field of [`last_origin`]: the origin for a row with a reset, else the
286/// incoming cache. `last_start_b` is `0` for a row with no reset, which gives
287/// the plain window.
288pub(crate) fn restart_window<const D: usize>(
289    x_all: Tensor<D>,
290    axis: usize,
291    end_b: Tensor<1, Int>,
292    width: usize,
293    last_start_b: Tensor<1, Int>,
294) -> Tensor<D> {
295    let [batch] = end_b.dims();
296    let device = x_all.device();
297    let start_bw = last_start_b.reshape([batch, 1]).expand([batch, width]);
298    // In `[start | last segment]`, the window starts at the real length of
299    // the segment (zero for a segment with no real row).
300    let local_end_bw = (end_b.reshape([batch, 1]).expand([batch, width]) - start_bw.clone()).clamp_min(0);
301    let local_bw = Tensor::<1, Int>::arange(0..width as i64, &device)
302        .reshape([1, width])
303        .expand([batch, width])
304        + local_end_bw;
305    // Back to `[start | row]`: a `start` entry stays, a segment entry moves
306    // by the start of the segment.
307    let in_cache_bw = local_bw.clone().lower_elem(width as i64);
308    let idx_bw = local_bw + start_bw.mask_fill(in_cache_bw, 0);
309    let mut shape = [1usize; D];
310    shape[0] = batch;
311    shape[axis] = width;
312    let mut dims = x_all.dims();
313    dims[axis] = width;
314    x_all.gather(axis, idx_bw.reshape(shape).expand(dims))
315}
