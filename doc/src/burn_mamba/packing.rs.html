1//! Packed segments inside a block: several sequences one after another in one
2//! row, each from one origin cache.
3//!
4//! `reset_bt` (`[batch, tokens]`, `true` at the first token of a segment)
5//! marks where a row starts a new sequence. At that token, the block starts
6//! again from the **origin** cache (`origin`, a fresh one for `origin = None`). The
7//! tokens before the first reset of a row continue from the incoming cache
8//! (`cache`), as in a plain call. So a stream of packed calls can continue a
9//! sequence across a call boundary. A reset at the first token of a call
10//! starts a new sequence.
11//!
12//! Each sequence gives what its own row gives in a batch of plain calls from
13//! the origin: the outputs, the last cache and the gradients. This is true for
14//! any split of the stream into calls. The previous segment gets no gradient
15//! through a reset. The origin gets the gradients of every segment that
16//! starts from it. `origin = cache` starts every segment from the incoming
17//! cache (for example, a shared prefix).
18//!
19//! # Only at a chunk start
20//!
21//! A reset must be at the first token of a chunk (`chunk_len` tokens, or
22//! `chunk_len / micro_steps` for Mamba-3). Then the chunkwise scan changes at
23//! one place only, the carry between chunks (K4). Every other site that reads
24//! an earlier sample (a conv tap, a trapezoid tap) crosses a reset exactly when
25//! two conditions are true:
26//!
27//! - its chunk starts with a reset,
28//! - it reads further back than its own index in the chunk.
29//!
30//! Such a read takes the value of the origin instead (`restart_heads`). So a
31//! per-chunk flag is all that those sites need, and no scan runs over the row
32//! for them. This needs a read that reaches back one chunk at most:
33//!
34//! - A Mamba-2 conv tap reads `conv_kernel − 1` positions back. So
35//!   `Mamba2::forward_packed` asserts `chunk_len ≥ conv_kernel − 1`.
36//! - A Mamba-3 tap reads `u` positions (one token) back at most.
37//!
38//! A segment can end with padded positions (`pad`), up to the next chunk start. The official Mamba-3 varlen kernels also start each
39//! sequence at a chunk start, with a padded last chunk.
40//!
41//! Some scans run over the whole row: the cumulative rotation of Mamba-3 and
42//! its positive systems. They start again at the first position of each segment,
43//! from the carry of the origin.
44//!
45//! The cache that a call returns is that of the last segment of each row. Its
46//! "last samples" fields (conv window, tap FIFO, carries) read the origin for
47//! the entries before the start of that segment (`last_origin`,
48//! `restart_window`).
49//!
50//! Only the serial SSD paths take resets. The inter-chunk decay of `Minimal`
51//! is a `nchunks × nchunks` matrix, which a long packed row makes large.
52//!
53//! The entry points are `Mamba2::forward_packed` and
54//! `Mamba3::forward_packed`.
55

66/// The resets of a packed row as an SSD pass takes them: the chunks that
67/// restart, and the state that they restart from.
68#[derive(Clone, Debug)]
69pub struct Restarts {
70    /// `[batch, nchunks]`: `true` at a chunk that starts a new segment.
71    pub reset_bn: Tensor<2, Bool>,
72    /// `[batch, nheads, per_head_dim, state_rank]`: the state of the origin
73    /// cache. Each chunk of `reset_bn` starts from it.
74    pub origin_bhpr: Tensor<4>,
75}
76
77impl Restarts {
78    /// The restarts as an SSD op takes them ([`OpRestarts`]). `keep_bn` gets
79    /// the dtype of `like`, and `like` is `[batch, nchunks, …]`.
80    pub(crate) fn op_inputs<const D: usize>(
81        restarts: Option<Self>,
82        like: &Tensor<D>,
83    ) -> OpRestarts<Dispatch> {
84        let Some(Restarts { reset_bn, origin_bhpr }) = restarts else {
85            return OpRestarts::None;
86        };
87        let dims = like.dims();
88        let options = (&like.device(), like.dtype());
89        let keep_bn = Tensor::<2>::ones([dims[0], dims[1]], options).mask_fill(reset_bn, 0.0);
90        OpRestarts::Some {
91            keep_bn: keep_bn.into_dispatch(),
92            origin_bhpr: origin_bhpr.into_dispatch(),
93        }
94    }
95}

97/// The restarts of a packed row as an SSD op takes them, on the primitives of
98/// `B`. An op argument of this type has `#[extension_type]`.
99///
100/// This enum takes the place of an `Option` of the two tensors. Burn has no
101/// `ExtensionType` for `Option`, and the orphan rule stops this crate from
102/// adding one.
103#[derive(ExtensionType)]
104pub enum OpRestarts<B: Backend> {
105    /// No chunk of the call restarts.
106    None,
107    /// The chunks that restart, and the state that they restart from.
108    Some {
109        /// `[batch, nchunks]`: `0` at a chunk that starts a new segment, `1`
110        /// elsewhere. It is a constant: the op registers no gradient for it.
111        keep_bn: FloatTensor<B>,
112        /// `[batch, nheads, per_head_dim, state_rank]`: the state that each
113        /// such chunk starts from.
114        origin_bhpr: FloatTensor<B>,
115    },
116}
117
118impl<B: Backend> OpRestarts<B> {
119    /// The pair `(keep_bn, origin_bhpr)` that the standard SSD scan takes.
120    pub(crate) fn into_option(self) -> Option<(F<B, 2>, F<B, 4>)> {
121        match self {
122            OpRestarts::None => None,
123            OpRestarts::Some { keep_bn, origin_bhpr } => {
124                Some((F::new(keep_bn), F::new(origin_bhpr)))
125            }
126        }
127    }
128
129    /// The restarts from the two parts that a recompute node keeps (see
130    /// [`OpRestarts::into_recompute`]). `origin_bhpr` stays unread when
131    /// `keep_bn` is `None`.
132    #[cfg(feature = "autodiff")]
133    pub(crate) fn from_recompute(keep_bn: Option<FloatTensor<B>>, origin_bhpr: FloatTensor<B>) -> Self {
134        match keep_bn {
135            Some(keep_bn) => OpRestarts::Some { keep_bn, origin_bhpr },
136            None => OpRestarts::None,
137        }
138    }
139}

142impl<B: Backend, C: CheckpointStrategy> OpRestarts<Autodiff<B, C>> {
143    /// The restarts as the `Autodiff` impl of an SSD op gives them to its
144    /// recompute node:
145    ///
146    /// - `keep_bn`, on the primitives of `B`: a constant, with no gradient.
147    /// - `origin_bhpr`: a tracked input. For `None`, it is a [`placeholder`].
148    pub(crate) fn into_recompute(
149        self,
150        like: &FloatTensor<Autodiff<B, C>>,
151    ) -> (Option<FloatTensor<B>>, FloatTensor<Autodiff<B, C>>) {
152        match self {
153            OpRestarts::Some { keep_bn, origin_bhpr } => (Some(keep_bn.into_primitive()), origin_bhpr),
154            OpRestarts::None => (None, placeholder::<B, C, 4>(like)),
155        }
156    }
157}

159/// The placeholder of a missing tracked input of a recompute node: the
160/// `origin_bhpr` or the `seed_bnhpr` of a call with no resets. The node takes
161/// a fixed set of tracked inputs. So the `Autodiff` impl of an SSD op gives it
162/// this tensor, which the op does not read. Each dimension is 1, and the
163/// device and the dtype are those of `like`.
164#[cfg(feature = "autodiff")]
165pub(crate) fn placeholder<B: Backend, C: CheckpointStrategy, const D: usize>(
166    like: &FloatTensor<Autodiff<B, C>>,
167) -> FloatTensor<Autodiff<B, C>> {
168    use burn::backend::TensorMetadata;
169    F::<Autodiff<B, C>, D>::zeros([1; D], &like.device(), like.dtype().into()).inner()
170}

172/// `cache`, where each row with a reset takes the row of `origin` instead:
173/// the field that the last segment of each row starts from. Axis 0 is the
174/// batch.
175pub(crate) fn last_origin<const D: usize>(
176    cache: Tensor<D>,
177    origin: Tensor<D>,
178    reset_bn: &Tensor<2, Bool>,
179) -> Tensor<D> {
180    let dims = cache.dims();
181    let mut shape = [1usize; D];
182    shape[0] = dims[0];
183    let restarted = reset_bn.clone().any_dim(1).reshape(shape).expand(dims);
184    cache.mask_where(restarted, origin)
185}

187/// The reset flag of each chunk, `[batch, nchunks]`: the flag of its first
188/// token. `nchunks = ⌈tokens / chunk_tokens⌉`, so a partial last chunk also
189/// has a flag.
190///
191/// With `debug_assertions`, it panics if a reset is not at a chunk start (this
192/// check reads the device).
193pub(crate) fn chunk_resets(reset_bt: Tensor<2, Bool>, chunk_tokens: usize) -> Tensor<2, Bool> {
194    let [batch, tokens] = reset_bt.dims();
195    let nchunks = tokens.div_ceil(chunk_tokens);
196    let pad = nchunks * chunk_tokens - tokens;
197    // The partial last chunk is padded with "no reset".
198    let whole_bt = if pad == 0 {
199        reset_bt
200    } else {
201        let device = reset_bt.device();
202        let none_bt = Tensor::<2, Int>::zeros([batch, pad], &device).bool();
203        Tensor::cat(vec![reset_bt, none_bt], 1)
204    };
205    let reset_bnt = whole_bt.reshape([batch, nchunks, chunk_tokens]);
206    if cfg!(debug_assertions) && chunk_tokens > 1 {
207        let inside = reset_bnt.clone().narrow(2, 1, chunk_tokens - 1).any();
208        assert!(
209            !inside.into_scalar::<bool>().to_bool(),
210            "a reset must be at the first token of a chunk (every {chunk_tokens} tokens)"
211        );
212    }
213    reset_bnt.narrow(2, 0, 1).reshape([batch, nchunks])
214}

216/// `[batch, len]`: `true` at the first `width` positions of every chunk that
217/// starts with a reset. `len ≤ nchunks · chunk_len`.
218///
219/// These are the positions whose reads `width` positions back cross the reset.
220pub(crate) fn chunk_heads(
221    reset_bn: &Tensor<2, Bool>,
222    chunk_len: usize,
223    width: usize,
224    len: usize,
225) -> Tensor<2, Bool> {
226    let [batch, nchunks] = reset_bn.dims();
227    let device = reset_bn.device();
228    let head_bnl = Tensor::<1, Int>::arange(0..chunk_len as i64, &device)
229        .lower_elem(width as i64)
230        .reshape([1, 1, chunk_len])
231        .expand([batch, nchunks, chunk_len]);
232    reset_bn
233        .clone()
234        .unsqueeze_dim::<3>(2)
235        .expand([batch, nchunks, chunk_len])
236        .bool_and(head_bnl)
237        .reshape([batch, nchunks * chunk_len])
238        .narrow(1, 0, len)
239}

241/// `x`, where `head` replaces the first `width` positions (along `axis`) of
242/// every chunk that starts with a reset. `head` holds the values of the origin
243/// cache that those positions read at a reset: `x` with `width` entries
244/// along `axis`, one set per row. Axis 0 is the batch.
245pub(crate) fn restart_heads<const D: usize>(
246    x: Tensor<D>,
247    axis: usize,
248    head: Tensor<D>,
249    reset_bn: &Tensor<2, Bool>,
250    chunk_len: usize,
251) -> Tensor<D> {
252    let dims = x.dims();
253    let len = dims[axis];
254    let width = head.dims()[axis];
255    // `head` at every position, by its index in the chunk (clamped: the
256    // positions past `width` are not selected). A gather, because its
257    // backward is a scatter-add, and the backward of `repeat_dim` is wrong in
258    // this Burn.
259    let mut line = [1usize; D];
260    line[axis] = len;
261    let idx = Tensor::<1, Int>::arange(0..len as i64, &x.device())
262        .remainder_scalar(chunk_len as i64)
263        .clamp_max(width as i64 - 1)
264        .reshape(line)
265        .expand(dims);
266    let tiled = head.gather(axis, idx);
267    let mut shape = [1usize; D];
268    shape[0] = dims[0];
269    shape[axis] = len;
270    let heads = chunk_heads(reset_bn, chunk_len, width, len)
271        .reshape(shape)
272        .expand(dims);
273    x.mask_where(heads, tiled)
274}

276/// `[batch, len]`: the first position of the segment that holds each
277/// position. It is `-1` where no reset comes before the position in its row:
278/// the incoming cache continues there.
279pub(crate) fn segment_start(reset_bn: &Tensor<2, Bool>, chunk_len: usize, len: usize) -> Tensor<2, Int> {
280    let [batch, nchunks] = reset_bn.dims();
281    let device = reset_bn.device();
282    let first_bn = (Tensor::<1, Int>::arange(0..nchunks as i64, &device) * chunk_len as i64)
283        .reshape([1, nchunks])
284        .expand([batch, nchunks]);
285    Tensor::<2, Int>::full([batch, nchunks], -1, &device)
286        .mask_where(reset_bn.clone(), first_bn)
287        .cummax(1)
288        .unsqueeze_dim::<3>(2)
289        .expand([batch, nchunks, chunk_len])
290        .reshape([batch, nchunks * chunk_len])
291        .narrow(1, 0, len)
292}

294/// `[batch]`: the first position of the last segment of each row, or `0` for
295/// a row with no reset.
296pub(crate) fn last_start_b(reset_bn: &Tensor<2, Bool>, chunk_len: usize) -> Tensor<1, Int> {
297    let [batch, nchunks] = reset_bn.dims();
298    let device = reset_bn.device();
299    (Tensor::<1, Int>::arange(0..nchunks as i64, &device) * chunk_len as i64)
300        .reshape([1, nchunks])
301        .expand([batch, nchunks])
302        .mask_fill(reset_bn.clone().bool_not(), 0)
303        .max_dim(1)
304        .reshape([batch])
305}

307/// The resets of a packed call on its write axis (Mamba-3: `micro_steps`
308/// positions per token), with what the sites that read the past need.
309pub(crate) struct Segments {
310    /// `[batch, nchunks]`: the chunks that start a segment.
311    pub reset_bn: Tensor<2, Bool>,
312    /// Positions per chunk.
313    pub chunk_len: usize,
314    /// `[batch, len]`: see [`segment_start`].
315    pub start_bs: Tensor<2, Int>,
316}
317
318impl Segments {
319    /// `len` is the number of positions of the call (not padded to a chunk).
320    pub fn new(reset_bn: Tensor<2, Bool>, chunk_len: usize, len: usize) -> Self {
321        let start_bs = segment_start(&reset_bn, chunk_len, len);
322        Self {
323            reset_bn,
324            chunk_len,
325            start_bs,
326        }
327    }
328
329    /// `[batch, len]`: `true` at the first `width` positions of each segment
330    /// (see [`chunk_heads`]).
331    pub fn heads(&self, width: usize) -> Tensor<2, Bool> {
332        let len = self.start_bs.dims()[1];
333        chunk_heads(&self.reset_bn, self.chunk_len, width, len)
334    }
335
336    /// See [`restart_heads`].
337    pub fn restart_heads<const D: usize>(&self, x: Tensor<D>, axis: usize, head: Tensor<D>) -> Tensor<D> {
338        restart_heads(x, axis, head, &self.reset_bn, self.chunk_len)
339    }
340
341    /// See [`last_start_b`].
342    pub fn last_start_b(&self) -> Tensor<1, Int> {
343        last_start_b(&self.reset_bn, self.chunk_len)
344    }
345
346    /// See [`last_origin`].
347    pub fn last_origin<const D: usize>(&self, cache: Tensor<D>, origin: Tensor<D>) -> Tensor<D> {
348        last_origin(cache, origin, &self.reset_bn)
349    }
350}

352/// [`crate::padding::window`] of `x_all` (`[start | row]` along `axis`, where
353/// the `start` field is `width` entries) at `end_b`, as the last segment of
354/// each row sees it. That segment starts from the `start` field, so an entry
355/// from before its start is read from that field instead. `start` is the
356/// field of [`last_origin`]: the origin for a row with a reset, else the
357/// incoming cache. `last_start_b` is `0` for a row with no reset, which gives
358/// the plain window.
359pub(crate) fn restart_window<const D: usize>(
360    x_all: Tensor<D>,
361    axis: usize,
362    end_b: Tensor<1, Int>,
363    width: usize,
364    last_start_b: Tensor<1, Int>,
365) -> Tensor<D> {
366    let [batch] = end_b.dims();
367    let device = x_all.device();
368    let start_bw = last_start_b.reshape([batch, 1]).expand([batch, width]);
369    // In `[start | last segment]`, the window starts at the real length of
370    // the segment (zero for a segment with no real token).
371    let local_end_bw = (end_b.reshape([batch, 1]).expand([batch, width]) - start_bw.clone()).clamp_min(0);
372    let local_bw = Tensor::<1, Int>::arange(0..width as i64, &device)
373        .reshape([1, width])
374        .expand([batch, width])
375        + local_end_bw;
376    // Back to `[start | row]`: a `start` entry stays, a segment entry moves
377    // by the start of the segment.
378    let in_cache_bw = local_bw.clone().lower_elem(width as i64);
379    let idx_bw = local_bw + start_bw.mask_fill(in_cache_bw, 0);
380    let mut shape = [1usize; D];
381    shape[0] = batch;
382    shape[axis] = width;
383    let mut dims = x_all.dims();
384    dims[axis] = width;
385    x_all.gather(axis, idx_bw.reshape(shape).expand(dims))
386}
