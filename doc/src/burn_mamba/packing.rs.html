1//! Packed segments inside a block: several sequences one after another in one
2//! row, each from one origin cache.
3//!
4//! `reset_bt` (`[batch, tokens]`, `true` at the first token of a segment)
5//! marks where a row starts a new sequence. At that token, the block starts
6//! again from the **origin** cache (`origin`, a fresh one for `origin = None`). The
7//! tokens before the first reset of a row continue from the incoming cache
8//! (`cache`), as in a plain call. So a stream of packed calls can continue a
9//! sequence across a call boundary. A reset at the first token of a call
10//! starts a new sequence.
11//!
12//! Each sequence gives what its own row gives in a batch of plain calls from
13//! the origin: the outputs, the last cache and the gradients. This is true for
14//! any split of the stream into calls. The previous segment gets no gradient
15//! through a reset. The origin gets the gradients of every segment that
16//! starts from it. `origin = cache` starts every segment from the incoming
17//! cache (for example, a shared prefix).
18//!
19//! # Only at a chunk start
20//!
21//! A reset must be at the first token of a chunk (`chunk_len` tokens, or
22//! `chunk_len / micro_steps` for Mamba-3). Then the chunkwise scan changes at
23//! one place only, the carry between chunks (K4). Every other site that reads
24//! an earlier sample (a conv tap, a trapezoid tap) crosses a reset exactly when
25//! two conditions are true:
26//!
27//! - its chunk starts with a reset,
28//! - it reads further back than its own index in the chunk.
29//!
30//! Such a read takes the value of the origin instead (`origin_heads`). So a
31//! per-chunk flag is all that those sites need, and no scan runs over the row
32//! for them. This needs a read that reaches back one chunk at most:
33//!
34//! - A Mamba-2 conv tap reads `conv_kernel − 1` positions back. So
35//!   `Mamba2::forward_packed` asserts `chunk_len ≥ conv_kernel − 1`.
36//! - A Mamba-3 tap reads `u` positions (one token) back at most.
37//!
38//! A segment can end with padded positions (`pad`), up to the next chunk start. The official Mamba-3 varlen kernels also start each
39//! sequence at a chunk start, with a padded last chunk.
40//!
41//! Some scans run over the whole row: the cumulative rotation of Mamba-3 and
42//! its positive systems. They start again at the first position of each segment,
43//! from the carry of the origin.
44//!
45//! The cache that a call returns is that of the last segment of each row. Its
46//! "last samples" fields (conv window, tap FIFO, carries) read the origin for
47//! the entries before the start of that segment (`crate::row_end`).
48//!
49//! Only the serial SSD paths take resets. The inter-chunk decay of `Minimal`
50//! is a `nchunks × nchunks` matrix, which a long packed row makes large.
51//!
52//! The entry points are `Mamba2::forward_packed` and
53//! `Mamba3::forward_packed`.
54

65/// The resets of a packed row as an SSD pass takes them: the chunks with a
66/// reset, and the state that they start from.
67#[derive(Clone, Debug)]
68pub struct SsdResets {
69    /// `[batch, nchunks]`: `true` at a chunk that starts a new segment.
70    pub reset_bn: Tensor<2, Bool>,
71    /// `[batch, nheads, per_head_dim, state_rank]`: the state of the origin
72    /// cache. Each chunk of `reset_bn` starts from it.
73    pub origin_bhpr: Tensor<4>,
74}
75
76impl SsdResets {
77    /// The resets as an SSD op takes them ([`OpResets`]). `keep_bn` gets the
78    /// dtype of `like`, and `like` is `[batch, nchunks, …]`.
79    pub(crate) fn op_inputs<const D: usize>(
80        resets: Option<Self>,
81        like: &Tensor<D>,
82    ) -> OpResets<Dispatch> {
83        let Some(SsdResets { reset_bn, origin_bhpr }) = resets else {
84            return OpResets::None;
85        };
86        let dims = like.dims();
87        let options = (&like.device(), like.dtype());
88        let keep_bn = Tensor::<2>::ones([dims[0], dims[1]], options).mask_fill(reset_bn, 0.0);
89        OpResets::Some {
90            keep_bn: keep_bn.into_dispatch(),
91            origin_bhpr: origin_bhpr.into_dispatch(),
92        }
93    }
94}

96/// The resets of a packed row as an SSD op takes them, on the primitives of
97/// `B`. An op argument of this type has `#[extension_type]`.
98///
99/// This enum takes the place of an `Option` of the two tensors. Burn has no
100/// `ExtensionType` for `Option`, and the orphan rule stops this crate from
101/// adding one.
102#[derive(ExtensionType)]
103pub enum OpResets<B: Backend> {
104    /// No chunk of the call has a reset.
105    None,
106    /// The chunks with a reset, and the state that they start from.
107    Some {
108        /// `[batch, nchunks]`: `0` at a chunk that starts a new segment, `1`
109        /// elsewhere. It is a constant: the op registers no gradient for it.
110        keep_bn: FloatTensor<B>,
111        /// `[batch, nheads, per_head_dim, state_rank]`: the state that each
112        /// such chunk starts from.
113        origin_bhpr: FloatTensor<B>,
114    },
115}
116
117impl<B: Backend> OpResets<B> {
118    /// The pair `(keep_bn, origin_bhpr)` that the standard SSD scan takes.
119    pub(crate) fn into_option(self) -> Option<(F<B, 2>, F<B, 4>)> {
120        match self {
121            OpResets::None => None,
122            OpResets::Some { keep_bn, origin_bhpr } => {
123                Some((F::new(keep_bn), F::new(origin_bhpr)))
124            }
125        }
126    }
127
128    /// The resets from the two parts that a recompute node keeps (see
129    /// [`OpResets::into_recompute`]). `origin_bhpr` stays unread when
130    /// `keep_bn` is `None`.
131    #[cfg(feature = "autodiff")]
132    pub(crate) fn from_recompute(keep_bn: Option<FloatTensor<B>>, origin_bhpr: FloatTensor<B>) -> Self {
133        match keep_bn {
134            Some(keep_bn) => OpResets::Some { keep_bn, origin_bhpr },
135            None => OpResets::None,
136        }
137    }
138}

141impl<B: Backend, C: CheckpointStrategy> OpResets<Autodiff<B, C>> {
142    /// The resets as the `Autodiff` impl of an SSD op gives them to its
143    /// recompute node:
144    ///
145    /// - `keep_bn`, on the primitives of `B`: a constant, with no gradient.
146    /// - `origin_bhpr`: a tracked input. For `None`, it is a [`placeholder`].
147    pub(crate) fn into_recompute(
148        self,
149        like: &FloatTensor<Autodiff<B, C>>,
150    ) -> (Option<FloatTensor<B>>, FloatTensor<Autodiff<B, C>>) {
151        match self {
152            OpResets::Some { keep_bn, origin_bhpr } => (Some(keep_bn.into_primitive()), origin_bhpr),
153            OpResets::None => (None, placeholder::<B, C, 4>(like)),
154        }
155    }
156}

158/// The placeholder of a missing tracked input of a recompute node: the
159/// `origin_bhpr` or the `seed_bnhpr` of a call with no resets. The node takes
160/// a fixed set of tracked inputs. So the `Autodiff` impl of an SSD op gives it
161/// this tensor, which the op does not read. Each dimension is 1, and the
162/// device and the dtype are those of `like`.
163#[cfg(feature = "autodiff")]
164pub(crate) fn placeholder<B: Backend, C: CheckpointStrategy, const D: usize>(
165    like: &FloatTensor<Autodiff<B, C>>,
166) -> FloatTensor<Autodiff<B, C>> {
167    use burn::backend::TensorMetadata;
168    F::<Autodiff<B, C>, D>::zeros([1; D], &like.device(), like.dtype().into()).inner()
169}

171/// The reset flag of each chunk, `[batch, nchunks]`: the flag of its first
172/// token. `nchunks = ⌈tokens / chunk_tokens⌉`, so a partial last chunk also
173/// has a flag.
174///
175/// With `debug_assertions`, it panics if a reset is not at a chunk start (this
176/// check reads the device).
177pub(crate) fn chunk_resets(reset_bt: Tensor<2, Bool>, chunk_tokens: usize) -> Tensor<2, Bool> {
178    let [batch, tokens] = reset_bt.dims();
179    let nchunks = tokens.div_ceil(chunk_tokens);
180    let pad = nchunks * chunk_tokens - tokens;
181    // The partial last chunk is padded with "no reset".
182    let whole_bt = if pad == 0 {
183        reset_bt
184    } else {
185        let device = reset_bt.device();
186        let none_bt = Tensor::<2, Int>::zeros([batch, pad], &device).bool();
187        Tensor::cat(vec![reset_bt, none_bt], 1)
188    };
189    let reset_bnt = whole_bt.reshape([batch, nchunks, chunk_tokens]);
190    if cfg!(debug_assertions) && chunk_tokens > 1 {
191        let inside = reset_bnt.clone().narrow(2, 1, chunk_tokens - 1).any();
192        assert!(
193            !inside.into_scalar::<bool>().to_bool(),
194            "a reset must be at the first token of a chunk (every {chunk_tokens} tokens)"
195        );
196    }
197    reset_bnt.narrow(2, 0, 1).reshape([batch, nchunks])
198}

200/// `[batch, len]`: `true` at the first `width` positions of every chunk that
201/// starts with a reset. `len ≤ nchunks · chunk_len`.
202///
203/// These are the positions whose reads `width` positions back cross the reset.
204pub(crate) fn chunk_heads(
205    reset_bn: &Tensor<2, Bool>,
206    chunk_len: usize,
207    width: usize,
208    len: usize,
209) -> Tensor<2, Bool> {
210    let [batch, nchunks] = reset_bn.dims();
211    let device = reset_bn.device();
212    let head_bnl = Tensor::<1, Int>::arange(0..chunk_len as i64, &device)
213        .lower_elem(width as i64)
214        .reshape([1, 1, chunk_len])
215        .expand([batch, nchunks, chunk_len]);
216    reset_bn
217        .clone()
218        .unsqueeze_dim::<3>(2)
219        .expand([batch, nchunks, chunk_len])
220        .bool_and(head_bnl)
221        .reshape([batch, nchunks * chunk_len])
222        .narrow(1, 0, len)
223}

225/// `x`, where `head` replaces the first `width` positions (along `axis`) of
226/// every chunk that starts with a reset. `head` holds the values of the origin
227/// cache that those positions read at a reset: `x` with `width` entries
228/// along `axis`, one set per row. Axis 0 is the batch.
229pub(crate) fn origin_heads<const D: usize>(
230    x: Tensor<D>,
231    axis: usize,
232    head: Tensor<D>,
233    reset_bn: &Tensor<2, Bool>,
234    chunk_len: usize,
235) -> Tensor<D> {
236    let dims = x.dims();
237    let len = dims[axis];
238    let width = head.dims()[axis];
239    // `head` at every position, by its index in the chunk (clamped: the
240    // positions past `width` are not selected). A gather, because its
241    // backward is a scatter-add, and the backward of `repeat_dim` is wrong in
242    // this Burn.
243    let mut line = [1usize; D];
244    line[axis] = len;
245    let idx = Tensor::<1, Int>::arange(0..len as i64, &x.device())
246        .remainder_scalar(chunk_len as i64)
247        .clamp_max(width as i64 - 1)
248        .reshape(line)
249        .expand(dims);
250    let tiled = head.gather(axis, idx);
251    let mut shape = [1usize; D];
252    shape[0] = dims[0];
253    shape[axis] = len;
254    let heads = chunk_heads(reset_bn, chunk_len, width, len)
255        .reshape(shape)
256        .expand(dims);
257    x.mask_where(heads, tiled)
258}

260/// `[batch, len]`: the first position of the segment that holds each
261/// position. It is `-1` where no reset comes before the position in its row:
262/// the incoming cache continues there.
263pub(crate) fn segment_start(reset_bn: &Tensor<2, Bool>, chunk_len: usize, len: usize) -> Tensor<2, Int> {
264    let [batch, nchunks] = reset_bn.dims();
265    let device = reset_bn.device();
266    let first_bn = (Tensor::<1, Int>::arange(0..nchunks as i64, &device) * chunk_len as i64)
267        .reshape([1, nchunks])
268        .expand([batch, nchunks]);
269    Tensor::<2, Int>::full([batch, nchunks], -1, &device)
270        .mask_where(reset_bn.clone(), first_bn)
271        .cummax(1)
272        .unsqueeze_dim::<3>(2)
273        .expand([batch, nchunks, chunk_len])
274        .reshape([batch, nchunks * chunk_len])
275        .narrow(1, 0, len)
276}

278/// The resets of a packed call on its write axis (Mamba-3: `micro_steps`
279/// positions per token), with what the sites that read the past need.
280pub(crate) struct Segments {
281    /// `[batch, nchunks]`: the chunks that start a segment.
282    pub reset_bn: Tensor<2, Bool>,
283    /// Positions per chunk.
284    pub chunk_len: usize,
285    /// `[batch, len]`: see [`segment_start`].
286    pub start_bs: Tensor<2, Int>,
287}
288
289impl Segments {
290    /// `len` is the number of positions of the call (not padded to a chunk).
291    pub fn new(reset_bn: Tensor<2, Bool>, chunk_len: usize, len: usize) -> Self {
292        let start_bs = segment_start(&reset_bn, chunk_len, len);
293        Self {
294            reset_bn,
295            chunk_len,
296            start_bs,
297        }
298    }
299
300    /// `[batch, len]`: `true` at the first `width` positions of each segment
301    /// (see [`chunk_heads`]).
302    pub fn heads(&self, width: usize) -> Tensor<2, Bool> {
303        let len = self.start_bs.dims()[1];
304        chunk_heads(&self.reset_bn, self.chunk_len, width, len)
305    }
306
307    /// See [`origin_heads`].
308    pub fn origin_heads<const D: usize>(&self, x: Tensor<D>, axis: usize, head: Tensor<D>) -> Tensor<D> {
309        origin_heads(x, axis, head, &self.reset_bn, self.chunk_len)
310    }
311}
