1//! Right padding inside a block: the part of the
2//! [`burn_stack::utils::padding`] contract that a family keeps.
3//!
4//! A padded row is absent: the outputs of the real rows and the returned cache
5//! are those of the real rows of each slot, run alone. Every family meets this
6//! contract in the same two ways:
7//!
8//! - **The state** goes through a padded row unchanged. The family zeroes the
9//!   decay and the writes of that step where the discretisation forms them
10//!   (`Δ = 0` gives `Ā = 1, B̄ = 0`, the same transparency that the internal
11//!   chunk padding uses). With right padding, no real row reads a padded one,
12//!   so nothing else has to know.
13//! - **Every cache field that holds the last few samples of a slot** (a conv
14//!   window, a tap FIFO, the carry of a scan) is read at the own end of each
15//!   slot: a gather at a per-slot offset ([`window`]), with a fixed shape for
16//!   any lengths.
17
18use burn::prelude::*;
19
20/// Real rows per slot, `[batch]`.
21pub(crate) fn real_len_b(pad_bs: &Tensor<2, Bool>) -> Tensor<1, Int> {
22    let [batch, _] = pad_bs.dims();
23    pad_bs.clone().bool_not().int().sum_dim(1).reshape([batch])
24}
25
26/// `width` consecutive positions of `x` along `axis`, from `start_b` in each
27/// slot (axis 0 is the batch): the per-slot `narrow(axis, start, width)`, as a
28/// fixed-shape gather.
29pub(crate) fn window<const D: usize>(
30    x: Tensor<D>,
31    axis: usize,
32    start_b: Tensor<1, Int>,
33    width: usize,
34) -> Tensor<D> {
35    let [batch] = start_b.dims();
36    let device = x.device();
37    let idx_bw = Tensor::<1, Int>::arange(0..width as i64, &device)
38        .reshape([1, width])
39        .expand([batch, width])
40        .add(start_b.reshape([batch, 1]).expand([batch, width]));
41    let mut shape = [1usize; D];
42    shape[0] = batch;
43    shape[axis] = width;
44    let mut dims = x.dims();
45    dims[axis] = width;
46    x.gather(axis, idx_bw.reshape(shape).expand(dims))
47}

49/// `pad_bt` with every row repeated `u` times in place: the padding of a token
50/// spread over its `u` folded micro-steps (`[batch, tokens]` →
51/// `[batch, tokens·u]`).
52pub(crate) fn repeat_rows(pad_bt: Tensor<2, Bool>, u: usize) -> Tensor<2, Bool> {
53    let [batch, tokens] = pad_bt.dims();
54    pad_bt
55        .unsqueeze_dim::<3>(2)
56        .expand([batch, tokens, u])
57        .reshape([batch, tokens * u])
58}

60/// `x` with the entries of every padded row replaced by `value`. `x` is
61/// `[batch, sequence, …]` and `pad_bs` is `[batch, sequence]`.
62pub(crate) fn fill_padded<const D: usize>(x: Tensor<D>, pad_bs: &Tensor<2, Bool>, value: f32) -> Tensor<D> {
63    let [batch, sequence] = pad_bs.dims();
64    let mut shape = [1usize; D];
65    shape[0] = batch;
66    shape[1] = sequence;
67    let dims = x.dims();
68    x.mask_fill(pad_bs.clone().reshape(shape).expand(dims), value)
69}
