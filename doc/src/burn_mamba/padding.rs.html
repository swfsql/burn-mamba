1//! Right padding inside a block: the part of the
2//! [`burn_stack::utils::padding`] contract that a family keeps.
3//!
4//! A padded position is absent: the outputs of the real tokens and the
5//! returned cache are those of the real tokens of each row, run alone. Every
6//! family meets this contract in the same two ways:
7//!
8//! - **The state** goes through a padded position unchanged. The family zeroes
9//!   the decay and the writes of that micro-step where the discretisation forms
10//!   them (`Δ = 0` gives `Ā = 1, B̄ = 0`, the same transparency that the
11//!   internal chunk padding uses). With right padding, no real token reads a
12//!   padded one, so nothing else has to know.
13//! - **Every cache field that holds the last few samples of a row** (a conv
14//!   window, a tap FIFO, the carry of a scan) is read at the row end of each
15//!   row ([`crate::row_end`]).
16

19/// `pad_bt` with every position repeated `u` times in place: the padding of a
20/// token spread over its `u` micro-steps on the write axis (`[batch, tokens]`
21/// → `[batch, tokens·u]`).
22#[cfg_attr(not(feature = "mamba3"), allow(dead_code))]
23pub(crate) fn repeat_positions(pad_bt: Tensor<2, Bool>, u: usize) -> Tensor<2, Bool> {
24    let [batch, tokens] = pad_bt.dims();
25    pad_bt
26        .unsqueeze_dim::<3>(2)
27        .expand([batch, tokens, u])
28        .reshape([batch, tokens * u])
29}

31/// `x` with the entries of every padded position replaced by `value`. `x` is
32/// `[batch, sequence, …]` and `pad_bs` is `[batch, sequence]`.
33pub(crate) fn fill_padded<const D: usize>(x: Tensor<D>, pad_bs: &Tensor<2, Bool>, value: f32) -> Tensor<D> {
34    let [batch, sequence] = pad_bs.dims();
35    let mut shape = [1usize; D];
36    shape[0] = batch;
37    shape[1] = sequence;
38    let dims = x.dims();
39    x.mask_fill(pad_bs.clone().reshape(shape).expand(dims), value)
40}
