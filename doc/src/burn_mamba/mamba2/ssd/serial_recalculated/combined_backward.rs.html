1//! # Recompute-based gradient math for the Mamba-2 SSD
2//!
3//! The analytic backward of the `SerialRecalculated` path. The
4//! [standard SSD scan](crate::standard_ssd) holds the gradient of the scan.
5//! This file holds what Mamba-2 adds to it. It forms the inputs `v = Δ·x` and
6//! `da = Δ·A`, and it calls the backward of the scan. Then it applies the chain
7//! rule back to its own inputs, with the backward of the `D` skip:
8//!
9//! ```text
10//! d_x = d_v·Δ + d_y·D
11//! d_Δ = Σ_p d_v·x + d_da·A
12//! d_A = Σ_{b,n,l} d_da·Δ
13//! d_D = Σ_{b,n,l,p} d_y·x
14//! ```
15//!
16//! The scan returns `d_b`, `d_c`, `d_initial_state` and `d_origin` directly.
17//! The length-1 `m` axis of the scan exists only at the call. ADR-0010 records
18//! why this path does not follow the reference line by line.
19//!
20//! All the math is on backend **primitives**, through the rank-tagged `F`
21//! wrapper. The [recompute node](burn_stack::utils::recompute_node) runs with a
22//! generic backend `B`, where the high-level `Tensor` (pinned to the global
23//! `Dispatch` backend) is not available.
24
25use super::serial_recalculated::scan_inputs;
26use crate::standard_ssd;
27use burn_stack::utils::fprim::{F, san};
28use burn::backend::Backend;
29
30/// Per-input gradients produced by [`combined_backward`] (one field per
31/// differentiable forward input).
32#[non_exhaustive]
33pub struct CombinedGrads<B: Backend> {
34    /// Gradient of the input `x`.
35    pub d_x_bnlhp: F<B, 5>,
36    /// Gradient of the step size `Δ` (`dt`).
37    pub d_dt_discretized_bhnl: F<B, 4>,
38    /// Gradient of the input projection `B`.
39    pub d_b_bnlhr: F<B, 5>,
40    /// Gradient of the output projection `C`.
41    pub d_c_bnlhr: F<B, 5>,
42    /// Gradient of the per-head skip term `D`.
43    pub d_d_h: F<B, 1>,
44    /// Gradient of the initial state.
45    pub d_initial_state_bhpr: F<B, 4>,
46    /// Gradient of the origin state of the resets (`None` without resets).
47    pub d_origin_bhpr: Option<F<B, 4>>,
48    /// Gradient of the per-head decay rate `A` (as `a_decay_h`).
49    pub d_a_decay_h: F<B, 1>,
50}
51
52/// Memory-efficient backward for the Mamba-2 chunkwise SSD.
53///
54/// # Arguments
55/// - `d_y_bnlhp` — upstream gradient of the SSD output
56/// - `d_final_bhpr` — upstream gradient of the final state
57/// - `x_bnlhp`, `dt_discretized_bhnl`, `b_bnlhr`, `c_bnlhr`, `d_h`,
58///   `initial_state_bhpr`, `a_decay_h` — the seven saved forward inputs
59/// - `restarts` — `(keep_bn, origin_bhpr)`, or `None`. `keep_bn` is the
60///   constant reset factor of the forward (`0` at a chunk that starts a new
61///   segment). `origin_bhpr` is the state that those chunks start from.
62///
63/// # Returns
64/// One [`CombinedGrads`] struct containing gradients for all 7 inputs, and
65/// for the origin state.
66#[allow(clippy::too_many_arguments)]
67pub fn combined_backward<B: Backend>(
68    d_y_bnlhp: F<B, 5>,
69    d_final_bhpr: F<B, 4>,
70    //
71    x_bnlhp: F<B, 5>,
72    dt_discretized_bhnl: F<B, 4>,
73    b_bnlhr: F<B, 5>,
74    c_bnlhr: F<B, 5>,
75    d_h: F<B, 1>,
76    initial_state_bhpr: F<B, 4>,
77    a_decay_h: F<B, 1>,
78    restarts: Option<(F<B, 2>, F<B, 4>)>,
79) -> CombinedGrads<B> {
80    let [batch, nchunks, chunk_len, nheads, per_head_dim] = x_bnlhp.dims();
81
82    san(&x_bnlhp);
83    san(&dt_discretized_bhnl);
84    san(&d_h);
85    san(&a_decay_h);
86
87    let dt_bnlh = dt_discretized_bhnl.permute([0, 2, 3, 1]);
88    let a_decay_111h = a_decay_h.unsqueeze_dims::<4>(&[0, 1, 2]);
89    let (v_bnlhp, da_bnlh) = scan_inputs(x_bnlhp.clone(), dt_bnlh.clone(), a_decay_111h.clone());
90
91    let grads = standard_ssd::combined_backward::<B>(
92        d_y_bnlhp.clone().unsqueeze_dim::<6>(3),
93        d_final_bhpr,
94        v_bnlhp.unsqueeze_dim::<6>(3),
95        da_bnlh,
96        b_bnlhr.unsqueeze_dim::<6>(3),
97        c_bnlhr.unsqueeze_dim::<6>(3),
98        initial_state_bhpr,
99        1,
100        restarts,
101    );
102    let d_v_bnlhp = grads.d_v_bnlmhp.squeeze_dim::<5>(3);
103    let d_da_bnlh = grads.d_da_bnlh;
104
105    // ── SKIP backward: y += D · x ─────────────────────────────────────────
106    let d_d_h = (d_y_bnlhp.clone() * x_bnlhp.clone())
107        .permute([3, 0, 1, 2, 4]) // _hbnlp
108        .reshape([nheads, batch * nchunks * chunk_len * per_head_dim]) // _hBNLP
109        .sum_dim(1) // _h1
110        .squeeze_dim::<1>(1); // _h
111    san(&d_d_h);
112    let d_x_skip_bnlhp = d_y_bnlhp * d_h.unsqueeze_dims::<5>(&[0, 1, 2, 4]);
113
114    // ── The chain rule ────────────────────────────────────────────────────
115    let d_x_bnlhp = d_v_bnlhp.clone() * dt_bnlh.clone().unsqueeze_dim::<5>(4) + d_x_skip_bnlhp;
116    san(&d_x_bnlhp);
117
118    let d_dt_discretized_bhnl = ((d_v_bnlhp * x_bnlhp).sum_dim(4).squeeze_dim::<4>(4)
119        + d_da_bnlh.clone() * a_decay_111h)
120        .permute([0, 3, 1, 2]);
121    san(&d_dt_discretized_bhnl);
122
123    let d_a_decay_h = (d_da_bnlh * dt_bnlh)
124        .permute([3, 0, 1, 2]) // _hbnl
125        .reshape([nheads, batch * nchunks * chunk_len]) // _hBNL
126        .sum_dim(1) // _h1
127        .squeeze_dim::<1>(1); // _h
128    san(&d_a_decay_h);
129
130    CombinedGrads {
131        d_x_bnlhp,
132        d_dt_discretized_bhnl,
133        d_b_bnlhr: grads.d_b_bnlmhr.squeeze_dim::<5>(3),
134        d_c_bnlhr: grads.d_c_bntmhr.squeeze_dim::<5>(3),
135        d_d_h,
136        d_initial_state_bhpr: grads.d_initial_state_bhpr,
137        d_origin_bhpr: grads.d_origin_bhpr,
138        d_a_decay_h,
139    }
140}
