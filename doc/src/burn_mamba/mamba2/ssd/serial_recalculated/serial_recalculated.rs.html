1//! # Serial SSD with a custom, memory-efficient backward (Mamba-2)
2//!
3//! This is the `SerialRecalculated` path. The forward is the same five-kernel
4//! serial scan as [`super::super::serial`], but it goes through the
5//! [`Mamba2BackendExt`] trait. Thus an `Autodiff` backend can use a **custom
6//! backward** that recomputes the per-chunk intermediates instead of storing
7//! them (see [`super::backward`] / [`super::combined_backward`]). This costs a
8//! little extra compute and saves ~⅓ of the training memory.
9//!
10//! Every plain (non-autodiff) backend uses the default body of the trait,
11//! which replays the [`super::super::serial`] kernels K1–K5. The
12//! [`burn_stack::impl_backend_ext_for_burn_backends!`] /
13//! [`burn_stack::decl_autodiff_backend_ext!`] macros make the per-backend
14//! impls and the autodiff marker trait.
15

23impl Mamba2SsdInput {
24    /// Forward pass for the Mamba-2 SSD module (recompute-backward path).
25    ///
26    /// Returns:
27    /// - `y_bnlhp`.
28    /// - `final_state_bhpr`.
29    #[allow(non_snake_case)]
30    pub fn ssd_serial_recalculated(self) -> (Tensor<5>, Tensor<4>) {
31        let input = self;
32        // This must be a backend-dependent method. On a plain backend, it
33        // replays `Mamba2SsdInput::ssd_serial`. On an autodiff backend, it
34        // calls the custom backward.
35
36        let [batch, nchunks, chunk_len, nheads, _per_head_dim] = input.x_bnlhp.dims();
37        assert!(nchunks > 0, "sequence length must be at least 1");
38
39        assert!(
40            input.init_state_hpr.is_none(),
41            "init_state_hpr not yet implemented"
42        );
43
44        // ── Permutes ──────────────────────────────────────────────────────────────────
45        // The reference computes Δ in Kernel 1. Here `Mamba2::forward` computes
46        // it (step 4), before the chunk padding.
47        let dt_discretized_bhnl = input.dt_bnlh.permute([0, 3, 1, 2]);
48        assert_eq!(
49            [batch, nheads, nchunks, chunk_len],
50            dt_discretized_bhnl.dims()
51        );
52
53        // The resets go into the op as a keep factor and the origin state (see
54        // `Restarts::op_inputs`).
55        let (keep_bn, origin_bhpr, resets) =
56            crate::packing::Restarts::op_inputs(input.restarts, &input.x_bnlhp);
57
58        // The custom op computes K1 itself (in forward and backward). It gets
59        // `a_decay_h` directly, so `da_cumsum` is not an autodiff-tracked
60        // intermediate that crosses the op boundary.
61        let (y_bnlhp, final_state_bhpr) = <Dispatch as Mamba2BackendExt>::ssd_serial_recalculated(
62            input.x_bnlhp.into_dispatch(),
63            dt_discretized_bhnl.into_dispatch(),
64            input.b_bnlhr.into_dispatch(),
65            input.c_bnlhr.into_dispatch(),
66            input.d_h.into_dispatch(),
67            input.initial_state_bhpr.into_dispatch(),
68            input.a_decay_h.into_dispatch(),
69            keep_bn.into_dispatch(),
70            origin_bhpr.into_dispatch(),
71            resets,
72        );
73        let y_bnlhp = Tensor::from_dispatch(y_bnlhp);
74        let final_state_bhpr = Tensor::from_dispatch(final_state_bhpr);
75        (y_bnlhp, final_state_bhpr)
76    }
77}

79/// Backend extension for the Mamba-2 SSD with a recompute backward.
80///
81/// The default body runs the forward on backend primitives. The `Autodiff<B>`
82/// impl (in [`super::backward`]) adds the custom backward node.
83#[backend_extension(
84    // Every cubecl runtime (CUDA, ROCm, Metal, Vulkan, WebGPU, wgpu, CPU) is
85    // this one backend. The device of a tensor tells which runtime it uses.
86    // The cfg mirrors burn's own `cube_backend`.
87    Cube: cfg(any(
88        feature = "backend-cpu",
89        feature = "backend-cuda",
90        feature = "backend-rocm",
91        feature = "backend-metal",
92        feature = "backend-vulkan",
93        feature = "backend-wgpu",
94        feature = "backend-webgpu"
95    )),
96    Flex:  cfg(feature = "backend-flex"),
97    NdArray:  cfg(feature = "backend-ndarray"),
98    LibTorch:  cfg(any(feature = "backend-tch-cpu", feature = "backend-tch-gpu")),
99    Autodiff:  cfg(feature = "autodiff"),
100)]
101pub trait Mamba2BackendExt: Backend {
102    /// `keep_bn` (`[batch, nchunks]`) is `0` at a chunk that starts a new
103    /// segment of a packed row, and `1` elsewhere. It is a constant: the op
104    /// registers no gradient for it. Such a chunk starts from `origin_bhpr`
105    /// (`[batch, nheads, per_head_dim, state_rank]`). The op reads both only
106    /// when `resets` is `true`.
107    ///
108    /// Returns:
109    /// - `y_bnlhp`.
110    /// - `final_state_bhpr`.
111    #[allow(clippy::too_many_arguments)]
112    fn ssd_serial_recalculated(
113        x_bnlhp: FloatTensor<Self>,
114        dt_discretized_bhnl: FloatTensor<Self>,
115        b_bnlhr: FloatTensor<Self>,
116        c_bnlhr: FloatTensor<Self>,
117        d_h: FloatTensor<Self>,
118        initial_state_bhpr: FloatTensor<Self>,
119        a_decay_h: FloatTensor<Self>,
120        keep_bn: FloatTensor<Self>,
121        origin_bhpr: FloatTensor<Self>,
122        resets: bool,
123    ) -> (FloatTensor<Self>, FloatTensor<Self>) {
124        // The default impl replicates Mamba2SsdInput::ssd_serial on backend
125        // primitives. This body runs under a generic `B`, where the high-level
126        // `Tensor` (pinned to `Dispatch`) is not available. So the math uses
127        // the rank-tagged [`F`] primitive wrapper and the primitive K-kernels
128        // below.
129        let x_bnlhp = F::<Self, 5>::new(x_bnlhp);
130        let dt_discretized_bhnl = F::<Self, 4>::new(dt_discretized_bhnl);
131        let b_bnlhr = F::<Self, 5>::new(b_bnlhr);
132        let c_bnlhr = F::<Self, 5>::new(c_bnlhr);
133        let d_h = F::<Self, 1>::new(d_h);
134        let initial_state_bhpr = F::<Self, 4>::new(initial_state_bhpr);
135        let a_decay_h = F::<Self, 1>::new(a_decay_h);
136        let restarts = resets.then(|| (F::<Self, 2>::new(keep_bn), F::<Self, 4>::new(origin_bhpr)));
137
138        let nchunks = x_bnlhp.dims()[1];
139        assert!(nchunks > 0, "sequence length must be at least 1");
140
141        // ── Kernel 1 ──────────────────────────────────────────────────────────
142        let (da_cumsum_bhnl, da_chunk_end_bhn) =
143            k1_ssd_chunk_cumsum::<Self>(dt_discretized_bhnl.clone(), a_decay_h);
144        san(&da_cumsum_bhnl);
145
146        // ── Kernel 2 ──────────────────────────────────────────────────────────
147        let cb_bnhll = k2_ssd_bmm::<Self>(c_bnlhr.clone(), b_bnlhr.clone());
148        san(&cb_bnhll);
149
150        // ── Kernel 3 ──────────────────────────────────────────────────────────
151        let intra_chunk_state_bnhpr = k3_ssd_chunk_state::<Self>(
152            x_bnlhp.clone(),
153            b_bnlhr,
154            da_cumsum_bhnl.clone(),
155            dt_discretized_bhnl.clone(),
156        );
157        san(&intra_chunk_state_bnhpr);
158
159        // ── Kernel 4 ──────────────────────────────────────────────────────────
160        let (chunk_input_state_bnhpr, final_state_bhpr) = k4_ssd_state_passing::<Self>(
161            intra_chunk_state_bnhpr,
162            da_chunk_end_bhn,
163            initial_state_bhpr,
164            restarts,
165        );
166        san(&chunk_input_state_bnhpr);
167        san(&final_state_bhpr);
168
169        // ── Kernel 5 ──────────────────────────────────────────────────────────
170        let y_bnlhp = k5_ssd_chunk_scan::<Self>(
171            da_cumsum_bhnl,
172            dt_discretized_bhnl,
173            x_bnlhp,
174            c_bnlhr,
175            cb_bnhll,
176            chunk_input_state_bnhpr,
177            d_h,
178        );
179        san(&y_bnlhp);
180
181        (y_bnlhp.inner(), final_state_bhpr.inner())
182    }
183}

185// ─── Primitive forward kernels (K1–K5) ───────────────────────────────────────
186// Primitive ports of the high-level [`crate::mamba2::ssd::serial`] kernels, on
187// the primitives of `B` through [`F`], so the default trait body can run under
188// a generic backend. The recompute backward in [`super::combined_backward`]
189// uses K1/K2/K4 again. K5 is forward-only: the backward computes the gradient
190// of K5 analytically instead of recomputing it.
191
192/// Primitive port of [`crate::mamba2::ssd::serial::k1_ssd_chunk_cumsum`].
193///
194/// Returns the intra-chunk cumsum `da_cumsum_bhnl` and the per-chunk last value
195/// `da_chunk_end_bhn`.
196pub(crate) fn k1_ssd_chunk_cumsum<B: Backend>(
197    dt_discretized_bhnl: F<B, 4>,
198    a_decay_h: F<B, 1>,
199) -> (F<B, 4>, F<B, 3>) {
200    let [batch, nheads, nchunks, chunk_len] = dt_discretized_bhnl.dims();
201    let da_cumsum_bhnl: F<B, 4> = {
202        let a_decay_bhnl = a_decay_h
203            .unsqueeze_dims::<4>(&[0, 2, 3])
204            .expand([batch, nheads, nchunks, chunk_len]);
205        (dt_discretized_bhnl * a_decay_bhnl).cumsum(3)
206    };
207    let da_chunk_end_bhn = da_cumsum_bhnl
208        .clone()
209        .slice(s![.., .., .., -1])
210        .squeeze_dim::<3>(3);
211    (da_cumsum_bhnl, da_chunk_end_bhn)
212}

214/// Primitive port of [`crate::mamba2::ssd::serial::k2_ssd_bmm`].
215///
216/// Returns the intra-chunk `C·Bᵀ` block matrix `cb_bnhll`.
217pub(crate) fn k2_ssd_bmm<B: Backend>(c_bnlhr: F<B, 5>, b_bnlhr: F<B, 5>) -> F<B, 5> {
218    let c_bnhlr = c_bnlhr.swap_dims(2, 3);
219    let b_bnhrl = b_bnlhr.permute([0, 1, 3, 4, 2]);
220    c_bnhlr.matmul(b_bnhrl)
221}

223/// Primitive port of [`crate::mamba2::ssd::serial::k3_ssd_chunk_state`] (lean:
224/// returns only the chunk-end state).
225///
226/// Returns `intra_chunk_state_bnhpr`: the contribution of each chunk to its
227/// end state, from a zero state at the chunk start.
228pub(crate) fn k3_ssd_chunk_state<B: Backend>(
229    x_bnlhp: F<B, 5>,
230    b_bnlhr: F<B, 5>,
231    da_cumsum_bhnl: F<B, 4>,
232    dt_discretized_bhnl: F<B, 4>,
233) -> F<B, 5> {
234    let [batch, nchunks, chunk_len, nheads, per_head_dim] = x_bnlhp.dims();
235    let [.., state_rank] = b_bnlhr.dims();
236
237    let x_bnhpl = x_bnlhp.permute([0, 1, 3, 4, 2]);
238    let b_bnhlr = b_bnlhr.swap_dims(2, 3);
239
240    // K3 scaling factor: dt · exp(cumA_last − cumA)
241    let da_cumsum_last_bhn1 = da_cumsum_bhnl.clone().slice(s![.., .., .., -1]);
242    let forward_decay_to_chunk_end_bhnl =
243        (da_cumsum_last_bhn1.expand([batch, nheads, nchunks, chunk_len]) - da_cumsum_bhnl).exp();
244    let b_bar_scale_bhnl = forward_decay_to_chunk_end_bhnl * dt_discretized_bhnl;
245
246    let b_bar_scale_bnhlr = b_bar_scale_bhnl
247        .swap_dims(1, 2) // b_bar_scale_bnhl
248        .unsqueeze_dim::<5>(4) // b_bar_scale_bnhl1
249        .expand([batch, nchunks, nheads, chunk_len, state_rank]);
250    let b_scaled_bnhlr = b_bnhlr * b_bar_scale_bnhlr;
251
252    let intra_chunk_state_bnhpr = x_bnhpl.matmul(b_scaled_bnhlr);
253    assert_eq!(
254        [batch, nchunks, nheads, per_head_dim, state_rank],
255        intra_chunk_state_bnhpr.dims()
256    );
257    intra_chunk_state_bnhpr
258}

260/// Primitive port of [`crate::mamba2::ssd::serial::k4_ssd_state_passing`].
261///
262/// Returns the per-chunk input-state stream `chunk_input_state_bnhpr` and the
263/// `final_state_bhpr`.
264///
265/// `restarts` is `(keep_bn, origin_bhpr)`: the reset of the high-level K4 as
266/// a factor, and the origin state. The state that goes into a chunk with
267/// `keep = 0` is the origin state.
268pub(crate) fn k4_ssd_state_passing<B: Backend>(
269    intra_chunk_state_bnhpr: F<B, 5>,
270    da_chunk_end_bhn: F<B, 3>,
271    initial_state_bhpr: F<B, 4>,
272    restarts: Option<(F<B, 2>, F<B, 4>)>,
273) -> (F<B, 5>, F<B, 4>) {
274    let [batch, nchunks, nheads, per_head_dim, state_rank] = intra_chunk_state_bnhpr.dims();
275
276    let mut running_state_bhpr = initial_state_bhpr.clone();
277    let mut chunk_input_state_vec_bhpr = Vec::with_capacity(nchunks + 1);
278    chunk_input_state_vec_bhpr.push(running_state_bhpr.clone());
279    let at = |t_bn: &F<B, 2>, i_chunk: usize| {
280        t_bn.clone()
281            .slice(s![.., i_chunk])
282            .reshape([batch, 1, 1, 1])
283            .expand([batch, nheads, per_head_dim, state_rank])
284    };
285    let restarts = restarts.map(|(keep_bn, origin_bhpr)| {
286        let restart_bn = restart_factor(&keep_bn);
287        (keep_bn, restart_bn, origin_bhpr)
288    });
289
290    for i_chunk in 0..nchunks {
291        // A chunk that starts a segment starts from the origin state. (A
292        // learnable `init_state_hpr` would join it, but only `Minimal` takes
293        // one.)
294        if let Some((keep_bn, restart_bn, origin_bhpr)) = &restarts {
295            running_state_bhpr = running_state_bhpr * at(keep_bn, i_chunk)
296                + origin_bhpr.clone() * at(restart_bn, i_chunk);
297            *chunk_input_state_vec_bhpr.last_mut().unwrap() = running_state_bhpr.clone();
298        }
299
300        let intra_state_bhpr = intra_chunk_state_bnhpr
301            .clone()
302            .slice(s![.., i_chunk, .., .., ..])
303            .squeeze_dim::<4>(1);
304        let decay_bhpr = da_chunk_end_bhn
305            .clone()
306            .slice(s![.., .., i_chunk])
307            .exp()
308            .unsqueeze_dim::<4>(3)
309            .expand([batch, nheads, per_head_dim, state_rank]);
310        running_state_bhpr = (decay_bhpr * running_state_bhpr) + intra_state_bhpr;
311        chunk_input_state_vec_bhpr.push(running_state_bhpr.clone());
312    }
313
314    let final_state_bhpr = chunk_input_state_vec_bhpr.pop().unwrap();
315    let chunk_input_state_bnhpr = F::stack(chunk_input_state_vec_bhpr, 1);
316    (chunk_input_state_bnhpr, final_state_bhpr)
317}

319/// `1 − keep`: `1` at a chunk that restarts from the initial state.
320pub(crate) fn restart_factor<B: Backend>(keep_bn: &F<B, 2>) -> F<B, 2> {
321    F::<B, 2>::full(keep_bn.dims(), 1.0, &keep_bn.device(), keep_bn.dtype()) - keep_bn.clone()
322}

324/// Primitive port of [`crate::mamba2::ssd::serial::k5_ssd_chunk_scan`].
325///
326/// Combines the intra-chunk (ORANGE, attention-like) and inter-chunk (BLUE,
327/// state-carried) contributions plus the `D` skip term into the output
328/// `y_bnlhp`. Forward-only.
329#[allow(clippy::too_many_arguments)]
330fn k5_ssd_chunk_scan<B: Backend>(
331    da_cumsum_bhnl: F<B, 4>,
332    dt_discretized_bhnl: F<B, 4>,
333    x_bnlhp: F<B, 5>,
334    c_bnlhr: F<B, 5>,
335    cb_bnhll: F<B, 5>,
336    chunk_input_state_bnhpr: F<B, 5>,
337    d_h: F<B, 1>,
338) -> F<B, 5> {
339    let [batch, nchunks, chunk_len, nheads, per_head_dim] = x_bnlhp.dims();
340    let device = x_bnlhp.device();
341
342    let da_cumsum_bnhl = da_cumsum_bhnl.swap_dims(1, 2);
343    let dt_bnhl = dt_discretized_bhnl.swap_dims(1, 2);
344    let x_bnhlp = x_bnlhp.clone().swap_dims(2, 3);
345    let c_bnhlr = c_bnlhr.swap_dims(2, 3);
346
347    // ── BLUE: exp(dA[l]) · C[l,:] @ state_in^T ─────────────────────────────
348    let exp_da_cumsum_bnhlp = da_cumsum_bnhl
349        .clone()
350        .exp()
351        .unsqueeze_dim::<5>(4) // exp_da_cumsum_bnhl1
352        .expand([batch, nchunks, nheads, chunk_len, per_head_dim]);
353    let chunk_input_state_bnhrp = chunk_input_state_bnhpr.transpose();
354    let blue_scaled_bnhlp = c_bnhlr.matmul(chunk_input_state_bnhrp) * exp_da_cumsum_bnhlp;
355    san(&blue_scaled_bnhlp);
356
357    // ── ORANGE: causal CB_weighted @ X ──────────────────────────────────────
358    let da_cumsum_target_bnhll = da_cumsum_bnhl
359        .clone()
360        .unsqueeze_dim::<5>(4) // da_cumsum_bnhl1
361        .expand([batch, nchunks, nheads, chunk_len, chunk_len]);
362    let da_cumsum_source_bnhll = da_cumsum_bnhl
363        .unsqueeze_dim::<5>(3) // da_cumsum_bnh1l
364        .expand([batch, nchunks, nheads, chunk_len, chunk_len]);
365    let da_cumsum_diff_bnhll = da_cumsum_target_bnhll - da_cumsum_source_bnhll;
366
367    // Causal mask + exp stabiliser: strictly-above-diagonal set to −∞.
368    let causal_mask_bnhll: Mask<B> = Mask::tril_mask(chunk_len, chunk_len, 0, &device)
369        .reshape([1, 1, 1, chunk_len, chunk_len])
370        .expand([batch, nchunks, nheads, chunk_len, chunk_len]);
371    let da_cumsum_diff_exp_bnhll = da_cumsum_diff_bnhll
372        .mask_fill(causal_mask_bnhll, f32::NEG_INFINITY)
373        .exp();
374    san(&da_cumsum_diff_exp_bnhll);
375
376    let dt_source_bnhll = dt_bnhl
377        .unsqueeze_dim::<5>(3) // dt_bnh1l
378        .expand([batch, nchunks, nheads, chunk_len, chunk_len]);
379    let orange_lhs_bnhll = cb_bnhll * da_cumsum_diff_exp_bnhll * dt_source_bnhll;
380    let orange_bnhlp = orange_lhs_bnhll.matmul(x_bnhlp);
381    san(&orange_bnhlp);
382
383    // ── SKIP: D[h] · x[l,p] ─────────────────────────────────────────────────
384    let skip_bnlhp = d_h
385        .unsqueeze_dims::<5>(&[0, 1, 2, 4]) // d_111h1
386        .expand([batch, nchunks, chunk_len, nheads, per_head_dim])
387        * x_bnlhp;
388
389    let y_partial_bnhlp = blue_scaled_bnhlp + orange_bnhlp;
390    let y_partial_bnlhp = y_partial_bnhlp.swap_dims(2, 3);
391    let y_bnlhp = y_partial_bnlhp + skip_bnlhp;
392    assert_eq!(
393        [batch, nchunks, chunk_len, nheads, per_head_dim],
394        y_bnlhp.dims()
395    );
396    y_bnlhp
397}
