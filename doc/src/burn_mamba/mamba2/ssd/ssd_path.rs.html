1//! # SSD path selection and input bundle (Mamba-2)
2//!
3//! [`Mamba2SsdPath`] selects which of the three exact SSD reformulations
4//! ([`minimal`] / [`serial`] / [`serial_recalculated`]) runs, and at what
5//! chunk length. [`Mamba2SsdInput`] bundles the pre-processed tensors
6//! that the scan uses (B/C already GQA-expanded to per-head).
7//! [`Mamba2SsdInput::run`] runs the SSD path that the selector names.
8
9use crate::mamba2::prelude::*;
10use burn::prelude::*;
11
12/// The SSD path of the Mamba-2 chunkwise SSD.
13///
14/// Each variant carries an optional chunk length. A larger value gives more
15/// intra-chunk GEMM work and a shorter inter-chunk scan. The optimal value is
16/// approximately `√(state_rank · per_head_dim)` (see
17/// [`Self::optimal_chunk_len`]). `None` uses that optimal value.
18#[derive(Debug, Clone)]
19pub enum Mamba2SsdPath {
20    /// Minimal SSD: mostly batched matmuls, with the autodiff backward.
21    ///
22    /// See [`Mamba2SsdInput::ssd_minimal`]. For training, prefer
23    /// [`Self::SerialRecalculated`]. This is the only path that supports
24    /// [`Mamba2SsdInput::init_state_hpr`].
25    ///
26    /// Based on `/mamba_ssm/modules/ssd_minimal.py` in the `state-spaces/mamba`
27    /// GitHub reference.
28    Minimal(Option<usize>),
29
30    /// (Hybrid) serial SSD: a serial loop over the chunks plus batched
31    /// matmuls, with the autodiff backward.
32    ///
33    /// See [`Mamba2SsdInput::ssd_serial`]. For a memory-saving custom backward,
34    /// see [`Self::SerialRecalculated`].
35    ///
36    /// Based on 5 kernels under `/mamba_ssm/ops/triton/` in the
37    /// `state-spaces/mamba` GitHub reference:
38    /// - `ssd_chunk_state.py` (K1, K3).
39    /// - `ssd_bmm.py` (K2).
40    /// - `ssd_state_passing.py` (K4).
41    /// - `ssd_chunk_scan.py` (K5).
42    Serial(Option<usize>),
43
44    /// (Hybrid) serial SSD with a custom, memory-efficient backward. The
45    /// backward recomputes the forward intermediates instead of storing them.
46    ///
47    /// See [`Mamba2SsdInput::ssd_serial_recalculated`]. For a plain autodiff
48    /// backward, see [`Self::Serial`].
49    ///
50    /// Based on the combined kernel `/mamba_ssm/ops/triton/ssd_combined.py` in
51    /// the `state-spaces/mamba` GitHub reference.
52    SerialRecalculated(Option<usize>),
53}
54
55/// SSD input.
56///
57/// All tensors are pre-processed: B/C are already GQA-expanded to per-head.
58/// `a_decay_h` is `A < 0`, not yet multiplied by `Δ`.
59pub struct Mamba2SsdInput {
60    /// # Shape
61    /// - `[batch, nchunks, chunk_len, nheads, per_head_dim]`
62    pub x_bnlhp: Tensor<5>,
63    /// # Shape
64    /// - `[batch, nchunks, chunk_len, nheads]`
65    pub dt_bnlh: Tensor<4>,
66    /// # Shape
67    /// - `[nheads]`
68    pub a_decay_h: Tensor<1>,
69    /// B tensor, expanded to per-head.
70    ///
71    /// # Shape
72    /// - `[batch, nchunks, chunk_len, nheads, state_rank]`
73    pub b_bnlhr: Tensor<5>,
74    /// C tensor, expanded to per-head.
75    ///
76    /// # Shape
77    /// - `[batch, nchunks, chunk_len, nheads, state_rank]`
78    pub c_bnlhr: Tensor<5>,
79    /// # Shape
80    /// - `[nheads]`
81    pub d_h: Tensor<1>,
82    /// The incoming state (from the cache).
83    ///
84    /// # Shape
85    /// - `[batch, nheads, per_head_dim, state_rank]`
86    pub initial_state_bhpr: Tensor<4>,
87    /// The learnable initial state, added to `initial_state_bhpr`. Only
88    /// [`Mamba2SsdPath::Minimal`] supports it. The two serial paths panic when
89    /// it is `Some`.
90    ///
91    /// # Shape
92    /// - `[nheads, per_head_dim, state_rank]`
93    pub init_state_hpr: Option<Tensor<3>>,
94    /// The chunks that start a new segment of a packed row, and the origin
95    /// state. The carry into each of these chunks restarts from the origin
96    /// (see [`crate::packing`]). The call itself starts from
97    /// `initial_state_bhpr`.
98    /// `None` ⇒ no reset. Only the two serial paths take it.
99    /// [`Mamba2SsdPath::Minimal`] panics when it is `Some`.
100    pub restarts: Option<crate::packing::Restarts>,
101}
102
103impl Mamba2SsdInput {
104    /// Run the [`NaN`/`Inf` guards](burn_stack::modules::misc::sanity) on every input tensor.
105    pub fn sanity(&self) {
106        use burn_stack::modules::sanity as san;
107        san(&self.x_bnlhp);
108        san(&self.dt_bnlh);
109        san(&self.a_decay_h);
110        san(&self.b_bnlhr);
111        san(&self.c_bnlhr);
112        san(&self.d_h);
113        san(&self.initial_state_bhpr);
114        if let Some(ref init_state_hpr) = self.init_state_hpr {
115            san(init_state_hpr);
116        }
117    }
118}
119
120impl Mamba2SsdPath {
121    /// Optimal chunk length, approximately `√(state_rank · per_head_dim)`,
122    /// rounded up to a multiple of 32 and capped at 512.
123    pub fn optimal_chunk_len(state_rank: usize, per_head_dim: usize) -> usize {
124        (state_rank * per_head_dim)
125            .isqrt()
126            .next_multiple_of(32) // rule-of-thumb: common plane dimension.
127            .min(512) // rule-of-thumb: ceiling at 512.
128    }
129
130    /// The chunk length carried by this variant, if any.
131    pub fn chunk_len(&self) -> Option<usize> {
132        match self {
133            Self::Minimal(chunk_len)
134            | Self::Serial(chunk_len)
135            | Self::SerialRecalculated(chunk_len) => *chunk_len,
136        }
137    }
138
139    /// The chunk length carried by this variant, or [`Self::optimal_chunk_len`]
140    /// when unset.
141    pub fn chunk_len_or_optimal(&self, state_rank: usize, per_head_dim: usize) -> usize {
142        self.chunk_len()
143            .unwrap_or_else(|| Self::optimal_chunk_len(state_rank, per_head_dim))
144    }
145
146    /// The recommended default path for a given block: [`Self::SerialRecalculated`]
147    /// with [`Self::optimal_chunk_len`] for the block's dimensions.
148    pub fn default_optimal_from_block(block: &Mamba2) -> Self {
149        let chunk_len = Self::optimal_chunk_len(block.state_rank, block.per_head_dim());
150        Self::SerialRecalculated(Some(chunk_len))
151    }
152}
153
154impl Mamba2SsdInput {
155    /// Run the selected SSD path on this input: `ssd_minimal`,
156    /// `ssd_serial`, or `ssd_serial_recalculated`, by [`Mamba2SsdPath`]
157    /// variant. The chunk length is already in the tensor shapes.
158    ///
159    /// # Returns
160    /// - `y_bnlhp`: `[batch, nchunks, chunk_len, nheads, per_head_dim]`
161    /// - `final_state_bhpr`: `[batch, nheads, per_head_dim, state_rank]`
162    pub fn run(self, path: &Mamba2SsdPath) -> (Tensor<5>, Tensor<4>) {
163        match path {
164            Mamba2SsdPath::Minimal(_) => {
165                assert!(
166                    self.restarts.is_none(),
167                    "Minimal does not take resets: use Serial or SerialRecalculated"
168                );
169                self.ssd_minimal()
170            }
171            Mamba2SsdPath::Serial(_) => self.ssd_serial(),
172            Mamba2SsdPath::SerialRecalculated(_) => self.ssd_serial_recalculated(),
173        }
174    }
175}
176
177impl Default for Mamba2SsdPath {
178    fn default() -> Mamba2SsdPath {
179        // Defaults to the SerialRecalculated SSD path with the optimal chunk length.
180        Mamba2SsdPath::SerialRecalculated(None)
181    }
182}
