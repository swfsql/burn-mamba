1//! # Mamba-2 Inference Caches
2//!
3//! The state that the block keeps between calls. In *training* or *prefill*,
4//! the full sequence is available at once and [`Mamba2::forward`] uses the
5//! chunked SSD algorithm. In *decoding*, [`Mamba2::step`] processes one token
6//! per call with the pure recurrent form:
7//!
8//! ```text
9//!   hₜ = Āₜ hₜ₋₁ + B̄ₜ xₜ        (state update)
10//!   yₜ = Cₜᵀ hₜ + D xₜ            (output)
11//! ```
12//!
13//! Each block keeps two pieces of state:
14//!
15//! 1. **Convolution cache** — the last `conv_kernel` inputs to the depthwise
16//!    Conv1d. With it, each decoding step applies the causal filter without
17//!    processing earlier tokens again.
18//!
19//! 2. **State** — the matrix `hₜ ∈ ℝ^{per_head_dim×state_rank}` per
20//!    head. It compresses the full past into a fixed size, for any number of
21//!    tokens. This is the memory advantage of SSMs over attention: the
22//!    KV-cache of a Transformer grows linearly with the sequence length, but
23//!    the state is always O(per_head_dim·state_rank).
24//!
25//! [`Mamba2::zero_cache`] builds the cache of a fresh sequence.
26

32// ---------------------------------------------------------------------------
33// Mamba2Caches  (one cache entry per layer)
34// ---------------------------------------------------------------------------
35
36/// A collection of per-layer caches for a complete Mamba-2 network.
37///
38/// Every `step` of the family-generic [`burn_stack::modules::Layers`] stack
39/// takes and returns one [`Mamba2Caches`]. Each element of `caches` is the
40/// cache of one (virtual) layer of the network.
41#[derive(Module, Debug)]
42pub struct Mamba2Caches {
43    /// Per-layer caches.
44    ///
45    /// Length: `n_caches`, the number of *virtual* layers. With shared weights
46    /// (a layer schedule), it can be larger than the number of *real* weight
47    /// layers.
48    pub caches: Vec<Mamba2Cache>,
49}
50
51// ---------------------------------------------------------------------------
52// Mamba2Cache  (cache for a single block)
53// ---------------------------------------------------------------------------
54
55/// The cache that a **single** Mamba-2 block carries between calls.
56///
57/// Each call to [`Mamba2::step`] or [`Mamba2::forward`] returns a new cache
58/// with both tensors updated.
59#[derive(Module, Debug)]
60pub struct Mamba2Cache {
61    /// **Convolution rolling window.**
62    ///
63    /// The last `conv_kernel` pre-activation feature vectors that went into
64    /// the depthwise Conv1d. Each step removes the oldest column and appends
65    /// the projection of the new token on the right.
66    ///
67    /// Shape: `[batch, conv_dim, conv_kernel]`
68    ///   - `conv_dim  = d_inner + 2 · ngroups · state_rank`
69    ///   - `conv_kernel` is typically 4
70    pub conv_bvk: Tensor<3>,
71
72    /// **State** `hₜ`.
73    ///
74    /// The O(per_head_dim·state_rank) compressed summary of all earlier
75    /// tokens. Each decoding step applies `hₜ = Āₜ hₜ₋₁ + B̄ₜ xₜ`.
76    ///
77    /// The layout `[…, per_head_dim, state_rank]` is the transpose of the
78    /// paper's `hₜ ∈ ℝ^{state_rank×per_head_dim}`. The content is the same.
79    ///
80    /// Shape: `[batch, nheads, per_head_dim, state_rank]`
81    pub ssm_bhpr: Tensor<4>,
82}
83
84impl Mamba2Cache {
85    /// Run the [`NaN`/`Inf` guards](burn_stack::modules::misc::sanity) on every cached tensor.
86    pub fn sanity(&self) {
87        san(&self.conv_bvk);
88        san(&self.ssm_bhpr);
89    }
90}
91
92impl Mamba2 {
93    /// The cache of a fresh sequence, for `batch` rows. A `forward` or a
94    /// `step` without a cache starts from it, and so does each segment of a
95    /// packed call without an origin. The shapes come from the block.
96    ///
97    /// Zeros are correct:
98    /// - A zero convolution cache means "no previous tokens" (the causal zero
99    ///   padding).
100    /// - A zero state is the standard initial condition `h₀ = 0`. If the
101    ///   block has a learnable initial state, [`Mamba2::forward`] adds it to
102    ///   this state.
103    pub fn zero_cache(&self, batch: usize, device: &Device) -> Mamba2Cache {
104        let [conv_dim, _, conv_kernel] = self.conv1d.weight.dims();
105        let conv_bvk = Tensor::zeros(Shape::new([batch, conv_dim, conv_kernel]), device);
106        let ssm_bhpr = Tensor::zeros(
107            Shape::new([batch, self.nheads(), self.per_head_dim(), self.state_rank]),
108            device,
109        );
110        Mamba2Cache { conv_bvk, ssm_bhpr }
111    }
112}
