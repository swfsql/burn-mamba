1//! [`MuonPlan::describe`]: which optimizer each of a model's parameters lands
2//! on — the quickest way to check a plan against a real model.
3
4use burn::module::{Module, ModuleVisitor, Param, ParamId};
5use burn::prelude::*;
6use core::fmt::Write as _;
7
8use super::MuonPlan;
9
10/// One visited parameter.
11struct Row {
12    path: String,
13    dims: Vec<usize>,
14    id: ParamId,
15}
16
17#[derive(Default)]
18struct Collect {
19    path: Vec<String>,
20    rows: Vec<Row>,
21}
22
23impl ModuleVisitor for Collect {
24    fn enter_module(&mut self, name: &str, _container_type: &str) {
25        self.path.push(name.to_string());
26    }
27    fn exit_module(&mut self, _name: &str, _container_type: &str) {
28        self.path.pop();
29    }
30    fn visit_float<const D: usize>(&mut self, param: &Param<Tensor<D>>) {
31        self.rows.push(Row {
32            path: self.path.join("."),
33            dims: param.val().shape().dims::<D>().to_vec(),
34            id: param.id,
35        });
36    }
37}
38
39impl MuonPlan {
40    /// A per-parameter report of this plan applied to `module`: path, shape, and
41    /// the owning optimizer (with the column segments for a fused weight, `*`
42    /// marking the ones left on AdamW), plus the share of parameters on Muon.
43    ///
44    /// Purely diagnostic — print it once to confirm the plan matches the model
45    /// you actually built.
46    pub fn describe(&self, module: &impl Module) -> String {
47        let mut collect = Collect::default();
48        module.visit(&mut collect);
49
50        let mut out = String::new();
51        let (mut total, mut on_muon) = (0usize, 0usize);
52
53        for row in &collect.rows {
54            let count: usize = row.dims.iter().product();
55            total += count;
56
57            let spec = self
58                .specs
59                .iter()
60                .rev()
61                .find(|spec| spec.param_group().matches(&row.id, Some(row.path.as_str())));
62
63            let owner = match spec {
64                None => "adamw".to_string(),
65                Some(spec) => {
66                    let muon_width: usize =
67                        spec.segments.iter().filter(|s| s.muon).map(|s| s.width).sum();
68                    on_muon += muon_width * row.dims[0];
69                    let segments = spec
70                        .segments
71                        .iter()
72                        .map(|s| {
73                            let mark = if s.muon { "" } else { "*" };
74                            format!("{}:{}{mark}", s.name, s.width)
75                        })
76                        .collect::<Vec<_>>()
77                        .join(" ");
78                    format!("muon[{segments}]")
79                }
80            };
81
82            let _ = writeln!(
83                out,
84                "{:>2}D {:<62} {:?} ({count})  {owner}",
85                row.dims.len(),
86                row.path,
87                row.dims
88            );
89        }
90
91        let share = if total == 0 {
92            0.0
93        } else {
94            100.0 * on_muon as f64 / total as f64
95        };
96        let _ = writeln!(
97            out,
98            "total params: {total}; on muon: {on_muon} ({share:.1}%)  (* = segment left on adamw)"
99        );
100        out
101    }
102}
