1//! Column layout of the fused projection weights — the data the Muon parameter
2//! groups are built from.
3//!
4//! Every family fuses several independent linear maps into one `Linear`, so a
5//! block's weight tensor is a *concatenation* of matrices along its output
6//! (column) axis. Muon orthogonalises a whole matrix at once, so it must be told
7//! where those seams are; a [`ProjSpec`] is that description, and each family
8//! config builds one per fused weight (`Mamba3Config::muon_projections` and
9//! friends), right next to the `d_in_proj()` that sizes them.
10
11use burn::module::ParamGroup;
12
13/// The field names a [`MambaBlock`](crate::modules::MambaBlock) is stored under.
14///
15/// [`Layer`](crate::modules::Layer) holds one (`mamba_block`);
16/// [`BidiLayerPair`](crate::modules::bidi::BidiLayerPair) holds a pair. Matching
17/// a block weight under any of them makes a [`ProjSpec`] independent of the
18/// container — one plan covers a plain stack, a virtual-layer stack, and a
19/// bidirectional stack alike, including hand-written models built from these
20/// pieces.
21pub const BLOCK_CONTAINERS: [&str; 3] = ["mamba_block.", "straight_block.", "reverse_block."];
22
23/// Where in the module tree a [`ProjSpec`]'s path is anchored.
24#[derive(Clone, Copy, Debug, PartialEq, Eq)]
25pub enum ProjScope {
26    /// A weight of the SSM block: the path is matched under each of
27    /// [`BLOCK_CONTAINERS`].
28    Block,
29    /// Any other weight: the path is matched as a plain substring of the
30    /// parameter path (the layer MLP, a bidirectional merge, …).
31    Path,
32}
33
34/// One contiguous column block of a fused projection weight.
35///
36/// `width` is the number of *columns* the block owns (Burn's `Linear` weight is
37/// `[d_input, d_output]`, so the fused axis is dim 1).
38#[derive(Clone, Debug, PartialEq, Eq)]
39pub struct ProjSegment {
40    /// Name of the sub-projection, for diagnostics (`"z"`, `"b"`, `"dt"`, …).
41    pub name: &'static str,
42    /// Number of columns this segment spans.
43    pub width: usize,
44    /// Whether Muon owns this segment (`false` ⇒ it stays on the fallback
45    /// optimizer, i.e. AdamW).
46    pub muon: bool,
47}
48
49impl ProjSegment {
50    /// A segment Muon orthogonalises on its own.
51    pub fn muon(name: &'static str, width: usize) -> Self {
52        Self { name, width, muon: true }
53    }
54
55    /// A segment left to the fallback optimizer.
56    ///
57    /// Used for the channels that produce *per-head scalars* rather than a
58    /// feature vector (Δ, `A`, `λ`): orthogonalising a `[d_model, nheads]` slab
59    /// would force the heads' step-size directions to be mutually orthogonal,
60    /// which is a constraint on a gain, not on a linear map. Same reasoning as
61    /// the usual "no Muon on biases, norm gains or embeddings" rule.
62    pub fn adamw(name: &'static str, width: usize) -> Self {
63        Self { name, width, muon: false }
64    }
65}
66
67/// One 2-D weight tensor and how its columns split into independent maps.
68#[derive(Clone, Debug, PartialEq, Eq)]
69pub struct ProjSpec {
70    /// Trailing part of the parameter path identifying the weight, e.g.
71    /// `"in_proj.weight"`, anchored per [`Self::scope`].
72    pub path: String,
73    /// Where `path` is anchored.
74    pub scope: ProjScope,
75    /// The column blocks, in order; their widths must sum to the weight's output
76    /// width.
77    pub segments: Vec<ProjSegment>,
78}
79
80impl ProjSpec {
81    /// A fused weight of the SSM block.
82    pub fn block(path: impl Into<String>, segments: Vec<ProjSegment>) -> Self {
83        Self { path: path.into(), scope: ProjScope::Block, segments }
84    }
85
86    /// An unfused weight of the SSM block, Muon owns it in full.
87    pub fn block_whole(path: impl Into<String>, width: usize) -> Self {
88        Self::block(path, vec![ProjSegment::muon("all", width)])
89    }
90
91    /// A fused weight matched by plain path substring.
92    pub fn path(path: impl Into<String>, segments: Vec<ProjSegment>) -> Self {
93        Self { path: path.into(), scope: ProjScope::Path, segments }
94    }
95
96    /// An unfused weight matched by plain path substring, Muon owns it in full.
97    pub fn path_whole(path: impl Into<String>, width: usize) -> Self {
98        Self::path(path, vec![ProjSegment::muon("all", width)])
99    }
100
101    /// Total output width (sum of the segment widths).
102    pub fn width(&self) -> usize {
103        self.segments.iter().map(|s| s.width).sum()
104    }
105
106    /// Whether any segment is Muon's.
107    pub fn has_muon(&self) -> bool {
108        self.segments.iter().any(|s| s.muon)
109    }
110
111    /// Whether Muon owns the whole tensor as a single block (so stock
112    /// [`Muon`](burn::optim::Muon) applies directly, no splitting needed).
113    pub fn is_whole_muon(&self) -> bool {
114        self.segments.len() == 1 && self.segments[0].muon
115    }
116
117    /// The path substrings this spec matches (one per [`BLOCK_CONTAINERS`]
118    /// entry for [`ProjScope::Block`], otherwise the path itself).
119    pub fn predicates(&self) -> Vec<String> {
120        match self.scope {
121            ProjScope::Block => BLOCK_CONTAINERS
122                .iter()
123                .map(|container| format!("{container}{}", self.path))
124                .collect(),
125            ProjScope::Path => vec![self.path.clone()],
126        }
127    }
128
129    /// The parameter group selecting this weight (OR over [`Self::predicates`]).
130    pub fn param_group(&self) -> ParamGroup {
131        ParamGroup::from_any_predicates(self.predicates())
132    }
133}
