1//! [`Segmented`]: run a different optimizer on each column block of a fused
2//! projection weight.
3//!
4//! Burn's [`Muon`](burn::optim::Muon) orthogonalises the *whole* matrix it is handed. Handing it a
5//! fused `in_proj` would couple the singular values of maps that have nothing to
6//! do with each other (the gate `z`, the values `x`, the SSM keys/queries `B`/`C`,
7//! the per-head Δ/`A`/`λ` scalars, …). [`Segmented`] slices the weight and its
8//! gradient along the fused axis, steps each block with its own optimizer, and
9//! concatenates the results — so each sub-matrix is orthogonalised (and
10//! shape-LR-adjusted) on its own, exactly as if it had been a separate `Linear`.
11//!
12//! Nothing about the *model* changes: the forward pass keeps one fused GEMM.
13//!
14//! Slicing along columns is exact for every optimizer used here: AdamW is
15//! elementwise, and Muon's Newton–Schulz is per-matrix. A [`Segmented`] whose
16//! blocks are all AdamW is therefore bit-comparable to plain AdamW on the whole
17//! tensor (asserted in `tests.rs`).
18

27/// The optimizer owning one column block.
28#[derive(Clone)]
29enum BlockOptim {
30    /// Muon — orthogonalised momentum-SGD, for genuine feature maps.
31    Muon(Muon),
32    /// AdamW — the fallback for scalar-producing (or otherwise unsuitable) blocks.
33    AdamW(AdamW),
34}
35
36/// One column block's optimizer state.
37// The AdamW variant is the bigger one; a handful of these exist per fused
38// weight, so the padding is not worth an extra indirection.
39#[allow(clippy::large_enum_variant)]
40#[derive(Clone)]
41pub enum BlockState<const D: usize> {
42    /// State of a [`Muon`] block.
43    Muon(MuonState<D>),
44    /// State of an [`AdamW`] block.
45    AdamW(AdamWState<D>),
46}
47
48/// State of a [`Segmented`] optimizer: one entry per column block, in order.
49#[derive(Clone)]
50pub struct SegmentedState<const D: usize> {
51    /// Per-block states, in the same order as the [`ProjSpec`] segments.
52    pub blocks: Vec<BlockState<D>>,
53}
54
55/// Per-column-block optimizer over a fused projection weight.
56///
57/// Built by [`ProjSpec`]-driven [`MuonPlan`](super::MuonPlan) assembly; the
58/// blocks' widths must sum to the parameter's size along [`Self::dim`].
59#[derive(Clone)]
60pub struct Segmented {
61    optims: Vec<BlockOptim>,
62    widths: Vec<usize>,
63    dim: usize,
64}
65
66impl Segmented {
67    /// Build the per-block optimizers for `spec`, splitting along `dim`
68    /// (`1` for a Burn `Linear` weight, whose layout is `[d_input, d_output]`).
69    pub fn new(spec: &ProjSpec, muon: Muon, adamw: AdamW, dim: usize) -> Self {
70        let optims = spec
71            .segments
72            .iter()
73            .map(|s| {
74                if s.muon {
75                    BlockOptim::Muon(muon.clone())
76                } else {
77                    BlockOptim::AdamW(adamw.clone())
78                }
79            })
80            .collect();
81        let widths = spec.segments.iter().map(|s| s.width).collect();
82        Self { optims, widths, dim }
83    }
84
85    /// The axis the blocks are laid out along.
86    pub fn dim(&self) -> usize {
87        self.dim
88    }
89}
90
91impl Optimizer for Segmented {
92    type State<const D: usize> = SegmentedState<D>;
93
94    fn step<const D: usize>(
95        &self,
96        lr: LearningRate,
97        tensor: Tensor<D>,
98        grad: Tensor<D>,
99        state: Option<Self::State<D>>,
100    ) -> (Tensor<D>, Option<Self::State<D>>) {
101        assert!(
102            self.dim < D,
103            "Segmented: split dim {} out of range for a {D}D parameter",
104            self.dim
105        );
106        let total: usize = self.widths.iter().sum();
107        assert_eq!(
108            tensor.shape().dims::<D>()[self.dim],
109            total,
110            "Segmented: the parameter's dim-{} width does not match the projection spec",
111            self.dim
112        );
113
114        let tensors = tensor.split_with_sizes(self.widths.clone(), self.dim);
115        let grads = grad.split_with_sizes(self.widths.clone(), self.dim);
116
117        // A missing (first-step) state, or one whose length drifted from the
118        // spec, restarts every block from scratch rather than mis-pairing them.
119        let mut prev: Vec<Option<BlockState<D>>> = match state {
120            Some(s) if s.blocks.len() == self.optims.len() => {
121                s.blocks.into_iter().map(Some).collect()
122            }
123            _ => (0..self.optims.len()).map(|_| None).collect(),
124        };
125
126        let mut out = Vec::with_capacity(self.optims.len());
127        let mut blocks = Vec::with_capacity(self.optims.len());
128
129        for (i, optim) in self.optims.iter().enumerate() {
130            let (t, g) = (tensors[i].clone(), grads[i].clone());
131            match optim {
132                BlockOptim::Muon(muon) => {
133                    let prev = match prev[i].take() {
134                        Some(BlockState::Muon(s)) => Some(s),
135                        _ => None,
136                    };
137                    let (t, s) = muon.step(lr, t, g, prev);
138                    out.push(t);
139                    blocks.extend(s.map(BlockState::Muon));
140                }
141                BlockOptim::AdamW(adamw) => {
142                    let prev = match prev[i].take() {
143                        Some(BlockState::AdamW(s)) => Some(s),
144                        _ => None,
145                    };
146                    let (t, s) = adamw.step(lr, t, g, prev);
147                    out.push(t);
148                    blocks.extend(s.map(BlockState::AdamW));
149                }
150            }
151        }
152
153        // Both inner optimizers always return a state, so `blocks` is complete.
154        assert_eq!(blocks.len(), self.optims.len());
155        (Tensor::cat(out, self.dim), Some(SegmentedState { blocks }))
156    }
157
158    fn to_device<const D: usize>(state: Self::State<D>, device: &Device) -> Self::State<D> {
159        let blocks = state
160            .blocks
161            .into_iter()
162            .map(|b| match b {
163                BlockState::Muon(s) => BlockState::Muon(Muon::to_device(s, device)),
164                BlockState::AdamW(s) => BlockState::AdamW(AdamW::to_device(s, device)),
165            })
166            .collect();
167        SegmentedState { blocks }
168    }
169}

171/// Hand-written because the `RecordState` derive covers `Vec<Tensor>` but not a
172/// `Vec` of nested states, and because the reload has no access to the spec: the
173/// two block kinds are told apart by their leaf names (`momentum.velocity` for
174/// Muon, `momentum.moment_1`/`moment_2` for AdamW), which never overlap.
175impl<const D: usize> RecordState for SegmentedState<D> {
176    fn state_flatten(&self, prefix: &str, out: &mut StateSink) {
177        for (i, block) in self.blocks.iter().enumerate() {
178            let prefix = join_index(prefix, i);
179            match block {
180                BlockState::Muon(s) => s.state_flatten(&prefix, out),
181                BlockState::AdamW(s) => s.state_flatten(&prefix, out),
182            }
183        }
184    }
185
186    fn state_unflatten(prefix: &str, src: &mut StateSource, device: &Device) -> Option<Self> {
187        let mut blocks = Vec::new();
188        for i in 0.. {
189            let prefix = join_index(prefix, i);
190            if !src.has_under(&prefix) {
191                break;
192            }
193            // A failed attempt consumes nothing (the leaf it looks for is
194            // absent), so trying Muon first is safe.
195            let block = MuonState::state_unflatten(&prefix, src, device)
196                .map(BlockState::Muon)
197                .or_else(|| {
198                    AdamWState::state_unflatten(&prefix, src, device).map(BlockState::AdamW)
199                })?;
200            blocks.push(block);
201        }
202
203        (!blocks.is_empty()).then_some(SegmentedState { blocks })
204    }
205}
