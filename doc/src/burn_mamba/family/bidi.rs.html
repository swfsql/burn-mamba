1//! Runtime-selectable bidirectional stacks: one enum over the
2//! [`BidiLayers`](burn_stack::modules::BidiLayers) monomorphisations of the
3//! three families. Its serializable config is in [`crate::family::config`].
4
5use crate::prelude::*;
6use burn::prelude::*;
7use burn_stack::modules::bidi::BidiLayers;
8use burn_stack::utils::ClassCursors;
9
10/// A runtime-selectable bidirectional stack: the same paired straight/reverse
11/// structure over any Mamba-x family, chosen at runtime. The forward-only
12/// counterpart of [`MambaLatentNet`] for non-autoregressive tasks.
13#[derive(Module, Debug)]
14pub enum MambaBidiLayers {
15    /// Mamba-1 bidirectional stack.
16    #[cfg(feature = "mamba1")]
17    Mamba1(BidiLayers<crate::mamba1::prelude::Mamba1>),
18    /// Mamba-2 bidirectional stack.
19    #[cfg(feature = "mamba2")]
20    Mamba2(BidiLayers<crate::mamba2::prelude::Mamba2>),
21    /// Mamba-3 bidirectional stack.
22    #[cfg(feature = "mamba3")]
23    Mamba3(BidiLayers<crate::mamba3::prelude::Mamba3>),
24}
25
26impl MambaBidiLayers {
27    /// Output positions of the stack-level class latents for an `orig_len`
28    /// input. With them, a caller can read a class latent back out of the
29    /// longer `forward` output (for example, as a pooled summary). A marker
30    /// that never lands (a `Custom` at or past the end) reports a position
31    /// past that output.
32    pub fn class_latent_output_indices(&self, orig_len: usize) -> Vec<usize> {
33        match self {
34            #[cfg(feature = "mamba1")]
35            Self::Mamba1(layers) => layers.class_latent_output_indices(orig_len),
36            #[cfg(feature = "mamba2")]
37            Self::Mamba2(layers) => layers.class_latent_output_indices(orig_len),
38            #[cfg(feature = "mamba3")]
39            Self::Mamba3(layers) => layers.class_latent_output_indices(orig_len),
40        }
41    }
42
43    /// Full-sequence bidirectional pass. The `ssd_path` must match the family
44    /// of the stack. A mismatch is a caller error and panics. `pad` marks a
45    /// right-padded batch ([`BidiLayers::forward`]).
46    pub fn forward(
47        &self,
48        x: Tensor<3>,
49        caches: Option<MambaCaches>,
50        ssd_path: MambaSsdPath,
51        class: Option<&mut ClassCursors>,
52        pad: Option<Tensor<2, Bool>>,
53    ) -> (Tensor<3>, MambaCaches) {
54        match self {
55            #[cfg(feature = "mamba1")]
56            Self::Mamba1(layers) => {
57                let caches = caches.map(|c| match c {
58                    MambaCaches::Mamba1(c) => c,
59                    #[allow(unreachable_patterns)]
60                    _ => panic!("cache family does not match Mamba-1 bidi stack"),
61                });
62                match ssd_path {
63                    MambaSsdPath::Mamba1 => {}
64                    #[allow(unreachable_patterns)]
65                    _ => panic!("ssd_path family does not match Mamba-1 bidi stack"),
66                }
67                let (y, c) = layers.forward(x, caches, (), class, pad);
68                (y, MambaCaches::Mamba1(c))
69            }
70            #[cfg(feature = "mamba2")]
71            Self::Mamba2(layers) => {
72                let caches = caches.map(|c| match c {
73                    MambaCaches::Mamba2(c) => c,
74                    #[allow(unreachable_patterns)]
75                    _ => panic!("cache family does not match Mamba-2 bidi stack"),
76                });
77                let path = match ssd_path {
78                    MambaSsdPath::Mamba2(p) => p,
79                    #[allow(unreachable_patterns)]
80                    _ => panic!("ssd_path family does not match Mamba-2 bidi stack"),
81                };
82                let (y, c) = layers.forward(x, caches, path, class, pad);
83                (y, MambaCaches::Mamba2(c))
84            }
85            #[cfg(feature = "mamba3")]
86            Self::Mamba3(layers) => {
87                let caches = caches.map(|c| match c {
88                    MambaCaches::Mamba3(c) => c,
89                    #[allow(unreachable_patterns)]
90                    _ => panic!("cache family does not match Mamba-3 bidi stack"),
91                });
92                let path = match ssd_path {
93                    MambaSsdPath::Mamba3(p) => p,
94                    #[allow(unreachable_patterns)]
95                    _ => panic!("ssd_path family does not match Mamba-3 bidi stack"),
96                };
97                let (y, c) = layers.forward(x, caches, path, class, pad);
98                (y, MambaCaches::Mamba3(c))
99            }
100        }
101    }
102}
